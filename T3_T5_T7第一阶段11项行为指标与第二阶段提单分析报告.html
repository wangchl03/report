综合结论：指标单独效果与组合效果

1. 期次选择是最强的单指标意图信号。 4 次以上期次切换时，T3/T5/T7 提单率为 87.56% / 84.68% / 79.66%，较 0 次高 22.80 / 25.50 / 25.62pp；但覆盖仅 1.28% / 1.69% / 2.04%，适合高价值、小规模优先触达。
2. 还款计划与金额切换更适合作为可规模化标签。 T7 中，查看还款计划 1 次以上覆盖 21.68%，提单率 65.68%~75.50%；金额切换 1 次以上覆盖 19.54%，提单率 66.92%~72.87%。它们兼具较高覆盖和显著梯度。
3. 进入提单页是广覆盖但需要与决策信号叠加的行为。 仅进入提单页的提单率为 T3/T5/T7 65.98%/59.35%/53.10%，只比无上述行为高 5.27/4.98/3.99pp；因此单独进入提单页并不足以代表强意图。
4. 决策行为组合显著提升提单率。 “提单页 + 三项决策行为”提单率为 81.53%/78.11%/73.72%，较无上述行为分别高 20.82/23.74/24.61pp；覆盖为 2.73%/3.46%/4.21%。可建立“高意图”人群，但规模有限。
5. 两项决策行为是覆盖与效果更均衡的运营层。 覆盖 T3/T5/T7 的 9.19%/10.75%/11.81%，提单率为 79.39%/75.26%/70.69%。相比最高组合覆盖更大、提单率仅低 2~3pp，适合作为主要转化触达层。
6. 时间窗口差异。 相同行为组合下，T3 提单率最高、T7 最低，例如两项决策行为为 79.39%/75.26%/70.69%；但 T7 高意图组合覆盖最高（4.21%），说明延长窗口会捕捉更多持续决策行为，同时整体后续转化概率递减。
7. 数据质量与策略边界。 三项优惠券指标全为 0 次，应先排查埋点。所有结果是关联关系，建议将“无行为 / 单项 / 两项 / 高意图组合”做随机实验，联合评估放款金额、逾期和最终盈利额。

/*
  T3/T5/T7 第一阶段 11 个行为指标分段与第二阶段提单表现。
  行为数值为第一阶段累计次数；活跃判定不使用这些指标，而使用 active_date 是否有记录。
  分母为干净结清、同日获额通过、且结清日和第一阶段未提单的可观察事件。
*/
WITH
all_orders AS (
    SELECT user_id, serial_id, apply_time, remit_time, repaid_time, repaid_date,
           remit_date, loan_status_code, is_remit, reg_date
    FROM wangchuanliang.order_loan_f_v2_copy
    WHERE reg_date BETWEEN DATE '2026-01-01' AND DATE '2026-07-15'
      AND apply_time IS NOT NULL
),
all_users AS (
    SELECT DISTINCT user_id FROM all_orders
    WHERE is_remit = 1 AND remit_date > DATE '2026-01-01'
),
settled_raw AS (
    SELECT o.user_id, o.serial_id, o.repaid_date, o.repaid_time,
           ROW_NUMBER() OVER (PARTITION BY o.user_id ORDER BY o.repaid_time, o.serial_id) AS settle_seq
    FROM all_orders o JOIN all_users u ON u.user_id = o.user_id
    WHERE o.is_remit = 1 AND o.loan_status_code = 7
      AND o.repaid_time IS NOT NULL AND o.remit_time IS NOT NULL
),
clean_settle AS (
    SELECT s.* FROM settled_raw s
    WHERE NOT EXISTS (
        SELECT 1 FROM all_orders o2
        WHERE o2.user_id = s.user_id AND o2.serial_id <> s.serial_id
          AND o2.apply_time < s.repaid_time
          AND (o2.loan_status_code IN (5, 8)
               OR (o2.loan_status_code = 7 AND o2.repaid_time > s.repaid_time))
    )
),
credit_pass AS (
    SELECT DISTINCT ON (v.user_id, v.vir_date)
           v.user_id, v.vir_date, to_timestamp(v.vir_unix) AS vir_time
    FROM wangchuanliang.order_vir_f_copy v
    WHERE v.is_pass1 = 1 AND v.vir_date > DATE '2026-01-01'
    ORDER BY v.user_id, v.vir_date, to_timestamp(v.vir_unix) DESC
),
eligible AS (
    SELECT cs.* FROM clean_settle cs
    JOIN credit_pass cp ON cp.user_id = cs.user_id
                       AND cp.vir_date = cs.repaid_date::date
                       AND cp.vir_time > cs.repaid_time
),
applies AS (
    SELECT user_id, apply_time FROM all_orders
    WHERE apply_time > TIMESTAMP '2026-01-01'
),
merged AS (
    SELECT user_id, repaid_time AS ts, 'settle' AS kind, settle_seq::float AS settle_seq FROM eligible
    UNION ALL
    SELECT user_id, apply_time, 'apply', NULL::float FROM applies
),
ranked AS (
    SELECT *, ROW_NUMBER() OVER (
        PARTITION BY user_id
        ORDER BY ts, CASE kind WHEN 'settle' THEN 0 ELSE 1 END
    ) AS rn
    FROM merged
),
pairs AS (
    SELECT s.user_id, s.ts AS repaid_time, s.settle_seq, a.ts AS next_apply_time
    FROM ranked s
    LEFT JOIN ranked a ON a.user_id = s.user_id AND a.rn = s.rn + 1 AND a.kind = 'apply'
    WHERE s.kind = 'settle'
),
thresholds AS (
    SELECT 'T3' AS cohort, 3 AS threshold_days
    UNION ALL SELECT 'T5', 5
    UNION ALL SELECT 'T7', 7
),
candidates AS (
    SELECT t.cohort, t.threshold_days, p.user_id, p.repaid_time::date AS repaid_date,
           CASE WHEN p.next_apply_time IS NULL THEN 0 ELSE 1 END AS phase2_apply,
           CASE WHEN p.next_apply_time IS NULL THEN NULL
                ELSE p.next_apply_time::date - p.repaid_time::date END AS gap_days
    FROM pairs p CROSS JOIN thresholds t
    WHERE p.settle_seq <= 10
      AND (p.next_apply_time IS NULL
           OR p.next_apply_time::date > p.repaid_time::date + t.threshold_days)
),
/* 每个结清事件汇总第一阶段 11 个行为次数。 */
behavior_metrics AS (
    SELECT
        c.*,
        COALESCE(SUM(e.active_status), 0) AS active_status,
        COALESCE(SUM(e.newpage), 0) AS newpage,
        COALESCE(SUM(e.amount_selection), 0) AS amount_selection,
        COALESCE(SUM(e.contract_newdetails), 0) AS contract_newdetails,
        COALESCE(SUM(e.term_selection), 0) AS term_selection,
        COALESCE(SUM(e.newconfirm), 0) AS newconfirm,
        COALESCE(SUM(e.plan_newdetails), 0) AS plan_newdetails,
        COALESCE(SUM(e.purpose_newdetails), 0) AS purpose_newdetails,
        COALESCE(SUM(e.coupons_back), 0) AS coupons_back,
        COALESCE(SUM(e.coupons_confirm), 0) AS coupons_confirm,
        COALESCE(SUM(e.coupons_page), 0) AS coupons_page
    FROM candidates c
    LEFT JOIN wangchuanliang.event_track e
      ON e.user_id = c.user_id
     AND e.active_date > c.repaid_date
     AND e.active_date <= c.repaid_date + c.threshold_days
    GROUP BY c.cohort, c.threshold_days, c.user_id,
             c.repaid_date, c.phase2_apply, c.gap_days
),
/* 将宽表转成长表，保证 11 个指标采用相同的分段统计逻辑。 */
metric_long AS (
    SELECT cohort, phase2_apply, gap_days, 'active_status' AS metric_code, '活跃度' AS metric_name, active_status::numeric AS metric_value FROM behavior_metrics
    UNION ALL SELECT cohort, phase2_apply, gap_days, 'newpage', '进入提单页次数', newpage::numeric FROM behavior_metrics
    UNION ALL SELECT cohort, phase2_apply, gap_days, 'amount_selection', '切换借款金额次数', amount_selection::numeric FROM behavior_metrics
    UNION ALL SELECT cohort, phase2_apply, gap_days, 'contract_newdetails', '查看借款合同次数', contract_newdetails::numeric FROM behavior_metrics
    UNION ALL SELECT cohort, phase2_apply, gap_days, 'term_selection', '切换借款期次次数', term_selection::numeric FROM behavior_metrics
    UNION ALL SELECT cohort, phase2_apply, gap_days, 'newconfirm', '确认提单次数', newconfirm::numeric FROM behavior_metrics
    UNION ALL SELECT cohort, phase2_apply, gap_days, 'plan_newdetails', '查看还款计划次数', plan_newdetails::numeric FROM behavior_metrics
    UNION ALL SELECT cohort, phase2_apply, gap_days, 'purpose_newdetails', '选择借款原因次数', purpose_newdetails::numeric FROM behavior_metrics
    UNION ALL SELECT cohort, phase2_apply, gap_days, 'coupons_back', '优惠券拦截关闭次数', coupons_back::numeric FROM behavior_metrics
    UNION ALL SELECT cohort, phase2_apply, gap_days, 'coupons_confirm', '优惠券拦截点击次数', coupons_confirm::numeric FROM behavior_metrics
    UNION ALL SELECT cohort, phase2_apply, gap_days, 'coupons_page', '进入优惠券挽留页次数', coupons_page::numeric FROM behavior_metrics
),
/* 次数分段：0、1、2-3、4+；active_status 按强度使用 0、1-10、11-30、31+。 */
bucketed AS (
    SELECT *,
        CASE
            WHEN metric_code = 'active_status' AND metric_value = 0 THEN '0'
            WHEN metric_code = 'active_status' AND metric_value <= 10 THEN '1-10'
            WHEN metric_code = 'active_status' AND metric_value <= 30 THEN '11-30'
            WHEN metric_code = 'active_status' THEN '31+'
            WHEN metric_value = 0 THEN '0 次'
            WHEN metric_value = 1 THEN '1 次'
            WHEN metric_value <= 3 THEN '2-3 次'
            ELSE '4 次及以上'
        END AS value_bucket,
        CASE
            WHEN metric_code = 'active_status' AND metric_value = 0 THEN 1
            WHEN metric_code = 'active_status' AND metric_value <= 10 THEN 2
            WHEN metric_code = 'active_status' AND metric_value <= 30 THEN 3
            WHEN metric_code = 'active_status' THEN 4
            WHEN metric_value = 0 THEN 1
            WHEN metric_value = 1 THEN 2
            WHEN metric_value <= 3 THEN 3
            ELSE 4
        END AS bucket_order
    FROM metric_long
)
SELECT
    metric_code,
    metric_name,
    cohort,
    value_bucket,
    COUNT(*) AS events,
    ROUND(COUNT(*) * 100.0 / SUM(COUNT(*)) OVER (PARTITION BY metric_code, cohort), 2) AS user_share_pct,
    SUM(phase2_apply) AS phase2_apply_events,
    ROUND(SUM(phase2_apply) * 100.0 / COUNT(*), 2) AS phase2_apply_rate,
    ROUND(AVG(metric_value)::numeric, 2) AS avg_metric_value,
    ROUND(AVG(CASE WHEN phase2_apply = 1 THEN gap_days END)::numeric, 2) AS avg_gap_days
FROM bucketed
GROUP BY metric_code, metric_name, cohort, value_bucket, bucket_order
ORDER BY metric_code, cohort, bucket_order;
