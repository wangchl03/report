五、结论与运营建议

1. T0 是自然转化主战场

T0 稳定占约半数。对这部分用户，应以顺畅获额、页面体验和必要提醒为主，避免默认发券，以防对本会即时提单的用户产生无效让利。

2. T1–T3 是低成本提醒优先窗口

T1 约 10%–12%，T2–T3 合计约 9%–10%。用户已较接近决策，可优先尝试非价格提醒、方案展示或轻量权益，而非直接高力度降息。

3. T4–T15 是券实验的重点对象

该人群占比随月份上升，反映更多用户存在延迟决策。应在已进入 App、存在提单页或方案比较行为时再进入分层随机实验，比较无券、3 天券与 7 天券的增量。

4. T15+ 需区分真实慢转化与观察不足

完整月份中 T15+ 约 10%–12%；7 月和 8 月偏低主要受数据观察窗口不足影响。后续看慢转化时，应至少保留 D+30 的成熟观察期。

附录：完整 PGSQL 查询代码（含口径注释）

以下代码用于生成本报告的日、周、月三个维度 T0–T15+ 首次提单时效分布。每个“用户 × 干净结清事件”仅归入一个首次提单时段。

/* ================================================================
   T0–T15+ 首次提单时效分布：日 / 周 / 月统计
   统计对象：2026-01-01 至 2026-08-01 的干净结清事件；
   获额条件：结清当日 is_pass1 = 1，且获额通过时间晚于结清时间。
   说明：一个“用户 × 干净结清事件”只取结清后的首次提单，
   因而 T0、T1、…、T15+ 互斥，单个周期的构成占比合计为 100%。
   ================================================================ */

WITH all_orders AS (
    /* 订单基础表：保留本次分析所需字段 */
    SELECT
        o.user_id,
        o.serial_id,
        o.apply_time,
        o.repaid_time,
        o.repaid_date,
        o.loan_status_code,
        o.is_remit
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    WHERE o.apply_time < CURRENT_DATE + INTERVAL '1 day'
),

settled_orders AS (
    /* 已结清订单；状态 7 表示当前已结清。
       订单结清前是否曾逾期不在这里限制，干净结清在后续时点判断。 */
    SELECT
        o.user_id,
        o.serial_id,
        o.repaid_time,
        o.repaid_date::date AS repaid_date
    FROM all_orders AS o
    WHERE o.is_remit = 1
      AND o.loan_status_code = 7
      AND o.repaid_time IS NOT NULL
      AND o.repaid_date::date >= DATE '2026-01-01'
      AND o.repaid_date::date <= DATE '2026-08-01'
),

last_settle_of_day AS (
    /* 同一用户同日多笔结清时，仅取结清时间最后的一笔。
       repaid_time 不会相同，因此不以 serial_id 参与排序。 */
    SELECT DISTINCT ON (
        s.user_id,
        s.repaid_date
    )
        s.user_id,
        s.serial_id,
        s.repaid_time,
        s.repaid_date
    FROM settled_orders AS s
    ORDER BY
        s.user_id,
        s.repaid_date,
        s.repaid_time DESC
),

clean_last_settles AS (
    /* 干净结清：目标订单结清的那一刻，用户不存在其他在途订单。
       当前仍为放款中(5)、逾期中(8)，或虽已结清但结清时间更晚(7)，
       均意味着在目标结清时点另一笔订单尚未结清，故排除。 */
    SELECT
        s.user_id,
        s.serial_id,
        s.repaid_time,
        s.repaid_date
    FROM last_settle_of_day AS s
    WHERE NOT EXISTS (
        SELECT
            1
        FROM all_orders AS o2
        WHERE o2.user_id = s.user_id
          AND o2.serial_id <> s.serial_id
          AND o2.apply_time < s.repaid_time
          AND (
                o2.loan_status_code = 5
             OR o2.loan_status_code = 8
             OR (
                    o2.loan_status_code = 7
                AND o2.repaid_time > s.repaid_time
                )
          )
    )
),

credit_pass AS (
    /* 获额通过记录；vir_unix 转为时间戳后用于与结清时刻比较。 */
    SELECT
        v.user_id,
        v.vir_date::date AS vir_date,
        to_timestamp(v.vir_unix) AS vir_time
    FROM wangchuanliang.order_vir_f_copy AS v
    WHERE v.is_pass1 = 1
),

eligible_events AS (
    /* 分母：干净结清后，当天获额通过、且通过时刻晚于结清时刻的事件。 */
    SELECT
        c.user_id,
        c.serial_id,
        c.repaid_time,
        c.repaid_date
    FROM clean_last_settles AS c
    WHERE EXISTS (
        SELECT
            1
        FROM credit_pass AS cp
        WHERE cp.user_id = c.user_id
          AND cp.vir_date = c.repaid_date
          AND cp.vir_time > c.repaid_time
    )
),

first_apply_after_settle AS (
    /* 每个合格事件只取结清后的第一次提单，确保时段唯一归属。 */
    SELECT
        e.user_id,
        e.serial_id,
        e.repaid_date,
        MIN(o.apply_time)::date AS first_apply_date
    FROM eligible_events AS e
    INNER JOIN all_orders AS o
        ON o.user_id = e.user_id
       AND o.apply_time > e.repaid_time
    GROUP BY
        e.user_id,
        e.serial_id,
        e.repaid_date
),

bucketed_first_apply AS (
    /* 按结清日期与首次提单日期的自然日间隔划分 T0–T15+。 */
    SELECT
        f.user_id,
        f.serial_id,
        f.repaid_date,
        CASE
            WHEN f.first_apply_date = f.repaid_date THEN 'T0'
            WHEN f.first_apply_date = f.repaid_date + 1 THEN 'T1'
            WHEN f.first_apply_date = f.repaid_date + 2 THEN 'T2'
            WHEN f.first_apply_date = f.repaid_date + 3 THEN 'T3'
            WHEN f.first_apply_date = f.repaid_date + 4 THEN 'T4'
            WHEN f.first_apply_date = f.repaid_date + 5 THEN 'T5'
            WHEN f.first_apply_date = f.repaid_date + 6 THEN 'T6'
            WHEN f.first_apply_date = f.repaid_date + 7 THEN 'T7'
            WHEN f.first_apply_date <= f.repaid_date + 15 THEN 'T8-T15'
            ELSE 'T15+'
        END AS time_bucket
    FROM first_apply_after_settle AS f
),

bucket_order AS (
    /* 固定输出顺序，并补齐某个周期不存在的时段。 */
    SELECT 1 AS sort_no, 'T0' AS time_bucket
    UNION ALL SELECT 2, 'T1'
    UNION ALL SELECT 3, 'T2'
    UNION ALL SELECT 4, 'T3'
    UNION ALL SELECT 5, 'T4'
    UNION ALL SELECT 6, 'T5'
    UNION ALL SELECT 7, 'T6'
    UNION ALL SELECT 8, 'T7'
    UNION ALL SELECT 9, 'T8-T15'
    UNION ALL SELECT 10, 'T15+'
),

periodized_events AS (
    /* 同一份首次提单明细展开为日、周、月三个统计粒度。 */
    SELECT
        '日' AS period_type,
        b.repaid_date AS period_start,
        b.time_bucket
    FROM bucketed_first_apply AS b

    UNION ALL

    SELECT
        '周' AS period_type,
        date_trunc('week', b.repaid_date)::date AS period_start,
        b.time_bucket
    FROM bucketed_first_apply AS b

    UNION ALL

    SELECT
        '月' AS period_type,
        date_trunc('month', b.repaid_date)::date AS period_start,
        b.time_bucket
    FROM bucketed_first_apply AS b
),

period_list AS (
    /* 每个粒度实际存在的日期、周、月份。 */
    SELECT DISTINCT
        p.period_type,
        p.period_start
    FROM periodized_events AS p
),

period_bucket_count AS (
    /* 统计各周期、各时段的首次提单人数。 */
    SELECT
        p.period_type,
        p.period_start,
        p.time_bucket,
        COUNT(*) AS apply_users
    FROM periodized_events AS p
    GROUP BY
        p.period_type,
        p.period_start,
        p.time_bucket
),

period_total AS (
    /* 分布分析的分母：该周期内所有已发生后续首次提单的事件数。 */
    SELECT
        p.period_type,
        p.period_start,
        COUNT(*) AS total_apply_users
    FROM periodized_events AS p
    GROUP BY
        p.period_type,
        p.period_start
)

SELECT
    pl.period_type AS 统计粒度,
    pl.period_start AS 结清周期起始日,
    bo.time_bucket AS 提单时段,
    COALESCE(pbc.apply_users, 0) AS 提单用户数,
    pt.total_apply_users AS 周期总提单用户数,
    ROUND(
        COALESCE(pbc.apply_users, 0)::numeric
        / NULLIF(pt.total_apply_users, 0),
        6
    ) AS 提单用户占比
FROM period_list AS pl
CROSS JOIN bucket_order AS bo
INNER JOIN period_total AS pt
    ON pt.period_type = pl.period_type
   AND pt.period_start = pl.period_start
LEFT JOIN period_bucket_count AS pbc
    ON pbc.period_type = pl.period_type
   AND pbc.period_start = pl.period_start
   AND pbc.time_bucket = bo.time_bucket
ORDER BY
    CASE pl.period_type
        WHEN '日' THEN 1
        WHEN '周' THEN 2
        WHEN '月' THEN 3
    END,
    pl.period_start,
    bo.sort_no;
