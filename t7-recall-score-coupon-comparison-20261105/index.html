WITH blacklist AS (

    VALUES (1, '637-1000'),
        (2, '561-636'),
        (3, '518-560'),
        (4, '460-517'),
        (5, '459-')
), source_users AS (
    SELECT DISTINCT churn_user_id::bigint AS user_id,
           TRIM(coupon_level_2::text) AS coupon_label, score
    FROM wangchuanliang.t7recalllist_0930
    WHERE churn_user_id IS NOT NULL
      AND NOT EXISTS (SELECT 1 FROM blacklist b WHERE b.user_id=churn_user_id::bigint)
), coupon_users AS (
    SELECT u.user_id, u.score, u.coupon_label, g.group_id
    FROM source_users u
    JOIN label_order g ON g.coupon_label=u.coupon_label
), group_sizes AS (
    SELECT g.group_id, g.coupon_label, COUNT(c.user_id) AS users
    FROM label_order g
    LEFT JOIN coupon_users c ON c.group_id=g.group_id
    GROUP BY g.group_id, g.coupon_label
), rank_bounds AS (
    SELECT group_id, coupon_label, users,
           SUM(users) OVER (ORDER BY group_id)-users+1 AS min_rank,
           SUM(users) OVER (ORDER BY group_id) AS max_rank
    FROM group_sizes
), ranked_score AS (
    SELECT user_id, score, coupon_label,
           ROW_NUMBER() OVER (ORDER BY score DESC, user_id ASC) AS score_rank
    FROM source_users
    WHERE score BETWEEN 0 AND 3000
), sample_members AS (
    SELECT 'coupon'::text AS scheme, c.group_id, c.user_id, c.score,
           c.coupon_label, NULL::bigint AS score_rank
    FROM coupon_users c
    UNION ALL
    SELECT 'score'::text AS scheme, g.group_id, s.user_id, s.score,
           s.coupon_label, s.score_rank
    FROM ranked_score s
    JOIN rank_bounds g ON s.score_rank BETWEEN g.min_rank AND g.max_rank
), candidate_users AS (
    SELECT DISTINCT user_id FROM sample_members
), first_application AS (
    SELECT o.user_id, MIN(o.apply_time) AS first_apply_time
    FROM wangchuanliang.order_loan_f_v2_copy o
    JOIN candidate_users u ON u.user_id=o.user_id
    WHERE o.apply_time >= TIMESTAMP '2026-09-30 00:00:00'
      AND o.apply_time < TIMESTAMP '2026-10-30 00:00:00'
      AND o.apply_date >= DATE '2026-09-30'
      AND o.apply_date < DATE '2026-10-30'
    GROUP BY o.user_id
)
SELECT m.scheme, COALESCE(m.group_id,0) AS group_id,
       COUNT(*) AS selected_users,
       COUNT(DISTINCT m.user_id) AS distinct_users,
       MIN(m.score_rank) AS min_rank, MAX(m.score_rank) AS max_rank,

"""Verify every published batch, window, group and cumulative result."""
from pathlib import Path
from decimal import Decimal
import json
x=json.loads((Path(__file__).resolve().parent/'02_对比统计数据.json').read_text())
checked=0
for b in x['batches']:
    groups=b['groups'];days=list(b['windows'])
    assert [g['users'] for g in groups]==b['group_sizes']
    assert sum(b['group_sizes'])==b['coupon_selected_users']==b['score_selected_users']
    assert b['source_users']-b['invalid_coupon_users']==b['coupon_selected_users']
    for i,g in enumerate(groups):
        assert g['score_rank_min']==sum(b['group_sizes'][:i])+1
        assert g['score_rank_max']==sum(b['group_sizes'][:i+1])
        for f in ['coupon','score']:
            counts=[g['windows'][d][f+'_applicants'] for d in days]
            assert all(0<=a<=g['users'] for a in counts)
            assert counts==sorted(counts)
            for d,a in zip(days,counts):
                assert Decimal(g['windows'][d][f+'_rate'])==Decimal(a)/g['users']
                checked+=1
    for r in b['cumulative']:
        part=groups[:r['top_groups']]
        assert r['users']==sum(g['users'] for g in part)
        for d,w in r['windows'].items():
            for f in ['coupon','score']:
                a=sum(g['windows'][d][f+'_applicants'] for g in part)
                assert w[f+'_applicants']==a
                assert Decimal(w[f+'_rate'])==Decimal(a)/r['users']
                checked+=1
            assert w['additional_observed_applicants']==w['score_applicants']-w['coupon_applicants']
            assert Decimal(w['difference_pp'])==Decimal(w['score_applicants']-w['coupon_applicants'])*100/r['users']
    print(b['batch'],'groups=',len(groups),'windows=',','.join(days),'verified')
assert [len(b['groups']) for b in x['batches']]==[5,5,5,5,3]
assert [list(b['windows']) for b in x['batches']]==[['3','7','15','30'],['3','7','15'],['3','7','15'],['3','7','15'],['3','7']]
print(json.dumps({'all_passed':True,'batches':len(x['batches']),'rates_verified':checked,'database_required':False},ensure_ascii=False))
