口径

T0 origin：2026-01-01～09-20；loan_type_code=2、is_pass1=1、recycle_type=3；20 秒批次留 serial_id 最大；干净结清；vir_time≥repaid_time；day_rn=1。人日不去重。
产品个数：order_vir_f_copy.risk_product 去空格后按逗号拆开，非空 product_no 个数；最多看到第 10 槽。全期只有 1 或 2，min=1、max=2。
当天提单（验证，不圈人）：获额后首次 apply，apply_time≥vir_time 且 apply_date=vir_date。分母是该类 T0 人日，不是提单订单数。
点确认：获额后 24 小时，Loan_newconfirm / Loan_newconfirm_select。1–8 月 95,023 / 177,133 = 53.65%。
1–8 月提单质量：获额后首次 apply，apply_date 在 1–8 月。加权息费 = 已放款 Σ(pre_amt+post_amt)/Σ remit_amt，不卡到期。到期 = is_remit=1 且 remit_amt>0 且 due_date < CURRENT_DATE。盈利率 = Σ(repayment_record_copy.amount，payin_date≤due_date) / Σ remit_amt − 1。整笔逾期 = repaid_date > due_date 或空/≤2000-01-01。只做验证，不圈人。

完整 PGSQL（点击打开） 
-- =============================================================================
-- T0 获额通过：提供的产品数量 × 当天提单率
-- 窗口：2026-01-01～09-20
-- 产品个数 = risk_product 去空格、按逗号拆开后非空 product_no 个数（最多看到第 10 槽）
-- 当天提单 = 获额后首次 apply 且 apply_date = vir_date 且 apply_time >= vir_time
-- 首次提单质量：申请日在 1–8 月
--   加权息费：已放款（不卡到期）Σ(pre_amt+post_amt)/Σ remit_amt
--   盈利率：已放款且 due_date < CURRENT_DATE，Σ(payin_date≤due_date 的还款) / Σ remit_amt − 1
--   整笔逾期：repaid_date > due_date，或 repaid_date 空/≤2000-01-01
-- =============================================================================

SET search_path TO wangchuanliang, public;
SET statement_timeout = 0;

DROP TABLE IF EXISTS tmp_m_origin;
CREATE TEMP TABLE tmp_m_origin AS
WITH params AS (
    SELECT DATE '2026-01-01' AS start_date, DATE '2026-09-21' AS end_date
), raw_offer AS (
    SELECT DISTINCT ON (v.user_id, v.serial_id)
        v.user_id, v.serial_id, v.vir_date::date AS vir_date,
        TO_TIMESTAMP(v.vir_unix) AS vir_time, v.risk_product
    FROM wangchuanliang.order_vir_f_copy v
    INNER JOIN wangchuanliang.side_recycle_type_copy r ON r.serial_id = v.serial_id
    CROSS JOIN params p
    WHERE v.is_pass1 = 1 AND v.loan_type_code = 2 AND r.recycle_type = 3
      AND v.vir_date >= p.start_date - 1 AND v.vir_date < p.end_date
    ORDER BY v.user_id, v.serial_id, v.vir_unix DESC
), lagged AS (
    SELECT r.*, LAG(r.vir_time) OVER (PARTITION BY r.user_id ORDER BY r.vir_time, r.serial_id) AS prev_vir_time
    FROM raw_offer r
), batched AS (
    SELECT l.*, SUM(CASE WHEN l.prev_vir_time IS NULL OR l.vir_time - l.prev_vir_time > INTERVAL '20 seconds' THEN 1 ELSE 0 END)
      OVER (PARTITION BY l.user_id ORDER BY l.vir_time, l.serial_id ROWS UNBOUNDED PRECEDING) AS batch_id
    FROM lagged l
), dedup_offer AS (
    SELECT user_id, serial_id, vir_date, vir_time, risk_product
    FROM (
        SELECT b.*, ROW_NUMBER() OVER (PARTITION BY b.user_id, b.batch_id ORDER BY b.serial_id DESC) AS batch_rn
        FROM batched b
    ) x
    CROSS JOIN params p
    WHERE batch_rn = 1 AND vir_date >= p.start_date AND vir_date < p.end_date
), candidate_users AS (SELECT DISTINCT user_id FROM dedup_offer),
all_orders AS (
    SELECT o.user_id, o.serial_id, o.apply_time, o.repaid_time, o.repaid_date::date AS repaid_date,
           o.loan_status_code, o.is_remit, o.is_repaid
    FROM wangchuanliang.order_loan_f_v2_copy o
    INNER JOIN candidate_users u ON u.user_id = o.user_id
), clean_settlement AS (
    SELECT o.user_id, o.serial_id, o.repaid_time, o.repaid_date
    FROM all_orders o
    WHERE o.is_remit = 1 AND (o.is_repaid = 1 OR o.loan_status_code = 7)
      AND o.repaid_time > TIMESTAMP '2000-01-01 00:00:00'
      AND NOT EXISTS (
          SELECT 1 FROM all_orders x
          WHERE x.user_id = o.user_id AND x.serial_id <> o.serial_id AND x.apply_time < o.repaid_time
            AND (x.loan_status_code IN (5, 8) OR (x.loan_status_code = 7 AND x.repaid_time > o.repaid_time))
      )
), paired AS (
    SELECT e.user_id, e.serial_id AS credit_serial_id, e.vir_date, e.vir_time, e.risk_product, s.repaid_time,
           ROW_NUMBER() OVER (
               PARTITION BY e.user_id, e.vir_date, e.serial_id
               ORDER BY s.repaid_time DESC, s.serial_id DESC
           ) AS s_rn
    FROM dedup_offer e
    INNER JOIN clean_settlement s
        ON s.user_id = e.user_id AND s.repaid_date = e.vir_date AND e.vir_time >= s.repaid_time
), day_pick AS (
    SELECT p.*, ROW_NUMBER() OVER (PARTITION BY p.user_id, p.vir_date ORDER BY p.vir_time ASC, p.credit_serial_id DESC) AS day_rn
    FROM paired p
    WHERE s_rn = 1
)
SELECT user_id, credit_serial_id, vir_date, vir_time, repaid_time, risk_product
FROM day_pick WHERE day_rn = 1;

-- 最多拆 10 槽，避免上次只拆 2 槽漏掉 3 个及以上
DROP TABLE IF EXISTS tmp_slots;
CREATE TEMP TABLE tmp_slots AS
SELECT
    t.user_id,
    t.vir_date,
    TRIM(split_part(regexp_replace(COALESCE(t.risk_product::text, ''), '\s', '', 'g'), ',', gs.n)) AS product_no
FROM tmp_m_origin t
CROSS JOIN generate_series(1, 10) AS gs(n)
WHERE NULLIF(TRIM(split_part(regexp_replace(COALESCE(t.risk_product::text, ''), '\s', '', 'g'), ',', gs.n)), '') IS NOT NULL;

DROP TABLE IF EXISTS tmp_nprod;
CREATE TEMP TABLE tmp_nprod AS
SELECT
    t.user_id,
    t.vir_date,
    DATE_TRUNC('month', t.vir_date)::date AS ym,
    regexp_replace(COALESCE(t.risk_product::text, ''), '\s', '', 'g') AS combo,
    COALESCE(s.n_prod, 0) AS n_prod
FROM tmp_m_origin t
LEFT JOIN (
    SELECT user_id, vir_date, COUNT(*) AS n_prod
    FROM tmp_slots
    GROUP BY user_id, vir_date
) s ON s.user_id = t.user_id AND s.vir_date = t.vir_date;

DROP TABLE IF EXISTS tmp_confirm;
CREATE TEMP TABLE tmp_confirm AS
SELECT t.user_id, t.vir_date, 1 AS f_confirm
FROM tmp_m_origin t
INNER JOIN wangchuanliang.event_track_record e
  ON e.user_id = t.user_id
 AND e.create_date >= DATE '2026-01-01'
 AND e.create_date <  DATE '2026-09-22'
 AND e.create_time >= t.vir_time
 AND e.create_time < t.vir_time + INTERVAL '24 hours'
 AND e.props_id IN ('Loan_newconfirm','Loan_newconfirm_select')
GROUP BY t.user_id, t.vir_date;

DROP TABLE IF EXISTS tmp_apply;
CREATE TEMP TABLE tmp_apply AS
SELECT z.user_id, z.vir_date, 1 AS f_apply
FROM (
    SELECT t.user_id, t.vir_date,
           ROW_NUMBER() OVER (PARTITION BY t.user_id, t.vir_date ORDER BY a.apply_time, a.serial_id) AS rn
    FROM tmp_m_origin t
    INNER JOIN wangchuanliang.order_loan_f_v2_copy a
      ON a.user_id = t.user_id
     AND a.apply_time >= t.vir_time
     AND a.apply_date = t.vir_date
) z WHERE rn = 1;

DROP TABLE IF EXISTS tmp_univ;
CREATE TEMP TABLE tmp_univ AS
SELECT
    n.user_id,
    n.vir_date,
    n.ym,
    n.n_prod,
    COALESCE(c.f_confirm, 0) AS f_confirm,
    COALESCE(a.f_apply, 0) AS f_apply
FROM tmp_nprod n
LEFT JOIN tmp_confirm c ON c.user_id = n.user_id AND c.vir_date = n.vir_date
LEFT JOIN tmp_apply a ON a.user_id = n.user_id AND a.vir_date = n.vir_date;

-- R0 点确认 + 全期产品个数 + 当天提单
SELECT
    COUNT(*) AS n_t0,
    SUM(f_confirm) AS n_confirm,
    ROUND(100.0 * AVG(f_confirm), 2) AS pct_confirm,
    SUM(CASE WHEN vir_date < DATE '2026-09-01' THEN 1 ELSE 0 END) AS n_t0_jan_aug,
    SUM(CASE WHEN vir_date < DATE '2026-09-01' THEN f_confirm ELSE 0 END) AS n_confirm_jan_aug,
    ROUND(100.0 * SUM(CASE WHEN vir_date < DATE '2026-09-01' THEN f_confirm ELSE 0 END)
        / NULLIF(SUM(CASE WHEN vir_date < DATE '2026-09-01' THEN 1 ELSE 0 END), 0), 2) AS pct_confirm_jan_aug,
    MAX(n_prod) AS max_n_prod,
    MIN(n_prod) AS min_n_prod,
    ROUND(AVG(n_prod)::numeric, 4) AS avg_n_prod,
    SUM(CASE WHEN n_prod = 0 THEN 1 ELSE 0 END) AS n_0,
    SUM(CASE WHEN n_prod = 1 THEN 1 ELSE 0 END) AS n_1,
    SUM(CASE WHEN n_prod = 2 THEN 1 ELSE 0 END) AS n_2,
    SUM(CASE WHEN n_prod >= 3 THEN 1 ELSE 0 END) AS n_3p,
    SUM(f_apply) AS n_apply,
    ROUND(100.0 * AVG(f_apply), 2) AS pct_apply,
    SUM(CASE WHEN n_prod = 1 THEN f_apply ELSE 0 END) AS n_apply_1,
    SUM(CASE WHEN n_prod = 2 THEN f_apply ELSE 0 END) AS n_apply_2,
    ROUND(100.0 * SUM(CASE WHEN n_prod = 1 THEN f_apply ELSE 0 END)
        / NULLIF(SUM(CASE WHEN n_prod = 1 THEN 1 ELSE 0 END), 0), 2) AS pct_apply_1,
    ROUND(100.0 * SUM(CASE WHEN n_prod = 2 THEN f_apply ELSE 0 END)
        / NULLIF(SUM(CASE WHEN n_prod = 2 THEN 1 ELSE 0 END), 0), 2) AS pct_apply_2,
    CURRENT_DATE::text AS as_of
FROM tmp_univ;

-- R1 月度：产品个数占比 + 当天提单
SELECT * FROM (
SELECT
    TO_CHAR(ym, 'YYYY-MM') AS ym,
    COUNT(*) AS n_t0,
    ROUND(AVG(n_prod)::numeric, 4) AS avg_n_prod,

    ROUND(100.0 * AVG(f_apply), 2) AS pct_apply,
    SUM(CASE WHEN n_prod = 1 THEN f_apply ELSE 0 END) AS n_apply_1,
    SUM(CASE WHEN n_prod = 2 THEN f_apply ELSE 0 END) AS n_apply_2,
    ROUND(100.0 * SUM(CASE WHEN n_prod = 1 THEN f_apply ELSE 0 END)
        / NULLIF(SUM(CASE WHEN n_prod = 1 THEN 1 ELSE 0 END), 0), 2) AS pct_apply_1,
    ROUND(100.0 * SUM(CASE WHEN n_prod = 2 THEN f_apply ELSE 0 END)
        / NULLIF(SUM(CASE WHEN n_prod = 2 THEN 1 ELSE 0 END), 0), 2) AS pct_apply_2,
    ROUND(100.0 * AVG(f_confirm), 2) AS pct_confirm

GROUP BY ym
UNION ALL
SELECT
    '合计',
    COUNT(*),
    ROUND(AVG(n_prod)::numeric, 4),

    SUM(CASE WHEN n_prod = 2 THEN 1 ELSE 0 END),
    ROUND(100.0 * AVG(CASE WHEN n_prod = 1 THEN 1 ELSE 0 END), 2),
    ROUND(100.0 * AVG(CASE WHEN n_prod = 2 THEN 1 ELSE 0 END), 2),
    SUM(f_apply),
    ROUND(100.0 * AVG(f_apply), 2),
    SUM(CASE WHEN n_prod = 1 THEN f_apply ELSE 0 END),
    SUM(CASE WHEN n_prod = 2 THEN f_apply ELSE 0 END),
    ROUND(100.0 * SUM(CASE WHEN n_prod = 1 THEN f_apply ELSE 0 END)
        / NULLIF(SUM(CASE WHEN n_prod = 1 THEN 1 ELSE 0 END), 0), 2),

        / NULLIF(SUM(CASE WHEN n_prod = 2 THEN 1 ELSE 0 END), 0), 2),
    ROUND(100.0 * AVG(f_confirm), 2)
FROM tmp_univ

CREATE TEMP TABLE tmp_fa AS

  AND apply_date >= DATE '2026-01-01'

DROP TABLE IF EXISTS tmp_repay;
CREATE TEMP TABLE tmp_repay AS

FROM tmp_fa f
INNER JOIN wangchuanliang.repayment_record_copy r ON r.serial_id = f.serial_id
WHERE f.is_remit = 1
  AND COALESCE(f.remit_amt, 0) > 0
  AND f.due_date < CURRENT_DATE
  AND r.payin_date <= f.due_date
GROUP BY f.serial_id;

DROP TABLE IF EXISTS tmp_q;

    f.user_id,

    CASE WHEN f.n_prod = 1 THEN '1个产品' WHEN f.n_prod = 2 THEN '2个产品' ELSE '其他' END AS prod_grp,
    1 AS n_apply,

    CASE WHEN f.is_remit = 1 AND COALESCE(f.remit_amt, 0) > 0 AND f.due_date < CURRENT_DATE

         THEN 1 ELSE 0 END AS is_od,
    CASE WHEN f.is_remit = 1 AND COALESCE(f.remit_amt, 0) > 0
         THEN COALESCE(f.pre_amt, 0) + COALESCE(f.post_amt, 0) ELSE 0 END AS fee_amt,
    CASE WHEN f.is_remit = 1 AND COALESCE(f.remit_amt, 0) > 0
         THEN f.remit_amt ELSE 0 END AS remit_loan,
    CASE WHEN f.is_remit = 1 AND COALESCE(f.remit_amt, 0) > 0 AND f.due_date < CURRENT_DATE
         THEN COALESCE(p.repay_amt, 0) ELSE 0 END AS repay_due,
    CASE WHEN f.is_remit = 1 AND COALESCE(f.remit_amt, 0) > 0 AND f.due_date < CURRENT_DATE
         THEN f.remit_amt ELSE 0 END AS remit_due
FROM tmp_fa f
LEFT JOIN tmp_repay p ON p.serial_id = f.serial_id;

    prod_grp,

    SUM(is_due) AS n_due,

    ROUND(100.0 * SUM(fee_amt) / NULLIF(SUM(remit_loan), 0), 2) AS fee_rate,
    ROUND(100.0 * SUM(is_od) / NULLIF(SUM(is_due), 0), 2) AS overdue_pct,

UNION ALL

    '合计',

    SUM(is_loan),
    SUM(is_due),
