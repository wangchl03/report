结论

1. 排序更准：高分用户贡献了更多提单
同样在流失用户里取 score 最高三分之一，这些人占全部提单的比例，迭代前 65.43%（564/862），迭代后 73.59%（209/284），提高 8.16pp。高分档大约只占名单 33%，迭代后贡献了约四分之三的提单。
2. 对齐触达后同样 5 天，提升还在
迭代前观察到 13 天、迭代后只有 5 天。后面几天会进来更多中低分提单，把前 1/3 占比拉低。两批都只看到 D0～D4：迭代前 71.01%，迭代后 73.59%，仍高 2.58pp。按触达后天数看累计占比，迭代后曲线更高、更稳。
3. 证明的是排序质量，不是名单整体提单率变高
对齐 5 日，名单整体提单率迭代前 5.22%、迭代后 3.28%，两批名单大小和运营条件不同，不能用来证伪或证实“提单变多”。前 1/3 档内提单率相对整批的倍数：迭代前 2.13 倍，迭代后 2.20 倍。两版 score 量纲不同（0910 约 5～115，0918 约 20～100），只比三分位排序，不比绝对分。
4. 经营含义
升级后的分更适合做触达排序和额度/券倾斜：同样覆盖前三分之一流失用户，能打到更大比例的实际提单人。今天（2026-09-22）库内提单尚未跑完，占比与截止昨天几乎相同。

口径

流失用户：名单 churn_user_id 去重。core 分 = score，越高越好；NTILE(3) ORDER BY score DESC, user_id，tertile=1 为前 1/3。
提单：order_loan_f_v2_copy.apply_date 在触达日～2026-09-22（含）。每人取首次提单日。
主指标：前 1/3 里提单人数 / 该批全部提单人数。
对齐 5 日：首次提单日落在触达日 + 0～4 天。0910 与 0918 的 score 量纲不同，不做绝对分对比。

R1/R4 总览（含 D0～D4）；R2/R5 三分位人数；R3/R6 按首次提单日。占比用对应列相除。统计日随 CURRENT_DATE 滚动。

-- =============================================================================
-- T7 召回提单模型：迭代前 t7recall_active0910 vs 迭代后 t7recalllist_active0918
-- 库：kaby_dw · schema wangchuanliang
-- 主指标：core 前 1/3 流失用户中的提单人数 / 该批全部提单人数
-- 提单窗口：各批触达日（9/10、9/18）起至 CURRENT_DATE（含）
-- 对齐窗口：触达日起连续 5 个日历日（D0～D4，0910 为 9/10～9/14，0918 为 9/18～9/22）
-- =============================================================================

-- ---------- 迭代前：t7recall_active0910，触达 2026-09-10 ----------
DROP TABLE IF EXISTS tmp_u_v1;
CREATE TEMP TABLE tmp_u_v1 AS
SELECT DISTINCT ON (churn_user_id)
    churn_user_id::bigint AS user_id,
    score::numeric AS score
FROM wangchuanliang.t7recall_active0910
WHERE churn_user_id IS NOT NULL
ORDER BY churn_user_id, score DESC NULLS LAST;

DROP TABLE IF EXISTS tmp_r_v1;
CREATE TEMP TABLE tmp_r_v1 AS
SELECT u.*, NTILE(3) OVER (ORDER BY score DESC NULLS LAST, user_id) AS tertile
FROM tmp_u_v1 u;

DROP TABLE IF EXISTS tmp_a_v1;
CREATE TEMP TABLE tmp_a_v1 AS
SELECT o.user_id, MIN(o.apply_date) AS first_apply
FROM tmp_u_v1 u
INNER JOIN wangchuanliang.order_loan_f_v2_copy o ON o.user_id = u.user_id
WHERE o.apply_date >= DATE '2026-09-10' AND o.apply_date <= CURRENT_DATE
GROUP BY o.user_id;

-- R1 迭代前总览（观察到今天 + D0～D4）
SELECT
    'v1_0910' AS batch,
    (SELECT COUNT(*) FROM tmp_r_v1) AS n_churn,
    (SELECT COUNT(*) FROM tmp_r_v1 WHERE tertile = 1) AS n_top,
    (SELECT MIN(score) FROM tmp_r_v1 WHERE tertile = 1) AS top_score_min,
    (SELECT MAX(score) FROM tmp_r_v1 WHERE tertile = 1) AS top_score_max,
    (SELECT COUNT(*) FROM tmp_a_v1) AS n_apply,
    (SELECT COUNT(*) FROM tmp_a_v1 a INNER JOIN tmp_r_v1 r ON r.user_id = a.user_id AND r.tertile = 1) AS n_apply_top,
    (SELECT COUNT(*) FROM tmp_a_v1 WHERE first_apply <= DATE '2026-09-10' + 4) AS n_apply_d4,
    (SELECT COUNT(*) FROM tmp_a_v1 a INNER JOIN tmp_r_v1 r ON r.user_id = a.user_id AND r.tertile = 1
      WHERE a.first_apply <= DATE '2026-09-10' + 4) AS n_apply_top_d4;

-- R2 迭代前三分位（观察到今天）
SELECT r.tertile,
       COUNT(*) AS n_churn,
       COUNT(a.user_id) AS n_apply
FROM tmp_r_v1 r
LEFT JOIN tmp_a_v1 a ON a.user_id = r.user_id
GROUP BY r.tertile
ORDER BY r.tertile;

-- R3 迭代前按首次提单日
SELECT a.first_apply::text AS d,
       COUNT(*) AS n_apply,
       SUM(CASE WHEN r.tertile = 1 THEN 1 ELSE 0 END) AS n_apply_top
FROM tmp_a_v1 a
INNER JOIN tmp_r_v1 r ON r.user_id = a.user_id
GROUP BY 1
ORDER BY 1;

-- ---------- 迭代后：t7recalllist_active0918，触达 2026-09-18 ----------

CREATE TEMP TABLE tmp_u_v2 AS

DROP TABLE IF EXISTS tmp_r_v2;

DROP TABLE IF EXISTS tmp_a_v2;
CREATE TEMP TABLE tmp_a_v2 AS
SELECT o.user_id, MIN(o.apply_date) AS first_apply
FROM tmp_u_v2 u
INNER JOIN wangchuanliang.order_loan_f_v2_copy o ON o.user_id = u.user_id

-- R4 迭代后总览
SELECT
    'v2_0918' AS batch,
    (SELECT COUNT(*) FROM tmp_r_v2) AS n_churn,
    (SELECT COUNT(*) FROM tmp_r_v2 WHERE tertile = 1) AS n_top,
    (SELECT MIN(score) FROM tmp_r_v2 WHERE tertile = 1) AS top_score_min,
    (SELECT MAX(score) FROM tmp_r_v2 WHERE tertile = 1) AS top_score_max,
    (SELECT COUNT(*) FROM tmp_a_v2) AS n_apply,

    (SELECT COUNT(*) FROM tmp_a_v2 WHERE first_apply <= DATE '2026-09-18' + 4) AS n_apply_d4,
    (SELECT COUNT(*) FROM tmp_a_v2 a INNER JOIN tmp_r_v2 r ON r.user_id = a.user_id AND r.tertile = 1
      WHERE a.first_apply <= DATE '2026-09-18' + 4) AS n_apply_top_d4;

-- R5 迭代后三分位（观察到今天）
SELECT r.tertile,
       COUNT(*) AS n_churn,
       COUNT(a.user_id) AS n_apply
FROM tmp_r_v2 r
