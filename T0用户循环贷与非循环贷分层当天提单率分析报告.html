附录：完整 PGSQL

以下代码覆盖本报告的第一、二、三层样本、总体结果与月度结果。当前版本对应 2026-01-01 至 2026-08-15 的 T0 用户。

模块一：完整取数与三层样本构建
/* ================================================================
   T0 循环贷与非循环贷分层当天提单率
   第一层：全量 T0，循环贷/非循环贷用户拆分
   第二层：循环贷用户中，多在途结清后转 T0
   第三层：第二层中，最终结清订单本身也是循环贷
   ================================================================ */

WITH params AS (
    SELECT
        DATE '2026-01-01' AS start_date,
        DATE '2026-08-15' AS end_date,
        TIMESTAMP '2026-08-16 00:00:00' AS data_cutoff_time
),

/* 1. 统计期间已结清订单；同日多笔结清时，后续仅保留最后结清的一笔。 */
settled_orders AS (
    SELECT
        o.user_id,
        o.serial_id,
        o.apply_time,
        o.repaid_time,
        o.repaid_date::date AS repaid_date
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    CROSS JOIN params AS p
    WHERE o.is_remit = 1
      AND o.loan_status_code = 7
      AND o.repaid_date >= p.start_date
      AND o.repaid_date <= p.end_date
),

candidate_users AS (
    SELECT DISTINCT
        s.user_id
    FROM settled_orders AS s
),

/* 2. 候选用户全量订单：用于结清时点是否存在其他在途订单的判断。 */
all_orders AS (
    SELECT
        o.user_id,
        o.serial_id,
        o.apply_time,
        o.repaid_time,
        o.repaid_date,
        o.loan_status_code,
        o.is_remit
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    INNER JOIN candidate_users AS u
        ON u.user_id = o.user_id
    CROSS JOIN params AS p
    WHERE o.apply_time < p.data_cutoff_time
),

last_settle_of_day AS (
    SELECT DISTINCT ON (
        s.user_id,
        s.repaid_date
    )
        s.user_id,
        s.serial_id,
        s.apply_time,
        s.repaid_time,
        s.repaid_date
    FROM settled_orders AS s
    ORDER BY
        s.user_id,
        s.repaid_date,
        s.repaid_time DESC
),

/* 3. 干净结清：在目标订单结清时点，没有另一笔已申请且仍在途的订单。
      状态 5、8 视作在途；状态 7 但结清时间更晚，同样在该时点尚未结清。 */
clean_last_settles AS (
    SELECT
        s.user_id,
        s.serial_id,
        s.apply_time,
        s.repaid_time,
        s.repaid_date
    FROM last_settle_of_day AS s
    WHERE NOT EXISTS (
        SELECT
            1
        FROM all_orders AS o2
        WHERE o2.user_id = s.user_id
          AND o2.serial_id <> s.serial_id
          AND o2.apply_time < s.repaid_time
          AND (
                o2.loan_status_code = 5
             OR o2.loan_status_code = 8
             OR (
                    o2.loan_status_code = 7
                AND o2.repaid_time > s.repaid_time
                )
          )
    )
),

/* 4. 同日获额通过，且获额时间晚于目标结清时间。 */
credit_pass AS (
    SELECT
        v.user_id,
        v.vir_date::date AS vir_date,
        TO_TIMESTAMP(v.vir_unix) AS vir_time
    FROM wangchuanliang.order_vir_f_copy AS v
    CROSS JOIN params AS p
    WHERE v.is_pass1 = 1
      AND v.vir_date >= p.start_date
      AND v.vir_date <= p.end_date
),

t0_samples AS (
    SELECT DISTINCT
        c.user_id,
        c.serial_id AS final_serial_id,
        c.apply_time AS final_apply_time,
        c.repaid_time AS final_repaid_time,
        c.repaid_date
    FROM clean_last_settles AS c
    INNER JOIN credit_pass AS cp
        ON cp.user_id = c.user_id
       AND cp.vir_date = c.repaid_date
       AND cp.vir_time > c.repaid_time
),

/* 5. 生命周期内任一已放款订单 recycle_type=2，即循环贷用户。 */
cycle_users AS (
    SELECT DISTINCT
        o.user_id
    FROM all_orders AS o
    INNER JOIN wangchuanliang.side_recycle_type_copy AS r
        ON r.serial_id = o.serial_id
       AND r.recycle_type = 2
    WHERE o.is_remit = 1
),

/* 6. 第一层：标记循环贷属性及结清当日是否出现新申请。 */
layer1_t0 AS (
    SELECT
        t.user_id,
        t.final_serial_id,
        t.final_apply_time,
        t.final_repaid_time,
        t.repaid_date,
        CASE
            WHEN cu.user_id IS NOT NULL THEN '循环贷用户'
            ELSE '非循环贷用户'
        END AS user_type,
        CASE
            WHEN EXISTS (
                SELECT
                    1
                FROM all_orders AS n
                WHERE n.user_id = t.user_id
                  AND n.apply_time > t.final_repaid_time
                  AND n.apply_time::date = t.repaid_date
            ) THEN 1
            ELSE 0
        END AS is_t0_apply
    FROM t0_samples AS t
    LEFT JOIN cycle_users AS cu
        ON cu.user_id = t.user_id
),

/* 7. 第二层：最终订单 B 在此前订单 A 结清时已在途；
      A 取 B 之前最近的一笔并行结清订单；A 至 B 结清之间无新提单。 */
layer2_path AS (
    SELECT DISTINCT
        t.user_id,
        t.final_serial_id,
        t.repaid_date,
        t.is_t0_apply
    FROM layer1_t0 AS t
    INNER JOIN all_orders AS a
        ON a.user_id = t.user_id
       AND a.serial_id <> t.final_serial_id
       AND a.is_remit = 1
       AND a.loan_status_code = 7
       AND a.repaid_time < t.final_repaid_time
       AND t.final_apply_time < a.repaid_time
       AND a.repaid_time = (
            SELECT
                MAX(a2.repaid_time)
            FROM all_orders AS a2
            WHERE a2.user_id = t.user_id
              AND a2.serial_id <> t.final_serial_id
              AND a2.is_remit = 1
              AND a2.loan_status_code = 7
              AND a2.repaid_time < t.final_repaid_time
              AND t.final_apply_time < a2.repaid_time
       )
    WHERE t.user_type = '循环贷用户'
      AND NOT EXISTS (
            SELECT
                1
            FROM all_orders AS n
            WHERE n.user_id = t.user_id
              AND n.apply_time > a.repaid_time
              AND n.apply_time < t.final_repaid_time
      )
),

/* 8. 第三层：第二层基础上，最终结清订单 B 自身 recycle_type=2。 */
layer3_strict_path AS (
    SELECT
        p.user_id,
        p.final_serial_id,
        p.repaid_date,
        p.is_t0_apply
    FROM layer2_path AS p
    INNER JOIN wangchuanliang.side_recycle_type_copy AS r
        ON r.serial_id = p.final_serial_id
       AND r.recycle_type = 2
),

/* 9. 输出第一层月度结果。 */
layer1_monthly AS (
    SELECT
        DATE_TRUNC('month', t.repaid_date)::date AS month,
        t.user_type AS statistic_object,
        COUNT(*) AS sample_cnt,
        SUM(t.is_t0_apply) AS apply_cnt,
        ROUND(
            SUM(t.is_t0_apply)::numeric
            / NULLIF(COUNT(*), 0) * 100,
            2
        ) AS t0_apply_rate_pct
    FROM layer1_t0 AS t
    GROUP BY
        DATE_TRUNC('month', t.repaid_date)::date,
        t.user_type
),

/* 10. 输出第二、三层月度结果。 */
path_monthly AS (
    SELECT
        DATE_TRUNC('month', p.repaid_date)::date AS month,
        '第二层：原始路径' AS statistic_object,
        COUNT(*) AS sample_cnt,
        SUM(p.is_t0_apply) AS apply_cnt,
        ROUND(
            SUM(p.is_t0_apply)::numeric
            / NULLIF(COUNT(*), 0) * 100,
            2
        ) AS t0_apply_rate_pct
    FROM layer2_path AS p
    GROUP BY
        DATE_TRUNC('month', p.repaid_date)::date

    UNION ALL

    SELECT
        DATE_TRUNC('month', p.repaid_date)::date AS month,
        '第三层：严格路径' AS statistic_object,
        COUNT(*) AS sample_cnt,
        SUM(p.is_t0_apply) AS apply_cnt,
        ROUND(
            SUM(p.is_t0_apply)::numeric
            / NULLIF(COUNT(*), 0) * 100,
            2
        ) AS t0_apply_rate_pct
    FROM layer3_strict_path AS p
    GROUP BY
        DATE_TRUNC('month', p.repaid_date)::date
)

SELECT
    '第一层' AS layer,
    m.month AS 结清月份,
    m.statistic_object AS 统计对象,
    m.sample_cnt AS 样本数,
    ROUND(
        m.sample_cnt::numeric
        / SUM(m.sample_cnt) OVER (
            PARTITION BY m.month
        ) * 100,
        2
    ) AS 当月样本占比_pct,
    m.apply_cnt AS 当天提单数,
    m.t0_apply_rate_pct AS 当天提单率_pct
FROM layer1_monthly AS m

UNION ALL

SELECT
    '第二三层' AS layer,
    m.month AS 结清月份,
    m.statistic_object AS 统计对象,
    m.sample_cnt AS 样本数,
    NULL::numeric AS 当月样本占比_pct,
    m.apply_cnt AS 当天提单数,
    m.t0_apply_rate_pct AS 当天提单率_pct
FROM path_monthly AS m
ORDER BY
    layer,
    结清月份,
    统计对象;
模块二：第一层指标——循环贷与非循环贷构成、当天提单率
/* 在模块一的 layer1_t0 CTE 后执行本模块；
   或将本模块 SELECT 替换至模块一最后的输出位置。 */

SELECT
    '总体' AS 统计周期,
    t.user_type AS 用户类型,
    COUNT(*) AS T0用户日期样本数,
    ROUND(
        COUNT(*)::numeric
        / SUM(COUNT(*)) OVER () * 100,
        2
    ) AS T0用户占比_pct,
    SUM(t.is_t0_apply) AS 当天提单用户数,
    ROUND(
        SUM(t.is_t0_apply)::numeric
        / NULLIF(COUNT(*), 0) * 100,
        2
    ) AS 当天提单率_pct
FROM layer1_t0 AS t
GROUP BY
    t.user_type

UNION ALL

SELECT
    TO_CHAR(
        DATE_TRUNC('month', t.repaid_date),
        'YYYY-MM'
    ) AS 统计周期,
    t.user_type AS 用户类型,
    COUNT(*) AS T0用户日期样本数,
    ROUND(
        COUNT(*)::numeric
        / SUM(COUNT(*)) OVER (
            PARTITION BY DATE_TRUNC('month', t.repaid_date)
        ) * 100,
        2
    ) AS T0用户占比_pct,
    SUM(t.is_t0_apply) AS 当天提单用户数,
    ROUND(
        SUM(t.is_t0_apply)::numeric
        / NULLIF(COUNT(*), 0) * 100,
        2
    ) AS 当天提单率_pct
FROM layer1_t0 AS t
GROUP BY
    DATE_TRUNC('month', t.repaid_date),
    t.user_type
ORDER BY
    统计周期,
    用户类型;
模块三：第二、三层指标——多在途结清转 T0 路径
/* 在模块一已构建 layer1_t0、layer2_path、layer3_strict_path 后执行。 */

SELECT
    '第二层：原始路径' AS 路径层级,
    COUNT(*) AS 路径样本数,
    ROUND(
        COUNT(*)::numeric
        / NULLIF(
            (
                SELECT
                    COUNT(*)
                FROM layer1_t0
                WHERE user_type = '循环贷用户'
            ),
            0
        ) * 100,
        2
    ) AS 占第一层循环贷T0_pct,
    SUM(p.is_t0_apply) AS 当天提单数,
    ROUND(
        SUM(p.is_t0_apply)::numeric
        / NULLIF(COUNT(*), 0) * 100,
        2
    ) AS 当天提单率_pct
FROM layer2_path AS p

UNION ALL

SELECT
    '第三层：严格路径' AS 路径层级,
    COUNT(*) AS 路径样本数,
    ROUND(
        COUNT(*)::numeric
        / NULLIF(
            (
                SELECT
                    COUNT(*)
                FROM layer1_t0
                WHERE user_type = '循环贷用户'
            ),
            0
        ) * 100,
        2
    ) AS 占第一层循环贷T0_pct,
    SUM(p.is_t0_apply) AS 当天提单数,
    ROUND(
        SUM(p.is_t0_apply)::numeric
        / NULLIF(COUNT(*), 0) * 100,
        2
    ) AS 当天提单率_pct
FROM layer3_strict_path AS p;
模块四：第二、三层月度提单率与严格路径覆盖率
/* 月度分布：第二层与第三层分别统计；第三层覆盖率以第二层为分母。 */

WITH layer2_monthly AS (
    SELECT
        DATE_TRUNC('month', p.repaid_date)::date AS month,
        COUNT(*) AS sample_cnt,
        SUM(p.is_t0_apply) AS apply_cnt
    FROM layer2_path AS p
    GROUP BY
        DATE_TRUNC('month', p.repaid_date)::date
),
layer3_monthly AS (
    SELECT
        DATE_TRUNC('month', p.repaid_date)::date AS month,
        COUNT(*) AS sample_cnt,
        SUM(p.is_t0_apply) AS apply_cnt
    FROM layer3_strict_path AS p
    GROUP BY
        DATE_TRUNC('month', p.repaid_date)::date
)
SELECT
    l2.month AS 结清月份,
    l2.sample_cnt AS 第二层样本数,
    l2.apply_cnt AS 第二层当天提单数,
    ROUND(
        l2.apply_cnt::numeric
        / NULLIF(l2.sample_cnt, 0) * 100,
        2
    ) AS 第二层当天提单率_pct,
    l3.sample_cnt AS 第三层样本数,
    l3.apply_cnt AS 第三层当天提单数,
    ROUND(
        l3.apply_cnt::numeric
        / NULLIF(l3.sample_cnt, 0) * 100,
        2
    ) AS 第三层当天提单率_pct,
    ROUND(
        l3.sample_cnt::numeric
        / NULLIF(l2.sample_cnt, 0) * 100,
        2
    ) AS 第三层覆盖第二层_pct
FROM layer2_monthly AS l2
LEFT JOIN layer3_monthly AS l3
    ON l3.month = l2.month
ORDER BY
    结清月份;
模块五：当天提单订单的加权利率与加权额度使用率
/* ================================================================
   本模块可独立执行。
   输出：循环贷/非循环贷 T0 用户的月度及整体可用分子、分母。
   利率 = SUM(pre_amt + post_amt) / SUM(remit_amt)
   额度使用率 = SUM(apply_amt) / SUM(credit_amt)
   ================================================================ */

WITH params AS (
    SELECT
        DATE '2026-01-01' AS start_date,
        DATE '2026-08-15' AS end_date,
        TIMESTAMP '2026-08-16 00:00:00' AS data_cutoff_time
),

settled_orders AS (
    SELECT
        o.user_id,
        o.serial_id,
        o.repaid_time,
        o.repaid_date::date AS repaid_date
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    CROSS JOIN params AS p
    WHERE o.is_remit = 1
      AND o.loan_status_code = 7
      AND o.repaid_date >= p.start_date
      AND o.repaid_date <= p.end_date
),

candidate_users AS (
    SELECT DISTINCT
        s.user_id
    FROM settled_orders AS s
),

all_orders AS (
    SELECT
        o.user_id,
        o.serial_id,
        o.apply_time,
        o.repaid_time,
        o.loan_status_code,
        o.is_remit,
        o.pre_amt,
        o.post_amt,
        o.remit_amt,
        o.apply_amt
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    INNER JOIN candidate_users AS u
        ON u.user_id = o.user_id
    CROSS JOIN params AS p
    WHERE o.apply_time < p.data_cutoff_time
),

last_settle_of_day AS (
    SELECT DISTINCT ON (
        s.user_id,
        s.repaid_date
    )
        s.user_id,
        s.serial_id,
        s.repaid_time,
        s.repaid_date
    FROM settled_orders AS s
    ORDER BY
        s.user_id,
        s.repaid_date,
        s.repaid_time DESC
),

clean_last_settles AS (
    SELECT
        s.user_id,
        s.serial_id,
        s.repaid_time,
        s.repaid_date
    FROM last_settle_of_day AS s
    WHERE NOT EXISTS (
        SELECT
            1
        FROM all_orders AS o2
        WHERE o2.user_id = s.user_id
          AND o2.serial_id <> s.serial_id
          AND o2.apply_time < s.repaid_time
          AND (
                o2.loan_status_code = 5
             OR o2.loan_status_code = 8
             OR (
                    o2.loan_status_code = 7
                AND o2.repaid_time > s.repaid_time
                )
          )
    )
),

credit_pass AS (
    SELECT
        v.user_id,
        v.vir_date::date AS vir_date,
        TO_TIMESTAMP(v.vir_unix) AS vir_time
    FROM wangchuanliang.order_vir_f_copy AS v
    CROSS JOIN params AS p
    WHERE v.is_pass1 = 1
      AND v.vir_date >= p.start_date
      AND v.vir_date <= p.end_date
),

t0_samples AS (
    SELECT DISTINCT
        c.user_id,
        c.repaid_date,
        c.repaid_time
    FROM clean_last_settles AS c
    INNER JOIN credit_pass AS cp
        ON cp.user_id = c.user_id
       AND cp.vir_date = c.repaid_date
       AND cp.vir_time > c.repaid_time
),

cycle_users AS (
    SELECT DISTINCT
        o.user_id
    FROM all_orders AS o
    INNER JOIN wangchuanliang.side_recycle_type_copy AS r
        ON r.serial_id = o.serial_id
       AND r.recycle_type = 2
    WHERE o.is_remit = 1
),

/* 同一 T0 样本有多笔当天提单时，只保留结清后的首次提单；
   若提单时间相同，则保留 serial_id 最大的一笔。 */
first_t0_apply AS (
    SELECT DISTINCT ON (
        t.user_id,
        t.repaid_date
    )
        t.user_id,
        t.repaid_date,
        a.serial_id,
        a.apply_time,
        a.is_remit,
        a.pre_amt,
        a.post_amt,
        a.remit_amt,
        a.apply_amt
    FROM t0_samples AS t
    INNER JOIN all_orders AS a
        ON a.user_id = t.user_id
       AND a.apply_time > t.repaid_time
       AND a.apply_time::date = t.repaid_date
    ORDER BY
        t.user_id,
        t.repaid_date,
        a.apply_time ASC,
        a.serial_id DESC
),

/* 每笔提单订单按 serial_id 取最新获额记录。 */
credit_by_order AS (
    SELECT DISTINCT ON (
        v.serial_id
    )
        v.serial_id,
        v.vir_date::date AS vir_date,
        v.max_credit_apply_amt,
        v.max_risk_credit_amount
    FROM wangchuanliang.order_vir_f_copy AS v
    INNER JOIN first_t0_apply AS a
        ON a.serial_id = v.serial_id
    ORDER BY
        v.serial_id,
        v.vir_unix DESC
),

metric_base AS (
    SELECT
        t.repaid_date,
        CASE
            WHEN cu.user_id IS NOT NULL THEN '循环贷用户'
            ELSE '非循环贷用户'
        END AS 用户类型,
        a.serial_id AS 提单订单号,
        a.is_remit,
        a.pre_amt,
        a.post_amt,
        a.remit_amt,
        a.apply_amt,
        CASE
            WHEN c.vir_date >= DATE '2025-04-01'
                THEN c.max_credit_apply_amt
            WHEN r.recycle_type = 2
                THEN c.max_credit_apply_amt
            ELSE c.max_risk_credit_amount
        END AS credit_amt
    FROM t0_samples AS t
    LEFT JOIN cycle_users AS cu
        ON cu.user_id = t.user_id
    LEFT JOIN first_t0_apply AS a
        ON a.user_id = t.user_id
       AND a.repaid_date = t.repaid_date
    LEFT JOIN credit_by_order AS c
        ON c.serial_id = a.serial_id
    LEFT JOIN wangchuanliang.side_recycle_type_copy AS r
        ON r.serial_id = a.serial_id
),

metric_aggregation AS (
    SELECT
        GROUPING(DATE_TRUNC('month', m.repaid_date)) AS g_month,
        DATE_TRUNC('month', m.repaid_date)::date AS 结清月份,
        m.用户类型,
        COUNT(m.提单订单号) AS 当天提单订单数,
        COUNT(CASE
            WHEN m.is_remit = 1
             AND m.remit_amt > 0
                THEN 1
        END) AS 利率有效放款订单数,
        SUM(CASE
            WHEN m.is_remit = 1
             AND m.remit_amt > 0
                THEN COALESCE(m.pre_amt, 0) + COALESCE(m.post_amt, 0)
            ELSE 0
        END) AS 利率分子_pre_amt加post_amt,
        SUM(CASE
            WHEN m.is_remit = 1
             AND m.remit_amt > 0
                THEN m.remit_amt
            ELSE 0
        END) AS 利率分母_remit_amt,
        COUNT(CASE
            WHEN m.apply_amt >= 200
             AND m.credit_amt >= 200
                THEN 1
        END) AS 额度使用率有效订单数,
        SUM(CASE
            WHEN m.apply_amt >= 200
             AND m.credit_amt >= 200
                THEN m.apply_amt
            ELSE 0
        END) AS 使用率分子_apply_amt,
        SUM(CASE
            WHEN m.apply_amt >= 200
             AND m.credit_amt >= 200
                THEN m.credit_amt
            ELSE 0
        END) AS 使用率分母_credit_amt
    FROM metric_base AS m
    GROUP BY GROUPING SETS (
        (DATE_TRUNC('month', m.repaid_date), m.用户类型),
        (m.用户类型)
    )
)

SELECT
    CASE
        WHEN a.g_month = 1 THEN '合计'
        ELSE TO_CHAR(a.结清月份, 'YYYY-MM')
    END AS 结清月份,
    a.用户类型,
    a.当天提单订单数,
    a.利率有效放款订单数,
    a.利率分子_pre_amt加post_amt,
    a.利率分母_remit_amt,
    ROUND(
        a.利率分子_pre_amt加post_amt::numeric
        / NULLIF(a.利率分母_remit_amt, 0) * 100,
        2
    ) AS 加权利率_pct,
    a.额度使用率有效订单数,
    a.使用率分子_apply_amt,
    a.使用率分母_credit_amt,
    ROUND(
        a.使用率分子_apply_amt::numeric
        / NULLIF(a.使用率分母_credit_amt, 0) * 100,
        2
    ) AS 加权额度使用率_pct
FROM metric_aggregation AS a
ORDER BY
    a.g_month,
    a.结清月份,
    CASE a.用户类型
        WHEN '循环贷用户' THEN 1
        WHEN '非循环贷用户' THEN 2
    END;
模块六：结清订单利率分组的月度当天提单率
/* ================================================================
   利率标签取 T0 当天刚结清订单，避免使用提单后的结果变量。
   结清订单利率 = (pre_amt + post_amt) / remit_amt
   分组：<30%、30%-40%、40%-50%、50%-60%、≥60%
   ================================================================ */

WITH params AS (
    SELECT
        DATE '2026-01-01' AS start_date,
        DATE '2026-08-15' AS end_date,
        TIMESTAMP '2026-08-16 00:00:00' AS data_cutoff_time
),

settled_orders AS (
    SELECT
        o.user_id,
        o.serial_id,
        o.repaid_time,
        o.repaid_date::date AS repaid_date,
        o.pre_amt,
        o.post_amt,
        o.remit_amt
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    CROSS JOIN params AS p
    WHERE o.is_remit = 1
      AND o.loan_status_code = 7
      AND o.repaid_date >= p.start_date
      AND o.repaid_date <= p.end_date
),

candidate_users AS (
    SELECT DISTINCT
        s.user_id
    FROM settled_orders AS s
),

all_orders AS (
    SELECT
        o.user_id,
        o.serial_id,
        o.apply_time,
        o.repaid_time,
        o.loan_status_code
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    INNER JOIN candidate_users AS u
        ON u.user_id = o.user_id
    CROSS JOIN params AS p
    WHERE o.apply_time < p.data_cutoff_time
),

last_settle_of_day AS (
    SELECT DISTINCT ON (
        s.user_id,
        s.repaid_date
    )
        s.user_id,
        s.serial_id,
        s.repaid_time,
        s.repaid_date,
        s.pre_amt,
        s.post_amt,
        s.remit_amt
    FROM settled_orders AS s
    ORDER BY
        s.user_id,
        s.repaid_date,
        s.repaid_time DESC
),

clean_last_settles AS (
    SELECT
        s.user_id,
        s.serial_id,
        s.repaid_time,
        s.repaid_date,
        s.pre_amt,
        s.post_amt,
        s.remit_amt
    FROM last_settle_of_day AS s
    WHERE NOT EXISTS (
        SELECT
            1
        FROM all_orders AS o2
        WHERE o2.user_id = s.user_id
          AND o2.serial_id <> s.serial_id
          AND o2.apply_time < s.repaid_time
          AND (
                o2.loan_status_code = 5
             OR o2.loan_status_code = 8
             OR (
                    o2.loan_status_code = 7
                AND o2.repaid_time > s.repaid_time
                )
          )
    )
),

credit_pass AS (
    SELECT
        v.user_id,
        v.vir_date::date AS vir_date,
        TO_TIMESTAMP(v.vir_unix) AS vir_time
    FROM wangchuanliang.order_vir_f_copy AS v
    CROSS JOIN params AS p
    WHERE v.is_pass1 = 1
      AND v.vir_date >= p.start_date
      AND v.vir_date <= p.end_date
),

t0_labeled AS (
    SELECT
        c.user_id,
        c.repaid_date,
        c.repaid_time,
        CASE
            WHEN c.remit_amt > 0
                THEN (COALESCE(c.pre_amt, 0) + COALESCE(c.post_amt, 0))
                     / c.remit_amt::numeric
        END AS settled_order_rate,
        CASE
            WHEN EXISTS (
                SELECT
                    1
                FROM all_orders AS a
                WHERE a.user_id = c.user_id
                  AND a.apply_time > c.repaid_time
                  AND a.apply_time::date = c.repaid_date
            ) THEN 1
            ELSE 0
        END AS is_t0_apply
    FROM clean_last_settles AS c
    WHERE EXISTS (
        SELECT
            1
        FROM credit_pass AS cp
        WHERE cp.user_id = c.user_id
          AND cp.vir_date = c.repaid_date
          AND cp.vir_time > c.repaid_time
    )
),

bucketed AS (
    SELECT
        t.*,
        CASE
            WHEN t.settled_order_rate < 0.30 THEN '30%以下'
            WHEN t.settled_order_rate < 0.40 THEN '30%-40%'
            WHEN t.settled_order_rate < 0.50 THEN '40%-50%'
            WHEN t.settled_order_rate < 0.60 THEN '50%-60%'
            WHEN t.settled_order_rate >= 0.60 THEN '60%以上'
        END AS 利率分组
    FROM t0_labeled AS t
)

SELECT
    TO_CHAR(
        DATE_TRUNC('month', b.repaid_date),
        'YYYY-MM'
    ) AS 结清月份,
    b.利率分组,
    COUNT(*) AS T0用户日期样本数,
    SUM(b.is_t0_apply) AS 当天提单用户数,
    ROUND(
        SUM(b.is_t0_apply)::numeric
        / NULLIF(COUNT(*), 0) * 100,
        2
    ) AS 当天提单率_pct
FROM bucketed AS b
WHERE b.利率分组 IS NOT NULL
GROUP BY
    DATE_TRUNC('month', b.repaid_date),
    b.利率分组
ORDER BY
    结清月份,
    CASE b.利率分组
        WHEN '30%以下' THEN 1
        WHEN '30%-40%' THEN 2
        WHEN '40%-50%' THEN 3
        WHEN '50%-60%' THEN 4
        WHEN '60%以上' THEN 5
    END;

        a.total_term
    FROM t0_samples AS t
    INNER JOIN all_orders AS a
        ON a.user_id = t.user_id
       AND a.apply_time > t.repaid_time
       AND a.apply_time::date = t.repaid_date
    ORDER BY
        t.user_id,
        t.repaid_date,
        a.apply_time ASC,
        a.serial_id DESC
)

SELECT
    TO_CHAR(
        DATE_TRUNC('month', f.repaid_date),
        'YYYY-MM'
    ) AS 结清月份,
    CASE
        WHEN f.total_term IS NULL THEN '期数缺失'
        ELSE f.total_term::text || '期'
    END AS 提单订单期数,
    COUNT(*) AS 当天提单订单数
FROM first_t0_apply AS f
GROUP BY
    DATE_TRUNC('month', f.repaid_date),
    提单订单期数,
    f.total_term
ORDER BY
    结清月份,
    f.total_term NULLS LAST;
口径说明：循环贷用户指统计截止日前存在任一笔已放款、且 recycle_type = 2 的订单的用户；非循环贷用户指其余 T0 用户。报告展示用户日期样本，同一用户在不同结清日可重复出现。
