按模块一至五在同一数据库会话顺序执行，可复现本报告全部样本、月度表格、图表指标与路径分层结果。

模块一：公共样本、20 秒去重、最早结清 0 在贷及路径临时表
CREATE TEMP TABLE raw_credit_pass AS
SELECT DISTINCT ON (v.user_id, v.serial_id)
       v.user_id,
       v.serial_id,
       v.vir_date::date AS vir_date,
       to_timestamp(v.vir_unix) AS vir_time
FROM wangchuanliang.order_vir_f_copy v
WHERE v.is_pass1 = 1
  AND v.vir_date >= DATE '2026-01-01'
  AND v.vir_date < DATE '2026-08-16'
ORDER BY v.user_id, v.serial_id, v.vir_unix DESC;
CREATE INDEX ON raw_credit_pass (user_id, vir_time);

CREATE TEMP TABLE credit_pass_batch AS
WITH ordered AS (
    SELECT *,
           LAG(vir_time) OVER (
               PARTITION BY user_id
               ORDER BY vir_time, serial_id
           ) AS prev_vir_time
    FROM raw_credit_pass
), batched AS (
    SELECT *,
           SUM(CASE WHEN prev_vir_time IS NULL
                         OR vir_time - prev_vir_time > INTERVAL '20 seconds'
                    THEN 1 ELSE 0 END)
           OVER (
               PARTITION BY user_id
               ORDER BY vir_time, serial_id
               ROWS UNBOUNDED PRECEDING
           ) AS batch_id
    FROM ordered
)
SELECT * FROM batched;
CREATE INDEX ON credit_pass_batch (user_id, batch_id);

CREATE TEMP TABLE dedup_credit_pass AS
SELECT user_id, serial_id, vir_date, vir_time
FROM (
    SELECT *,
           ROW_NUMBER() OVER (
               PARTITION BY user_id, batch_id
               ORDER BY serial_id DESC
           ) AS rn
    FROM credit_pass_batch
) x
WHERE rn = 1;
CREATE INDEX ON dedup_credit_pass (user_id, vir_date, vir_time);

CREATE TEMP TABLE settled_orders AS
SELECT o.user_id,
       o.serial_id,
       o.apply_time,
       o.repaid_time,
       o.repaid_date::date AS repaid_date
FROM wangchuanliang.order_loan_f_v2_copy o
WHERE o.is_remit = 1
  AND o.loan_status_code = 7
  AND o.repaid_time > TIMESTAMP '2000-01-01 00:00:00'
  AND o.repaid_date >= DATE '2026-01-01'
  AND o.repaid_date < DATE '2026-08-16';
CREATE INDEX ON settled_orders (user_id, repaid_date, repaid_time);

CREATE TEMP TABLE candidate_users AS
SELECT DISTINCT user_id FROM settled_orders;
CREATE INDEX ON candidate_users (user_id);

CREATE TEMP TABLE all_orders AS
SELECT o.user_id,
       o.serial_id,
       o.apply_time,
       o.repaid_time,
       o.repaid_date::date AS repaid_date,
       o.loan_status_code,
       o.is_remit,
       o.total_term,
       o.pre_amt,
       o.post_amt,
       o.remit_amt,
       o.apply_amt
FROM wangchuanliang.order_loan_f_v2_copy o
JOIN candidate_users u ON u.user_id = o.user_id
WHERE o.apply_time < TIMESTAMP '2026-08-16 00:00:00';
CREATE INDEX ON all_orders (user_id, apply_time);
CREATE INDEX ON all_orders (user_id, repaid_time);

CREATE TEMP TABLE clean_settles AS
SELECT s.*
FROM settled_orders s
WHERE NOT EXISTS (
    SELECT 1
    FROM all_orders o2
    WHERE o2.user_id = s.user_id
      AND o2.serial_id <> s.serial_id
      AND o2.apply_time < s.repaid_time
      AND (
          o2.loan_status_code IN (5, 8)
          OR (o2.loan_status_code = 7 AND o2.repaid_time > s.repaid_time)
      )
);
CREATE INDEX ON clean_settles (user_id, repaid_date, repaid_time);

/* 匹配去重后的同日获额；同一用户同日仅保留最早满足条件的结清。 */
CREATE TEMP TABLE t0_samples AS
SELECT user_id,
       final_serial_id,
       final_apply_time,
       final_repaid_time,
       repaid_date,
       credit_serial_id,
       credit_vir_time
FROM (
    SELECT c.user_id,
           c.serial_id AS final_serial_id,
           c.apply_time AS final_apply_time,
           c.repaid_time AS final_repaid_time,
           c.repaid_date,
           cp.serial_id AS credit_serial_id,
           cp.vir_time AS credit_vir_time,
           ROW_NUMBER() OVER (
               PARTITION BY c.user_id, c.repaid_date
               ORDER BY c.repaid_time ASC, cp.vir_time ASC, cp.serial_id DESC
           ) AS rn
    FROM clean_settles c
    JOIN dedup_credit_pass cp
      ON cp.user_id = c.user_id
     AND cp.vir_date = c.repaid_date
     AND cp.vir_time >= c.repaid_time
) x
WHERE rn = 1;
CREATE INDEX ON t0_samples (user_id, repaid_date);

CREATE TEMP TABLE cycle_users AS
SELECT DISTINCT o.user_id
FROM all_orders o
JOIN wangchuanliang.side_recycle_type_copy r
  ON r.serial_id = o.serial_id
 AND r.recycle_type = 2
WHERE o.is_remit = 1;
CREATE INDEX ON cycle_users (user_id);

CREATE TEMP TABLE layer1_t0 AS
SELECT t.user_id,
       t.final_serial_id,
       t.final_apply_time,
       t.final_repaid_time,
       t.repaid_date,
       t.credit_serial_id,
       t.credit_vir_time,
       CASE WHEN c.user_id IS NOT NULL THEN '循环贷用户' ELSE '非循环贷用户' END AS user_type,
       CASE WHEN EXISTS (
           SELECT 1
           FROM all_orders n
           WHERE n.user_id = t.user_id
             AND n.apply_time > t.final_repaid_time
             AND n.apply_time::date = t.repaid_date
       ) THEN 1 ELSE 0 END AS is_t0_apply
FROM t0_samples t
LEFT JOIN cycle_users c ON c.user_id = t.user_id;
CREATE INDEX ON layer1_t0 (user_type, repaid_date);

CREATE TEMP TABLE layer2_path AS
SELECT DISTINCT t.user_id,
       t.final_serial_id,
       t.repaid_date,
       t.is_t0_apply
FROM layer1_t0 t
JOIN all_orders a
  ON a.user_id = t.user_id
 AND a.serial_id <> t.final_serial_id
 AND a.is_remit = 1
 AND a.loan_status_code = 7
 AND a.repaid_time < t.final_repaid_time
 AND t.final_apply_time < a.repaid_time
 AND a.repaid_time = (
     SELECT MAX(a2.repaid_time)
     FROM all_orders a2
     WHERE a2.user_id = t.user_id
       AND a2.serial_id <> t.final_serial_id
       AND a2.is_remit = 1
       AND a2.loan_status_code = 7
       AND a2.repaid_time < t.final_repaid_time
       AND t.final_apply_time < a2.repaid_time
 )
WHERE t.user_type = '循环贷用户'
  AND NOT EXISTS (
      SELECT 1
      FROM all_orders n
      WHERE n.user_id = t.user_id
        AND n.apply_time > a.repaid_time
        AND n.apply_time < t.final_repaid_time
  );
CREATE INDEX ON layer2_path (user_id, final_serial_id);

CREATE TEMP TABLE layer3_path AS
SELECT p.*
FROM layer2_path p
JOIN wangchuanliang.side_recycle_type_copy r
  ON r.serial_id = p.final_serial_id
 AND r.recycle_type = 2;
CREATE INDEX ON layer3_path (user_id, final_serial_id);
模块二：第一层循环贷/非循环贷 T0 样本、占比与当天提单率
WITH m AS (
    SELECT to_char(date_trunc('month', repaid_date), 'YYYY-MM') AS month,
           user_type,
           COUNT(*) AS sample_cnt,
           SUM(is_t0_apply) AS apply_cnt
    FROM layer1_t0
    GROUP BY GROUPING SETS ((to_char(date_trunc('month', repaid_date), 'YYYY-MM'), user_type), (user_type))
)
SELECT COALESCE(month, '总计') AS 结清月份,
       user_type AS 用户类型,
       sample_cnt AS T0样本数,
       ROUND(sample_cnt::numeric / SUM(sample_cnt) OVER (PARTITION BY month) * 100, 2) AS 样本占比_pct,
       apply_cnt AS 当天提单数,
       ROUND(apply_cnt::numeric / NULLIF(sample_cnt, 0) * 100, 2) AS 当天提单率_pct
FROM m
ORDER BY CASE WHEN month IS NULL THEN 0 ELSE 1 END, month, user_type;
模块三：第二层和第三层多在途结清转 T0 路径指标
WITH x AS (
    SELECT '第二层：原始路径' AS 路径层级,
           to_char(date_trunc('month', repaid_date), 'YYYY-MM') AS month,
           COUNT(*) AS sample_cnt,
           SUM(is_t0_apply) AS apply_cnt
    FROM layer2_path
    GROUP BY GROUPING SETS ((to_char(date_trunc('month', repaid_date), 'YYYY-MM')), ())
    UNION ALL
    SELECT '第三层：严格路径',
           to_char(date_trunc('month', repaid_date), 'YYYY-MM'),
           COUNT(*), SUM(is_t0_apply)
    FROM layer3_path
    GROUP BY GROUPING SETS ((to_char(date_trunc('month', repaid_date), 'YYYY-MM')), ())
), cycle AS (
    SELECT to_char(date_trunc('month', repaid_date), 'YYYY-MM') AS month,
           COUNT(*) AS cycle_t0_cnt
    FROM layer1_t0
    WHERE user_type = '循环贷用户'
    GROUP BY GROUPING SETS ((to_char(date_trunc('month', repaid_date), 'YYYY-MM')), ())
)
SELECT COALESCE(x.month, '总计') AS 结清月份,
       x.路径层级,
       x.sample_cnt AS 样本数,
       x.apply_cnt AS 当天提单数,
       ROUND(x.apply_cnt::numeric / NULLIF(x.sample_cnt, 0) * 100, 2) AS 当天提单率_pct,
       ROUND(x.sample_cnt::numeric / NULLIF(c.cycle_t0_cnt, 0) * 100, 2) AS 占循环贷T0_pct
FROM x JOIN cycle c ON c.month IS NOT DISTINCT FROM x.month
ORDER BY CASE WHEN x.month IS NULL THEN 0 ELSE 1 END, x.month, x.路径层级;
模块四：当天首次提单订单的利率与额度使用率
CREATE TEMP TABLE first_t0_apply AS
SELECT user_id, repaid_date, user_type, total_term, is_remit, pre_amt, post_amt, remit_amt, apply_amt
FROM (
    SELECT t.user_id, t.repaid_date, t.user_type,
           a.total_term, a.is_remit, a.pre_amt, a.post_amt, a.remit_amt, a.apply_amt,
           ROW_NUMBER() OVER (
             PARTITION BY t.user_id, t.repaid_date
             ORDER BY a.apply_time, a.serial_id DESC
           ) AS rn
    FROM layer1_t0 t
    JOIN all_orders a ON a.user_id = t.user_id
                     AND a.apply_time > t.final_repaid_time
                     AND a.apply_time::date = t.repaid_date
) x WHERE rn = 1;
CREATE INDEX ON first_t0_apply (user_type, repaid_date);

WITH m AS (
  SELECT to_char(date_trunc('month',repaid_date),'YYYY-MM') AS month,
         user_type,
         COUNT(CASE WHEN is_remit=1 AND remit_amt>0 THEN 1 END) AS remit_order_cnt,
         SUM(CASE WHEN is_remit=1 AND remit_amt>0 THEN COALESCE(pre_amt,0)+COALESCE(post_amt,0) ELSE 0 END)::numeric
           / NULLIF(SUM(CASE WHEN is_remit=1 AND remit_amt>0 THEN remit_amt ELSE 0 END),0) * 100 AS weighted_interest_rate_pct,
         COUNT(CASE WHEN apply_amt>=200 AND remit_amt>=200 THEN 1 END) AS usage_order_cnt,
         SUM(CASE WHEN apply_amt>=200 AND remit_amt>=200 THEN remit_amt ELSE 0 END)::numeric
           / NULLIF(SUM(CASE WHEN apply_amt>=200 AND remit_amt>=200 THEN apply_amt ELSE 0 END),0) * 100 AS weighted_utilization_rate_pct
  FROM first_t0_apply
  GROUP BY GROUPING SETS ((to_char(date_trunc('month',repaid_date),'YYYY-MM'),user_type),(user_type))
)
SELECT COALESCE(month,'总计') AS 结清月份, user_type AS 用户类型,
       remit_order_cnt AS 利率订单数, ROUND(weighted_interest_rate_pct,2) AS 加权利率_pct,
       usage_order_cnt AS 使用率订单数, ROUND(weighted_utilization_rate_pct,2) AS 加权额度使用率_pct
FROM m
ORDER BY CASE WHEN month IS NULL THEN 0 ELSE 1 END,month,user_type;
模块五：结清订单利率分组、当天首次提单期数及不同期数利率
WITH settled_rate AS (
    SELECT t.user_id,
           t.repaid_date,
           t.user_type,
           t.is_t0_apply,
           CASE WHEN o.remit_amt > 0
                THEN (COALESCE(o.pre_amt, 0) + COALESCE(o.post_amt, 0))
                     / o.remit_amt::numeric
           END AS settled_rate
    FROM layer1_t0 t
    JOIN all_orders o ON o.user_id = t.user_id
                     AND o.serial_id = t.final_serial_id
), rate_bucket AS (
    SELECT user_type,
           CASE
             WHEN settled_rate < 0.30 THEN '30%以下'
             WHEN settled_rate < 0.40 THEN '30%-40%'
             WHEN settled_rate < 0.50 THEN '40%-50%'
             WHEN settled_rate < 0.60 THEN '50%-60%'
             ELSE '60%以上'
           END AS metric_group,
           COUNT(*) AS denominator,
           SUM(is_t0_apply) AS numerator,
           ROUND(SUM(is_t0_apply)::numeric / NULLIF(COUNT(*),0) * 100,2) AS metric_value
    FROM settled_rate
    WHERE settled_rate IS NOT NULL
    GROUP BY user_type, metric_group
), term_distribution AS (
    SELECT user_type,
           total_term::text || '期' AS metric_group,
           COUNT(*) AS denominator,
           COUNT(*) AS numerator,
           COUNT(*)::numeric AS metric_value
    FROM first_t0_apply
    GROUP BY user_type, total_term
), term_interest AS (
    SELECT user_type,
           total_term::text || '期' AS metric_group,
           COUNT(CASE WHEN is_remit=1 AND remit_amt>0 THEN 1 END) AS denominator,
           COUNT(*) AS numerator,
           ROUND(
              SUM(CASE WHEN is_remit=1 AND remit_amt>0
                       THEN COALESCE(pre_amt,0)+COALESCE(post_amt,0) ELSE 0 END)::numeric
              / NULLIF(SUM(CASE WHEN is_remit=1 AND remit_amt>0 THEN remit_amt ELSE 0 END),0) * 100,
              2
           ) AS metric_value
    FROM first_t0_apply
    GROUP BY user_type, total_term
)
SELECT '结清订单利率分组当天提单率' AS 指标,
       user_type AS 用户类型,
       metric_group AS 分组,
       denominator AS 分母样本数,
       numerator AS 分子订单数,
       metric_value AS 指标值_pct
FROM rate_bucket
UNION ALL
SELECT '当天首次提单订单期数分布', user_type, metric_group,
       denominator, numerator, metric_value
FROM term_distribution
UNION ALL
SELECT '当天首次提单订单期数加权利率', user_type, metric_group,
       denominator, numerator, metric_value
FROM term_interest
ORDER BY 指标, 用户类型, 分组;
