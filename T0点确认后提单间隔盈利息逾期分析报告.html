结论

1. 点确认占比在下降，四档提单率只在点确认里算
1–8 月点确认 95,023 / 177,133 = 53.65%。1 月 60.46%，8 月 47.18%。确认后 24 小时内提单 96.89%；其中 3 分钟内 89.14%，3分钟–1小时 5.07%，1–3小时 1.03%，3–24小时 1.63%。正式提单高度集中在确认后 3 分钟。
2. 盈利不是「越晚越差」，真正偏弱的是 3 分钟–1 小时
合计到期盈利率：3 分钟内 6.55%，3分钟–1小时 4.50%（低 2.05pp），1–3小时 5.15%，3–24小时 7.44%。
3 分钟内是量的主体（到期 82,739 单），走势就是总体：1–2 月 2.6–3.9%，3–5 月抬到 8.8–10.4%，6–7 月回落到 5.4–5.9%。
3 分钟–1 小时（到期 4,668 单）1–4 月还好于或接近 3 分钟内，6 月 -1.95%、8 月 -4.18%，下半年明显转差。
后两档到期只有 946 / 1,506 单，单月盈利从约 -8% 到 +17%，3–24 小时合计 7.44% 是小样本拉高，不能读成「晚提更好」。7–8 月到期未完全成熟。
3. 四档逾期很接近，间隔几乎分不开风险
合计逾期：3 分钟内 38.27%，3分钟–1小时 38.13%，1–3小时 39.96%，3–24小时 39.04%，都在 38–40%。
逾期用结清日对到期日，四档几乎叠在一起；3分钟–1小时盈利更差，主要不是逾期标签拉开的，而是到期前回款相对放款更少。
4. 经营含义
确认后 3 分钟内提单同时贡献了量和相对稳定的盈利。催提应优先把 3 分钟未提的人尽快提进来（这 10.86% 里有一半会在 1 小时内补上）。
要单独盯的是「过了 3 分钟、又在 1 小时内才提」这一档，6 月以后盈利转负。不要因为 3–24 小时合计盈利偏高就放任晚提。24 小时仍未提单的 2,958 人不进这四档盈利表。

口径

T0 获额通过：2026-01-01～08-31；is_pass1=1、loan_type_code=2、recycle_type=3；20 秒批次；当天干净结清且 vir_time≥repaid_time；day_rn=1。
点确认：获额后 24 小时内第一次 Loan_newconfirm / Loan_newconfirm_select。
提单：确认后 24 小时内最早一笔 apply。四档互斥：<3 分钟；≥3 分钟且 <1 小时；≥1 小时且 <3 小时；≥3 小时且 <24 小时。
到期样本：该笔提单 is_remit=1、remit_amt>0、due_date < CURRENT_DATE。
逾期：NOT (repaid_date > 2000-01-01 AND repaid_date ≤ due_date)。
盈利率：Σ(payin_date 在 apply_date～due_date 的 repayment amount) / Σremit_amt − 1。

附录：完整 PGSQL（可复现全部指标）

把 __START_DATE__ / __END_DATE__ 换成 01月 2026-01-01～2026-02-01，…，08月 2026-08-01～2026-09-01。R1 是漏斗人数，R2 是四档到期还款与逾期。各月人数和金额相加即合计，比率用合计重算。

-- 1) T0 origin（人日）
DROP TABLE IF EXISTS tmp_m_origin;
CREATE TEMP TABLE tmp_m_origin AS
WITH params AS (
    SELECT DATE '__START_DATE__' AS start_date, DATE '__END_DATE__' AS end_date
), raw_offer AS (
    SELECT DISTINCT ON (v.user_id, v.serial_id)
        v.user_id, v.serial_id, v.vir_date::date AS vir_date, TO_TIMESTAMP(v.vir_unix) AS vir_time
    FROM wangchuanliang.order_vir_f_copy v
    INNER JOIN wangchuanliang.side_recycle_type_copy r ON r.serial_id = v.serial_id
    CROSS JOIN params p
    WHERE v.is_pass1 = 1 AND v.loan_type_code = 2 AND r.recycle_type = 3
      AND v.vir_date >= p.start_date - 1 AND v.vir_date < p.end_date
    ORDER BY v.user_id, v.serial_id, v.vir_unix DESC
), lagged AS (
    SELECT r.*, LAG(r.vir_time) OVER (PARTITION BY r.user_id ORDER BY r.vir_time, r.serial_id) AS prev_vir_time
    FROM raw_offer r
), batched AS (
    SELECT l.*, SUM(CASE WHEN l.prev_vir_time IS NULL OR l.vir_time - l.prev_vir_time > INTERVAL '20 seconds' THEN 1 ELSE 0 END)
      OVER (PARTITION BY l.user_id ORDER BY l.vir_time, l.serial_id ROWS UNBOUNDED PRECEDING) AS batch_id
    FROM lagged l
), dedup_offer AS (
    SELECT user_id, serial_id, vir_date, vir_time
    FROM (
        SELECT b.*, ROW_NUMBER() OVER (PARTITION BY b.user_id, b.batch_id ORDER BY b.vir_time DESC, b.serial_id DESC) AS batch_rn
        FROM batched b
    ) x
    CROSS JOIN params p
    WHERE batch_rn = 1 AND vir_date >= p.start_date AND vir_date < p.end_date
), candidate_users AS (SELECT DISTINCT user_id FROM dedup_offer),
all_orders AS (
    SELECT o.user_id, o.serial_id, o.apply_time, o.repaid_time, o.repaid_date::date AS repaid_date,
           o.loan_status_code, o.is_remit, o.is_repaid
    FROM wangchuanliang.order_loan_f_v2_copy o
    INNER JOIN candidate_users u ON u.user_id = o.user_id
), clean_settlement AS (
    SELECT o.user_id, o.serial_id, o.repaid_time, o.repaid_date
    FROM all_orders o
    WHERE o.is_remit = 1 AND (o.is_repaid = 1 OR o.loan_status_code = 7)
      AND o.repaid_time > TIMESTAMP '2000-01-01 00:00:00'
      AND NOT EXISTS (
          SELECT 1 FROM all_orders x
          WHERE x.user_id = o.user_id AND x.serial_id <> o.serial_id AND x.apply_time < o.repaid_time
            AND (x.loan_status_code IN (5, 8) OR (x.loan_status_code = 7 AND x.repaid_time > o.repaid_time))
      )
), paired AS (
    SELECT e.user_id, e.serial_id AS credit_serial_id, e.vir_date, e.vir_time,
           ROW_NUMBER() OVER (PARTITION BY e.user_id, e.vir_date ORDER BY e.vir_time ASC, e.serial_id DESC) AS day_rn
    FROM dedup_offer e
    INNER JOIN clean_settlement s
        ON s.user_id = e.user_id AND s.repaid_date = e.vir_date AND e.vir_time >= s.repaid_time
)
SELECT user_id, credit_serial_id, vir_date, vir_time FROM paired WHERE day_rn = 1;

-- 2) 获额后 24h 内第一次点确认
DROP TABLE IF EXISTS tmp_confirm;
CREATE TEMP TABLE tmp_confirm AS
SELECT t.user_id, t.vir_date, t.vir_time, MIN(e.create_time) AS first_confirm
FROM tmp_m_origin t
INNER JOIN wangchuanliang.event_track_record e
  ON e.user_id = t.user_id
 AND e.create_date >= (SELECT MIN(vir_date) FROM tmp_m_origin)
 AND e.create_date <= (SELECT MAX(vir_date) FROM tmp_m_origin) + 1
 AND e.create_date >= t.vir_date AND e.create_date <= t.vir_date + 1
 AND e.create_time >= t.vir_time
 AND e.create_time < t.vir_time + INTERVAL '24 hours'
 AND e.props_id IN ('Loan_newconfirm', 'Loan_newconfirm_select')
GROUP BY t.user_id, t.vir_date, t.vir_time;

-- 3) 确认后 24h 内最早一笔提单（带间隔分档、订单字段）
DROP TABLE IF EXISTS tmp_first_apply;
CREATE TEMP TABLE tmp_first_apply AS
SELECT
    z.user_id, z.vir_date, z.first_confirm, z.serial_id, z.apply_time, z.apply_date,
    z.is_remit, z.remit_amt, z.due_date, z.repaid_date, z.gap
FROM (
    SELECT
        c.user_id, c.vir_date, c.first_confirm,
        o.serial_id, o.apply_time, o.apply_date::date AS apply_date,
        o.is_remit, o.remit_amt, o.due_date::date AS due_date, o.repaid_date::date AS repaid_date,
        CASE
            WHEN o.apply_time < c.first_confirm + INTERVAL '3 minutes' THEN '3m'
            WHEN o.apply_time < c.first_confirm + INTERVAL '1 hour' THEN '3m_1h'
            WHEN o.apply_time < c.first_confirm + INTERVAL '3 hours' THEN '1h_3h'
            ELSE '3h_24h'
        END AS gap,
        ROW_NUMBER() OVER (PARTITION BY c.user_id, c.vir_date ORDER BY o.apply_time, o.serial_id) AS rn
    FROM tmp_confirm c
    INNER JOIN wangchuanliang.order_loan_f_v2_copy o
      ON o.user_id = c.user_id
     AND o.apply_time >= c.first_confirm
     AND o.apply_time < c.first_confirm + INTERVAL '24 hours'
) z
WHERE z.rn = 1;

-- 4) 到期日及之前还款流水（整笔订单）
DROP TABLE IF EXISTS tmp_repay;
CREATE TEMP TABLE tmp_repay AS
SELECT a.serial_id, SUM(r.amount) AS repay_amt
FROM tmp_first_apply a
INNER JOIN wangchuanliang.repayment_record_copy r ON r.serial_id = a.serial_id
WHERE a.is_remit = 1
  AND COALESCE(a.remit_amt, 0) > 0
  AND a.due_date < CURRENT_DATE
  AND r.payin_date >= a.apply_date
  AND r.payin_date <= a.due_date
GROUP BY a.serial_id;

-- R1 漏斗：点确认占比 + 四档互斥提单人数
SELECT
    (SELECT COUNT(*) FROM tmp_m_origin) AS n_t0,
    (SELECT COUNT(*) FROM tmp_confirm) AS n_confirm,
    COUNT(*) AS n_apply_24h,
    SUM(CASE WHEN gap = '3m' THEN 1 ELSE 0 END) AS n_3m,
    SUM(CASE WHEN gap = '3m_1h' THEN 1 ELSE 0 END) AS n_3m_1h,
    SUM(CASE WHEN gap = '1h_3h' THEN 1 ELSE 0 END) AS n_1h_3h,
    SUM(CASE WHEN gap = '3h_24h' THEN 1 ELSE 0 END) AS n_3h_24h
FROM tmp_first_apply;

-- R2 四档：到期单、逾期、到期前回款 / 放款
SELECT
    a.gap,
    COUNT(*) AS n_apply,
    SUM(CASE WHEN a.is_remit = 1 AND COALESCE(a.remit_amt, 0) > 0 AND a.due_date < CURRENT_DATE THEN 1 ELSE 0 END) AS n_due,
    SUM(CASE
            WHEN a.is_remit = 1 AND COALESCE(a.remit_amt, 0) > 0 AND a.due_date < CURRENT_DATE
             AND NOT (a.repaid_date > DATE '2000-01-01' AND a.repaid_date <= a.due_date)
            THEN 1 ELSE 0 END) AS n_od,
    SUM(CASE WHEN a.is_remit = 1 AND COALESCE(a.remit_amt, 0) > 0 AND a.due_date < CURRENT_DATE
             THEN a.remit_amt ELSE 0 END) AS due_remit,
    SUM(CASE WHEN a.is_remit = 1 AND COALESCE(a.remit_amt, 0) > 0 AND a.due_date < CURRENT_DATE
             THEN COALESCE(p.repay_amt, 0) ELSE 0 END) AS due_repay
FROM tmp_first_apply a
LEFT JOIN tmp_repay p ON p.serial_id = a.serial_id
GROUP BY a.gap;

数据截至库内 CURRENT_DATE。7–8 月到期未完全成熟。源表：order_vir_f_copy、side_recycle_type_copy、order_loan_f_v2_copy、event_track_record、repayment_record_copy。
