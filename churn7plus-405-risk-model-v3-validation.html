四、第三项：长期流失优质客群的概率校准

检验方法

  • 独立使用历史向前预测，不使用模型在自身训练样本上的预测。
  • 比较整体校准与“405四档×流失是否≥30天”的8个分层校准；小客群的调整幅度向整体曲线收缩。

五、新发现：重点客群风险关系随时间变化

3—6月前向校准验证的长期流失优质客群有756条记录，实际D7逾期率22.35%；8—9月测试同一分组317条记录，实际为35.96%。两段都是按相同405档位和流失条件划分，但不是同一批用户。

这意味着什么

  • 用较早的风险关系做校准，无法自动纠正后续月份明显升高的实际风险。
  • 更多历史还款细分可以帮助部分月份排序，却不等于捕捉到了用户当前风险变化。
  • 该差异提示需要检查客群组成、产品/账期和审批选择的时间变化；不能直接解释为同一个用户因流失而恶化。

这仍是相关性诊断。只对回流且成功放款订单有风险标签；未回流、未放款用户不能直接套用同等风险解释。

八、模型决策

保留V1，不替换正式模型

  • 本轮三项检验已完成，不能宣称V3提高了最终预测准确度。
  • 保留“最近1—5笔借还细分”作为后续候选，其余特征组不凭数量加入。
  • 下一步若继续，需要用新成熟月份作前瞻验收，并检查长期流失客群的产品、账期和风险关系变化。不得用当前测试实际逾期率反向校准测试预测。

def digest(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for block in iter(lambda:f.read(1048576),b''):h.update(block)
    return h.hexdigest()


def metrics(frame, probability):
    y=frame.target_d7.to_numpy(int);p=np.clip(np.asarray(probability,float),1e-6,1-1e-6)
    n=len(y)
    if not n:return {'n':0,'bad':0,'actual':None,'prediction':None,'auc':None,'brier':None}
    r={'n':n,'users':int(frame.user_id.nunique()),'bad':int(y.sum()),'actual':float(y.mean()),
       'prediction':float(p.mean()),'bias_pp':float((p.mean()-y.mean())*100),
       'brier':float(brier_score_loss(y,p)),'log_loss':float(log_loss(y,p,labels=[0,1]))}
    if len(np.unique(y))==2:
        f,t,_=roc_curve(y,p);top=np.argsort(-p,kind='stable')[:max(1,int(np.ceil(n*.2)))]
        r.update(auc=float(roc_auc_score(y,p)),ks=float(max(t-f)),ap=float(average_precision_score(y,p)),
                 top20_bad_rate=float(y[top].mean()),top20_bad_recall=float(y[top].sum()/y.sum()))
    else:r['auc']=None
    return r

def trim_new_fields(frame, requested, base):
    """Never alter the 57 original baseline fields. Fit trimming on training only."""
    keep=list(base);removed=[]
    arrays={f:frame[f].to_numpy(float) for f in keep}
    fingerprints={}
    for f in keep:
        fp=hashlib.sha256(np.nan_to_num(arrays[f],nan=-1e300).tobytes()).hexdigest()
        fingerprints.setdefault(fp,[]).append(f)
    for f in requested:
        if f in keep:continue
        a=frame[f].to_numpy(float)
        if frame[f].nunique(dropna=False)<=1:
            removed.append({'field':f,'reason':'训练期常量','duplicate_of':None});continue
        fp=hashlib.sha256(np.nan_to_num(a,nan=-1e300).tobytes()).hexdigest()
        match=next((g for g in fingerprints.get(fp,[]) if np.array_equal(a,arrays[g],equal_nan=True)),None)
        if match is not None:
            removed.append({'field':f,'reason':'训练期完全重复','duplicate_of':match});continue
        keep.append(f);arrays[f]=a;fingerprints.setdefault(fp,[]).append(f)
    return keep,removed


def groups(allcols,base):
    out={k:[] for k in GROUP_LABELS}
    for f in allcols:
        if f in base:continue
        if f.startswith('score405'):g='score'
        elif f.startswith(('app_','days_since_app')):g='app'
        elif f.startswith('nonpay_'):g='nonpay'
        elif f.startswith(('last1_','last2_','last3_','last4_','last5_')) and not any(s in f for s in ('minus','_count','_rate')):g='recent'
        elif f.startswith('last') and f.endswith('_count'):g='windows'
        elif f.startswith('detail_') and any(f.endswith('_'+str(w)+'d') for w in (90,180,365)) and '_rate_' not in f:g='windows'
        elif '_rate' in f or f=='detail_cycle_funded_share':g='rates'
        elif f.startswith('detail_'):g='history'
        else:g='trends'
        out[g].append(f)
    assert set(sum(out.values(),[]))==set(allcols)-set(base)
    assert len(sum(out.values(),[]))==len(set(allcols)-set(base))
    return out

    for k in nonapp:out['drop_'+k]=base+sum((g[j] for j in nonapp if j!=k),[])

def fit_fold(tag, cols, base, train, validation, start, cuts):
    file=ROOT/'models'/f'fold_{start.strftime("%Y%m")}_{tag}.joblib'
    kept,removed=trim_new_fields(train,cols,base)
    if file.exists():
        bundle=joblib.load(file)
        assert bundle['features']==kept and bundle['train_ids_sha']==digest_ids(train)
    else:
        model=HistGradientBoostingClassifier(**PARAMS)
        model.fit(train[kept],train.target_d7)
        bundle={'model':model,'features':kept,'removed':removed,'params':PARAMS,
                'train_ids_sha':digest_ids(train),'score_feature_cuts':cuts}
        joblib.dump(bundle,file)
    p=bundle['model'].predict_proba(validation[kept])[:,1]
    detail=validation[['event_id','user_id','obs_date','target_due_date','target_d7','risk_score_405','churn_days']].copy()
    detail['fold']=start.strftime('%Y-%m');detail['raw_probability']=p
    detail.to_csv(ROOT/'data'/f'oof_{start.strftime("%Y%m")}_{tag}.csv.gz',index=False,compression='gzip')
    stat={'tag':tag,'fold':start.strftime('%Y-%m'),'train_n':len(train),'train_bad':int(train.target_d7.sum()),
          'train_label_cutoff':str(start.date()),'validation_label_cutoff':str(LOCK_DATE.date()),
          'features':len(kept),'requested_features':len(cols),'removed':removed,
          'overall':metrics(validation,p),'focus':metrics(validation.loc[focus(validation)],p[focus(validation)]),
          'long_churn':metrics(validation.loc[validation.churn_days>=30],p[validation.churn_days>=30])}
    return stat


def digest_ids(frame):
    return hashlib.sha256('\n'.join(frame.event_id.astype(str)).encode()).hexdigest()


def summarize(rows):
    base={x['fold']:x for x in rows if x['tag']=='base'}
    out=[]
    for tag in sorted({x['tag'] for x in rows}):
        rr=[x for x in rows if x['tag']==tag]
        delta=np.array([x['overall']['auc']-base[x['fold']]['overall']['auc'] for x in rr])
        out.append({'tag':tag,'mean_auc':float(np.mean([x['overall']['auc'] for x in rr])),
                    'mean_delta_auc':float(delta.mean()),'positive_months':int((delta>0).sum()),
                    'worst_delta_auc':float(delta.min()),'best_delta_auc':float(delta.max()),
                    'mean_focus_auc':float(np.mean([x['focus']['auc'] for x in rr if x['focus']['auc'] is not None])),
                    'stable_gain':bool(delta.mean()>=.002 and (delta>0).sum()>=4 and delta.min()>=-.02)})
    return out


def load_oof(tag):
    frames=[]
    for p in sorted((ROOT/'data').glob('oof_2026*_'+tag+'.csv.gz')):
        frames.append(pd.read_csv(p,dtype={'event_id':str},parse_dates=['obs_date','target_due_date'],float_precision='round_trip'))
    out=pd.concat(frames,ignore_index=True)
    assert out.event_id.is_unique
    return out


def bootstrap(frame, a, b, repeats=1000):
    rng=np.random.default_rng(SEED)
    users,inv=np.unique(frame.user_id,return_inverse=True);y=frame.target_d7.to_numpy(int)
    pa=frame[a].to_numpy();pb=frame[b].to_numpy();values=[];brier=[]
    ea=(pa-y)**2;eb=(pb-y)**2
    for _ in range(repeats):
        c=np.bincount(rng.integers(0,len(users),len(users)),minlength=len(users));w=c[inv]
        values.append(roc_auc_score(y,pa,sample_weight=w)-roc_auc_score(y,pb,sample_weight=w))
        brier.append(np.average(ea-eb,weights=w))
    return {'repeats':repeats,'unit':'user_id','auc_delta_low95':float(np.quantile(values,.025)),
            'auc_delta_high95':float(np.quantile(values,.975)),
            'brier_delta_low95':float(np.quantile(brier,.025)),'brier_delta_high95':float(np.quantile(brier,.975))}

"""Portable, regularized probability calibration; no post-observation inputs."""
import numpy as np
from scipy.optimize import minimize
from scipy.special import expit, logit


def risk_cells(frame, cuts=(487., 472., 456.)):
    score = frame.risk_score_405.to_numpy(float)
    grade = np.select([score >= cuts[0], score >= cuts[1], score >= cuts[2]], [0, 1, 2], default=3)
    return grade * 2 + frame.churn_days.ge(30).to_numpy(int)


def fit_calibration(probability, frame, asof, penalty=None):
    """Positive slope. Cell offsets shrink toward the global calibration curve.

    Training data must be out-of-sample predictions with labels mature by asof.
    The last 180 days are used; 90-day half-life is fixed before test evaluation.
    """
    import pandas as pd
    asof = pd.Timestamp(asof)
    age = (asof - frame.obs_date).dt.days.to_numpy(float)
    assert (age >= 0).all()
    keep = age <= 180
    f = frame.loc[keep].copy()
    p = np.clip(np.asarray(probability)[keep], 1e-6, 1-1e-6)
    y = f.target_d7.to_numpy(float)
    assert len(y) >= 200 and len(np.unique(y)) == 2
    weight = np.power(.5, age[keep] / 90.)
    z = logit(p)
    cell = risk_cells(f)
    nparam = 2 if penalty is None else 10
    def loss(theta):
        eta = theta[0]*z + theta[1]
        if penalty is not None:
            eta = eta + theta[2:][cell]
        q = expit(eta)
        value = np.sum(weight*(np.logaddexp(0, eta)-y*eta))
        gradient = np.zeros(nparam)
        residual = weight*(q-y)
        gradient[0] = np.dot(residual, z)
        gradient[1] = residual.sum()
        # Modest slope regularization, shared by global and grouped variants.
        value += .5*(theta[0]-1)**2
        gradient[0] += theta[0]-1
        if penalty is not None:
            value += .5*penalty*np.dot(theta[2:], theta[2:])
            gradient[2:] = np.bincount(cell, weights=residual, minlength=8)+penalty*theta[2:]
        return float(value), gradient
    initial = np.zeros(nparam); initial[0] = 1.
    result = minimize(loss, initial, jac=True, method='L-BFGS-B',
                      bounds=[(.05, 5.), (None, None)]+[(None, None)]*(nparam-2),
                      options={'maxiter': 2000, 'ftol': 1e-12, 'gtol': 1e-7})
    assert result.success, result.message
    return {'theta':result.x, 'penalty':penalty, 'asof':str(asof.date()),
            'rows':len(f), 'users':int(f.user_id.nunique()), 'bad':int(y.sum()),
            'cell_counts':np.bincount(cell, minlength=8).tolist(),
            'positive_slope':float(result.x[0]), 'window_days':180, 'half_life_days':90,
            'optimizer_converged':bool(result.success)}


def apply_calibration(probability, frame, calibrator):
    z = logit(np.clip(np.asarray(probability), 1e-6, 1-1e-6))
    theta = calibrator['theta']
    eta = theta[0]*z + theta[1]
    if calibrator['penalty'] is not None:
        eta = eta + theta[2:][risk_cells(frame)]
    return np.clip(expit(eta), 1e-6, 1-1e-6)


def refresh_score_features(frame, cuts):
    frame = frame.copy()
    score = frame.risk_score_405
    grade = np.select([score>=cuts[0], score>=cuts[1], score>=cuts[2]], [0,1,2], default=3)
    frame['score405_times_churn_log'] = (score-cuts[1])/25*frame.churn_log1p
    frame['score405_times_early_rate'] = (score-cuts[1])/25*frame.hist_early_rate
    for g in range(4):
        frame[f'score405_band_{g}'] = (grade==g).astype(float)
        frame[f'score405_band_{g}_churn_log'] = (grade==g)*frame.churn_log1p
        for days in (15,30,60,90):
            frame[f'score405_band_{g}_churn_ge{days}'] = ((grade==g)&(frame.churn_days>=days)).astype(float)
    return frame


def predict_bundle(bundle, frame, mode='selected'):
    f = refresh_score_features(frame, bundle['score_feature_cuts'])
    raw = bundle['model'].predict_proba(f[bundle['features']])[:,1]
    key = 'global_calibrator' if mode=='global' else 'selected_calibrator'
    return apply_calibration(raw, f, bundle[key])
