DROP TABLE IF EXISTS tmp_u;
CREATE TEMP TABLE tmp_u AS
SELECT DISTINCT churn_user_id::bigint AS user_id
FROM wangchuanliang.t7recalllist_0818_0819
WHERE recall_date = DATE '2026-08-18' AND churn_user_id IS NOT NULL;

DROP TABLE IF EXISTS tmp_ord;
CREATE TEMP TABLE tmp_ord AS
SELECT o.user_id, o.serial_id, o.apply_time, o.apply_date::date AS apply_date,
       o.is_remit, COALESCE(o.remit_amt,0) AS remit_amt,
       o.loan_status_code, o.is_repaid, o.repaid_time, o.repaid_date::date AS repaid_date
FROM wangchuanliang.order_loan_f_v2_copy o
WHERE o.user_id IN (SELECT user_id FROM tmp_u);
CREATE INDEX ON tmp_ord (user_id);

DROP TABLE IF EXISTS tmp_first;
CREATE TEMP TABLE tmp_first AS
SELECT user_id, serial_id, repaid_time, repaid_date
FROM (
    SELECT o.*, ROW_NUMBER() OVER (PARTITION BY o.user_id ORDER BY o.apply_time, o.serial_id) AS rn
    FROM tmp_ord o
    WHERE o.apply_date >= DATE '2026-08-18' AND o.is_remit = 1 AND o.remit_amt > 0
) z
WHERE rn = 1 AND loan_status_code = 7
  AND repaid_time > TIMESTAMP '2000-01-01 00:00:00'
  AND repaid_date > DATE '2000-01-01';

DROP TABLE IF EXISTS tmp_rec;
CREATE TEMP TABLE tmp_rec AS
WITH params AS (
    SELECT DATE '2026-08-18' AS start_date, CURRENT_DATE + 1 AS end_date
), raw_offer AS (
    SELECT DISTINCT ON (v.user_id, v.serial_id)
        v.user_id, v.serial_id, v.vir_date::date AS vir_date, TO_TIMESTAMP(v.vir_unix) AS vir_time
    FROM wangchuanliang.order_vir_f_copy v
    INNER JOIN wangchuanliang.side_recycle_type_copy r ON r.serial_id = v.serial_id
    INNER JOIN tmp_first f ON f.user_id = v.user_id
    CROSS JOIN params p
    WHERE v.is_pass1 = 1 AND v.loan_type_code = 2 AND r.recycle_type = 3
      AND v.vir_date >= p.start_date - 1 AND v.vir_date < p.end_date
    ORDER BY v.user_id, v.serial_id, v.vir_unix DESC
), lagged AS (
    SELECT r.*, LAG(r.vir_time) OVER (PARTITION BY r.user_id ORDER BY r.vir_time, r.serial_id) AS prev_vir_time
    FROM raw_offer r
), batched AS (
    SELECT l.*, SUM(CASE WHEN l.prev_vir_time IS NULL OR l.vir_time - l.prev_vir_time > INTERVAL '20 seconds' THEN 1 ELSE 0 END)
      OVER (PARTITION BY l.user_id ORDER BY l.vir_time, l.serial_id ROWS UNBOUNDED PRECEDING) AS batch_id
    FROM lagged l
), dedup_offer AS (
    SELECT user_id, serial_id, vir_date, vir_time
    FROM (
        SELECT b.*, ROW_NUMBER() OVER (PARTITION BY b.user_id, b.batch_id ORDER BY b.vir_time DESC, b.serial_id DESC) AS batch_rn
        FROM batched b
    ) x
    CROSS JOIN params p
    WHERE batch_rn = 1 AND vir_date >= p.start_date AND vir_date < p.end_date
), clean_settlement AS (
    SELECT o.user_id, o.serial_id, o.repaid_time, o.repaid_date
    FROM tmp_ord o
    WHERE o.is_remit = 1 AND (o.is_repaid = 1 OR o.loan_status_code = 7)
      AND o.repaid_time > TIMESTAMP '2000-01-01 00:00:00'
      AND NOT EXISTS (
          SELECT 1 FROM tmp_ord x
          WHERE x.user_id = o.user_id AND x.serial_id <> o.serial_id AND x.apply_time < o.repaid_time
            AND (x.loan_status_code IN (5, 8) OR (x.loan_status_code = 7 AND x.repaid_time > o.repaid_time))
      )
), paired AS (
    SELECT e.user_id, e.serial_id AS credit_serial_id, e.vir_date, e.vir_time,
           ROW_NUMBER() OVER (PARTITION BY e.user_id, e.vir_date ORDER BY e.vir_time DESC, e.serial_id DESC) AS day_rn
    FROM dedup_offer e
    INNER JOIN tmp_first f
        ON f.user_id = e.user_id AND f.repaid_date = e.vir_date AND e.vir_time >= f.repaid_time
    INNER JOIN clean_settlement s
        ON s.user_id = f.user_id AND s.serial_id = f.serial_id
)
SELECT user_id, credit_serial_id, vir_date, vir_time
FROM paired
WHERE day_rn = 1 AND vir_date <= CURRENT_DATE - 7
;

DROP TABLE IF EXISTS tmp_sc;
CREATE TEMP TABLE tmp_sc AS
SELECT t.user_id, t.vir_date,
       CASE WHEN m.kabybmerge410fpd7cre > 0 THEN m.kabybmerge410fpd7cre END AS score,
       a.apply_d,
       CASE
         WHEN a.apply_d = 0 THEN '当天'
         WHEN a.apply_d BETWEEN 1 AND 3 THEN '1-3日'
         WHEN a.apply_d BETWEEN 4 AND 7 THEN '4-7日'
         ELSE '7日内未提单'
       END AS bucket
FROM __SRC__ t
LEFT JOIN wangchuanliang.model_result_copy m ON m.serial_id = t.credit_serial_id
LEFT JOIN (
    SELECT user_id, vir_date, apply_d FROM (
        SELECT t.user_id, t.vir_date, (a.apply_date::date - t.vir_date) AS apply_d,
               ROW_NUMBER() OVER (PARTITION BY t.user_id, t.vir_date ORDER BY a.apply_time, a.serial_id) AS rn
        FROM __SRC__ t
        INNER JOIN wangchuanliang.order_loan_f_v2_copy a
          ON a.user_id = t.user_id AND a.apply_time >= t.vir_time
    ) z WHERE rn = 1
) a ON a.user_id = t.user_id AND a.vir_date = t.vir_date;

SELECT bucket, COUNT(*) AS n,
       SUM(CASE WHEN score IS NULL THEN 1 ELSE 0 END) AS n_miss,
       ROUND(AVG(score)::numeric, 1) AS avg_s,
       ROUND(PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY score)::numeric, 1) AS p50,
       SUM(CASE WHEN score < 460 THEN 1 ELSE 0 END) AS b1,
       SUM(CASE WHEN score >= 460 AND score < 480 THEN 1 ELSE 0 END) AS b2,
       SUM(CASE WHEN score >= 480 AND score < 500 THEN 1 ELSE 0 END) AS b3,
       SUM(CASE WHEN score >= 500 AND score < 520 THEN 1 ELSE 0 END) AS b4,
       SUM(CASE WHEN score >= 520 AND score < 540 THEN 1 ELSE 0 END) AS b5,
       SUM(CASE WHEN score >= 540 AND score < 560 THEN 1 ELSE 0 END) AS b6,
       SUM(CASE WHEN score >= 560 THEN 1 ELSE 0 END) AS b7
FROM (
    SELECT '全部' AS bucket, score FROM tmp_sc
    UNION ALL
    SELECT bucket, score FROM tmp_sc
) z
GROUP BY bucket
