口径

V2 名单wangchuanliang.t7recalllist_active0918。字段 churn_user_id、score。同一 churn_user_id 按 score DESC NULLS LAST 保留一行。表内无 recall_date，触达日取 2026-09-18。
V1 名单wangchuanliang.t7recall_active0910。去重规则与 V2 相同。库内无 t7recalllist_active0910。触达日取 2026-09-10。
提单用户order_loan_f_v2_copy.apply_date ≥ 该版本触达日，且 apply_date ≤ 2026-10-13。同一 user_id 有至少 1 笔订单记为提单用户。不筛选 is_remit，不要求 remit_amt > 0。
提单率档内提单用户数 / 档内名单用户数。分母为去重后的名单人数，分子为上述提单用户与该档的交集。
五档切分版本内按 score DESC、user_id 升序编号。pct = (row_number − 1) / 名单人数。S：pct < 0.10；A：pct < 0.20；B：pct < 0.40；C：pct < 0.70；D：其余。正文五档表只列 V2。
前10%版本内 NTILE(10) = 1，排序与五档相同。该集合与 S 档人数一致。覆盖率 = 该集合中的提单用户数 / 该版本全部提单用户数。
前三分之一版本内 NTILE(3) = 1。覆盖率分母同样为该版本全部提单用户数。
S / DS 档提单率 ÷ D 档提单率。在各版本内部计算，不跨版本比较绝对分数。
S 档相对本批S 档提单率 ÷ 该版本整体提单率。
数据日kaby_dw CURRENT_DATE = 2026-10-13。V1 观察 34 个自然日，V2 观察 26 个自然日，均含起止日。

DROP TABLE IF EXISTS tmp_u;
CREATE TEMP TABLE tmp_u AS
SELECT 'V1' AS model, '2026-09-10'::date AS start_dt, user_id, score FROM (
    SELECT DISTINCT ON (churn_user_id)
        churn_user_id::bigint AS user_id, score::numeric AS score
    FROM wangchuanliang.t7recall_active0910
    WHERE churn_user_id IS NOT NULL
    ORDER BY churn_user_id, score DESC NULLS LAST
) a
UNION ALL
SELECT 'V2', '2026-09-18'::date, user_id, score FROM (
    SELECT DISTINCT ON (churn_user_id)
        churn_user_id::bigint AS user_id, score::numeric AS score
    FROM wangchuanliang.t7recalllist_active0918
    WHERE churn_user_id IS NOT NULL
    ORDER BY churn_user_id, score DESC NULLS LAST
) b;

DROP TABLE IF EXISTS tmp_g;
CREATE TEMP TABLE tmp_g AS
SELECT model, start_dt, user_id, score,
       CASE
           WHEN pct < 0.10 THEN 'S'
           WHEN pct < 0.20 THEN 'A'
           WHEN pct < 0.40 THEN 'B'
           WHEN pct < 0.70 THEN 'C'
           ELSE 'D'
       END AS grp,
       NTILE(3) OVER (PARTITION BY model ORDER BY score DESC NULLS LAST, user_id) AS tertile,
       NTILE(10) OVER (PARTITION BY model ORDER BY score DESC NULLS LAST, user_id) AS decile
FROM (
    SELECT model, start_dt, user_id, score,
           (ROW_NUMBER() OVER (PARTITION BY model ORDER BY score DESC NULLS LAST, user_id) - 1)::numeric
               / NULLIF(COUNT(*) OVER (PARTITION BY model), 0) AS pct
    FROM tmp_u
) u;

DROP TABLE IF EXISTS tmp_a;
CREATE TEMP TABLE tmp_a AS
SELECT u.model, u.user_id
FROM tmp_u u
INNER JOIN wangchuanliang.order_loan_f_v2_copy o ON o.user_id = u.user_id
WHERE o.apply_date >= u.start_dt AND o.apply_date <= CURRENT_DATE
GROUP BY u.model, u.user_id;

SELECT CURRENT_DATE AS as_of;

SELECT g.model, g.grp,
       COUNT(*) AS n_user,
       COUNT(a.user_id) AS n_apply,
       MIN(g.score) AS smin,
       MAX(g.score) AS smax
FROM tmp_g g
LEFT JOIN tmp_a a ON a.model = g.model AND a.user_id = g.user_id
GROUP BY g.model, g.grp
ORDER BY g.model, CASE g.grp WHEN 'S' THEN 1 WHEN 'A' THEN 2 WHEN 'B' THEN 3 WHEN 'C' THEN 4 ELSE 5 END;

SELECT g.model,
       COUNT(*) AS n_user,
       COUNT(a.user_id) AS n_apply,
       SUM(CASE WHEN g.decile = 1 THEN 1 ELSE 0 END) AS n_top10,
       SUM(CASE WHEN g.decile = 1 AND a.user_id IS NOT NULL THEN 1 ELSE 0 END) AS apply_top10,
       MIN(CASE WHEN g.decile = 1 THEN g.score END) AS smin10,
       MAX(CASE WHEN g.decile = 1 THEN g.score END) AS smax10,
       SUM(CASE WHEN g.tertile = 1 THEN 1 ELSE 0 END) AS n_top3,
       SUM(CASE WHEN g.tertile = 1 AND a.user_id IS NOT NULL THEN 1 ELSE 0 END) AS apply_top3,
       MIN(CASE WHEN g.tertile = 1 THEN g.score END) AS smin3,
       MAX(CASE WHEN g.tertile = 1 THEN g.score END) AS smax3
FROM tmp_g g
LEFT JOIN tmp_a a ON a.model = g.model AND a.user_id = g.user_id
GROUP BY g.model
ORDER BY g.model;
