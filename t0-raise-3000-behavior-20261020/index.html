WITH params AS (
    SELECT DATE '2026-10-01' AS start_date,
           DATE '2026-10-09' AS end_date
), raw_offer AS (
    SELECT DISTINCT ON (v.user_id, v.serial_id)
        v.user_id,
        v.serial_id AS credit_serial_id,
        v.vir_date::date AS vir_date,
        TO_TIMESTAMP(v.vir_unix) AS vir_time,
        v.max_credit_apply_amt::numeric AS current_credit_amt
    FROM wangchuanliang.order_vir_f_copy v
    CROSS JOIN params p
    WHERE v.is_pass1 = 1
      AND v.loan_type_code = 2
      AND EXISTS (SELECT 1 FROM wangchuanliang.side_recycle_type_copy r WHERE r.serial_id=v.serial_id AND r.recycle_type=3)
      AND v.vir_date >= p.start_date - 1
      AND v.vir_date < p.end_date + 1
    ORDER BY v.user_id, v.serial_id, v.vir_unix DESC
), lagged_offer AS (
    SELECT r.*,
      LAG(r.vir_time) OVER (
        PARTITION BY r.user_id ORDER BY r.vir_time, r.credit_serial_id
      ) AS prev_vir_time
    FROM raw_offer r
), batched_offer AS (
    SELECT l.*,
      SUM(
        CASE WHEN l.prev_vir_time IS NULL
                   OR l.vir_time - l.prev_vir_time > INTERVAL '20 seconds'
             THEN 1 ELSE 0 END
      ) OVER (
        PARTITION BY l.user_id
        ORDER BY l.vir_time, l.credit_serial_id
        ROWS UNBOUNDED PRECEDING
      ) AS batch_id
    FROM lagged_offer l
), dedup_offer AS (
    SELECT user_id, credit_serial_id, vir_date, vir_time,
           current_credit_amt, batch_id
    FROM (
      SELECT b.*,
        ROW_NUMBER() OVER (
          PARTITION BY b.user_id, b.batch_id
          ORDER BY b.vir_time DESC, b.credit_serial_id DESC
        ) AS batch_rn
      FROM batched_offer b
    ) x
    CROSS JOIN params p
    WHERE x.batch_rn = 1
      AND x.vir_date >= p.start_date
      AND x.vir_date < p.end_date
), candidate_users AS (
    SELECT DISTINCT user_id FROM dedup_offer
), all_orders AS (
    SELECT o.user_id, o.serial_id, o.apply_time, o.apply_date,
           o.remit_time, o.remit_amt, o.is_remit, o.apply_amt, o.product_no,
           o.repaid_time, o.repaid_date::date AS repaid_date,
           o.is_repaid, o.loan_status_code
    FROM wangchuanliang.order_loan_f_v2_copy o
    JOIN candidate_users u ON u.user_id = o.user_id
), clean_settlement AS (
    SELECT o.user_id, o.serial_id AS settled_serial_id,
           o.remit_amt::numeric AS settled_remit_amt,
           o.repaid_time, o.repaid_date
    FROM all_orders o
    WHERE o.is_remit = 1
      AND COALESCE(o.remit_amt, 0) > 0
      AND (o.is_repaid = 1 OR o.loan_status_code = 7)
      AND o.repaid_time > TIMESTAMP '2000-01-01 00:00:00'
      AND NOT EXISTS (
        SELECT 1
        FROM all_orders x
        WHERE x.user_id = o.user_id
          AND x.serial_id <> o.serial_id
          AND x.apply_time < o.repaid_time
          AND (
            x.loan_status_code IN (5, 8)
            OR (x.loan_status_code = 7 AND x.repaid_time > o.repaid_time)
          )
      )
), offer_settlement_pair AS (
    SELECT e.user_id, e.credit_serial_id, e.vir_date, e.vir_time,
           e.current_credit_amt, e.batch_id,
           s.settled_serial_id, s.settled_remit_amt, s.repaid_time,
      ROW_NUMBER() OVER (
        PARTITION BY e.user_id, e.credit_serial_id, e.vir_time
        ORDER BY s.repaid_time DESC, s.settled_serial_id DESC
      ) AS settlement_rn
    FROM dedup_offer e
    JOIN clean_settlement s
      ON s.user_id = e.user_id
     AND s.repaid_date = e.vir_date
     AND e.vir_time >= s.repaid_time
), strict_t0 AS (
    SELECT user_id, credit_serial_id, vir_date, vir_time,
           current_credit_amt, batch_id,
           settled_serial_id, settled_remit_amt, repaid_time
    FROM offer_settlement_pair
    WHERE settlement_rn = 1
), previous_credit_ranked AS (

           v.max_credit_apply_amt::numeric AS previous_credit_amt,

        PARTITION BY s.user_id, s.credit_serial_id, s.vir_time
        ORDER BY v.vir_unix DESC, v.serial_id DESC

    JOIN wangchuanliang.order_vir_f_copy v

def standardized(f,control='其他T0合计',strata=None,minimum=(10,20),name='产品条件可比'):
 strata=strata or ['month','loan_stage','credit_band','rate_band']
 keep=f.group.eq('提额≥3000') | (f.group.ne('提额≥3000') if control=='其他T0合计' else f.group.eq(control))
 x=f.loc[keep].copy();x['is_target']=x.group.eq('提额≥3000')
 x['cell']=x[strata].astype(str).agg('|'.join,axis=1)
 counts=x.groupby(['cell','is_target']).size().unstack(fill_value=0)
 common=counts.index[(counts.get(True,0)>=minimum[0])&(counts.get(False,0)>=minimum[1])]
 original_target=int(x.is_target.sum());x=x.loc[x.cell.isin(common)].copy()
 t=x.loc[x.is_target];c=x.loc[~x.is_target]
 meta={'analysis':name,'control':control,'strata':strata,'common_cells':len(common),
       'target_n':len(t),'control_n':len(c),'target_coverage':len(t)/original_target if original_target else 0,
       'control_coverage':len(c)/(keep.sum()-original_target) if keep.sum()>original_target else 0,
       'minimum_target_cell':minimum[0],'minimum_control_cell':minimum[1]}
 out=[]
 if not len(x):return out,meta
 target_counts=t.groupby('cell').size();weights=target_counts/len(t)
 for field in PRIMARY+['pre7_active_days','pre7_entry_days']:
  tm=t.groupby('cell')[field].mean();cm=c.groupby('cell')[field].mean()
  mt=float((weights*tm).sum());mc=float((weights*cm).sum());delta=mt-mc
  assert np.isclose(weights.sum(),1.) and np.isclose(mt,t[field].mean())
  if not field.startswith('pre7'):assert 0<=mt<=1 and 0<=mc<=1
  # Fixed common-support target weights; users may contribute both target and
  # control events. Aggregate both signs within user before calculating SE.
  influence=pd.Series(0.,index=x.index)
  for target,means,part,sign in [(True,tm,t,1),(False,cm,c,-1)]:
   ns=part.groupby('cell').size()
   influence.loc[part.index]=sign*part.cell.map(weights/ns)*(part[field]-part.cell.map(means))
  user_influence=influence.groupby(x.user_id).sum();n_users=len(user_influence)
  se=float(np.sqrt(np.square(user_influence).sum()*n_users/max(n_users-1,1)))
  out.append({'analysis':name,'control':control,'metric':field,'metric_label':MLABEL[field],
              'target':mt,'control_rate':mc,'delta':delta,'low95':delta-1.96*se,'high95':delta+1.96*se,
              'target_n':len(t),'control_n':len(c),'target_coverage':meta['target_coverage'],'common_cells':len(common)})
 return out,meta

def finish_report(document, source_files, source_directory):
    """Embed complete, HTML-escaped SQL/Python files with source SHA256.

    source_directory is the report-relative download folder. The caller copies
    these exact source files into that folder when packaging its experiment.
    No generated HTML or raw input data is embedded as source code.
    """
    files = sorted((Path(p) for p in source_files),
                   key=lambda p: (0 if p.suffix == '.sql' else 1, p.name))
    assert files and len({p.name for p in files}) == len(files)
    assert all(p.is_file() and p.suffix in {'.py', '.sql'} for p in files)
    sql_n = sum(p.suffix == '.sql' for p in files)
    py_n = len(files) - sql_n
    parts = ['<section id="code-appendix"><h2>附录：完整SQL与Python代码</h2>',
             f'<p>共 {sql_n} 个SQL文件、{py_n} 个Python文件，包含数据提取、事件与提单标记、行为构造、对比分析、核验及报告生成的实际源文件。以下逐文件完整收录，没有省略代码；可展开阅读或下载原文件。</p>',
             '<p class="muted">SQL保留脚本实际使用的参数占位符，参数由Python绑定。数据库连接沿用原提取脚本的本机配置或环境变量；附录不含连接密码。冻结输入、环境依赖和运行顺序见复现数据包。代码中的工作区路径按原分析保留。</p>',
             '<div class="source-controls"><label for="sourceSearch">代码文件</label><input id="sourceSearch" type="search" placeholder="搜索文件名，例如 train.py 或 daily_query.sql"><label for="sourceLanguage">语言</label><select id="sourceLanguage"><option value="">全部语言</option><option value="sql">SQL</option><option value="python">Python</option></select><button id="sourceExpand" type="button">展开筛选结果</button><button id="sourceCollapse" type="button">全部收起</button></div>',
             f'<p id="sourceCount" class="source-count" aria-live="polite">显示 {len(files)} / {len(files)} 个完整代码文件</p>']
    for i, path in enumerate(files, 1):
        raw = path.read_bytes()
        source = raw.decode('utf-8')
        digest = hashlib.sha256(raw).hexdigest()
        lang = 'sql' if path.suffix == '.sql' else 'python'
        label = 'SQL' if lang == 'sql' else 'Python'
        name = html.escape(path.name, quote=True)
        href = html.escape(quote(source_directory + '/' + path.name), quote=True)
        parts.append(
            f'<details class="source-file" id="source-file-{i:02d}" data-source="{name}" data-language="{lang}" data-sha256="{digest}">'
            f'<summary><span class="source-lang">{label}</span>{name}</summary>'
            f'<div class="source-meta"><span>{len(source.splitlines()):,} 行 · {len(raw):,} 字节</span><a href="{href}" download>下载完整源文件</a><span class="source-hash">SHA256：{digest}</span></div>'
            f'<pre><code class="language-{lang}">{html.escape(source, quote=False)}</code></pre></details>')
    parts.append('</section>')
    assert 'id="code-appendix"' not in document
    head_style_end = document.index('</style>')
    document = document[:head_style_end] + DARK_CSS + document[head_style_end:]
    nav_end = document.index('</nav>')
    document = document[:nav_end] + '<a href="#code-appendix">完整代码附录</a>' + document[nav_end:]
    main_end = document.rindex('</main>')
    return (document[:main_end] + ''.join(parts) + '<script>' + APPENDIX_JS +
            '</script>' + document[main_end:])
