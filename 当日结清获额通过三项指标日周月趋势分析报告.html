附录：PGSQL 查询代码 
/*
口径说明：
1. eligible_users：不限制干净结清；结清后当天获额通过。
2. clean_eligible_users：沿用此前 T0 分析的干净结清口径；结清后当天获额通过。
3. eligible_can_apply_users：不限制干净结清；获额订单 recycle_type 为 2 或 3。
*/

WITH all_orders AS (
    SELECT
        o.user_id,
        o.serial_id,
        o.apply_time,
        o.repaid_time,
        o.repaid_date,
        o.loan_status_code,
        o.is_remit
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    WHERE o.apply_time < TIMESTAMP '2026-08-16 00:00:00'
),

settled_orders AS (
    SELECT
        o.user_id,
        o.serial_id,
        o.repaid_time,
        o.repaid_date::date AS repaid_date
    FROM all_orders AS o
    WHERE o.is_remit = 1
      AND o.repaid_time IS NOT NULL
      AND o.repaid_date >= DATE '2026-01-01'
      AND o.repaid_date <= DATE '2026-08-15'
),

credit_pass AS (
    SELECT
        v.user_id,
        v.serial_id,
        v.vir_date::date AS vir_date,
        to_timestamp(v.vir_unix) AS vir_time
    FROM wangchuanliang.order_vir_f_copy AS v
    WHERE v.is_pass1 = 1
      AND v.vir_date >= DATE '2026-01-01'
      AND v.vir_date <= DATE '2026-08-15'
),

eligible_user_days AS (
    SELECT DISTINCT
        s.user_id,
        s.repaid_date
    FROM settled_orders AS s
    INNER JOIN credit_pass AS cp
        ON cp.user_id = s.user_id
       AND cp.vir_date = s.repaid_date
       AND cp.vir_time > s.repaid_time
),

eligible_daily AS (
    SELECT
        e.repaid_date,
        COUNT(DISTINCT e.user_id) AS eligible_users
    FROM eligible_user_days AS e
    GROUP BY
        e.repaid_date
),

clean_settled_orders AS (
    SELECT
        o.user_id,
        o.serial_id,
        o.repaid_time,
        o.repaid_date::date AS repaid_date
    FROM all_orders AS o
    WHERE o.is_remit = 1
      AND o.loan_status_code = 7
      AND o.repaid_time IS NOT NULL
      AND o.repaid_date >= DATE '2026-01-01'
      AND o.repaid_date <= DATE '2026-08-15'
),

last_settle_of_day AS (
    SELECT DISTINCT ON (
        s.user_id,
        s.repaid_date
    )
        s.user_id,
        s.serial_id,
        s.repaid_time,
        s.repaid_date
    FROM clean_settled_orders AS s
    ORDER BY
        s.user_id,
        s.repaid_date,
        s.repaid_time DESC
),

clean_last_settles AS (
    SELECT
        s.user_id,
        s.serial_id,
        s.repaid_time,
        s.repaid_date
    FROM last_settle_of_day AS s
    WHERE NOT EXISTS (
        SELECT
            1
        FROM all_orders AS o2
        WHERE o2.user_id = s.user_id
          AND o2.serial_id <> s.serial_id
          AND o2.apply_time < s.repaid_time
          AND (
                o2.loan_status_code = 5
             OR o2.loan_status_code = 8
             OR (
                    o2.loan_status_code = 7
                AND o2.repaid_time > s.repaid_time
                )
          )
    )
),

clean_t0_eligible AS (
    SELECT
        c.user_id,
        c.repaid_time,
        c.repaid_date
    FROM clean_last_settles AS c
    WHERE EXISTS (
        SELECT
            1
        FROM credit_pass AS cp
        WHERE cp.user_id = c.user_id
          AND cp.vir_date = c.repaid_date
          AND cp.vir_time > c.repaid_time
    )
),

clean_t0_by_user_date AS (
    SELECT
        t.user_id,
        t.repaid_date,
        MIN(a.apply_time) AS first_apply_time_t0
    FROM clean_t0_eligible AS t
    LEFT JOIN all_orders AS a
        ON a.user_id = t.user_id
       AND a.apply_time > t.repaid_time
       AND a.apply_time::date = t.repaid_date
    GROUP BY
        t.user_id,
        t.repaid_date
),

clean_eligible_daily AS (
    SELECT
        t.repaid_date,
        COUNT(DISTINCT t.user_id) AS clean_eligible_users
    FROM clean_t0_by_user_date AS t
    GROUP BY
        t.repaid_date
),

can_apply_credit_passes AS (
    SELECT
        cp.user_id,
        cp.vir_date,
        cp.vir_time
    FROM credit_pass AS cp
    INNER JOIN wangchuanliang.side_recycle_type_copy AS r
        ON r.serial_id = cp.serial_id
    WHERE r.recycle_type IN (
        2,
        3
    )
),

eligible_can_apply_user_days AS (
    SELECT DISTINCT
        s.user_id,
        s.repaid_date
    FROM settled_orders AS s
    INNER JOIN can_apply_credit_passes AS cp
        ON cp.user_id = s.user_id
       AND cp.vir_date = s.repaid_date
       AND cp.vir_time > s.repaid_time
),

eligible_can_apply_daily AS (
    SELECT
        e.repaid_date,
        COUNT(DISTINCT e.user_id) AS eligible_can_apply_users
    FROM eligible_can_apply_user_days AS e
    GROUP BY
        e.repaid_date
),

date_range AS (
    SELECT
        generate_series(
            DATE '2026-01-01',
            DATE '2026-08-15',
            INTERVAL '1 day'
        )::date AS repaid_date
)

SELECT
    d.repaid_date,
    COALESCE(e.eligible_users, 0) AS eligible_users,
    COALESCE(c.clean_eligible_users, 0) AS clean_eligible_users,
    COALESCE(a.eligible_can_apply_users, 0) AS eligible_can_apply_users
FROM date_range AS d
LEFT JOIN eligible_daily AS e
    ON e.repaid_date = d.repaid_date
LEFT JOIN clean_eligible_daily AS c
    ON c.repaid_date = d.repaid_date
LEFT JOIN eligible_can_apply_daily AS a
    ON a.repaid_date = d.repaid_date
ORDER BY
    d.repaid_date;
