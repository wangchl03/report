结论

1. 获额通过后，点确认的人在下降
1–8 月合计点确认 95,023 / 177,133 = 53.65%。1 月 60.46%，8 月 47.18%。后面窗口提单率只在点确认的人里算，不要拿 T0 全量当确认后转化分母。
2. 点确认后，绝大多数 3 分钟内就提单；再往后增量很小
确认后提单率：3 分钟 89.14%，30 分钟 93.36%，1 小时 94.22%，3 小时 95.25%，24 小时 96.89%。3 分钟到 24 小时只再抬 7.75pp。正式提单通过几乎都压在确认后极短窗口。
3. 3 分钟内未提单的人，后面还能补上一部分，但补不完
3 分钟未提单 10,315 人（占点确认 10.86%）。其中半小时内补上 4,010（38.88%），1 小时内 4,821（46.74%），3 小时内 5,804（56.27%），24 小时内 7,357（71.32%）。仍有 2,958 人 24 小时未提。
4. 确认后 3 分钟内绝大多数已经进活体
点确认后 3 分钟内有活体进页 92,227 人，占点确认 97.06%；24 小时内进过活体 99.31%。确认之后下一步就是活体，和产品链路一致。
5. 「点确认、没进活体、也没提单」存在，但不是主流
确认后 24 小时既未进活体、也未提单：600 人，占点确认 0.63%。进了活体仍 24 小时未提单 2,358 人；没进活体却提了单 53 人（埋点丢失或跳过活体页）。未提单主要仍是进了活体没走完，而不是确认后完全没进活体。

口径

T0 获额通过：2026-01-01～08-31；is_pass1=1、loan_type_code=2、recycle_type=3；20 秒批次去重；获额当天干净结清且 vir_time≥repaid_time；同一用户日 day_rn=1。
点确认：获额后 24 小时内第一次 Loan_newconfirm / Loan_newconfirm_select。
提单：确认后 24 小时内最早 apply_time；窗口含 3 分钟、30 分钟、1 小时、3 小时、24 小时（落在窗口内即算该窗口提单）。
3 分钟未提：apply 为空或 ≥ 确认+3分钟。后续转化仍要求提单落在确认后 24 小时内。
活体：确认后 24 小时内第一次 guide_live_ness_page_exposure 且 event_type=page_view_in。未进活体且未提单 = 24 小时内无活体进页且无提单。

完整 PGSQL（点击收起） 
-- =============================================================================
-- T0 获额通过 → 点确认 → 确认后窗口提单 / 3分钟未提后续转化 / 活体
-- 库：kaby_dw
-- 区间：把 __START_DATE__ / __END_DATE__ 换成各月起止（例 2026-01-01 / 2026-02-01），月人数相加得合计
-- 口径与既有 T0 origin + first_confirm 一致；提单时钟从 first_confirm 起算 24 小时
-- =============================================================================

-- 1) T0 origin（人日）
DROP TABLE IF EXISTS tmp_m_origin;
CREATE TEMP TABLE tmp_m_origin AS
WITH params AS (
    SELECT DATE '__START_DATE__' AS start_date, DATE '__END_DATE__' AS end_date
), raw_offer AS (
    SELECT DISTINCT ON (v.user_id, v.serial_id)
        v.user_id, v.serial_id, v.vir_date::date AS vir_date, TO_TIMESTAMP(v.vir_unix) AS vir_time
    FROM wangchuanliang.order_vir_f_copy v
    INNER JOIN wangchuanliang.side_recycle_type_copy r ON r.serial_id = v.serial_id
    CROSS JOIN params p
    WHERE v.is_pass1 = 1 AND v.loan_type_code = 2 AND r.recycle_type = 3
      AND v.vir_date >= p.start_date - 1 AND v.vir_date < p.end_date
    ORDER BY v.user_id, v.serial_id, v.vir_unix DESC
), lagged AS (
    SELECT r.*, LAG(r.vir_time) OVER (PARTITION BY r.user_id ORDER BY r.vir_time, r.serial_id) AS prev_vir_time
    FROM raw_offer r
), batched AS (
    SELECT l.*, SUM(CASE WHEN l.prev_vir_time IS NULL OR l.vir_time - l.prev_vir_time > INTERVAL '20 seconds' THEN 1 ELSE 0 END)
      OVER (PARTITION BY l.user_id ORDER BY l.vir_time, l.serial_id ROWS UNBOUNDED PRECEDING) AS batch_id
    FROM lagged l
), dedup_offer AS (
    SELECT user_id, serial_id, vir_date, vir_time
    FROM (
        SELECT b.*, ROW_NUMBER() OVER (PARTITION BY b.user_id, b.batch_id ORDER BY b.vir_time DESC, b.serial_id DESC) AS batch_rn
        FROM batched b
    ) x
    CROSS JOIN params p
    WHERE batch_rn = 1 AND vir_date >= p.start_date AND vir_date < p.end_date
), candidate_users AS (SELECT DISTINCT user_id FROM dedup_offer),
all_orders AS (
    SELECT o.user_id, o.serial_id, o.apply_time, o.repaid_time, o.repaid_date::date AS repaid_date,
           o.loan_status_code, o.is_remit, o.is_repaid
    FROM wangchuanliang.order_loan_f_v2_copy o
    INNER JOIN candidate_users u ON u.user_id = o.user_id
), clean_settlement AS (
    SELECT o.user_id, o.serial_id, o.repaid_time, o.repaid_date
    FROM all_orders o
    WHERE o.is_remit = 1 AND (o.is_repaid = 1 OR o.loan_status_code = 7)
      AND o.repaid_time > TIMESTAMP '2000-01-01 00:00:00'
      AND NOT EXISTS (
          SELECT 1 FROM all_orders x
          WHERE x.user_id = o.user_id AND x.serial_id <> o.serial_id AND x.apply_time < o.repaid_time
            AND (x.loan_status_code IN (5, 8) OR (x.loan_status_code = 7 AND x.repaid_time > o.repaid_time))
      )
), paired AS (
    SELECT e.user_id, e.serial_id AS credit_serial_id, e.vir_date, e.vir_time,
           ROW_NUMBER() OVER (PARTITION BY e.user_id, e.vir_date ORDER BY e.vir_time ASC, e.serial_id DESC) AS day_rn
    FROM dedup_offer e
    INNER JOIN clean_settlement s
        ON s.user_id = e.user_id AND s.repaid_date = e.vir_date AND e.vir_time >= s.repaid_time
)
SELECT user_id, credit_serial_id, vir_date, vir_time FROM paired WHERE day_rn = 1;

-- 2) 获额后 24h 内第一次点确认
DROP TABLE IF EXISTS tmp_confirm;
CREATE TEMP TABLE tmp_confirm AS
SELECT t.user_id, t.vir_date, t.vir_time, MIN(e.create_time) AS first_confirm
FROM tmp_m_origin t
INNER JOIN wangchuanliang.event_track_record e
  ON e.user_id = t.user_id
 AND e.create_date >= (SELECT MIN(vir_date) FROM tmp_m_origin)
 AND e.create_date <= (SELECT MAX(vir_date) FROM tmp_m_origin) + 1
 AND e.create_date >= t.vir_date AND e.create_date <= t.vir_date + 1
 AND e.create_time >= t.vir_time
 AND e.create_time < t.vir_time + INTERVAL '24 hours'
 AND e.props_id IN ('Loan_newconfirm', 'Loan_newconfirm_select')
GROUP BY t.user_id, t.vir_date, t.vir_time;

-- 3) 确认后 24h 内最早提单
DROP TABLE IF EXISTS tmp_apply;
CREATE TEMP TABLE tmp_apply AS
SELECT c.user_id, c.vir_date, c.first_confirm, MIN(o.apply_time) AS apply_after
FROM tmp_confirm c
LEFT JOIN wangchuanliang.order_loan_f_v2_copy o
  ON o.user_id = c.user_id
 AND o.apply_time >= c.first_confirm
 AND o.apply_time < c.first_confirm + INTERVAL '24 hours'
GROUP BY c.user_id, c.vir_date, c.first_confirm;

-- 4) 确认后 24h 内最早进活体页（page_view_in）
DROP TABLE IF EXISTS tmp_live;
CREATE TEMP TABLE tmp_live AS
SELECT c.user_id, c.vir_date, MIN(e.create_time) AS first_live
FROM tmp_confirm c
INNER JOIN wangchuanliang.event_track_record e
  ON e.user_id = c.user_id
 AND e.create_date >= (SELECT MIN(vir_date) FROM tmp_m_origin)
 AND e.create_date <= (SELECT MAX(vir_date) FROM tmp_m_origin) + 1
 AND e.create_date >= c.vir_date AND e.create_date <= c.vir_date + 1
 AND e.create_time >= c.first_confirm
 AND e.create_time < c.first_confirm + INTERVAL '24 hours'
 AND e.event_type = 'page_view_in'
 AND e.props_id = 'guide_live_ness_page_exposure'
GROUP BY c.user_id, c.vir_date;

-- 5) 人日明细 + 月汇总（报告全部指标）
DROP TABLE IF EXISTS tmp_grain;
CREATE TEMP TABLE tmp_grain AS
SELECT
    a.user_id,
    a.vir_date,
    a.first_confirm,
    a.apply_after,
    l.first_live,
    CASE WHEN a.apply_after IS NOT NULL AND a.apply_after < a.first_confirm + INTERVAL '3 minutes' THEN 1 ELSE 0 END AS f_apply_3m,
    CASE WHEN a.apply_after IS NOT NULL AND a.apply_after < a.first_confirm + INTERVAL '30 minutes' THEN 1 ELSE 0 END AS f_apply_30m,
    CASE WHEN a.apply_after IS NOT NULL AND a.apply_after < a.first_confirm + INTERVAL '1 hour' THEN 1 ELSE 0 END AS f_apply_1h,
    CASE WHEN a.apply_after IS NOT NULL AND a.apply_after < a.first_confirm + INTERVAL '3 hours' THEN 1 ELSE 0 END AS f_apply_3h,
    CASE WHEN a.apply_after IS NOT NULL THEN 1 ELSE 0 END AS f_apply_24h,
    CASE WHEN a.apply_after IS NULL OR a.apply_after >= a.first_confirm + INTERVAL '3 minutes' THEN 1 ELSE 0 END AS f_no3m,
    CASE WHEN l.first_live IS NOT NULL AND l.first_live < a.first_confirm + INTERVAL '3 minutes' THEN 1 ELSE 0 END AS f_live_3m,
    CASE WHEN l.first_live IS NOT NULL THEN 1 ELSE 0 END AS f_live_24h
FROM tmp_apply a
LEFT JOIN tmp_live l ON l.user_id = a.user_id AND l.vir_date = a.vir_date;

-- R1 当月漏斗
SELECT
    (SELECT COUNT(*) FROM tmp_m_origin) AS n_t0,
    COUNT(*) AS n_confirm,
    ROUND(100.0 * COUNT(*) / NULLIF((SELECT COUNT(*) FROM tmp_m_origin), 0), 2) AS pct_confirm,
    SUM(f_apply_3m) AS n_apply_3m,
    ROUND(100.0 * SUM(f_apply_3m) / COUNT(*), 2) AS pct_apply_3m,
    SUM(f_apply_30m) AS n_apply_30m,
    ROUND(100.0 * SUM(f_apply_30m) / COUNT(*), 2) AS pct_apply_30m,
    SUM(f_apply_1h) AS n_apply_1h,
    ROUND(100.0 * SUM(f_apply_1h) / COUNT(*), 2) AS pct_apply_1h,
    SUM(f_apply_3h) AS n_apply_3h,
    ROUND(100.0 * SUM(f_apply_3h) / COUNT(*), 2) AS pct_apply_3h,
    SUM(f_apply_24h) AS n_apply_24h,
    ROUND(100.0 * SUM(f_apply_24h) / COUNT(*), 2) AS pct_apply_24h,
    SUM(f_no3m) AS n_no3m,
    SUM(CASE WHEN f_no3m = 1 AND f_apply_30m = 1 THEN 1 ELSE 0 END) AS no3m_apply_30m,
    SUM(CASE WHEN f_no3m = 1 AND f_apply_1h = 1 THEN 1 ELSE 0 END) AS no3m_apply_1h,
    SUM(CASE WHEN f_no3m = 1 AND f_apply_3h = 1 THEN 1 ELSE 0 END) AS no3m_apply_3h,
    SUM(CASE WHEN f_no3m = 1 AND f_apply_24h = 1 THEN 1 ELSE 0 END) AS no3m_apply_24h,
    SUM(f_live_3m) AS n_live_3m,
    ROUND(100.0 * SUM(f_live_3m) / COUNT(*), 2) AS pct_live_3m,
    SUM(f_live_24h) AS n_live_24h,
    ROUND(100.0 * SUM(f_live_24h) / COUNT(*), 2) AS pct_live_24h,
    SUM(CASE WHEN f_live_24h = 0 AND f_apply_24h = 0 THEN 1 ELSE 0 END) AS n_no_live_no_apply,
    ROUND(100.0 * SUM(CASE WHEN f_live_24h = 0 AND f_apply_24h = 0 THEN 1 ELSE 0 END) / COUNT(*), 2) AS pct_no_live_no_apply,
    SUM(CASE WHEN f_live_24h = 1 AND f_apply_24h = 0 THEN 1 ELSE 0 END) AS n_live_no_apply,
    SUM(CASE WHEN f_live_24h = 0 AND f_apply_24h = 1 THEN 1 ELSE 0 END) AS n_apply_no_live
FROM tmp_grain;
