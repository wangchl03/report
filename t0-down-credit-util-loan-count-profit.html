           s.settled_remit_amt::numeric / NULLIF(p.previous_credit_amt, 0) AS previous_util_ratio,
           l.successful_loan_n
    FROM strict_t0 s
    JOIN previous_credit p
      ON p.user_id = s.user_id
     AND p.credit_serial_id = s.credit_serial_id
     AND p.vir_time = s.vir_time
     AND p.settled_serial_id = s.settled_serial_id
    JOIN loan_count_at_settlement l
      ON l.user_id = s.user_id
     AND l.credit_serial_id = s.credit_serial_id
     AND l.vir_time = s.vir_time
    WHERE s.current_credit_amt > 0
      AND p.previous_credit_amt > 0
      AND s.settled_remit_amt > 0
      AND s.current_credit_amt < p.previous_credit_amt
), eligible_snapshot AS (
    SELECT d.*
    FROM down_credit_snapshot d
    WHERE NOT EXISTS (
      SELECT 1
      FROM all_orders a
      WHERE a.user_id = d.user_id
        AND a.apply_time >= d.vir_time
        AND a.apply_date = d.vir_date
    )
), candidate_future_orders AS (
    SELECT e.user_id, e.credit_serial_id, e.vir_date, e.vir_time,
           o.serial_id, o.apply_time, o.apply_date,
           o.is_remit, o.remit_time, o.remit_amt, o.due_date,
      ROW_NUMBER() OVER (
        PARTITION BY o.serial_id
        ORDER BY e.vir_time DESC, e.credit_serial_id DESC
      ) AS event_match_rn
    FROM eligible_snapshot e
    JOIN all_orders o
      ON o.user_id = e.user_id
     AND o.apply_date BETWEEN e.vir_date + 1 AND e.vir_date + 5
     AND o.apply_time > e.vir_time
), assigned_future_orders AS (
    SELECT * FROM candidate_future_orders WHERE event_match_rn = 1
), first_d1_d5_apply AS (
    SELECT user_id, credit_serial_id, vir_time,
           serial_id AS next_apply_serial_id,
           apply_time AS next_apply_time
    FROM (
      SELECT o.*,
        ROW_NUMBER() OVER (
          PARTITION BY o.user_id, o.credit_serial_id, o.vir_time
          ORDER BY o.apply_time ASC NULLS LAST, o.serial_id ASC
        ) AS rn
      FROM assigned_future_orders o
    ) x
    WHERE rn = 1
), first_d1_d5_remit AS (
    SELECT user_id, credit_serial_id, vir_time,
           serial_id AS next_remit_serial_id,
           apply_time AS next_remit_apply_time,
           remit_time AS next_remit_time,
           remit_amt::numeric AS next_remit_amt,
           due_date AS next_due_date
    FROM (
      SELECT o.*,
        ROW_NUMBER() OVER (
          PARTITION BY o.user_id, o.credit_serial_id, o.vir_time
          ORDER BY o.apply_time ASC NULLS LAST, o.serial_id ASC
        ) AS rn
      FROM assigned_future_orders o
      WHERE o.is_remit = 1
        AND COALESCE(o.remit_amt, 0) > 0
    ) x
    WHERE rn = 1
), repayment_by_due AS (
    SELECT n.next_remit_serial_id,
           SUM(CASE WHEN r.payin_date <= n.next_due_date THEN r.amount ELSE 0 END) AS repaid_by_due_amt
    FROM first_d1_d5_remit n
    LEFT JOIN wangchuanliang.repayment_record_copy r
      ON r.serial_id = n.next_remit_serial_id
    WHERE n.next_due_date < CURRENT_DATE - 1
    GROUP BY n.next_remit_serial_id
)
SELECT
    DATE_TRUNC('month', e.vir_date)::date AS t0_vir_month,

      WHEN e.previous_util_ratio < 0.999 THEN '80%-100%'
      WHEN e.previous_util_ratio <= 1.001 THEN '接近100%'
      ELSE '>100%'
    END AS previous_util_band,
    e.current_credit_amt,
    e.credit_change_amt,
    e.credit_change_rate,
    e.successful_loan_n,
    CASE
      WHEN e.successful_loan_n = 1 THEN '1次'
      WHEN e.successful_loan_n = 2 THEN '2次'
      WHEN e.successful_loan_n = 3 THEN '3次'
      WHEN e.successful_loan_n = 4 THEN '4次'
      WHEN e.successful_loan_n = 5 THEN '5次'
      ELSE '6次及以上'
    END AS loan_count_bucket,
    CASE WHEN a.next_apply_serial_id IS NULL THEN 0 ELSE 1 END AS has_d1_d5_apply,
    a.next_apply_serial_id,
    a.next_apply_time,
    CASE WHEN n.next_remit_serial_id IS NULL THEN 0 ELSE 1 END AS has_d1_d5_remit,
    n.next_remit_serial_id,
    n.next_remit_apply_time,
    n.next_remit_time,
    n.next_remit_amt,
    n.next_due_date,
    n.next_remit_amt / NULLIF(e.current_credit_amt, 0) AS next_util_ratio,
    CASE WHEN p.next_remit_serial_id IS NULL THEN 0 ELSE 1 END AS is_matured,
    CASE WHEN p.next_remit_serial_id IS NULL THEN 0 ELSE n.next_remit_amt END AS matured_remit_amt,

 AND a.vir_time = e.vir_time
LEFT JOIN first_d1_d5_remit n
  ON n.user_id = e.user_id
 AND n.credit_serial_id = e.credit_serial_id
 AND n.vir_time = e.vir_time
LEFT JOIN repayment_by_due p
  ON p.next_remit_serial_id = n.next_remit_serial_id;
