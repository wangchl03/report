HISTORICAL COHORT · ELIGIBILITY-AWARE ANALYSIS

长期不活跃回流用户与比较活跃用户
提单及盈利质量对比分析报告

样本：首笔成功放款且结清的历史用户；注册日期 2025-01-01 至 2026-04-01。D0 为重新活跃日，D0–D30 为完整观察窗口。提单主分母仅包含 is_pass1=1 且 recycle_type∈(2,3) 的可提单用户。

一、核心结论

结论 1｜长期沉默回流用户并非“更容易总转化”，但其提单明显更集中在 D0。
长期组 D0 提单率 34.84%，比比较活跃组高 +5.54pp；但 D0–D30 累计提单率为 44.88%，较比较活跃组低 1.18pp。这更像“需求被重新激活后立即决策”，而不是总意愿更强。
结论 2｜当前样本不支持“长期沉默回流后，放款质量更差”。
长期组放款订单逾期率 31.09%，低于比较活跃组 33.19%（-2.10pp）；盈利率 14.48%，高 +7.15pp。这是资格筛选后的订单质量比较，不能外推到所有长期沉默用户。
结论 3｜长期沉默组的最大约束是“资格”，而非仅是触达。
长期组重新活跃用户中仅 60.15% 在 D0 可提单，低于比较活跃组的 77.55%（-17.40pp）。系统不可提单用户占长期组 39.85%，不应被混入主观不提单率。

分析口径的关键控制

  • 首笔放款先全量排序，再判断是否结清，避免把“首笔未结清、后续结清”的用户错误纳入。

  • D0 同日提单计为 D0；D0–D7 为累计 8 个自然日，D8–D30 为后续新增。

八、运营含义与边界

对长期沉默回流用户：先判断是否可提单。对已满足获额通过且 recycle_type 为 2/3 的用户，D0 是最高价值的响应窗口，可优先配置即时、清晰的回流引导；对不具资格者，不应按“不愿转化”处理，而应单列为资格/风控/产品路径问题。
对比较活跃用户：D0–D7 累计提单率略高，持续触达和正常路径承接更重要。不能因其绝对提单人数大，就省略与长期组相同的资格拆分。
使用边界：本分析为历史观察性比较，不是随机实验。用户分组、资格与后续质量均可能受共同的风险和需求因素影响；若要确认运营动作的增量效果，应在“长期组 × 可提单资格”内随机分配触达策略。

/* 查询一：两类用户的提单、放款、风险与盈利汇总 */

WITH all_orders AS (
    /* 订单基础：只读取已注册于目标区间内的订单用户。 */
    SELECT
        o.user_id,
        o.serial_id,
        o.reg_date,
        o.apply_date,
        o.apply_time,
        o.remit_time,
        o.repaid_date,
        o.repaid_time,
        o.is_remit,
        o.is_repaid,
        o.loan_status_code,
        o.remit_amt,
        o.repaid_amt
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    WHERE o.reg_date >= DATE '2025-01-01'
      AND o.reg_date <= DATE '2026-04-01'
),

first_remit_ranked AS (
    /* 先在全部成功放款订单中取用户首笔，不能先过滤结清订单。 */
    SELECT
        o.user_id,
        o.serial_id,
        o.reg_date,
        o.repaid_date,
        o.repaid_time,
        o.is_repaid,
        o.loan_status_code,
        ROW_NUMBER() OVER (
            PARTITION BY o.user_id
            ORDER BY
                o.remit_time ASC,
                o.serial_id ASC
        ) AS first_remit_rn
    FROM all_orders AS o
    WHERE o.is_remit = 1
      AND o.remit_time IS NOT NULL
),

first_settled_loan AS (
    /* 首笔成功放款已结清；is_repaid 优先，状态7为兼容判定；
       repaid_date 仅保障能确定结清日这个时间锚点。 */
    SELECT
        f.user_id,
        f.serial_id AS first_loan_serial_id,
        f.reg_date,
        f.repaid_date::date AS first_repaid_date
    FROM first_remit_ranked AS f
    WHERE f.first_remit_rn = 1
      AND (
            f.is_repaid = 1
         OR f.loan_status_code = 7
      )
      AND f.repaid_date > DATE '2000-01-01'
),

active_days AS (
    /* event_track 存在 active_date 即代表当日进入 App；不使用 active_status。 */
    SELECT DISTINCT
        e.user_id,
        e.active_date
    FROM wangchuanliang.event_track AS e
    INNER JOIN first_settled_loan AS f
        ON f.user_id = e.user_id
    WHERE e.active_date > f.first_repaid_date
      AND e.active_date <= CURRENT_DATE - 30
),

activity_gaps AS (
    /* 对每个用户的活跃日计算上次活跃日。 */
    SELECT
        a.user_id,
        a.active_date,
        LAG(a.active_date) OVER (
            PARTITION BY a.user_id
            ORDER BY
                a.active_date
        ) AS previous_active_date
    FROM active_days AS a
),

activity_events AS (
    /* 第一次活跃的前序基准为首笔结清日；沉默天数不包含两端日期。 */
    SELECT
        f.user_id,
        g.active_date AS d0_date,
        COALESCE(
            g.previous_active_date,
            f.first_repaid_date
        ) AS previous_active_date,
        g.active_date
            - COALESCE(
                g.previous_active_date,
                f.first_repaid_date
            )
            - 1 AS silence_days
    FROM first_settled_loan AS f
    INNER JOIN activity_gaps AS g
        ON g.user_id = f.user_id
),

long_candidates AS (
    SELECT
        e.user_id,
        e.d0_date,
        e.previous_active_date,
        e.silence_days,
        ROW_NUMBER() OVER (
            PARTITION BY e.user_id
            ORDER BY
                e.d0_date ASC
        ) AS rn
    FROM activity_events AS e
    WHERE e.silence_days >= 21
),

long_anchor AS (
    SELECT
        l.user_id,
        l.d0_date,
        l.previous_active_date,
        l.silence_days,
        '长期不活跃回流用户'::text AS user_group
    FROM long_candidates AS l
    WHERE l.rn = 1
),

short_candidates AS (
    SELECT
        e.user_id,
        e.d0_date,
        e.previous_active_date,
        e.silence_days,
        ROW_NUMBER() OVER (
            PARTITION BY e.user_id
            ORDER BY
                e.d0_date ASC
        ) AS rn
    FROM activity_events AS e
    WHERE e.silence_days BETWEEN 0 AND 7
),

short_anchor AS (
    /* 长沉默用户优先归入长期组，保证主比较组互斥。 */
    SELECT
        s.user_id,
        s.d0_date,
        s.previous_active_date,
        s.silence_days,
        '比较活跃用户'::text AS user_group
    FROM short_candidates AS s
    WHERE s.rn = 1
      AND NOT EXISTS (
            SELECT
                1
            FROM long_anchor AS l
            WHERE l.user_id = s.user_id
      )
),

anchors AS (
    SELECT
        l.user_id,
        l.d0_date,
        l.previous_active_date,
        l.silence_days,
        l.user_group
    FROM long_anchor AS l

    UNION ALL

    SELECT
        s.user_id,
        s.d0_date,
        s.previous_active_date,
        s.silence_days,
        s.user_group
    FROM short_anchor AS s
),

credit_batch_ranked AS (
    /* 同一秒多笔获额只保留 serial_id 最大的订单。 */
    SELECT
        v.user_id,
        v.serial_id,
        v.vir_unix,
        v.vir_date,
        TO_TIMESTAMP(v.vir_unix) AS vir_time,
        v.is_pass1,
        r.recycle_type,
        ROW_NUMBER() OVER (
            PARTITION BY
                v.user_id,
                v.vir_unix
            ORDER BY
                v.serial_id DESC
        ) AS batch_rn
    FROM wangchuanliang.order_vir_f_copy AS v
    LEFT JOIN wangchuanliang.side_recycle_type_copy AS r
        ON r.serial_id = v.serial_id
    WHERE v.vir_unix IS NOT NULL
),

credit_batches AS (
    SELECT
        c.user_id,
        c.serial_id,
        c.vir_unix,
        c.vir_date,
        c.vir_time,
        c.is_pass1,
        c.recycle_type
    FROM credit_batch_ranked AS c
    WHERE c.batch_rn = 1
),

anchor_credit_ranked AS (
    /* active_date 只有日粒度，故取 D0 当日及之前最近获额批次。 */
    SELECT
        a.user_id,
        a.d0_date,
        a.previous_active_date,
        a.silence_days,
        a.user_group,
        b.serial_id AS credit_serial_id,
        b.vir_time AS credit_time,
        b.is_pass1,
        b.recycle_type,
        ROW_NUMBER() OVER (
            PARTITION BY
                a.user_id,
                a.d0_date
            ORDER BY
                b.vir_unix DESC NULLS LAST,
                b.serial_id DESC NULLS LAST
        ) AS credit_rn
    FROM anchors AS a
    LEFT JOIN credit_batches AS b
        ON b.user_id = a.user_id
       AND b.vir_date <= a.d0_date
),

anchor_credit AS (
    SELECT
        a.user_id,
        a.d0_date,
        a.previous_active_date,
        a.silence_days,
        a.user_group,
        a.credit_serial_id,
        a.credit_time,
        a.is_pass1,
        a.recycle_type
    FROM anchor_credit_ranked AS a
    WHERE a.credit_rn = 1
),

eligible_anchor AS (
    /* 同时满足获额通过和类型可提单，才纳入主提单率分母。 */
    SELECT
        a.user_id,
        a.d0_date,
        a.previous_active_date,
        a.silence_days,
        a.user_group,
        a.credit_serial_id,
        a.credit_time,
        a.is_pass1,
        a.recycle_type,
        CASE
            WHEN a.is_pass1 = 1
             AND a.recycle_type IN (2, 3)
                THEN 1
            ELSE 0
        END AS can_apply
    FROM anchor_credit AS a
),

matched_order AS (
    /* 只将与 D0 最近获额订单 serial_id 完全一致的提单/放款结果归因。 */
    SELECT
        e.user_id,
        e.d0_date,
        e.user_group,
        e.silence_days,
        e.credit_serial_id,
        e.credit_time,
        e.is_pass1,
        e.recycle_type,
        e.can_apply,
        o.apply_date,
        o.apply_time,
        o.is_remit,
        o.loan_status_code,
        o.remit_amt,
        o.repaid_amt
    FROM eligible_anchor AS e
    LEFT JOIN all_orders AS o
        ON o.user_id = e.user_id
       AND o.serial_id = e.credit_serial_id
       AND o.apply_date >= e.d0_date
       AND o.apply_date <= e.d0_date + 30
),

d0_behavior AS (
    /* D0 当天行为强度，仅作解释变量。 */
    SELECT
        a.user_id,
        SUM(COALESCE(e.newpage, 0)) AS newpage_cnt,
        SUM(COALESCE(e.amount_selection, 0)) AS amount_selection_cnt,
        SUM(COALESCE(e.term_selection, 0)) AS term_selection_cnt,
        SUM(COALESCE(e.contract_newdetails, 0)) AS contract_view_cnt,
        SUM(COALESCE(e.plan_newdetails, 0)) AS plan_view_cnt,
        SUM(COALESCE(e.newconfirm, 0)) AS confirm_cnt,
        SUM(COALESCE(e.active_duration, 0)) AS active_duration,
        SUM(COALESCE(e.apply_duration, 0)) AS apply_duration,
        SUM(COALESCE(e.non_apply_duration, 0)) AS non_apply_duration
    FROM anchors AS a
    LEFT JOIN wangchuanliang.event_track AS e
        ON e.user_id = a.user_id
       AND e.active_date = a.d0_date
    GROUP BY
        a.user_id
)



SELECT
    m.user_group,
    COUNT(DISTINCT m.user_id) AS reactivated_users,
    COUNT(DISTINCT CASE
        WHEN m.can_apply = 1 THEN m.user_id
    END) AS can_apply_users,
    COUNT(DISTINCT CASE
        WHEN m.can_apply = 0 THEN m.user_id
    END) AS cannot_apply_users,
    AVG(m.silence_days)::numeric AS avg_silence_days,
    COUNT(DISTINCT CASE
        WHEN m.can_apply = 1
         AND m.apply_date = m.d0_date THEN m.user_id
    END) AS d0_apply_users,
    COUNT(DISTINCT CASE
        WHEN m.can_apply = 1
         AND m.apply_date = m.d0_date + 1 THEN m.user_id
    END) AS d1_apply_users,
    COUNT(DISTINCT CASE
        WHEN m.can_apply = 1
         AND m.apply_date = m.d0_date + 2 THEN m.user_id
    END) AS d2_apply_users,
    COUNT(DISTINCT CASE
        WHEN m.can_apply = 1
         AND m.apply_date = m.d0_date + 3 THEN m.user_id
    END) AS d3_apply_users,
    COUNT(DISTINCT CASE
        WHEN m.can_apply = 1
         AND m.apply_date <= m.d0_date + 7 THEN m.user_id
    END) AS d7_apply_users,
    COUNT(DISTINCT CASE
        WHEN m.can_apply = 1
         AND m.apply_date BETWEEN m.d0_date + 8 AND m.d0_date + 30 THEN m.user_id
    END) AS d8_30_apply_users,
    COUNT(DISTINCT CASE
        WHEN m.can_apply = 1
         AND m.apply_date <= m.d0_date + 30 THEN m.user_id
    END) AS d30_apply_users,
    COUNT(DISTINCT CASE
        WHEN m.can_apply = 1
         AND m.apply_date IS NOT NULL
         AND m.is_remit = 1 THEN m.user_id
    END) AS remit_users,
    COUNT(DISTINCT CASE
        WHEN m.can_apply = 1
         AND m.apply_date IS NOT NULL
         AND m.is_remit = 1 THEN m.credit_serial_id
    END) AS remit_orders,
    COUNT(DISTINCT CASE
        WHEN m.can_apply = 1
         AND m.apply_date IS NOT NULL
         AND m.is_remit = 1
         AND m.loan_status_code = 8 THEN m.credit_serial_id
    END) AS overdue_orders,
    SUM(CASE
        WHEN m.can_apply = 1
         AND m.apply_date IS NOT NULL
         AND m.is_remit = 1
         AND m.loan_status_code IN (7, 8) THEN m.repaid_amt
        ELSE 0
    END) AS profit_repaid_amt,
    SUM(CASE
        WHEN m.can_apply = 1
         AND m.apply_date IS NOT NULL
         AND m.is_remit = 1
         AND m.loan_status_code IN (7, 8) THEN m.remit_amt
        ELSE 0
    END) AS profit_remit_amt,
    SUM(CASE
        WHEN m.can_apply = 1
         AND m.apply_date IS NOT NULL
         AND m.is_remit = 1
         AND m.loan_status_code = 8 THEN m.repaid_amt
        ELSE 0
    END) AS overdue_repaid_amt,
    SUM(CASE
        WHEN m.can_apply = 1
         AND m.apply_date IS NOT NULL
         AND m.is_remit = 1
         AND m.loan_status_code = 8 THEN m.remit_amt
        ELSE 0
    END) AS overdue_remit_amt,
    AVG(CASE
        WHEN m.can_apply = 1 THEN COALESCE(b.newpage_cnt, 0)
    END)::numeric AS avg_newpage_cnt,
    AVG(CASE
        WHEN m.can_apply = 1 THEN COALESCE(b.amount_selection_cnt, 0)
    END)::numeric AS avg_amount_selection_cnt,
    AVG(CASE
        WHEN m.can_apply = 1 THEN COALESCE(b.term_selection_cnt, 0)
    END)::numeric AS avg_term_selection_cnt,
    AVG(CASE
        WHEN m.can_apply = 1 THEN COALESCE(b.contract_view_cnt, 0)
    END)::numeric AS avg_contract_view_cnt,
    AVG(CASE
        WHEN m.can_apply = 1 THEN COALESCE(b.plan_view_cnt, 0)
    END)::numeric AS avg_plan_view_cnt,
    AVG(CASE
        WHEN m.can_apply = 1 THEN COALESCE(b.confirm_cnt, 0)
    END)::numeric AS avg_confirm_cnt,
    AVG(CASE
        WHEN m.can_apply = 1 THEN COALESCE(b.active_duration, 0)
    END)::numeric AS avg_active_duration,
    AVG(CASE
        WHEN m.can_apply = 1 THEN COALESCE(b.apply_duration, 0)
    END)::numeric AS avg_apply_duration,
    AVG(CASE
        WHEN m.can_apply = 1 THEN COALESCE(b.non_apply_duration, 0)
    END)::numeric AS avg_non_apply_duration
FROM matched_order AS m
LEFT JOIN d0_behavior AS b
    ON b.user_id = m.user_id
GROUP BY
    m.user_group
ORDER BY
    CASE m.user_group
        WHEN '长期不活跃回流用户' THEN 1
        WHEN '比较活跃用户' THEN 2
    END;


/* 查询二：可提单用户中，提单与未提单用户的 D0 行为差异 */

WITH all_orders AS (
    /* 订单基础：只读取已注册于目标区间内的订单用户。 */
    SELECT
        o.user_id,
        o.serial_id,
        o.reg_date,
        o.apply_date,
        o.apply_time,
        o.remit_time,
        o.repaid_date,
        o.repaid_time,
        o.is_remit,
        o.is_repaid,
        o.loan_status_code,
        o.remit_amt,
        o.repaid_amt
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    WHERE o.reg_date >= DATE '2025-01-01'
      AND o.reg_date <= DATE '2026-04-01'
),

first_remit_ranked AS (
    /* 先在全部成功放款订单中取用户首笔，不能先过滤结清订单。 */
    SELECT
        o.user_id,
        o.serial_id,
        o.reg_date,
        o.repaid_date,
        o.repaid_time,
        o.is_repaid,
        o.loan_status_code,
        ROW_NUMBER() OVER (
            PARTITION BY o.user_id
            ORDER BY
                o.remit_time ASC,
                o.serial_id ASC
        ) AS first_remit_rn
    FROM all_orders AS o
    WHERE o.is_remit = 1
      AND o.remit_time IS NOT NULL
),

first_settled_loan AS (
    /* 首笔成功放款已结清；is_repaid 优先，状态7为兼容判定；
       repaid_date 仅保障能确定结清日这个时间锚点。 */
    SELECT
        f.user_id,
        f.serial_id AS first_loan_serial_id,
        f.reg_date,
        f.repaid_date::date AS first_repaid_date
    FROM first_remit_ranked AS f
    WHERE f.first_remit_rn = 1
      AND (
            f.is_repaid = 1
         OR f.loan_status_code = 7
      )
      AND f.repaid_date > DATE '2000-01-01'
),

active_days AS (
    /* event_track 存在 active_date 即代表当日进入 App；不使用 active_status。 */
    SELECT DISTINCT
        e.user_id,
        e.active_date
    FROM wangchuanliang.event_track AS e
    INNER JOIN first_settled_loan AS f
        ON f.user_id = e.user_id
    WHERE e.active_date > f.first_repaid_date
      AND e.active_date <= CURRENT_DATE - 30
),

activity_gaps AS (
    /* 对每个用户的活跃日计算上次活跃日。 */
    SELECT
        a.user_id,
        a.active_date,
        LAG(a.active_date) OVER (
            PARTITION BY a.user_id
            ORDER BY
                a.active_date
        ) AS previous_active_date
    FROM active_days AS a
),

activity_events AS (
    /* 第一次活跃的前序基准为首笔结清日；沉默天数不包含两端日期。 */
    SELECT
        f.user_id,
        g.active_date AS d0_date,
        COALESCE(
            g.previous_active_date,
            f.first_repaid_date
        ) AS previous_active_date,
        g.active_date
            - COALESCE(
                g.previous_active_date,
                f.first_repaid_date
            )
            - 1 AS silence_days
    FROM first_settled_loan AS f
    INNER JOIN activity_gaps AS g
        ON g.user_id = f.user_id
),

long_candidates AS (
    SELECT
        e.user_id,
        e.d0_date,
        e.previous_active_date,
        e.silence_days,
        ROW_NUMBER() OVER (
            PARTITION BY e.user_id
            ORDER BY
                e.d0_date ASC
        ) AS rn
    FROM activity_events AS e
    WHERE e.silence_days >= 21
),

long_anchor AS (
    SELECT
        l.user_id,
        l.d0_date,
        l.previous_active_date,
        l.silence_days,
        '长期不活跃回流用户'::text AS user_group
    FROM long_candidates AS l
    WHERE l.rn = 1
),

short_candidates AS (
    SELECT
        e.user_id,
        e.d0_date,
        e.previous_active_date,
        e.silence_days,
        ROW_NUMBER() OVER (
            PARTITION BY e.user_id
            ORDER BY
                e.d0_date ASC
        ) AS rn
    FROM activity_events AS e
    WHERE e.silence_days BETWEEN 0 AND 7
),

short_anchor AS (
    /* 长沉默用户优先归入长期组，保证主比较组互斥。 */
    SELECT
        s.user_id,
        s.d0_date,
        s.previous_active_date,
        s.silence_days,
        '比较活跃用户'::text AS user_group
    FROM short_candidates AS s
    WHERE s.rn = 1
      AND NOT EXISTS (
            SELECT
                1
            FROM long_anchor AS l
            WHERE l.user_id = s.user_id
      )
),

anchors AS (
    SELECT
        l.user_id,
        l.d0_date,
        l.previous_active_date,
        l.silence_days,
        l.user_group
    FROM long_anchor AS l

    UNION ALL

    SELECT
        s.user_id,
        s.d0_date,
        s.previous_active_date,
        s.silence_days,
        s.user_group
    FROM short_anchor AS s
),

credit_batch_ranked AS (
    /* 同一秒多笔获额只保留 serial_id 最大的订单。 */
    SELECT
        v.user_id,
        v.serial_id,
        v.vir_unix,
        v.vir_date,
        TO_TIMESTAMP(v.vir_unix) AS vir_time,
        v.is_pass1,
        r.recycle_type,
        ROW_NUMBER() OVER (
            PARTITION BY
                v.user_id,
                v.vir_unix
            ORDER BY
                v.serial_id DESC
        ) AS batch_rn
    FROM wangchuanliang.order_vir_f_copy AS v
    LEFT JOIN wangchuanliang.side_recycle_type_copy AS r
        ON r.serial_id = v.serial_id
    WHERE v.vir_unix IS NOT NULL
),

credit_batches AS (
    SELECT
        c.user_id,
        c.serial_id,
        c.vir_unix,
        c.vir_date,
        c.vir_time,
        c.is_pass1,
        c.recycle_type
    FROM credit_batch_ranked AS c
    WHERE c.batch_rn = 1
),

anchor_credit_ranked AS (
    /* active_date 只有日粒度，故取 D0 当日及之前最近获额批次。 */
    SELECT
        a.user_id,
        a.d0_date,
        a.previous_active_date,
        a.silence_days,
        a.user_group,
        b.serial_id AS credit_serial_id,
        b.vir_time AS credit_time,
        b.is_pass1,
        b.recycle_type,
        ROW_NUMBER() OVER (
            PARTITION BY
                a.user_id,
                a.d0_date
            ORDER BY
                b.vir_unix DESC NULLS LAST,
                b.serial_id DESC NULLS LAST
        ) AS credit_rn
    FROM anchors AS a
    LEFT JOIN credit_batches AS b
        ON b.user_id = a.user_id
       AND b.vir_date <= a.d0_date
),

anchor_credit AS (
    SELECT
        a.user_id,
        a.d0_date,
        a.previous_active_date,
        a.silence_days,
        a.user_group,
        a.credit_serial_id,
        a.credit_time,
        a.is_pass1,
        a.recycle_type
    FROM anchor_credit_ranked AS a
    WHERE a.credit_rn = 1
),

eligible_anchor AS (
    /* 同时满足获额通过和类型可提单，才纳入主提单率分母。 */
    SELECT
        a.user_id,
        a.d0_date,
        a.previous_active_date,
        a.silence_days,
        a.user_group,
        a.credit_serial_id,
        a.credit_time,
        a.is_pass1,
        a.recycle_type,
        CASE
            WHEN a.is_pass1 = 1
             AND a.recycle_type IN (2, 3)
                THEN 1
            ELSE 0
        END AS can_apply
    FROM anchor_credit AS a
),

matched_order AS (
    /* 只将与 D0 最近获额订单 serial_id 完全一致的提单/放款结果归因。 */
    SELECT
        e.user_id,
        e.d0_date,
        e.user_group,
        e.silence_days,
        e.credit_serial_id,
        e.credit_time,
        e.is_pass1,
        e.recycle_type,
        e.can_apply,
        o.apply_date,
        o.apply_time,
        o.is_remit,
        o.loan_status_code,
        o.remit_amt,
        o.repaid_amt
    FROM eligible_anchor AS e
    LEFT JOIN all_orders AS o
        ON o.user_id = e.user_id
       AND o.serial_id = e.credit_serial_id
       AND o.apply_date >= e.d0_date
       AND o.apply_date <= e.d0_date + 30
),

d0_behavior AS (
    /* D0 当天行为强度，仅作解释变量。 */
    SELECT
        a.user_id,
        SUM(COALESCE(e.newpage, 0)) AS newpage_cnt,
        SUM(COALESCE(e.amount_selection, 0)) AS amount_selection_cnt,
        SUM(COALESCE(e.term_selection, 0)) AS term_selection_cnt,
        SUM(COALESCE(e.contract_newdetails, 0)) AS contract_view_cnt,
        SUM(COALESCE(e.plan_newdetails, 0)) AS plan_view_cnt,
        SUM(COALESCE(e.newconfirm, 0)) AS confirm_cnt,
        SUM(COALESCE(e.active_duration, 0)) AS active_duration,
        SUM(COALESCE(e.apply_duration, 0)) AS apply_duration,
        SUM(COALESCE(e.non_apply_duration, 0)) AS non_apply_duration
    FROM anchors AS a
    LEFT JOIN wangchuanliang.event_track AS e
        ON e.user_id = a.user_id
       AND e.active_date = a.d0_date
    GROUP BY
        a.user_id
)



SELECT
    m.user_group,
    CASE
        WHEN m.apply_date IS NOT NULL
            THEN 'D0-D30提单用户'
        ELSE 'D0-D30未提单用户'
    END AS apply_group,
    COUNT(DISTINCT m.user_id) AS user_cnt,
    AVG(COALESCE(b.newpage_cnt, 0))::numeric AS avg_newpage_cnt,
    AVG(COALESCE(b.amount_selection_cnt, 0))::numeric AS avg_amount_selection_cnt,
    AVG(COALESCE(b.term_selection_cnt, 0))::numeric AS avg_term_selection_cnt,
    AVG(COALESCE(b.contract_view_cnt, 0))::numeric AS avg_contract_view_cnt,
    AVG(COALESCE(b.plan_view_cnt, 0))::numeric AS avg_plan_view_cnt,
    AVG(COALESCE(b.confirm_cnt, 0))::numeric AS avg_confirm_cnt,
    AVG(COALESCE(b.active_duration, 0))::numeric AS avg_active_duration,
    AVG(COALESCE(b.apply_duration, 0))::numeric AS avg_apply_duration,
    AVG(COALESCE(b.non_apply_duration, 0))::numeric AS avg_non_apply_duration
FROM matched_order AS m
LEFT JOIN d0_behavior AS b
    ON b.user_id = m.user_id
WHERE m.can_apply = 1
GROUP BY
    m.user_group,
    CASE
        WHEN m.apply_date IS NOT NULL
            THEN 'D0-D30提单用户'
        ELSE 'D0-D30未提单用户'
    END
ORDER BY
    CASE m.user_group
        WHEN '长期不活跃回流用户' THEN 1
        WHEN '比较活跃用户' THEN 2
    END,
    apply_group;
