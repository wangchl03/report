五、结论与后续动作

01

7 天券对 D3–D6 提单有正向拉动信号。

两组 D0–D2 基本一致；仅实验组仍有券的 D3–D6，实验组提单率高 3.07pp。这是券时长延长的直接观察窗口。

02

累计提单率更高，但统计把握度仍不足。

实验组累计提单率 52.59%，对照组 49.21%，相差 +3.38pp；当前 p=0.170，建议继续积累同口径批次，而非立即把该差异视为确定增量。

03

质量是当前最大约束。

实验组当前到期订单的逾期率和盈利率均弱于对照组。应持续追踪 D3–D6 拉动的订单在到期后的逾期与盈利，判断延长券期带来的增量是否覆盖让利与风险成本。

04

建议下一轮保留 3 天/7 天随机对照，并预先设置决策阈值。

建议以 D3–D6 提单率为主转化指标，同时以到期订单盈利率、逾期率_金额和逾期率_订单作为质量闸门；累计到足够到期样本后再决定是否全量延长至 7 天。

六、统计口径

实验名单

四张 no_apply_* 表直接 UNION ALL；四表无重复用户。treatment=7天券，control=3天券。

提单率

窗口提单率 = 窗口内有任意提单的用户数 ÷ 对应 AB 组样本数。窗口间不互斥，和参考报告保持一致。

到期质量

仅统计已到期且状态为 7/8 的放款分期；未到期或在途分期不计入质量分母。

附录：完整 PGSQL（可复现核心指标）

以下代码按模块顺序执行。模块一建立四批合并样本；模块二输出提单率；模块三输出到期质量；模块四输出基线和批次趋势。

模块一：合并四张实验表
CREATE TEMP TABLE tmp_coupon_ab_base AS
SELECT '0730_0804' AS batch, DATE '2026-08-04' AS recall_date,
       churn_user_id::bigint AS user_id, ab_group,
       high_interest_rate, overdue_cnt, total_remit_rnb
FROM wangchuanliang.no_apply_0730_0804_30pct
UNION ALL
SELECT '0731_0805', DATE '2026-08-05', churn_user_id::bigint, ab_group,
       high_interest_rate, overdue_cnt, total_remit_rnb
FROM wangchuanliang.no_apply_0731_0805_30pct
UNION ALL
SELECT '0801_0806', DATE '2026-08-06', churn_user_id::bigint, ab_group,
       high_interest_rate, overdue_cnt, total_remit_rnb
FROM wangchuanliang.no_apply_0801_0806_30pct
UNION ALL
SELECT '0802_0807', DATE '2026-08-07', churn_user_id::bigint, ab_group,
       high_interest_rate, overdue_cnt, total_remit_rnb
FROM wangchuanliang.no_apply_0802_0807_30pct;

-- 约定：treatment=7天30%降息券；control=3天30%降息券。
SELECT ab_group, COUNT(*) AS 样本用户数
FROM tmp_coupon_ab_base
GROUP BY ab_group;
模块二：D0–D2、D3–D6、后续窗口与累计提单率
WITH user_apply_flag AS (
    SELECT
        b.batch,
        b.recall_date,
        b.user_id,
        b.ab_group,
        MAX(CASE WHEN o.apply_date BETWEEN b.recall_date
                                    AND b.recall_date + 2
                 THEN 1 ELSE 0 END) AS d0_d2_apply,
        MAX(CASE WHEN o.apply_date BETWEEN b.recall_date + 3
                                    AND b.recall_date + 6
                 THEN 1 ELSE 0 END) AS d3_d6_apply,
        MAX(CASE WHEN o.apply_date BETWEEN b.recall_date + 7
                                    AND b.recall_date + 13
                 THEN 1 ELSE 0 END) AS d7_d13_apply,
        MAX(CASE WHEN o.apply_date BETWEEN b.recall_date + 14
                                    AND b.recall_date + 20
                 THEN 1 ELSE 0 END) AS d14_d20_apply,
        MAX(CASE WHEN o.apply_date >= b.recall_date + 21
                 THEN 1 ELSE 0 END) AS d21_plus_apply,
        MAX(CASE WHEN o.apply_date IS NOT NULL THEN 1 ELSE 0 END) AS total_apply
    FROM tmp_coupon_ab_base b
    LEFT JOIN wangchuanliang.order_loan_f_v2_copy o
      ON o.user_id = b.user_id
     AND o.apply_date >= b.recall_date
    GROUP BY b.batch, b.recall_date, b.user_id, b.ab_group
)
SELECT
    ab_group,
    COUNT(*) AS 样本用户数,
    SUM(d0_d2_apply) AS D0_D2提单用户,
    ROUND(SUM(d0_d2_apply)::numeric / COUNT(*) * 100, 2) AS D0_D2提单率_pct,
    SUM(d3_d6_apply) AS D3_D6提单用户,
    ROUND(SUM(d3_d6_apply)::numeric / COUNT(*) * 100, 2) AS D3_D6提单率_pct,
    SUM(d7_d13_apply) AS D7_D13提单用户,
    ROUND(SUM(d7_d13_apply)::numeric / COUNT(*) * 100, 2) AS D7_D13提单率_pct,
    SUM(d14_d20_apply) AS D14_D20提单用户,
    ROUND(SUM(d14_d20_apply)::numeric / COUNT(*) * 100, 2) AS D14_D20提单率_pct,
    SUM(d21_plus_apply) AS D21plus提单用户,
    ROUND(SUM(d21_plus_apply)::numeric / COUNT(*) * 100, 2) AS D21plus提单率_pct,
    SUM(total_apply) AS 累计提单用户,
    ROUND(SUM(total_apply)::numeric / COUNT(*) * 100, 2) AS 累计提单率_pct
FROM user_apply_flag
GROUP BY ab_group
ORDER BY ab_group;
模块三：到期订单盈利率与逾期率
WITH due_installment AS (
    SELECT
        b.ab_group,
        b.recall_date,
        CASE
            WHEN p.apply_date BETWEEN b.recall_date AND b.recall_date + 2
                THEN 'D0-D2（共同券期）'
            WHEN p.apply_date BETWEEN b.recall_date + 3 AND b.recall_date + 6
                THEN 'D3-D6（仅7天券有效）'
            WHEN p.apply_date BETWEEN b.recall_date + 7 AND b.recall_date + 13
                THEN 'D7-D13'
            WHEN p.apply_date BETWEEN b.recall_date + 14 AND b.recall_date + 20
                THEN 'D14-D20'
            WHEN p.apply_date >= b.recall_date + 21 THEN 'D21+'
        END AS apply_window,
        p.user_id, p.serial_id, p.loan_status_code_period,
        p.remit_amt_period, p.repaid_amt_period
    FROM tmp_coupon_ab_base b
    JOIN wangchuanliang.repayment_plan_v2_copy p
      ON p.user_id = b.user_id
     AND p.remit_date >= b.recall_date
    WHERE p.is_remit = 1
      AND p.due_date < CURRENT_DATE
      AND p.loan_status_code_period IN (7, 8)
)
SELECT
    ab_group,
    COUNT(DISTINCT serial_id) AS 到期订单数,
    COUNT(DISTINCT user_id) AS 到期用户数,
    ROUND((SUM(repaid_amt_period) - SUM(remit_amt_period))::numeric
          / NULLIF(SUM(remit_amt_period), 0) * 100, 2) AS 盈利率_pct,
    ROUND(SUM(CASE WHEN loan_status_code_period = 8
                   THEN remit_amt_period ELSE 0 END)::numeric
          / NULLIF(SUM(remit_amt_period), 0) * 100, 2) AS 逾期率_金额_pct,
    ROUND(COUNT(CASE WHEN loan_status_code_period = 8 THEN 1 END)::numeric
          / NULLIF(COUNT(*), 0) * 100, 2) AS 逾期率_分期_pct,
    ROUND(COUNT(DISTINCT CASE WHEN loan_status_code_period = 8 THEN serial_id END)::numeric
          / NULLIF(COUNT(DISTINCT serial_id), 0) * 100, 2) AS 逾期率_订单_pct
FROM due_installment
GROUP BY ab_group
ORDER BY ab_group;
模块四：按窗口到期质量、历史基线及批次趋势
-- 按窗口到期质量：复用模块三 due_installment CTE，将最后一段替换为：
SELECT
    apply_window,
    ab_group,
    COUNT(DISTINCT serial_id) AS 到期订单数,
    ROUND((SUM(repaid_amt_period) - SUM(remit_amt_period))::numeric
          / NULLIF(SUM(remit_amt_period), 0) * 100, 2) AS 盈利率_pct,
    ROUND(SUM(CASE WHEN loan_status_code_period = 8
                   THEN remit_amt_period ELSE 0 END)::numeric
          / NULLIF(SUM(remit_amt_period), 0) * 100, 2) AS 逾期率_金额_pct,
    ROUND(COUNT(DISTINCT CASE WHEN loan_status_code_period = 8 THEN serial_id END)::numeric
          / NULLIF(COUNT(DISTINCT serial_id), 0) * 100, 2) AS 逾期率_订单_pct
FROM due_installment
WHERE apply_window IS NOT NULL
GROUP BY apply_window, ab_group
ORDER BY apply_window, ab_group;

-- 召回前基线：
SELECT ab_group, COUNT(*) AS 用户数,
       ROUND(AVG(high_interest_rate), 3) AS 历史高息率,
       ROUND(AVG(overdue_cnt), 3) AS 历史逾期订单数,
       ROUND(AVG(total_remit_rnb), 2) AS 历史成功放款次数
FROM tmp_coupon_ab_base
GROUP BY ab_group;

-- 批次趋势：复用模块二 user_apply_flag CTE，按 batch、ab_group 分组即可。
SELECT batch, ab_group, COUNT(*) AS 用户数,
       ROUND(SUM(d0_d2_apply)::numeric / COUNT(*) * 100, 2) AS D0_D2提单率_pct,
       ROUND(SUM(d3_d6_apply)::numeric / COUNT(*) * 100, 2) AS D3_D6提单率_pct,
       ROUND(SUM(total_apply)::numeric / COUNT(*) * 100, 2) AS 累计提单率_pct
FROM user_apply_flag
GROUP BY batch, ab_group
ORDER BY batch, ab_group;

说明：当前质量统计为截至报告生成日已到期分期的快照；随更多订单到期，质量指标会更新。窗口提单率按用户在对应窗口“是否至少提单一次”统计，与参考报告一致，窗口之间并非互斥。
