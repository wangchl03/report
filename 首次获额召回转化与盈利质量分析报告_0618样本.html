实验组逾期率 39.4%，高于 8.4 个百分点 对照组。高额度使用率用户风险较高，改为先增加风控门槛、额度限额或更温和的触达方案，再单独验证。

WITH sample_users AS (
    /* 4932 与 3149 的共同流失用户样本：
       分组改用 3149 表的 ab_group_2、user_type_2；
       额度使用率沿用 4932 表的历史额度使用率。 */
    SELECT
        a.churn_user_id AS user_id,
        b.ab_group_2 AS ab_group,
        b.user_type_2 AS user_type,
        CASE
            WHEN a.utilization_rate >= 0.9 THEN 'high'
            ELSE 'low'
        END AS utilization_level
    FROM wangchuanliang.churned_user_4932 AS a
    INNER JOIN wangchuanliang.churned_user_3149 AS b
        ON b.churn_user_id_2 = a.churn_user_id::integer
),

first_credit_ranked AS (
    /* 2026-06-18 起的获额记录：
       先按获额时间取最早；同一时间戳多笔获额时取 serial_id 最大的一笔。 */
    SELECT
        v.user_id,
        v.serial_id,
        v.vir_date,
        v.vir_unix,
        v.is_pass1,
        v.max_credit_apply_amt,
        ROW_NUMBER() OVER (
            PARTITION BY v.user_id
            ORDER BY
                v.vir_unix ASC,
                v.serial_id DESC
        ) AS rn
    FROM wangchuanliang.order_vir_f_copy AS v
    INNER JOIN sample_users AS s
        ON s.user_id = v.user_id
    WHERE v.vir_date >= '2026-06-18'
      AND v.vir_unix IS NOT NULL
),

first_credit AS (
    SELECT
        f.user_id,
        f.serial_id,
        f.vir_date,
        TO_TIMESTAMP(f.vir_unix) AS vir_time,
        f.is_pass1,
        f.max_credit_apply_amt
    FROM first_credit_ranked AS f
    WHERE f.rn = 1
),

eligible_first_credit AS (
    /* 仅保留从 2026-06-18 召回开始至首次获额前从未提单的用户。 */
    SELECT
        f.user_id,
        f.serial_id,
        f.vir_date,
        f.vir_time,
        f.is_pass1,
        f.max_credit_apply_amt
    FROM first_credit AS f
    WHERE NOT EXISTS (
        SELECT
            1
        FROM wangchuanliang.order_loan_f_v2_copy AS o
        WHERE o.user_id = f.user_id
          AND o.apply_date IS NOT NULL
          AND o.apply_time >= TIMESTAMP '2026-06-18 00:00:00'
          AND o.apply_time < f.vir_time
    )
),

credit_by_order AS (
    /* 历史订单按 serial_id 保留唯一获额记录，以避免一笔订单重复计入额度。 */
    SELECT DISTINCT ON (
        v.serial_id
    )
        v.serial_id,
        v.vir_date,
        v.max_credit_apply_amt,
        v.max_risk_credit_amount
    FROM wangchuanliang.order_vir_f_copy AS v
    ORDER BY
        v.serial_id,
        v.vir_unix DESC
),

history_order_amount AS (
    /* 召回前历史订单：只用于计算召回前额度使用率。 */
    SELECT
        f.user_id,
        o.apply_amt,
        CASE
            WHEN v.vir_date >= DATE '2025-04-01'
                THEN v.max_credit_apply_amt
            WHEN r.recycle_type = 2
                THEN v.max_credit_apply_amt
            ELSE v.max_risk_credit_amount
        END AS credit_amt
    FROM eligible_first_credit AS f
    INNER JOIN wangchuanliang.order_loan_f_v2_copy AS o
        ON o.user_id = f.user_id
       AND o.apply_date IS NOT NULL
       AND o.apply_time < f.vir_time
    INNER JOIN credit_by_order AS v
        ON v.serial_id = o.serial_id
    LEFT JOIN wangchuanliang.side_recycle_type_copy AS r
        ON r.serial_id = o.serial_id
),

pre_utilization_by_user AS (
    /* 仅当提单金额、获额金额均不少于 200 时，历史订单参与额度使用率。 */
    SELECT
        h.user_id,
        SUM(
            CASE
                WHEN h.apply_amt >= 200
                 AND h.credit_amt >= 200
                    THEN h.apply_amt
                ELSE 0
            END
        ) AS pre_apply_amt,
        SUM(
            CASE
                WHEN h.apply_amt >= 200
                 AND h.credit_amt >= 200
                    THEN h.credit_amt
                ELSE 0
            END
        ) AS pre_credit_amt
    FROM history_order_amount AS h
    GROUP BY
        h.user_id
),

evaluation_base AS (
    /* 原始样本左连最终样本：原始流失用户数不受前置提单过滤影响。 */
    SELECT
        s.user_id,
        s.ab_group,
        s.user_type,
        s.utilization_level,
        f.serial_id AS first_credit_serial_id,
        f.is_pass1,
        f.max_credit_apply_amt,
        o.serial_id AS first_apply_serial_id,
        o.apply_amt,
        o.is_remit,
        o.loan_status_code,
        o.remit_amt,
        o.repaid_amt,
        p.pre_apply_amt,
        p.pre_credit_amt
    FROM sample_users AS s
    LEFT JOIN eligible_first_credit AS f
        ON f.user_id = s.user_id
    LEFT JOIN wangchuanliang.order_loan_f_v2_copy AS o
        ON o.serial_id = f.serial_id
    LEFT JOIN pre_utilization_by_user AS p
        ON p.user_id = s.user_id
),

metric_aggregation AS (
    /* 各层级均直接聚合原始人数、订单金额分子和分母。
       禁止对下级的百分比作简单平均。 */
    SELECT
        GROUPING(e.user_type) AS g_user_type,
        GROUPING(e.utilization_level) AS g_utilization_level,
        GROUPING(e.ab_group) AS g_ab_group,
        e.user_type,
        e.utilization_level,
        e.ab_group,
        COUNT(DISTINCT e.user_id) AS churn_users,
        COUNT(DISTINCT CASE
            WHEN e.first_credit_serial_id IS NOT NULL THEN e.user_id
        END) AS first_credit_users,
        COUNT(DISTINCT CASE
            WHEN e.is_pass1 = 1 THEN e.user_id
        END) AS first_credit_pass_users,
        COUNT(DISTINCT CASE
            WHEN e.is_pass1 = 1
             AND e.first_apply_serial_id IS NOT NULL THEN e.user_id
        END) AS first_apply_users,
        COUNT(DISTINCT CASE
            WHEN e.is_pass1 = 1
             AND e.first_apply_serial_id IS NOT NULL
             AND e.is_remit = 1 THEN e.user_id
        END) AS first_remit_users,
        COUNT(DISTINCT CASE
            WHEN e.is_pass1 = 1
             AND e.first_apply_serial_id IS NOT NULL
             AND e.is_remit = 1 THEN e.first_apply_serial_id
        END) AS remit_orders,
        COUNT(DISTINCT CASE
            WHEN e.is_pass1 = 1
             AND e.first_apply_serial_id IS NOT NULL
             AND e.is_remit = 1
             AND e.loan_status_code = 8 THEN e.first_apply_serial_id
        END) AS overdue_orders,
        SUM(CASE
            WHEN e.is_pass1 = 1
             AND e.first_apply_serial_id IS NOT NULL
             AND e.is_remit = 1
             AND e.loan_status_code IN (7, 8) THEN e.repaid_amt
            ELSE 0
        END) AS profit_repaid_amt,
        SUM(CASE
            WHEN e.is_pass1 = 1
             AND e.first_apply_serial_id IS NOT NULL
             AND e.is_remit = 1
             AND e.loan_status_code IN (7, 8) THEN e.remit_amt
            ELSE 0
        END) AS profit_remit_amt,
        SUM(CASE
            WHEN e.is_pass1 = 1
             AND e.first_apply_serial_id IS NOT NULL
             AND e.is_remit = 1
             AND e.loan_status_code = 8 THEN e.repaid_amt
            ELSE 0
        END) AS overdue_repaid_amt,
        SUM(CASE
            WHEN e.is_pass1 = 1
             AND e.first_apply_serial_id IS NOT NULL
             AND e.is_remit = 1
             AND e.loan_status_code = 8 THEN e.remit_amt
            ELSE 0
        END) AS overdue_remit_amt,
        SUM(CASE
            WHEN e.is_pass1 = 1
             AND e.first_apply_serial_id IS NOT NULL THEN e.pre_apply_amt
            ELSE 0
        END) AS pre_apply_amt,
        SUM(CASE
            WHEN e.is_pass1 = 1
             AND e.first_apply_serial_id IS NOT NULL THEN e.pre_credit_amt
            ELSE 0
        END) AS pre_credit_amt,
        SUM(CASE
            WHEN e.is_pass1 = 1
             AND e.first_apply_serial_id IS NOT NULL
             AND e.apply_amt >= 200
             AND e.max_credit_apply_amt >= 200 THEN e.apply_amt
            ELSE 0
        END) AS post_apply_amt,
        SUM(CASE
            WHEN e.is_pass1 = 1
             AND e.first_apply_serial_id IS NOT NULL
             AND e.apply_amt >= 200
             AND e.max_credit_apply_amt >= 200 THEN e.max_credit_apply_amt
            ELSE 0
        END) AS post_credit_amt
    FROM evaluation_base AS e
    GROUP BY GROUPING SETS (
        (e.user_type, e.utilization_level, e.ab_group),
        (e.user_type, e.utilization_level),
        (e.user_type),
        (e.ab_group),
        ()
    )
),

metrics AS (
    SELECT
        m.*,
        m.first_credit_pass_users::numeric
            / NULLIF(m.first_credit_users, 0) AS pass_rate,
        m.first_apply_users::numeric
            / NULLIF(m.first_credit_pass_users, 0) AS apply_rate,
        m.first_remit_users::numeric
            / NULLIF(m.first_apply_users, 0) AS remit_success_rate,
        m.overdue_orders::numeric
            / NULLIF(m.remit_orders, 0) AS overdue_rate,
        m.profit_repaid_amt
            / NULLIF(m.profit_remit_amt, 0) - 1 AS profit_rate,
        m.overdue_repaid_amt
            / NULLIF(m.overdue_remit_amt, 0) AS overdue_repay_rate,
        m.pre_apply_amt
            / NULLIF(m.pre_credit_amt, 0) AS pre_utilization_rate,
        m.post_apply_amt
            / NULLIF(m.post_credit_amt, 0) AS post_utilization_rate
    FROM metric_aggregation AS m
),

final_rows AS (
    /* 总计/AB 组：用于第一张表。 */
    SELECT
        '总计与AB组' AS sheet_type,
        CASE
            WHEN m.g_ab_group = 1 THEN 1
            WHEN m.ab_group = 'control' THEN 2
            WHEN m.ab_group = 'treatment' THEN 3
            ELSE 9
        END AS sort_order,
        CASE
            WHEN m.g_ab_group = 1 THEN '总计'
            WHEN m.ab_group = 'control' THEN '对照组'
            WHEN m.ab_group = 'treatment' THEN '实验组'
            ELSE m.ab_group
        END AS level_1,
        NULL::text AS level_2,
        NULL::text AS level_3,
        m.*
    FROM metrics AS m
    WHERE m.g_user_type = 1
      AND m.g_utilization_level = 1

    UNION ALL

    /* 分类明细：总计 → 用户类型汇总 → 额度分类汇总 → 组别。 */
    SELECT
        '层级分类' AS sheet_type,
        CASE
            WHEN m.g_user_type = 1 THEN 1
            WHEN m.user_type = 'type_1' AND m.g_utilization_level = 1 THEN 10
            WHEN m.user_type = 'type_1' AND m.utilization_level = 'high' AND m.g_ab_group = 1 THEN 11
            WHEN m.user_type = 'type_1' AND m.utilization_level = 'high' AND m.ab_group = 'control' THEN 12
            WHEN m.user_type = 'type_1' AND m.utilization_level = 'high' AND m.ab_group = 'treatment' THEN 13
            WHEN m.user_type = 'type_1' AND m.utilization_level = 'low' AND m.g_ab_group = 1 THEN 14
            WHEN m.user_type = 'type_1' AND m.utilization_level = 'low' AND m.ab_group = 'control' THEN 15
            WHEN m.user_type = 'type_1' AND m.utilization_level = 'low' AND m.ab_group = 'treatment' THEN 16
            WHEN m.user_type = 'type_2' AND m.g_utilization_level = 1 THEN 20
            WHEN m.user_type = 'type_2' AND m.utilization_level = 'high' AND m.g_ab_group = 1 THEN 21
            WHEN m.user_type = 'type_2' AND m.utilization_level = 'high' AND m.ab_group = 'control' THEN 22
            WHEN m.user_type = 'type_2' AND m.utilization_level = 'high' AND m.ab_group = 'treatment' THEN 23
            WHEN m.user_type = 'type_2' AND m.utilization_level = 'low' AND m.g_ab_group = 1 THEN 24
            WHEN m.user_type = 'type_2' AND m.utilization_level = 'low' AND m.ab_group = 'control' THEN 25
            WHEN m.user_type = 'type_2' AND m.utilization_level = 'low' AND m.ab_group = 'treatment' THEN 26
            WHEN m.user_type = 'type_3' AND m.g_utilization_level = 1 THEN 30
            WHEN m.user_type = 'type_3' AND m.utilization_level = 'high' AND m.g_ab_group = 1 THEN 31
            WHEN m.user_type = 'type_3' AND m.utilization_level = 'high' AND m.ab_group = 'control' THEN 32
            WHEN m.user_type = 'type_3' AND m.utilization_level = 'high' AND m.ab_group = 'treatment' THEN 33
            WHEN m.user_type = 'type_3' AND m.utilization_level = 'low' AND m.g_ab_group = 1 THEN 34
            WHEN m.user_type = 'type_3' AND m.utilization_level = 'low' AND m.ab_group = 'control' THEN 35
            WHEN m.user_type = 'type_3' AND m.utilization_level = 'low' AND m.ab_group = 'treatment' THEN 36
            ELSE 99
        END AS sort_order,
        CASE
            WHEN m.g_user_type = 1 THEN '总计'
            WHEN m.user_type = 'type_1' THEN '流失前非活跃用户'
            WHEN m.user_type = 'type_2' THEN '流失90天以上用户'
            WHEN m.user_type = 'type_3' THEN '流失8-90天用户'
            ELSE m.user_type
        END AS level_1,
        CASE
            WHEN m.g_user_type = 1 THEN '全部额度分类'
            WHEN m.g_utilization_level = 1 THEN '额度分类汇总'
            WHEN m.utilization_level = 'high' THEN '高额度使用率用户'
            WHEN m.utilization_level = 'low' THEN '低额度使用率用户'
            ELSE m.utilization_level
        END AS level_2,
        CASE
            WHEN m.g_ab_group = 1 THEN '组别汇总'
            WHEN m.ab_group = 'control' THEN '对照组'
            WHEN m.ab_group = 'treatment' THEN '实验组'
            ELSE m.ab_group
        END AS level_3,
        m.*
    FROM metrics AS m
    WHERE NOT (
        m.g_user_type = 1
        AND m.g_utilization_level = 1
        AND m.g_ab_group = 0
    )
)

SELECT
    f.sheet_type,
    f.sort_order,
    f.level_1,
    f.level_2,
    f.level_3,
    f.churn_users,
    f.first_credit_users,
    f.first_credit_pass_users,
    f.pass_rate,
    f.first_apply_users,
    f.apply_rate,
    f.first_remit_users,
    f.remit_success_rate,
    f.remit_orders,
    f.overdue_orders,
    f.overdue_rate,
    f.profit_rate,
    f.overdue_repay_rate,
    f.pre_utilization_rate,
    f.post_utilization_rate,
    f.post_utilization_rate - f.pre_utilization_rate AS utilization_rate_change,
    f.profit_repaid_amt,
    f.profit_remit_amt,
    f.overdue_repaid_amt,
    f.overdue_remit_amt,
    f.pre_apply_amt,
    f.pre_credit_amt,
    f.post_apply_amt,
    f.post_credit_amt
FROM final_rows AS f
ORDER BY
    f.sheet_type,
    f.sort_order;
