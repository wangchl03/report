-- T0 model V1. All tables created here are session-local TEMP tables.
-- Business timezone: America/Belize. Labels: user follow-up (definition 1).
-- Source data complete through 2026-10-06. Events: Jan 1 through Sep 30.
-- Model scores are keyed by serial_id; source has no score generation timestamp.

-- STEP params
CREATE TEMP TABLE tmp_t0ml_params AS
SELECT DATE '2026-01-01' AS start_date, DATE '2026-10-01' AS end_date,
       TIMESTAMP '2026-10-07 00:00:00' AS data_cutoff_ts;

-- STEP offer
CREATE TEMP TABLE tmp_t0ml_offer AS
SELECT DISTINCT ON (v.user_id,v.serial_id)
       v.user_id,v.serial_id,v.vir_date::date AS vir_date,
       (TO_TIMESTAMP(v.vir_unix) AT TIME ZONE 'America/Belize') AS vir_time,
       v.max_credit_apply_amt::float8 AS current_credit_amt,
       v.total_term::float8 AS current_total_term,
       v.risk_product,v.vir_version
FROM wangchuanliang.order_vir_f_copy v CROSS JOIN tmp_t0ml_params p
WHERE v.is_pass1=1 AND v.loan_type_code=2
  AND v.vir_date>=p.start_date-1 AND v.vir_date<p.data_cutoff_ts::date+1
  AND EXISTS (SELECT 1 FROM wangchuanliang.side_recycle_type_copy r
              WHERE r.serial_id=v.serial_id AND r.recycle_type=3)
ORDER BY v.user_id,v.serial_id,v.vir_unix DESC;

-- STEP dedup
CREATE TEMP TABLE tmp_t0ml_dedup AS
WITH lagged AS (
 SELECT v.*,LAG(vir_time) OVER(PARTITION BY user_id ORDER BY vir_time,serial_id) AS previous_time
 FROM tmp_t0ml_offer v
), batched AS (
 SELECT l.*,SUM(CASE WHEN previous_time IS NULL OR vir_time-previous_time>INTERVAL '20 seconds'
                    THEN 1 ELSE 0 END)
 OVER(PARTITION BY user_id ORDER BY vir_time,serial_id ROWS UNBOUNDED PRECEDING) AS batch_id
 FROM lagged l
), ranked AS (
 SELECT b.*,ROW_NUMBER() OVER(PARTITION BY user_id,batch_id ORDER BY vir_time DESC,serial_id DESC) AS rn
 FROM batched b
)
SELECT r.* FROM ranked r CROSS JOIN tmp_t0ml_params p
WHERE rn=1 AND vir_date>=p.start_date AND vir_time<p.data_cutoff_ts;

-- STEP orders
CREATE TEMP TABLE tmp_t0ml_orders AS
SELECT DISTINCT ON(o.user_id,o.serial_id)
 o.user_id,o.serial_id,o.apply_time,o.apply_date,o.apply_amt::float8,
 o.remit_time,o.remit_date,o.remit_amt::float8,o.is_remit,
 o.repaid_time,o.repaid_date,o.is_repaid,o.loan_status_code,
 o.due_date,o.total_term::float8,o.loan_day::float8,
 o.pre_amt::float8,o.post_amt::float8,o.remit_succ_rnb,
 o.reg_date,o.product_no,o.loan_type,o.repaid_amt::float8
FROM wangchuanliang.order_loan_f_v2_copy o
JOIN (SELECT DISTINCT user_id FROM tmp_t0ml_dedup) u ON u.user_id=o.user_id
ORDER BY o.user_id,o.serial_id,o.apply_time ASC;

-- STEP settlement
CREATE TEMP TABLE tmp_t0ml_settlement AS
SELECT o.* FROM tmp_t0ml_orders o CROSS JOIN tmp_t0ml_params p
WHERE o.is_remit=1 AND o.remit_amt>0
 AND (o.is_repaid=1 OR o.loan_status_code=7)
 AND o.repaid_time> TIMESTAMP '2000-01-01'
 AND o.repaid_date>=p.start_date AND o.repaid_time<p.data_cutoff_ts
 AND NOT EXISTS (
  SELECT 1 FROM tmp_t0ml_orders x
  WHERE x.user_id=o.user_id AND x.serial_id<>o.serial_id
    AND x.is_remit=1 AND x.remit_amt>0 AND x.remit_time<=o.repaid_time
    AND (x.repaid_time IS NULL OR x.repaid_time<=TIMESTAMP '2000-01-01'
         OR x.repaid_time>o.repaid_time)
 );

-- STEP events
CREATE TEMP TABLE tmp_t0ml_events_all AS
WITH matched AS (
 SELECT e.*,s.serial_id AS settled_serial_id,s.repaid_time,
  ROW_NUMBER() OVER(PARTITION BY e.user_id,e.serial_id ORDER BY s.repaid_time DESC,s.serial_id DESC) AS srn
 FROM tmp_t0ml_dedup e JOIN tmp_t0ml_settlement s
 ON s.user_id=e.user_id AND s.repaid_date=e.vir_date AND s.repaid_time<=e.vir_time
)
SELECT m.*,LEAD(vir_time) OVER(PARTITION BY user_id ORDER BY vir_time,serial_id) AS next_t0_time
FROM matched m WHERE srn=1;

-- STEP labels
CREATE TEMP TABLE tmp_t0ml_events AS
SELECT e.*,MIN(o.apply_time) AS first_apply_time,
 LEAST(COALESCE(e.next_t0_time,p.data_cutoff_ts),p.data_cutoff_ts,
       e.vir_date+INTERVAL '16 days') AS followup_end,
 p.data_cutoff_ts
FROM tmp_t0ml_events_all e CROSS JOIN tmp_t0ml_params p
LEFT JOIN tmp_t0ml_orders o ON o.user_id=e.user_id
 AND o.apply_time>=e.vir_time AND o.apply_time<p.data_cutoff_ts
 AND o.apply_time<e.vir_date+INTERVAL '16 days'
 AND (e.next_t0_time IS NULL OR o.apply_time<e.next_t0_time)
WHERE e.vir_date<p.end_date
GROUP BY e.user_id,e.serial_id,e.vir_date,e.vir_time,e.current_credit_amt,
 e.current_total_term,e.risk_product,e.vir_version,e.previous_time,e.batch_id,
 e.rn,e.settled_serial_id,e.repaid_time,e.srn,e.next_t0_time,p.data_cutoff_ts;

-- STEP previous_offer
CREATE TEMP TABLE tmp_t0ml_order_credit AS
SELECT DISTINCT ON (o.user_id,o.serial_id)
 o.user_id,o.serial_id,v.max_credit_apply_amt::float8 AS credit_amt,
 (TO_TIMESTAMP(v.vir_unix) AT TIME ZONE 'America/Belize') AS credit_time
FROM tmp_t0ml_orders o JOIN wangchuanliang.order_vir_f_copy v
 ON v.user_id=o.user_id AND v.serial_id=o.serial_id AND v.is_pass1=1
 AND (TO_TIMESTAMP(v.vir_unix) AT TIME ZONE 'America/Belize')<=o.apply_time
ORDER BY o.user_id,o.serial_id,v.vir_unix DESC;

-- STEP history
CREATE TEMP TABLE tmp_t0ml_order_history AS
WITH ranked AS (
 SELECT e.serial_id AS event_id,o.*,
  v.credit_amt,v.credit_time,
  ROW_NUMBER() OVER(PARTITION BY e.serial_id ORDER BY o.apply_time DESC,o.serial_id DESC) AS arn,
  ROW_NUMBER() OVER(PARTITION BY e.serial_id
    ORDER BY CASE WHEN o.is_remit=1 AND o.remit_time<=e.vir_time THEN o.remit_time END DESC NULLS LAST,
             o.serial_id DESC) AS rrn,
  CASE WHEN o.is_remit=1 AND o.remit_amt>0 AND o.remit_time<=e.vir_time THEN 1 ELSE 0 END AS funded,
  CASE WHEN o.repaid_time>TIMESTAMP '2000-01-01' AND o.repaid_time<=e.vir_time THEN 1 ELSE 0 END AS settled
 FROM tmp_t0ml_events e JOIN tmp_t0ml_orders o ON o.user_id=e.user_id AND o.apply_time<e.vir_time
 LEFT JOIN tmp_t0ml_order_credit v ON v.user_id=o.user_id AND v.serial_id=o.serial_id
), calculated AS (
 SELECT r.*,
  CASE WHEN apply_amt>0 AND credit_amt>0 THEN apply_amt/credit_amt END AS utilization,
  CASE WHEN credit_time IS NOT NULL THEN EXTRACT(EPOCH FROM (apply_time-credit_time))/86400.0 END AS credit_apply_gap_days
 FROM ranked r
)
SELECT event_id,COUNT(*)::float8 AS historical_apply_count,
 SUM(funded)::float8 AS successful_loan_count,
 SUM(CASE WHEN funded=1 THEN remit_amt ELSE 0 END)::float8 AS total_remit_amt,
 SUM(CASE WHEN settled=1 AND repaid_date<due_date THEN 1 ELSE 0 END)::float8 AS early_settle_count,
 SUM(CASE WHEN settled=1 AND repaid_date>due_date THEN 1 ELSE 0 END)::float8 AS late_settle_count,
 MAX(CASE WHEN settled=1 THEN GREATEST(repaid_date-due_date,0) END)::float8 AS max_past_late_days,
 MIN(reg_date) AS registration_date,
 MAX(CASE WHEN arn=1 THEN apply_amt END) AS apply_amt_l1,
 MAX(CASE WHEN arn=2 THEN apply_amt END) AS apply_amt_l2,
 MAX(CASE WHEN arn=3 THEN apply_amt END) AS apply_amt_l3,
 MAX(CASE WHEN arn=1 THEN utilization END) AS apply_util_l1,
 MAX(CASE WHEN arn=2 THEN utilization END) AS apply_util_l2,
 MAX(CASE WHEN arn=3 THEN utilization END) AS apply_util_l3,
 MAX(CASE WHEN arn=1 THEN credit_apply_gap_days END) AS apply_gap_l1,
 MAX(CASE WHEN arn=2 THEN credit_apply_gap_days END) AS apply_gap_l2,
 MAX(CASE WHEN arn=3 THEN credit_apply_gap_days END) AS apply_gap_l3,
 MAX(CASE WHEN funded=1 AND rrn=1 THEN remit_amt END) AS remit_amt_l1,
 MAX(CASE WHEN funded=1 AND rrn=2 THEN remit_amt END) AS remit_amt_l2,
 MAX(CASE WHEN funded=1 AND rrn=3 THEN remit_amt END) AS remit_amt_l3,
 MAX(apply_time) AS last_apply_time,
 MAX(CASE WHEN funded=1 THEN remit_time END) AS last_remit_time
FROM calculated GROUP BY event_id;

-- STEP credit_history
CREATE TEMP TABLE tmp_t0ml_credit_history AS
WITH eligible AS (
 SELECT e.serial_id AS event_id,v.serial_id,v.vir_unix,v.is_pass1,
 v.max_credit_apply_amt::float8 AS credit_amt,
 ROW_NUMBER() OVER(PARTITION BY e.serial_id,v.serial_id ORDER BY v.vir_unix DESC) AS srn
 FROM tmp_t0ml_events e JOIN wangchuanliang.order_vir_f_copy v
 ON v.user_id=e.user_id
 AND (TO_TIMESTAMP(v.vir_unix) AT TIME ZONE 'America/Belize')<=e.vir_time
), ranked AS (
 SELECT a.*,ROW_NUMBER() OVER(PARTITION BY event_id
   ORDER BY CASE WHEN is_pass1=1 THEN vir_unix END DESC NULLS LAST,serial_id DESC) AS prn
 FROM eligible a WHERE srn=1
)
SELECT event_id,COUNT(*)::float8 AS credit_event_count,
 SUM(CASE WHEN is_pass1=1 THEN 1 ELSE 0 END)::float8 AS credit_pass_count,
 MAX(CASE WHEN is_pass1=1 AND prn=1 THEN credit_amt END) AS credit_amt_l1,
 MAX(CASE WHEN is_pass1=1 AND prn=2 THEN credit_amt END) AS credit_amt_l2,
 MAX(CASE WHEN is_pass1=1 AND prn=3 THEN credit_amt END) AS credit_amt_l3
FROM ranked GROUP BY event_id;

-- STEP features
CREATE TEMP TABLE tmp_t0ml_features AS
SELECT e.serial_id AS event_id,e.user_id,e.serial_id AS credit_serial_id,e.settled_serial_id,
 e.vir_date,e.vir_time AS as_of_time,e.first_apply_time,e.followup_end,e.next_t0_time,e.data_cutoff_ts,
 e.current_credit_amt,e.current_total_term,e.risk_product,e.vir_version,
 c.credit_amt AS previous_credit_amt,

 NULLIF(m.kabybxgboost405apd7crecyc,-9999999) AS risk_score_405,

 h.historical_apply_count,h.successful_loan_count,h.total_remit_amt,
 h.early_settle_count,h.late_settle_count,h.max_past_late_days,
 (e.vir_date-h.registration_date)::float8 AS user_age_days,
 h.apply_amt_l1,h.apply_amt_l2,h.apply_amt_l3,
 h.apply_util_l1,h.apply_util_l2,h.apply_util_l3,
 h.apply_gap_l1,h.apply_gap_l2,h.apply_gap_l3,
 h.remit_amt_l1,h.remit_amt_l2,h.remit_amt_l3,

        hit=frame.first_apply_time.notna()&(frame.first_apply_time>=left)&(frame.first_apply_time<right)

        part['interval_length']=end-start
        part['hazard_label']=hit.loc[selected].astype(int)

def predict_raw(model,x):
    if hasattr(model,'booster_'):
        p=model.booster_.predict(x,num_threads=4)
    else:
        with np.errstate(over='ignore',invalid='ignore',divide='ignore'):
            p=model.predict_proba(x)[:,1]
    if not np.isfinite(p).all():raise AssertionError('Non-finite model probabilities')
    return p


def predict_cumulative(bundle,frame):
    hazards=[]
    for i,(start,end) in enumerate(zip(START_DAYS,END_DAYS)):
        current=frame.copy();current['interval_id']=i
        current['interval_start_day']=start;current['interval_length']=end-start
        x=bundle['preprocessor'].transform(current[bundle['features']].replace([np.inf,-np.inf],np.nan))
        p=predict_raw(bundle['model'],x)
        calibrator=bundle.get('calibrators',{}).get(i)
        if calibrator:
            if calibrator['kind']=='sigmoid':
                z=np.log(np.clip(p,1e-6,1-1e-6)/np.clip(1-p,1e-6,1))
                p=calibrator['model'].predict_proba(z.reshape(-1,1))[:,1]
            elif calibrator['kind']=='isotonic':p=calibrator['model'].predict(p)
        hazards.append(np.clip(p,1e-7,1-1e-7))
    hazards=np.column_stack(hazards)
    cumulative=1-np.cumprod(1-hazards,axis=1)
    if not np.isfinite(cumulative).all():raise AssertionError('Non-finite cumulative probabilities')
    if np.any(np.diff(cumulative,axis=1)<-1e-12):raise AssertionError('Probabilities are not monotone')
    return cumulative,hazards

def metric(y,p):
    valid=np.isfinite(y);y=np.asarray(y)[valid].astype(int);p=np.asarray(p)[valid]
    if len(y)<2 or len(np.unique(y))<2:return {'n':int(len(y))}
    top=max(1,int(np.ceil(len(y)*.2)))
    bins=pd.qcut(pd.Series(p),q=10,duplicates='drop') if len(np.unique(p))>1 else pd.Series(0,index=range(len(p)))
    cal=pd.DataFrame({'y':y,'p':p,'bin':bins}).groupby('bin',observed=True).agg(n=('y','size'),actual=('y','mean'),predicted=('p','mean'))
    fpr,tpr,_=roc_curve(y,p)
    top10=max(1,int(np.ceil(len(y)*.1)))
    def captured(k):
        cut=np.partition(p,len(p)-k)[len(p)-k]
        stronger=p>cut;ties=p==cut
        return float(y[stronger].sum()+(k-stronger.sum())/ties.sum()*y[ties].sum())
    captured20=captured(top);captured10=captured(top10)
    return {'n':int(len(y)),'positive':int(y.sum()),'actual_rate':float(y.mean()),'predicted_rate':float(p.mean()),
            'auc':float(roc_auc_score(y,p)),'pr_auc':float(average_precision_score(y,p)),
            'ks':float(np.max(tpr-fpr)),
            'brier':float(brier_score_loss(y,p)),'log_loss':float(log_loss(y,p)),
            'top10_lift':float(captured10/top10/y.mean()),
            'top10_coverage':float(captured10/y.sum()),
            'top20_lift':float(captured20/top/y.mean()),
            'top20_coverage':float(captured20/y.sum()),
            'ece':float((cal.n*(cal.actual-cal.predicted).abs()).sum()/len(y))}

def drift_metrics(training,test,columns):
    records=[]
    for c in columns:
        if c.startswith('interval_'):continue
        a=training[c].replace([np.inf,-np.inf],np.nan);b=test[c].replace([np.inf,-np.inf],np.nan)
        if c in CATEGORIES:

            aa=a.fillna('MISSING').map(lambda v:v if v in common else 'OTHER')
            bb=b.fillna('MISSING').map(lambda v:v if v in common else 'OTHER')
            categories=list(common)+['OTHER']
        else:
            values=pd.to_numeric(a,errors='coerce').dropna()
            breaks=np.unique(values.quantile(np.linspace(.1,.9,9)).to_numpy()) if len(values) else np.array([])
            edges=np.r_[-np.inf,breaks,np.inf]
            aa=pd.cut(a,edges,labels=False,include_lowest=True).fillna(-1)
            bb=pd.cut(b,edges,labels=False,include_lowest=True).fillna(-1)
            categories=list(range(len(edges)-1))+[-1]
        pa=aa.value_counts(normalize=True).reindex(categories,fill_value=0).to_numpy()+1e-4
        pb=bb.value_counts(normalize=True).reindex(categories,fill_value=0).to_numpy()+1e-4
        pa/=pa.sum();pb/=pb.sum()
        records.append({'feature':c,'psi':float(np.sum((pb-pa)*np.log(pb/pa))),
            'train_missing':float(a.isna().mean()),'test_missing':float(b.isna().mean())})
    return sorted(records,key=lambda r:r['psi'],reverse=True)

def current_streak(values):

def maximum_streak(values):

def describe(out,name,values,lags=False):
    a=np.asarray(values,dtype=float);v=finite(a)
    out[name+'_valid_n']=len(v)
    for kind in ['mean','std','min','max']:
        out[name+'_'+kind]=float(getattr(np,kind)(v)) if len(v) else np.nan
    out[name+'_last_first_delta']=v[-1]-v[0] if len(v)>=2 else np.nan
    ix=np.flatnonzero(np.isfinite(a));center=ix-ix.mean() if len(ix) else ix
    out[name+'_slope']=float(np.sum(center*(v-v.mean()))/np.sum(center**2)) if len(v)>=2 else np.nan

    assert not orders.serial_id.duplicated().any(),'Serial IDs must be globally unique for source paging'

    credits=credits.sort_values(['user_id','credit_time','serial_id']).reset_index(drop=True)
    payments=pd.read_csv(ROOT/'sources/payments.csv.gz');parse_times(payments,['payin_date'])
    unix=pd.to_numeric(payments.payin_time,errors='coerce');seconds=unix.where(unix<10**11,unix/1000)
    payments['payment_time']=pd.to_datetime(seconds,unit='s',utc=True,errors='coerce').dt.tz_convert('America/Belize').dt.tz_localize(None)
    payments['timestamp_valid']=(payments.payment_time.dt.normalize()==payments.payin_date)&(seconds>946684800)
    # Invalid/inconsistent intraday timestamps do not silently become midnight events.
    payment_audit={'ledger_rows':len(payments),'source_rows':int(payments.source_rows.sum()),
                   'invalid_time_n':int((~payments.timestamp_valid).sum()),'timestamp_unit':'seconds; millisecond values normalized',
                   'definition':'positive posted repayment_record_copy, not a separate payment success flag'}
    payments=payments.sort_values(['user_id','payment_time','serial_id']).reset_index(drop=True)
    approved=credits.loc[credits.is_pass1.eq(1),['user_id','serial_id','credit_time','credit_amt']].sort_values('credit_time')
    joined=pd.merge_asof(orders[['user_id','serial_id','apply_time']].assign(original_row=np.arange(len(orders))).sort_values('apply_time'),
                         approved,left_on='apply_time',right_on='credit_time',by=['user_id','serial_id'],direction='backward')
    orders['credit_amt']=joined.sort_values('original_row').credit_amt.to_numpy()
    orders['utilization']=(orders.apply_amt/orders.credit_amt.where(orders.credit_amt>0)).where(orders.apply_amt>0)

    orders=orders.merge(aggregate,left_on='serial_id',right_index=True,how='left',validate='one_to_one')
    bad=orders.flow_settle_time.dt.normalize().ne(orders.repaid_date)
    orders.loc[bad,'flow_settle_time']=pd.NaT
    orders.loc[bad,['settle_cash','settle_payment_n','settle_payment_days']]=np.nan

    orders['apply_settle_days']=(orders.flow_settle_time-orders.apply_time).dt.total_seconds()/86400
    orders['remit_settle_days']=(orders.flow_settle_time-orders.remit_time).dt.total_seconds()/86400
    orders['settle_cash_remit_ratio']=orders.settle_cash/orders.remit_amt.where(orders.remit_amt>0)
    orders.to_pickle(ROOT/'datasets/order_lifecycle_enriched.pkl.gz',compression='gzip')
    payment_audit['settled_orders_with_verified_cash_day']=int(orders.flow_settle_time.notna().sum())
    payment_audit['settled_orders_total']=int(((orders.is_repaid==1)|(orders.loan_status_code==7)).sum())
    (ROOT/'analysis/payment_source_audit.json').write_text(json.dumps(payment_audit,ensure_ascii=False,indent=2))
    return orders,credits,payments,None

def loan_features(orders,payments,credits,t,settled_serial,current,previous):
    out={}; hist=orders.loc[orders.apply_time<t]

    numeric=['apply_amt','remit_amt','credit_amt','utilization','fee_rate','daily_rate','loan_day','total_term']

    for col in ['early_days','apply_settle_days','remit_settle_days','settle_payment_n','settle_payment_days','settle_cash_remit_ratio']:
        describe(out,'repay_'+col,closed[col].to_numpy(),True)

    out['repay_ontime_to_early_n']=int(((early_values[:-1]==0)&(early_values[1:]>0)).sum()) if len(early_values)>1 else 0
    out['repay_early_to_late_n']=int(((early_values[:-1]>0)&(early_values[1:]<0)).sum()) if len(early_values)>1 else 0
    out['repay_late_to_early_n']=int(((early_values[:-1]<0)&(early_values[1:]>0)).sum()) if len(early_values)>1 else 0

    for at in hist.apply_time:

    for w in [1,3,7,14,30,90,180]:
        g=cash.loc[cash.payment_time.ge(t-pd.Timedelta(days=w))]
        out[f'cash_rows_{w}d']=g.source_rows.sum();out[f'cash_days_{w}d']=g.payin_date.nunique();out[f'cash_amount_{w}d']=g.amount.sum()
    eligible=credits.loc[credits.credit_time.le(t)].drop_duplicates('serial_id',keep='last')

    out['offer_current_time_record_missing']=int(not current_records.any())
    if current_records.any():
        batch=(all_passed.credit_time.diff().gt(pd.Timedelta(seconds=20))|all_passed.credit_time.diff().isna()).cumsum()
        start=all_passed.loc[batch.eq(batch.iloc[-1]),'credit_time'].min()
        out['offer_current_batch_known_products_n']=int(batch.eq(batch.iloc[-1]).sum())
        # Earlier products in this same automatic batch are current conditions, not prior offer history.
        offer=eligible.loc[eligible.credit_time.lt(start)]
    else:
        out['offer_current_batch_known_products_n']=np.nan

    out['trigger_flow_time_missing']=int(not len(trigger) or pd.isna(trigger.flow_settle_time.iloc[-1]))
    return out

def behavior_features(daily,prefix,payments,t,reg,same_second_settled_serials=()):
    out={};day=t.normalize();hist=daily.loc[daily.create_date<day]
    # Only verified payment times determine repayment days; invalid timestamps remain an audited exclusion.
    known_payment_cutoff=payments.payment_time.lt(t)
    if same_second_settled_serials and 'serial_id' in payments:
        known_payment_cutoff=known_payment_cutoff|(payments.payment_time.eq(t)&payments.serial_id.isin(same_second_settled_serials))
    paid_dates=set(payments.loc[payments.timestamp_valid&known_payment_cutoff&payments.payin_date.lt(day),'payin_date'])
    if payments.loc[payments.timestamp_valid&known_payment_cutoff&payments.payin_date.eq(day)].shape[0]:paid_dates.add(day)
    if prefix is not None:
        dates=np.r_[hist.create_date.to_numpy('datetime64[ns]'),day.to_datetime64()]
        values=np.vstack([hist[BEH].to_numpy(float),[prefix[c] for c in BEH]])
        last_time=np.datetime64(prefix['last_active_time'],'ns') if pd.notna(prefix['last_active_time']) else np.datetime64('NaT','ns')
        last=np.r_[hist.last_active_time.to_numpy('datetime64[ns]'),last_time]
    else:
        dates=hist.create_date.to_numpy('datetime64[ns]');values=hist[BEH].to_numpy(float);last=hist.last_active_time.to_numpy('datetime64[ns]')
    observation_start=max(pd.Timestamp('2024-01-01'),reg) if pd.notna(reg) else pd.Timestamp('2024-01-01')
    out['app_history_left_censored']=int(pd.notna(reg) and reg<pd.Timestamp('2024-01-01'))
    out['app_history_observation_days']=max(0,(day-observation_start).days+1)
    active=(values[:,0]>0)&~np.isnat(last)
    out['app_days_since_last_active']=float((t.to_datetime64()-last[active].max())/np.timedelta64(1,'D')) if active.any() else np.nan
    repayflags=np.isin(dates,np.asarray(list(paid_dates),dtype='datetime64[ns]'))
    for branch in ['all','nonpay']:
        allowed=np.ones(len(dates),dtype=bool) if branch=='all' else ~repayflags
        for w in WINDOWS+['life']:
            start=observation_start if w=='life' else max(observation_start,day-pd.Timedelta(days=w-1))
            ix=allowed&(dates>=start.to_datetime64());sums=dict(zip(BEH,values[ix].sum(axis=0)));days=max(0,(day-start).days+1)
            excluded=sum(start<=d<=day for d in paid_dates) if branch=='nonpay' else 0
            eligible=max(0,days-excluded);pre=f'app_{branch}_{w}d_'
            out[pre+'eligible_days']=eligible;out[pre+'empty_eligible']=int(eligible==0)
            out[pre+'active_days']=int((values[ix,0]>0).sum())
            out[pre+'active_day_share']=safe(out[pre+'active_days'],eligible)
            for col in BEH:out[pre+col]=float(sums[col]) if eligible>0 else np.nan
            out[pre+'non_apply_duration']=float(sums['duration']-sums['apply_duration']) if eligible>0 else np.nan
            out[pre+'events_per_eligible_day']=safe(sums['events'],eligible)
            out[pre+'duration_per_active_day']=safe(sums['duration'],out[pre+'active_days'])
            out[pre+'apply_duration_share']=safe(sums['apply_duration'],sums['duration'])
            out[pre+'amount_per_newpage']=safe(sums['amount'],sums['newpage'])
            out[pre+'plan_per_newpage']=safe(sums['plan'],sums['newpage'])
            out[pre+'confirm_per_newpage']=safe(sums['confirm'],sums['newpage'])
            out[pre+'coupon_accept_share']=safe(sums['coupon_confirm'],sums['coupon_confirm']+sums['coupon_back'])
        prior_mask=allowed&(dates>=(day-pd.Timedelta(days=13)).to_datetime64())&(dates<(day-pd.Timedelta(days=6)).to_datetime64())
        prior=dict(zip(BEH,values[prior_mask].sum(axis=0)))
        for col in ['events','newpage','amount','plan','term','confirm','duration']:
            now=out[f'app_{branch}_7d_{col}'];before=prior[col]
            out[f'app_{branch}_{col}_7d_delta_prior7']=now-before
            out[f'app_{branch}_{col}_7d_ratio_prior7']=safe(now,before)
        for col in ['events','newpage','amount','plan','duration']:
            out[f'app_{branch}_{col}_3d_perday_vs14d']=safe(out[f'app_{branch}_3d_{col}'],out[f'app_{branch}_3d_eligible_days'])-safe(out[f'app_{branch}_14d_{col}'],out[f'app_{branch}_14d_eligible_days'])
    for w in [7,30,90]:
        for col in ['events','newpage','amount','plan','duration']:
            out[f'app_nonpay_{col}_share_{w}d']=safe(out[f'app_nonpay_{w}d_{col}'],out[f'app_all_{w}d_{col}'])
    out['app_t0_day_repaid_before_prediction']=int(day in paid_dates)
    return out

def dedup(frame):
    """Only call with records already < as_of. No future serial update can alter past."""
    if len(frame)==0:return frame.copy()
    x=frame.sort_values(['credit_time','serial_id','export_id']).drop_duplicates('serial_id',keep='last')
    x=x.sort_values(['credit_time','serial_id']);batch=x.credit_time.diff().gt(pd.Timedelta(seconds=20)).cumsum()
    return x.assign(batch=batch.to_numpy()).groupby('batch',sort=False).tail(1).drop(columns='batch')

def scope_mask(times,scope,t):
    n=len(times);m=np.ones(n,bool)
    if scope.startswith('last'):m[:max(0,n-int(scope[4:]))]=False
    elif scope.endswith('d'):m=np.asarray(times>=t-pd.Timedelta(days=int(scope[:-1])))
    return m

def known_orders(orders,t):
    return orders.loc[orders.apply_time.lt(t)].sort_values(['apply_time','serial_id'])

def t0_history(offers,orders,t):
    candidates=dedup(offers.loc[offers.has_type3.eq(1)&offers.loan_type_code.eq(2)])
    funded=orders.loc[orders.is_remit.eq(1)&orders.remit_amt.gt(0)&orders.remit_time.le(t)]
    closed=funded.loc[((funded.is_repaid==1)|(funded.loan_status_code==7))&funded.repaid_time.gt(pd.Timestamp('2000-01-01'))&funded.repaid_time.lt(t)]
    dates={d:g.sort_values(['repaid_time','serial_id']) for d,g in closed.groupby('repaid_date',sort=False)}
    good=[]
    for c in candidates.itertuples():
        g=dates.get(c.vir_date)
        if g is None:continue
        g=g.loc[g.repaid_time.le(c.credit_time)]
        eligible=[]
        for o in g.itertuples():
            active=funded.loc[funded.serial_id.ne(o.serial_id)&funded.remit_time.le(o.repaid_time)&
                  (funded.repaid_time.isna()|funded.repaid_time.le(pd.Timestamp('2000-01-01'))|funded.repaid_time.gt(o.repaid_time))]
            if len(active)==0:eligible.append(o)
        if eligible:
            p=eligible[-1];good.append({'serial_id':c.serial_id,'credit_time':c.credit_time,'vir_date':c.vir_date,'settled_serial_id':p.serial_id})
    return pd.DataFrame(good,columns=['serial_id','credit_time','vir_date','settled_serial_id'])

class Profiles:
    def __init__(self,frame):
        frame=frame.copy();frame.profile_date=pd.to_datetime(frame.profile_date)
        self.data={}
        for code,g in frame.groupby('product_no',sort=False):
            g=g.sort_values('profile_date')
            numeric=['orders','remit_sum','fee_sum','day_sum','day_n','term_sum','term_n']
            self.data[str(code)]=(g.profile_date.to_numpy('datetime64[D]'),g[numeric].to_numpy(float).cumsum(axis=0))

    @lru_cache(maxsize=100000)
    def get(self,code,day):
        if code not in self.data:return (np.nan,np.nan,np.nan,np.nan,0)
        dates,agg=self.data[code];i=np.searchsorted(dates,np.datetime64(day,'D'),side='left')-1
        if i<0:return (np.nan,np.nan,np.nan,np.nan,0)
        n,remit,fee,days,dn,terms,tn=agg[i]
        fr=ratio(fee,remit);dy=ratio(days,dn);tm=ratio(terms,tn)
        return fr,ratio(fr,dy),dy,tm,n

def fit_encoder(frame):
    counts={}
    for s in frame.risk_product:
        for p in tokens(s):counts[p]=counts.get(p,0)+1
    known=sorted(counts)
    combos=sorted({','.join(tokens(s)) for s in frame.risk_product})
    mapping={c:{v:i+1 for i,v in enumerate(sorted((frame[c].map(lambda s:','.join(tokens(s))) if c=='risk_product' else frame[c].fillna('SOURCE_MISSING').astype(str)).unique()))} for c in CATS}
    return {'product_codes':known,'product_counts':counts,'product_combos':combos,'categories':mapping,
        'unknown_category_code':0,'note':'0 reserved for unseen; flags distinguish source absence; no future-category fit'}

def encode(frame,encoder):
    codepos={p:j for j,p in enumerate(encoder['product_codes'])};known=set(codepos);combos=set(encoder['product_combos'])
    names=['product_present_'+p for p in encoder['product_codes']]+['product_unknown_count','product_unknown_share','product_combo_unseen']
    names += [x for c in CATS for x in [c+'_encoded',c+'_unseen',c+'_source_missing']]
    x=np.zeros((len(frame),len(names)),dtype='float32');offset=len(codepos)
    for i,s in enumerate(frame.risk_product):
        ps=tokens(s)
        for p in ps:
            if p in codepos:x[i,codepos[p]]=1
        unseen=sum(p not in known for p in ps)
        x[i,offset:offset+3]=[unseen,unseen/len(ps) if ps else np.nan,int(','.join(ps) not in combos)]
    for j,c in enumerate(CATS):
        source=frame[c];vals=source.map(lambda s:','.join(tokens(s))) if c=='risk_product' else source.fillna('SOURCE_MISSING').astype(str);mp=encoder['categories'][c]
        at=offset+3+3*j
        x[:,at]=vals.map(mp).fillna(0).to_numpy(float)
        x[:,at+1]=(~vals.isin(mp)).to_numpy(int)
        x[:,at+2]=(source.isna()|vals.isin(['','SOURCE_MISSING','MISSING'])).to_numpy(int)

def apply_calibration(hazards,calibrators):
    z=hazards.copy()
    for j,model in calibrators.items():
        if model is None:continue
        p=np.clip(z[:,int(j)],1e-7,1-1e-7);logit=np.log(p/(1-p))
        z[:,int(j)]=model.predict_proba(logit[:,None])[:,1]
    return np.clip(z,1e-7,1-1e-7)

    pos={n:i for i,n in enumerate(bundle['features'])}

def main():
    path=ROOT/'models/t0_application_timing_v4.joblib'
    bundle=joblib.load(path);calibrators=bundle.get('calibrators',{})
    tag=bundle['selected_experiment']['tag'];df=base_frame();ys=[];ps=[];intervals=[]
    for name in ['fold1','fold2']:
        z=np.load(ROOT/'analysis'/f'{tag}__{name}_oof.npz')
        r=expand(df.iloc[z['rows']].assign(local_row=np.arange(len(z['rows']))))
        ps.append(z['hazard'][r.local_row.to_numpy(int),r.interval_id.to_numpy(int)])
        ys.append(r.hazard_label.to_numpy(float));intervals.append(r.interval_id.to_numpy(int))
    ph=np.concatenate(ps);target=np.concatenate(ys);iv=np.concatenate(intervals)
    assert np.isfinite(ph).all() and np.isfinite(target).all()
    checks=[]
    for j,model in calibrators.items():
        mask=iv==int(j);p=np.clip(ph[mask],1e-7,1-1e-7);x=np.log(p/(1-p));y=target[mask]
        theta=np.r_[model.coef_.ravel(),model.intercept_.ravel()]
        assert theta.shape==(2,) and np.isfinite(theta).all() and theta[0]>0
        # Same sklearn C=1 L2 objective; scalar reductions avoid the warning's matrix multiply.
        def objective(t):
            linear=t[0]*x+t[1]
            return float(np.logaddexp(0,linear).sum()-np.sum(y*linear)+.5*t[0]**2/model.C)
        def gradient(t):
            residual=expit(t[0]*x+t[1])-y
            return np.array([np.sum(residual*x)+t[0]/model.C,np.sum(residual)])
        reference=minimize(objective,np.array([1.,0.]),jac=gradient,method='L-BFGS-B',
            options={'maxiter':500,'ftol':1e-14,'gtol':1e-8})
        assert np.isfinite(reference.x).all()
        saved=expit(theta[0]*x+theta[1]);ref=expit(reference.x[0]*x+reference.x[1])
        error=float(np.max(np.abs(saved-ref)))
        loss_gap=float((objective(theta)-objective(reference.x))/len(y))
        assert error<1e-3 and loss_gap<1e-7,(j,error,loss_gap)
        direct=model.predict_proba(x[:,None])[:,1]
        assert np.isfinite(direct).all() and np.max(np.abs(direct-saved))<1e-10
        checks.append(dict(interval=int(j),n=len(y),coefficients_finite=True,slope_positive=True,
            max_probability_difference_from_independent_fit=error,mean_objective_gap=loss_gap,
            stable_mean_gradient_max=float(np.max(np.abs(gradient(theta)))/len(y))))
