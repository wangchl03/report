六、结论与运营方向

结论 1：获额后的即时阶段是当天提单的核心窗口。
获额后 3 小时内贡献 41.59% 个百分点的提单率，占 24 小时总提单率的 78.60%。应将获额通知、落地页跳转、金额和期数选择、风控确认等链路作为优先排查对象。
结论 2：6–24 小时窗口仍有可运营承接空间。
该时段累计贡献 11.33% 个百分点。可针对已活跃、已进入提单页、已浏览额度但未提交的用户定向测试降息券或提额券，不宜对所有未提单用户统一发券。
结论 3：月度变化需拆解到小时段判断原因。
0–1 小时段下降更明显，优先排查即时产品吸引力和页面链路。

七、统计口径

结清 0 在贷样本：获额通过订单先按用户 20 秒获额批次去重，每批保留订单号最大的获额订单；再要求用户结清时无其他在途订单、结清日等于获额通过日、获额时间大于等于结清时间；同一用户同日多次满足时，仅保留获额时间最早的符合条件订单。

首次提单：从对应获额通过时间起，取 24 小时内最早的一笔提单；按 FLOOR((apply_time − credit_vir_time) / 3600) 分入 0–1、1–2 … 23–24 小时桶。每个用户日期样本最多归入一个小时桶。各桶提单率 = 对应小时桶首次提单用户数 ÷ 当天结清 0 在贷用户数；24 小时总提单率为 24 个小时桶首次提单人数之和 ÷ 同一分母。

附录：完整 PGSQL

以下代码按模块一、模块二、模块三顺序在同一 PostgreSQL 数据库会话内执行，可复现本报告的结清 0 在贷样本、获额后 0–24 小时首次提单人数、各小时段提单率、24 小时总提单率，以及日度、月度、总计数据。

模块一：结清 0 在贷样本（20 秒获额去重、结清时点无在途判断）
-- 模块一：结清 0 在贷样本
-- 统计期：2026-01-01 至 2026-08-30；apply_end 额外保留至 2026-09-01，
-- 用于观察 8 月 30 日获额后的完整 24 小时提单。

CREATE TEMP TABLE params AS
SELECT DATE '2026-01-01' AS start_date,
       DATE '2026-08-31' AS end_date,
       TIMESTAMP '2026-09-01 00:00:00' AS apply_end;

-- 1. 只保留统计期内实际发生结清的“用户 × 结清日”，用于同日获额预筛。
CREATE TEMP TABLE settle_candidate_days AS
SELECT DISTINCT
       o.user_id,
       o.repaid_date::date AS repaid_date
FROM wangchuanliang.order_loan_f_v2_copy o
CROSS JOIN params p
WHERE o.is_remit = 1
  AND o.loan_status_code = 7
  AND o.repaid_time > TIMESTAMP '2000-01-01 00:00:00'
  AND o.repaid_date >= p.start_date
  AND o.repaid_date < p.end_date;

CREATE INDEX ON settle_candidate_days (user_id, repaid_date);

-- 2. 获额通过订单：同一 user_id + serial_id 仅留最新获额记录。
CREATE TEMP TABLE raw_credit_pass AS
SELECT DISTINCT ON (v.user_id, v.serial_id)
       v.user_id,
       v.serial_id,
       v.vir_date::date AS vir_date,
       to_timestamp(v.vir_unix) AS vir_time
FROM wangchuanliang.order_vir_f_copy v
JOIN settle_candidate_days d
  ON d.user_id = v.user_id
 AND d.repaid_date = v.vir_date::date
CROSS JOIN params p
WHERE v.is_pass1 = 1
  AND v.vir_date >= p.start_date
  AND v.vir_date < p.end_date
ORDER BY v.user_id,
         v.serial_id,
         v.vir_unix DESC;

CREATE INDEX ON raw_credit_pass (user_id, vir_time);

-- 3. 20 秒获额批次：相邻获额间隔不超过 20 秒视为同一批次。
CREATE TEMP TABLE credit_pass_batch AS
WITH ordered AS (
    SELECT r.*,
           LAG(r.vir_time) OVER (
               PARTITION BY r.user_id
               ORDER BY r.vir_time, r.serial_id
           ) AS prev_vir_time
    FROM raw_credit_pass r
), batched AS (
    SELECT o.*,
           SUM(
               CASE
                   WHEN o.prev_vir_time IS NULL
                     OR o.vir_time - o.prev_vir_time > INTERVAL '20 seconds'
                   THEN 1
                   ELSE 0
               END
           ) OVER (
               PARTITION BY o.user_id
               ORDER BY o.vir_time, o.serial_id
           ) AS batch_id
    FROM ordered o
)
SELECT *
FROM batched;

CREATE INDEX ON credit_pass_batch (user_id, batch_id);

-- 4. 每个 20 秒批次仅保留 serial_id 最大的获额订单。
CREATE TEMP TABLE dedup_credit_pass AS
SELECT user_id,
       serial_id,
       vir_date,
       vir_time
FROM (
    SELECT b.*,
           ROW_NUMBER() OVER (
               PARTITION BY b.user_id, b.batch_id
               ORDER BY b.serial_id DESC, b.vir_time DESC
           ) AS rn
    FROM credit_pass_batch b
) x
WHERE x.rn = 1;

CREATE INDEX ON dedup_credit_pass (user_id, vir_date, vir_time);

-- 5. 统计期内结清的放款订单。
CREATE TEMP TABLE settled_orders AS
SELECT o.user_id,
       o.serial_id,
       o.apply_time,
       o.repaid_time,
       o.repaid_date::date AS repaid_date
FROM wangchuanliang.order_loan_f_v2_copy o
CROSS JOIN params p
WHERE o.is_remit = 1
  AND o.loan_status_code = 7
  AND o.repaid_time > TIMESTAMP '2000-01-01 00:00:00'
  AND o.repaid_date >= p.start_date
  AND o.repaid_date < p.end_date;

CREATE INDEX ON settled_orders (user_id, repaid_date, repaid_time);

CREATE TEMP TABLE candidate_users AS
SELECT DISTINCT user_id
FROM settled_orders;

-- 6. 用户订单全集。apply_end 至 9 月 1 日，保证 8 月 30 日样本可观察完整 24 小时。
CREATE TEMP TABLE all_orders AS
SELECT o.user_id,
       o.serial_id,
       o.apply_time,
       o.repaid_time,
       o.repaid_date::date AS repaid_date,
       o.loan_status_code,
       o.is_remit
FROM wangchuanliang.order_loan_f_v2_copy o
JOIN candidate_users u
  ON u.user_id = o.user_id
CROSS JOIN params p
WHERE o.apply_time < p.apply_end;

CREATE INDEX ON all_orders (user_id, apply_time);
CREATE INDEX ON all_orders (user_id, repaid_time);

-- 7. 干净结清：结清时点不存在其他在途订单。
CREATE TEMP TABLE clean_settles AS
SELECT s.*
FROM settled_orders s
WHERE NOT EXISTS (
    SELECT 1
    FROM all_orders o2
    WHERE o2.user_id = s.user_id
      AND o2.serial_id <> s.serial_id
      AND o2.apply_time < s.repaid_time
      AND (
             o2.loan_status_code IN (5, 8)
          OR (
                 o2.loan_status_code = 7
             AND o2.repaid_time > s.repaid_time
             )
      )
);

CREATE INDEX ON clean_settles (user_id, repaid_date, repaid_time);

-- 8. 结清 0 在贷：同日获额通过且获额时间不早于结清时间；
--    同一用户同日多次满足时，仅保留获额时间最早的一笔。
CREATE TEMP TABLE t0_samples AS
SELECT user_id,
       final_serial_id,
       final_apply_time,
       final_repaid_time,
       repaid_date,
       credit_serial_id,
       credit_vir_time
FROM (
    SELECT c.user_id,
           c.serial_id AS final_serial_id,
           c.apply_time AS final_apply_time,
           c.repaid_time AS final_repaid_time,
           c.repaid_date,
           cp.serial_id AS credit_serial_id,
           cp.vir_time AS credit_vir_time,
           ROW_NUMBER() OVER (
               PARTITION BY c.user_id, c.repaid_date
               ORDER BY cp.vir_time ASC,
                        cp.serial_id DESC,
                        c.repaid_time ASC
           ) AS rn
    FROM clean_settles c
    JOIN dedup_credit_pass cp
      ON cp.user_id = c.user_id
     AND cp.vir_date = c.repaid_date
     AND cp.vir_time >= c.repaid_time
) x
WHERE x.rn = 1;

CREATE INDEX ON t0_samples (user_id, repaid_date);
模块二：获额后 24 小时首次提单、小时分布、日度/月度/总计明细
-- 模块二：获额后 24 小时首次提单、小时分布、日度/月度/总计汇总
-- 请在模块一完成后、同一数据库会话内执行。

-- 1. 每个“用户 × 结清日样本”仅保留获额后 24 小时内最早的一笔提单。
CREATE TEMP TABLE first_apply_24h AS
SELECT user_id,
       repaid_date,
       credit_vir_time,
       first_apply_time,
       FLOOR(
           EXTRACT(EPOCH FROM (first_apply_time - credit_vir_time)) / 3600.0
       )::int AS hour_bucket
FROM (
    SELECT t.user_id,
           t.repaid_date,
           t.credit_vir_time,
           a.apply_time AS first_apply_time,
           ROW_NUMBER() OVER (
               PARTITION BY t.user_id, t.repaid_date
               ORDER BY a.apply_time, a.serial_id
           ) AS rn
    FROM t0_samples t
    JOIN all_orders a
      ON a.user_id = t.user_id
     AND a.apply_time >= t.credit_vir_time
     AND a.apply_time < t.credit_vir_time + INTERVAL '24 hours'
) x
WHERE x.rn = 1;

-- 2. 每日结清 0 在贷样本量。
CREATE TEMP TABLE daily_base AS
SELECT repaid_date,
       COUNT(*) AS t0_sample_users
FROM t0_samples
GROUP BY repaid_date;

-- 3. 日度 24 个小时桶；没有提单的小时桶也补 0。
CREATE TEMP TABLE daily_hourly AS
SELECT d.repaid_date,
       h.hour_bucket,
       d.t0_sample_users,
       COUNT(f.user_id) AS first_apply_users
FROM daily_base d
CROSS JOIN generate_series(0, 23) AS h(hour_bucket)
LEFT JOIN first_apply_24h f
  ON f.repaid_date = d.repaid_date
 AND f.hour_bucket = h.hour_bucket
GROUP BY d.repaid_date,
         h.hour_bucket,
         d.t0_sample_users;

-- 4. 一次性输出报告所用的总计、月度与日度数据。
WITH daily AS (
    SELECT '日度'::text AS statistic_level,
           repaid_date::text AS statistic_period,
           repaid_date,
           hour_bucket,
           t0_sample_users,
           first_apply_users
    FROM daily_hourly
), monthly AS (
    SELECT '月度'::text AS statistic_level,
           to_char(date_trunc('month', repaid_date), 'YYYY-MM') AS statistic_period,
           NULL::date AS repaid_date,
           hour_bucket,
           SUM(t0_sample_users) AS t0_sample_users,
           SUM(first_apply_users) AS first_apply_users
    FROM daily_hourly
    GROUP BY to_char(date_trunc('month', repaid_date), 'YYYY-MM'),
             hour_bucket
), total AS (
    SELECT '总计'::text AS statistic_level,
           '总计'::text AS statistic_period,
           NULL::date AS repaid_date,
           hour_bucket,
           SUM(t0_sample_users) AS t0_sample_users,
           SUM(first_apply_users) AS first_apply_users
    FROM daily_hourly
    GROUP BY hour_bucket
), result AS (
    SELECT * FROM total
    UNION ALL
    SELECT * FROM monthly
    UNION ALL
    SELECT * FROM daily
)
SELECT statistic_level AS "统计层级",
       statistic_period AS "统计期间",
       hour_bucket AS "获额后小时桶",
       t0_sample_users AS "结清0在贷用户数",
       first_apply_users AS "首次提单用户数",
       ROUND(
           first_apply_users::numeric
           / NULLIF(t0_sample_users, 0) * 100,
           4
       ) AS "该小时段提单率_pct",
       ROUND(
           SUM(first_apply_users) OVER (
               PARTITION BY statistic_level, statistic_period
           )::numeric
           / NULLIF(
               MAX(t0_sample_users) OVER (
                   PARTITION BY statistic_level, statistic_period
               ),
               0
           ) * 100,
           4
       ) AS "获额后24小时总提单率_pct"
FROM result
ORDER BY CASE statistic_level
             WHEN '总计' THEN 1
             WHEN '月度' THEN 2
             ELSE 3
         END,
         statistic_period,
         hour_bucket;
模块三：报告数字卡、月度趋势、四段时效指标与总计小时分布
-- 模块三：报告数字卡、月度趋势和四段时效指标
-- 请在模块一、模块二完成后、同一数据库会话内执行。
-- 本模块直接输出报告中的核心数字卡、月度可核对表和小时分布图数据。

-- 1. 月度分母：结清 0 在贷用户日样本。
CREATE TEMP TABLE monthly_t0_base AS
SELECT to_char(date_trunc('month', repaid_date), 'YYYY-MM') AS settle_month,
       SUM(t0_sample_users) AS t0_sample_users
FROM daily_base
GROUP BY to_char(date_trunc('month', repaid_date), 'YYYY-MM');

-- 2. 月度分子：按报告的四个时效区间汇总首次提单人数。
CREATE TEMP TABLE monthly_t0_apply AS
SELECT to_char(date_trunc('month', repaid_date), 'YYYY-MM') AS settle_month,
       SUM(CASE WHEN hour_bucket = 0 THEN first_apply_users ELSE 0 END)
           AS apply_user_0_1h,
       SUM(CASE WHEN hour_bucket BETWEEN 1 AND 2 THEN first_apply_users ELSE 0 END)
           AS apply_user_1_3h,
       SUM(CASE WHEN hour_bucket BETWEEN 3 AND 5 THEN first_apply_users ELSE 0 END)
           AS apply_user_3_6h,
       SUM(CASE WHEN hour_bucket BETWEEN 6 AND 23 THEN first_apply_users ELSE 0 END)
           AS apply_user_6_24h,
       SUM(first_apply_users) AS apply_user_24h
FROM daily_hourly
GROUP BY to_char(date_trunc('month', repaid_date), 'YYYY-MM');

-- 3. 月度表与“24 小时总提单率”“0–1 小时提单率”等趋势图数据。
SELECT b.settle_month AS "结清月份",
       b.t0_sample_users AS "结清0在贷用户日样本",
       a.apply_user_0_1h AS "0_1小时首次提单用户数",
       ROUND(a.apply_user_0_1h::numeric / NULLIF(b.t0_sample_users, 0) * 100, 4)
           AS "0_1小时提单率_pct",
       a.apply_user_1_3h AS "1_3小时首次提单用户数",
       ROUND(a.apply_user_1_3h::numeric / NULLIF(b.t0_sample_users, 0) * 100, 4)
           AS "1_3小时提单率_pct",
       a.apply_user_3_6h AS "3_6小时首次提单用户数",
       ROUND(a.apply_user_3_6h::numeric / NULLIF(b.t0_sample_users, 0) * 100, 4)
           AS "3_6小时提单率_pct",
       a.apply_user_6_24h AS "6_24小时首次提单用户数",
       ROUND(a.apply_user_6_24h::numeric / NULLIF(b.t0_sample_users, 0) * 100, 4)
           AS "6_24小时提单率_pct",
       a.apply_user_24h AS "24小时首次提单用户数",
       ROUND(a.apply_user_24h::numeric / NULLIF(b.t0_sample_users, 0) * 100, 4)
           AS "24小时总提单率_pct"
FROM monthly_t0_base b
JOIN monthly_t0_apply a
  ON a.settle_month = b.settle_month
ORDER BY b.settle_month;

-- 4. 全期间核心数字卡及四段时效汇总。
WITH total_base AS (
    SELECT SUM(t0_sample_users) AS t0_sample_users
    FROM daily_base
), total_apply AS (
    SELECT SUM(CASE WHEN hour_bucket = 0 THEN first_apply_users ELSE 0 END)
               AS apply_user_0_1h,
           SUM(CASE WHEN hour_bucket BETWEEN 1 AND 2 THEN first_apply_users ELSE 0 END)
               AS apply_user_1_3h,
           SUM(CASE WHEN hour_bucket BETWEEN 3 AND 5 THEN first_apply_users ELSE 0 END)
               AS apply_user_3_6h,
           SUM(CASE WHEN hour_bucket BETWEEN 6 AND 23 THEN first_apply_users ELSE 0 END)
               AS apply_user_6_24h,
           SUM(first_apply_users) AS apply_user_24h
    FROM daily_hourly
)
SELECT b.t0_sample_users AS "结清0在贷用户日样本",
       a.apply_user_24h AS "24小时首次提单用户数",
       ROUND(a.apply_user_24h::numeric / NULLIF(b.t0_sample_users, 0) * 100, 4)
           AS "24小时总提单率_pct",
       a.apply_user_0_1h AS "0_1小时首次提单用户数",
       ROUND(a.apply_user_0_1h::numeric / NULLIF(b.t0_sample_users, 0) * 100, 4)
           AS "0_1小时提单率_pct",
       ROUND(
           (a.apply_user_0_1h + a.apply_user_1_3h)::numeric
           / NULLIF(b.t0_sample_users, 0) * 100,
           4
       ) AS "0_3小时累计提单率_pct",
       ROUND(a.apply_user_3_6h::numeric / NULLIF(b.t0_sample_users, 0) * 100, 4)
           AS "3_6小时提单率_pct",
       ROUND(a.apply_user_6_24h::numeric / NULLIF(b.t0_sample_users, 0) * 100, 4)
           AS "6_24小时提单率_pct"
FROM total_base b
CROSS JOIN total_apply a;

-- 5. 总计小时分布图和小时明细表数据。
WITH total_base AS (
    SELECT SUM(t0_sample_users) AS t0_sample_users
    FROM daily_base
)
SELECT h.hour_bucket AS "获额后小时桶",
       SUM(h.first_apply_users) AS "首次提单用户数",
       ROUND(
           SUM(h.first_apply_users)::numeric
           / NULLIF(MAX(b.t0_sample_users), 0) * 100,
           4
       ) AS "该小时段提单率_pct"
FROM daily_hourly h
CROSS JOIN total_base b
GROUP BY h.hour_bucket
ORDER BY h.hour_bucket;
