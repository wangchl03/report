三关键对比

覆盖量：非提单页 0–30 秒覆盖 35.66%～37.53%，远高于提单页 120–300 秒的 4.18%～6.42%。前者适合大规模识别潜在意图，后者是更小的强意图人群。
转化强度：提单页长停留（120–300 秒）提单率较高（T3/T5/T7：81.40%/77.08%/72.58%）；非提单页 60–120 秒已达 79.87%/77.32%/72.27%，与之非常接近。300 秒以上分段样本占比不足 2%，提单率波动大，不作稳定比较。
行为路径：仅提单页停留的样本仅 0.05%～0.07%，不能视为稳定客群。绝大部分有提单页停留者同时浏览了非提单页，用户并非直达提单页，而是先/同时经历信息浏览和决策过程。

四结论与行动建议

结论 1：非提单页浏览是独立且规模可观的强信号。 仅非提单页停留的第二阶段提单率为 T3 76.04%、T5 71.07%、T7 67.04%，比两类页面均无停留高 16.88 / 18.62 / 20.15 个百分点，且覆盖约 11%。这说明额度、期次、计划等信息浏览本身反映了明确借款意图。
结论 2：两类页面均浏览覆盖最大、转化更快。 该组覆盖 34.28% / 38.06% / 40.03%，提单率 73.29% / 68.23% / 62.92%；平均提单间隔为 14.57 / 17.98 / 21.54 天，较“仅非提单页”分别快 4.67 / 5.89 / 6.48 天。可视为更接近完成决策的用户。
结论 3：不要把提单页作为唯一意图判断标准。 非提单页 0–30 秒已经带来 14.03～15.45 个百分点的提单率提升，且样本是提单页长停留群的约 5～9 倍。运营可先用非提单页短时浏览构建广覆盖池，再叠加提单页行为提升触达优先级。
实验设计标准：① 两类均无停留：召回与基础提醒；② 仅非提单页：额度、期次、还款计划引导；③ 两类均停留或提单页 ≥60 秒：低频、高优先级的转化提醒；④ 后续需联合评估放款、逾期和最终盈利额的净增量，避免只优化条件提单率。

这是行为关联，不代表延长页面停留本身会造成提单；资质、额度和触达等因素也可能同时影响行为与提单。

数据来源：order_loan_f_v2_copy、order_vir_f_copy、event_track。

附录：提单页与非提单页停留时长完整 PGSQL

展开完整代码与注释说明
/*
  提单页 / 非提单页第一阶段停留时长与第二阶段提单
  自然日口径：D0=结清日；T3/T5/T7 第一阶段分别为 D+1~D+3、D+1~D+5、D+1~D+7。
*/
WITH
/* 1. 注册时间范围内的全部已提单订单。 */
all_orders AS (
    SELECT
        user_id,
        serial_id,
        apply_time,
        remit_time,
        repaid_time,
        repaid_date,
        remit_date,
        loan_status_code,
        is_remit,
        reg_date
    FROM wangchuanliang.order_loan_f_v2_copy
    WHERE reg_date >= DATE '2026-01-01'
      AND reg_date <= DATE '2026-07-15'
      AND apply_time IS NOT NULL
),

/* 2. 排除仅提单、未放款的用户。 */
all_users AS (
    SELECT DISTINCT user_id
    FROM all_orders
    WHERE is_remit = 1
      AND remit_date > DATE '2026-01-01'
),

/* 3. 正常结清订单，按用户结清时间编号。 */
settled_raw AS (
    SELECT
        o.user_id,
        o.serial_id,
        o.repaid_date,
        o.repaid_time,
        ROW_NUMBER() OVER (
            PARTITION BY o.user_id
            ORDER BY o.repaid_time, o.serial_id
        ) AS settle_seq
    FROM all_orders o
    INNER JOIN all_users u ON u.user_id = o.user_id
    WHERE o.is_remit = 1
      AND o.loan_status_code = 7
      AND o.repaid_time IS NOT NULL
      AND o.remit_time IS NOT NULL
),

/* 4. 干净结清：结清时不存在在途、坏账或尚未结清的更早订单。 */
clean_settle AS (
    SELECT s.*
    FROM settled_raw s
    WHERE NOT EXISTS (
        SELECT 1
        FROM all_orders o2
        WHERE o2.user_id = s.user_id
          AND o2.serial_id <> s.serial_id
          AND o2.apply_time < s.repaid_time
          AND (
              o2.loan_status_code IN (5, 8)
              OR (o2.loan_status_code = 7 AND o2.repaid_time > s.repaid_time)
          )
    )
),

/* 5. 同一用户同一获额日期只保留最晚一次通过记录。 */
credit_pass AS (
    SELECT DISTINCT ON (v.user_id, v.vir_date)
        v.user_id,
        v.vir_date,
        to_timestamp(v.vir_unix) AS vir_time
    FROM wangchuanliang.order_vir_f_copy v
    WHERE v.is_pass1 = 1
      AND v.vir_date > DATE '2026-01-01'
    ORDER BY v.user_id, v.vir_date, to_timestamp(v.vir_unix) DESC
),

/* 6. 结清当日、且结清后获额通过的事件。 */
eligible AS (
    SELECT cs.*
    FROM clean_settle cs
    INNER JOIN credit_pass cp
        ON cp.user_id = cs.user_id
       AND cp.vir_date = cs.repaid_date::date
       AND cp.vir_time > cs.repaid_time
),

/* 7. 构造结清/提单统一时间线。 */
applies AS (
    SELECT user_id, apply_time
    FROM all_orders
    WHERE apply_time > TIMESTAMP '2026-01-01'
),
merged AS (
    SELECT user_id, repaid_time AS ts, 'settle' AS kind, settle_seq::float AS settle_seq
    FROM eligible
    UNION ALL
    SELECT user_id, apply_time AS ts, 'apply' AS kind, NULL::float AS settle_seq
    FROM applies
),
ranked AS (
    SELECT
        *,
        ROW_NUMBER() OVER (
            PARTITION BY user_id
            ORDER BY ts, CASE kind WHEN 'settle' THEN 0 ELSE 1 END
        ) AS rn
    FROM merged
),

/* 8. 每笔结清只匹配时间线中紧邻的下一次提单。 */
pairs AS (
    SELECT
        s.user_id,
        s.ts AS repaid_time,
        s.settle_seq,
        a.ts AS next_apply_time
    FROM ranked s
    LEFT JOIN ranked a
        ON a.user_id = s.user_id
       AND a.rn = s.rn + 1
       AND a.kind = 'apply'
    WHERE s.kind = 'settle'
),

/* 9. 三个自然日窗口参数。 */
thresholds AS (
    SELECT 'T3' AS cohort, 3 AS threshold_days
    UNION ALL SELECT 'T5', 5
    UNION ALL SELECT 'T7', 7
),

/* 10. 排除结清当天和第一阶段已提单；保留第二阶段提单或后续未提单事件。 */
candidates AS (
    SELECT
        t.cohort,
        t.threshold_days,
        p.user_id,
        p.repaid_time::date AS repaid_date,
        CASE WHEN p.next_apply_time IS NULL THEN 0 ELSE 1 END AS phase2_apply,
        CASE
            WHEN p.next_apply_time IS NULL THEN NULL
            ELSE p.next_apply_time::date - p.repaid_time::date
        END AS gap_days
    FROM pairs p
    CROSS JOIN thresholds t
    WHERE p.settle_seq <= 10
      AND (
          p.next_apply_time IS NULL
          OR p.next_apply_time::date > p.repaid_time::date + t.threshold_days
      )
),

/* 11. 汇总第一阶段内两类页面停留时长（秒）。 */
durations AS (
    SELECT
        c.*,
        COUNT(DISTINCT e.active_date) AS active_days,
        COALESCE(SUM(e.apply_duration), 0) AS apply_duration,
        COALESCE(SUM(e.non_apply_duration), 0) AS non_apply_duration
    FROM candidates c
    LEFT JOIN wangchuanliang.event_track e
        ON e.user_id = c.user_id
       AND e.active_date > c.repaid_date
       AND e.active_date <= c.repaid_date + c.threshold_days
    GROUP BY
        c.cohort, c.threshold_days, c.user_id,
        c.repaid_date, c.phase2_apply, c.gap_days
),

/* 12. 转成长表，以同一逻辑分别统计提单页和非提单页。 */
page_duration_long AS (
    SELECT *, '提单页停留时长' AS page_type, apply_duration AS duration
    FROM durations
    UNION ALL
    SELECT *, '非提单页停留时长', non_apply_duration
    FROM durations
),

/* 13. 对页面停留时长分段；负值、NaN 单列为数据质量异常。 */
bucketed AS (
    SELECT
        *,
        CASE
            WHEN duration::text = 'NaN' THEN '时长缺失/NaN'
            WHEN duration < 0 THEN '负值异常'
            WHEN duration = 0 THEN '0 秒'
            WHEN duration <= 30 THEN '0-30 秒'
            WHEN duration <= 60 THEN '30-60 秒'
            WHEN duration <= 120 THEN '60-120 秒'
            WHEN duration <= 300 THEN '120-300 秒'
            WHEN duration <= 600 THEN '300-600 秒'
            ELSE '600 秒以上'
        END AS duration_bucket,
        CASE
            WHEN duration::text = 'NaN' THEN 0
            WHEN duration < 0 THEN 1
            WHEN duration = 0 THEN 2
            WHEN duration <= 30 THEN 3
            WHEN duration <= 60 THEN 4
            WHEN duration <= 120 THEN 5
            WHEN duration <= 300 THEN 6
            WHEN duration <= 600 THEN 7
            ELSE 8
        END AS bucket_order
    FROM page_duration_long
),

/* 14. 页面联合路径：用于比较“仅非提单页”和“两类页面均停留”。 */
joint AS (
    SELECT
        cohort,
        CASE
            WHEN apply_duration > 0 AND non_apply_duration > 0 THEN '两类页面均停留'
            WHEN apply_duration > 0 THEN '仅提单页停留'
            WHEN non_apply_duration > 0 THEN '仅非提单页停留'
            ELSE '两类页面均无停留'
        END AS behavior_group,
        COUNT(*) AS events,
        SUM(phase2_apply) AS phase2_apply_events,
        ROUND(SUM(phase2_apply) * 100.0 / COUNT(*), 2) AS phase2_apply_rate,
        ROUND(AVG(CASE WHEN phase2_apply = 1 THEN gap_days END)::numeric, 2) AS avg_gap_days,
        ROUND(AVG(apply_duration)::numeric, 1) AS avg_apply_duration,
        ROUND(AVG(non_apply_duration)::numeric, 1) AS avg_non_apply_duration
    FROM durations
    GROUP BY
        cohort,
        CASE
            WHEN apply_duration > 0 AND non_apply_duration > 0 THEN '两类页面均停留'
            WHEN apply_duration > 0 THEN '仅提单页停留'
            WHEN non_apply_duration > 0 THEN '仅非提单页停留'
            ELSE '两类页面均无停留'
        END
)

/* 15. 输出页面时长分段和页面联合路径两类结果。 */
SELECT
    '页面时长分段' AS section,
    page_type,
    cohort,
    duration_bucket,
    COUNT(*) AS events,
    ROUND(COUNT(*) * 100.0 / SUM(COUNT(*)) OVER (PARTITION BY page_type, cohort), 2) AS share_pct,
    SUM(phase2_apply) AS phase2_apply_events,
    ROUND(SUM(phase2_apply) * 100.0 / COUNT(*), 2) AS phase2_apply_rate,
    ROUND(AVG(duration)::numeric, 1) AS avg_duration_sec,
    ROUND(AVG(CASE WHEN phase2_apply = 1 THEN gap_days END)::numeric, 2) AS avg_gap_days,
    NULL::numeric AS avg_apply_duration,
    NULL::numeric AS avg_non_apply_duration
FROM bucketed
GROUP BY page_type, cohort, duration_bucket, bucket_order

UNION ALL

SELECT
    '页面联合分组' AS section,
    NULL AS page_type,
    cohort,
    behavior_group AS duration_bucket,
    events,
    ROUND(events * 100.0 / SUM(events) OVER (PARTITION BY cohort), 2) AS share_pct,
    phase2_apply_events,
    phase2_apply_rate,
    NULL::numeric AS avg_duration_sec,
    avg_gap_days,
    avg_apply_duration,
    avg_non_apply_duration
FROM joint
ORDER BY section, cohort, duration_bucket;

查询结构

  • 订单基础样本
  • 正常/干净结清筛选
  • 同日获额通过匹配
  • 首次后续提单匹配
  • 自然日 T3/T5/T7 窗口划分
  • 页面时长分段、页面联合路径

指标口径

  • D0 是结清日，不计入第一阶段。
  • 页面与活跃分析以 active_date 判断活跃记录。
  • 第二阶段提单率为条件提单率。
  • 注释对应 SQL 中的 CTE 编号。
