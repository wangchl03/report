二、统计口径说明

基础样本用户结清时没有其他在途订单；结清当天获额通过（is_pass1=1）；且获额发生在结清之后。同一用户同日只保留一次。
D0、D1、D2按结清后的首次提单日期划分：结清当日为 D0，次日为 D1，第 2 日为 D2。
T3 / T5 / T7+分别从第 3 天、第 5 天、第 8 天开始观察后续 30 天的首次提单。三组窗口有重叠，人数不能直接相加。

八、建议与后续验证

优先定位 T0 的下滑环节

应将 T0 拆为“获额通过后未进入提单页、进入提单页未提单、提单后未放款”等步骤，判断是产品价格、资格、页面、触达还是需求变化导致。

保留并严谨验证延迟召回

T3/T5/T7+ 固定窗口均呈增长，建议继续分层随机验证，并把增量提单、放款、逾期及盈利同时纳入判定，而不只看提单率。

避免把嵌套窗口重复贡献

T5 是 T3 中从 D5 起的子窗口，T7+ 又是更后段子窗口。对大盘贡献应使用互斥区间（D0、D1、D2、D3+），不能将 T3/T5/T7+ 三者直接相加。

持续使用完整观察口径

大盘趋势优先使用 D0–D29 完整窗口；“截至取数日总提单率”适合作补充观察，不能单独用于判断最近月份的真实变化。

附录：完整 PGSQL

以下三个模块均可独立执行：每个模块都包含完整的基础样本 CTE、首次提单识别与最终统计。统计范围为 2026-01-01 至 2026-08-20 的 T0 用户日期样本。

模块一：总体首次提单时效统计
/* ================================================================
   T0 用户首次提单时效总体统计
   统计范围：2026-01-01 至 2026-08-20

   基础分母：干净结清 + 结清后当日获额通过 + 获额时间晚于结清时间。
   同一用户同日保留最早合格结清订单。
   ================================================================ */
WITH params AS (
    SELECT
        DATE '2026-01-01' AS start_date,
        DATE '2026-08-20' AS end_date,
        CURRENT_DATE AS data_cutoff_date,
        CURRENT_TIMESTAMP AS data_cutoff_time
),

settled_orders AS (
    SELECT
        o.user_id,
        o.serial_id,
        o.repaid_time,
        o.repaid_date::date AS repaid_date
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    CROSS JOIN params AS p
    WHERE o.is_remit = 1
      AND (
            o.is_repaid = 1
         OR o.loan_status_code = 7
         OR o.repaid_date::date > DATE '2000-01-01'
      )
      AND o.repaid_time > TIMESTAMP '2000-01-01 00:00:00'
      AND o.repaid_date::date >= p.start_date
      AND o.repaid_date::date <= p.end_date
),

candidate_users AS (
    SELECT DISTINCT
        s.user_id
    FROM settled_orders AS s
),

all_orders AS (
    SELECT
        o.user_id,
        o.serial_id,
        o.apply_time,
        o.repaid_time,
        o.loan_status_code
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    INNER JOIN candidate_users AS u
        ON u.user_id = o.user_id
    CROSS JOIN params AS p
    WHERE o.apply_time < p.data_cutoff_time
),

clean_settles AS (
    /* 结清瞬间没有任何其他在途订单。 */
    SELECT
        s.user_id,
        s.serial_id,
        s.repaid_time,
        s.repaid_date
    FROM settled_orders AS s
    WHERE NOT EXISTS (
        SELECT
            1
        FROM all_orders AS o2
        WHERE o2.user_id = s.user_id
          AND o2.serial_id <> s.serial_id
          AND o2.apply_time < s.repaid_time
          AND (
                o2.loan_status_code = 5
             OR o2.loan_status_code = 8
             OR (
                    o2.loan_status_code = 7
                AND o2.repaid_time > s.repaid_time
                )
          )
    )
),

same_day_credit_pass AS (
    SELECT
        v.user_id,
        v.vir_date::date AS vir_date,
        TO_TIMESTAMP(v.vir_unix) AS vir_time
    FROM wangchuanliang.order_vir_f_copy AS v
    CROSS JOIN params AS p
    WHERE v.is_pass1 = 1
      AND v.vir_date::date >= p.start_date
      AND v.vir_date::date <= p.end_date
),

denominator_candidates AS (
    SELECT
        c.user_id,
        c.serial_id,
        c.repaid_time,
        c.repaid_date
    FROM clean_settles AS c
    WHERE EXISTS (
        SELECT
            1
        FROM same_day_credit_pass AS cp
        WHERE cp.user_id = c.user_id
          AND cp.vir_date = c.repaid_date
          AND cp.vir_time > c.repaid_time
    )
),

base_sample AS (
    /* 同一用户同日取最早一笔合格结清。 */
    SELECT DISTINCT ON (
        d.user_id,
        d.repaid_date
    )
        d.user_id,
        d.serial_id,
        d.repaid_time,
        d.repaid_date
    FROM denominator_candidates AS d
    ORDER BY
        d.user_id,
        d.repaid_date,
        d.repaid_time ASC
),

first_apply AS (
    /* 仅识别目标结清时间之后的首次提单。 */
    SELECT
        b.user_id,
        b.serial_id,
        b.repaid_date,
        MIN(a.apply_time) AS first_apply_time
    FROM base_sample AS b
    LEFT JOIN all_orders AS a
        ON a.user_id = b.user_id
       AND a.serial_id <> b.serial_id
       AND a.apply_time > b.repaid_time
    GROUP BY
        b.user_id,
        b.serial_id,
        b.repaid_date
),

sample_flags AS (
    SELECT
        f.repaid_date,
        CASE
            WHEN f.first_apply_time IS NOT NULL THEN
                f.first_apply_time::date - f.repaid_date
        END AS first_apply_day
    FROM first_apply AS f
),

metric_rows AS (
    SELECT
        1 AS sort_order,
        'D0 首次提单' AS 指标,
        COUNT(*) AS 分母用户数,
        SUM(
            CASE
                WHEN first_apply_day = 0 THEN 1
                ELSE 0
            END
        ) AS 首次提单用户数
    FROM sample_flags

    UNION ALL

    SELECT
        2 AS sort_order,
        'D1 首次提单' AS 指标,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 1 FROM params) THEN 1
                ELSE 0
            END
        ) AS 分母用户数,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 1 FROM params)
                 AND first_apply_day = 1 THEN 1
                ELSE 0
            END
        ) AS 首次提单用户数
    FROM sample_flags

    UNION ALL

    SELECT
        3 AS sort_order,
        'D2 首次提单' AS 指标,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 2 FROM params) THEN 1
                ELSE 0
            END
        ) AS 分母用户数,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 2 FROM params)
                 AND first_apply_day = 2 THEN 1
                ELSE 0
            END
        ) AS 首次提单用户数
    FROM sample_flags

    UNION ALL

    SELECT
        4 AS sort_order,
        'D3 及以后首次提单' AS 指标,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 3 FROM params) THEN 1
                ELSE 0
            END
        ) AS 分母用户数,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 3 FROM params)
                 AND first_apply_day >= 3 THEN 1
                ELSE 0
            END
        ) AS 首次提单用户数
    FROM sample_flags

    UNION ALL

    SELECT
        5 AS sort_order,
        'D3 至 D32 首次提单（30天）' AS 指标,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 32 FROM params) THEN 1
                ELSE 0
            END
        ) AS 分母用户数,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 32 FROM params)
                 AND first_apply_day BETWEEN 3 AND 32 THEN 1
                ELSE 0
            END
        ) AS 首次提单用户数
    FROM sample_flags

    UNION ALL

    SELECT
        6 AS sort_order,
        'D5 至 D34 首次提单（30天）' AS 指标,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 34 FROM params) THEN 1
                ELSE 0
            END
        ) AS 分母用户数,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 34 FROM params)
                 AND first_apply_day BETWEEN 5 AND 34 THEN 1
                ELSE 0
            END
        ) AS 首次提单用户数
    FROM sample_flags

    UNION ALL

    SELECT
        7 AS sort_order,
        'D8 至 D37 首次提单（30天）' AS 指标,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 37 FROM params) THEN 1
                ELSE 0
            END
        ) AS 分母用户数,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 37 FROM params)
                 AND first_apply_day BETWEEN 8 AND 37 THEN 1
                ELSE 0
            END
        ) AS 首次提单用户数
    FROM sample_flags
)

SELECT
    指标,
    分母用户数,
    首次提单用户数,
    ROUND(
        首次提单用户数::numeric / NULLIF(分母用户数, 0),
        4
    ) AS 首次提单率
FROM metric_rows
ORDER BY
    sort_order;

    UNION ALL

    SELECT
        TO_CHAR(repaid_date, 'YYYY-MM') AS 结清月份,
        3 AS sort_order,
        'D2 首次提单' AS 指标,
        SUM(CASE WHEN repaid_date <= (SELECT data_cutoff_date - 2 FROM params) THEN 1 ELSE 0 END),
        SUM(CASE WHEN repaid_date <= (SELECT data_cutoff_date - 2 FROM params)
                  AND first_apply_day = 2 THEN 1 ELSE 0 END)
    FROM sample_flags
    GROUP BY TO_CHAR(repaid_date, 'YYYY-MM')

    UNION ALL

    SELECT
        TO_CHAR(repaid_date, 'YYYY-MM') AS 结清月份,
        4 AS sort_order,
        'D3 及以后首次提单' AS 指标,
        SUM(CASE WHEN repaid_date <= (SELECT data_cutoff_date - 3 FROM params) THEN 1 ELSE 0 END),
        SUM(CASE WHEN repaid_date <= (SELECT data_cutoff_date - 3 FROM params)
                  AND first_apply_day >= 3 THEN 1 ELSE 0 END)
    FROM sample_flags
    GROUP BY TO_CHAR(repaid_date, 'YYYY-MM')

    UNION ALL

    SELECT
        TO_CHAR(repaid_date, 'YYYY-MM') AS 结清月份,
        5 AS sort_order,
        'D3 至 D32 首次提单（30天）' AS 指标,
        SUM(CASE WHEN repaid_date <= (SELECT data_cutoff_date - 32 FROM params) THEN 1 ELSE 0 END),
        SUM(CASE WHEN repaid_date <= (SELECT data_cutoff_date - 32 FROM params)
                  AND first_apply_day BETWEEN 3 AND 32 THEN 1 ELSE 0 END)
    FROM sample_flags
    GROUP BY TO_CHAR(repaid_date, 'YYYY-MM')

    UNION ALL

    SELECT
        TO_CHAR(repaid_date, 'YYYY-MM') AS 结清月份,
        6 AS sort_order,
        'D5 至 D34 首次提单（30天）' AS 指标,
        SUM(CASE WHEN repaid_date <= (SELECT data_cutoff_date - 34 FROM params) THEN 1 ELSE 0 END),
        SUM(CASE WHEN repaid_date <= (SELECT data_cutoff_date - 34 FROM params)
                  AND first_apply_day BETWEEN 5 AND 34 THEN 1 ELSE 0 END)
    FROM sample_flags
    GROUP BY TO_CHAR(repaid_date, 'YYYY-MM')

    UNION ALL

    SELECT
        TO_CHAR(repaid_date, 'YYYY-MM') AS 结清月份,
        7 AS sort_order,
        'D8 至 D37 首次提单（30天）' AS 指标,
        SUM(CASE WHEN repaid_date <= (SELECT data_cutoff_date - 37 FROM params) THEN 1 ELSE 0 END),
        SUM(CASE WHEN repaid_date <= (SELECT data_cutoff_date - 37 FROM params)
                  AND first_apply_day BETWEEN 8 AND 37 THEN 1 ELSE 0 END)
    FROM sample_flags
    GROUP BY TO_CHAR(repaid_date, 'YYYY-MM')
)

SELECT
    结清月份,
    指标,
    分母用户数,
    首次提单用户数,
    ROUND(
        首次提单用户数::numeric / NULLIF(分母用户数, 0),
        4
    ) AS 首次提单率
FROM month_metric_rows
ORDER BY
    结清月份,
    sort_order;
模块三：按结清月份统计 D0–D29 总提单率
/* ================================================================
   T0 用户首次提单时效总体统计
   统计范围：2026-01-01 至 2026-08-20

   基础分母：干净结清 + 结清后当日获额通过 + 获额时间晚于结清时间。
   同一用户同日保留最早合格结清订单。
   ================================================================ */
WITH params AS (
    SELECT
        DATE '2026-01-01' AS start_date,
        DATE '2026-08-20' AS end_date,
        CURRENT_DATE AS data_cutoff_date,
        CURRENT_TIMESTAMP AS data_cutoff_time
),

settled_orders AS (
    SELECT
        o.user_id,
        o.serial_id,
        o.repaid_time,
        o.repaid_date::date AS repaid_date
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    CROSS JOIN params AS p
    WHERE o.is_remit = 1
      AND (
            o.is_repaid = 1
         OR o.loan_status_code = 7
         OR o.repaid_date::date > DATE '2000-01-01'
      )
      AND o.repaid_time > TIMESTAMP '2000-01-01 00:00:00'
      AND o.repaid_date::date >= p.start_date
      AND o.repaid_date::date <= p.end_date
),

candidate_users AS (
    SELECT DISTINCT
        s.user_id
    FROM settled_orders AS s
),

all_orders AS (
    SELECT
        o.user_id,
        o.serial_id,
        o.apply_time,
        o.repaid_time,
        o.loan_status_code
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    INNER JOIN candidate_users AS u
        ON u.user_id = o.user_id
    CROSS JOIN params AS p
    WHERE o.apply_time < p.data_cutoff_time
),

clean_settles AS (
    /* 结清瞬间没有任何其他在途订单。 */
    SELECT
        s.user_id,
        s.serial_id,
        s.repaid_time,
        s.repaid_date
    FROM settled_orders AS s
    WHERE NOT EXISTS (
        SELECT
            1
        FROM all_orders AS o2
        WHERE o2.user_id = s.user_id
          AND o2.serial_id <> s.serial_id
          AND o2.apply_time < s.repaid_time
          AND (
                o2.loan_status_code = 5
             OR o2.loan_status_code = 8
             OR (
                    o2.loan_status_code = 7
                AND o2.repaid_time > s.repaid_time
                )
          )
    )
),

same_day_credit_pass AS (
    SELECT
        v.user_id,
        v.vir_date::date AS vir_date,
        TO_TIMESTAMP(v.vir_unix) AS vir_time
    FROM wangchuanliang.order_vir_f_copy AS v
    CROSS JOIN params AS p
    WHERE v.is_pass1 = 1
      AND v.vir_date::date >= p.start_date
      AND v.vir_date::date <= p.end_date
),

denominator_candidates AS (
    SELECT
        c.user_id,
        c.serial_id,
        c.repaid_time,
        c.repaid_date
    FROM clean_settles AS c
    WHERE EXISTS (
        SELECT
            1
        FROM same_day_credit_pass AS cp
        WHERE cp.user_id = c.user_id
          AND cp.vir_date = c.repaid_date
          AND cp.vir_time > c.repaid_time
    )
),

base_sample AS (
    /* 同一用户同日取最早一笔合格结清。 */
    SELECT DISTINCT ON (
        d.user_id,
        d.repaid_date
    )
        d.user_id,
        d.serial_id,
        d.repaid_time,
        d.repaid_date
    FROM denominator_candidates AS d
    ORDER BY
        d.user_id,
        d.repaid_date,
        d.repaid_time ASC
),

first_apply AS (
    /* 仅识别目标结清时间之后的首次提单。 */
    SELECT
        b.user_id,
        b.serial_id,
        b.repaid_date,
        MIN(a.apply_time) AS first_apply_time
    FROM base_sample AS b
    LEFT JOIN all_orders AS a
        ON a.user_id = b.user_id
       AND a.serial_id <> b.serial_id
       AND a.apply_time > b.repaid_time
    GROUP BY
        b.user_id,
        b.serial_id,
        b.repaid_date
),

sample_flags AS (
    SELECT
        f.repaid_date,
        CASE
            WHEN f.first_apply_time IS NOT NULL THEN
                f.first_apply_time::date - f.repaid_date
        END AS first_apply_day
    FROM first_apply AS f
),

monthly_d0_d29 AS (
    SELECT
        TO_CHAR(repaid_date, 'YYYY-MM') AS 结清月份,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 29 FROM params) THEN 1
                ELSE 0
            END
        ) AS D0至D29完整观察用户数,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 29 FROM params)
                 AND first_apply_day BETWEEN 0 AND 29 THEN 1
                ELSE 0
            END
        ) AS D0至D29首次提单用户数
    FROM sample_flags
    GROUP BY
        TO_CHAR(repaid_date, 'YYYY-MM')
),
overall_d0_d29 AS (
    SELECT
        '总计' AS 结清月份,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 29 FROM params) THEN 1
                ELSE 0
            END
        ) AS D0至D29完整观察用户数,
        SUM(
            CASE
                WHEN repaid_date <= (SELECT data_cutoff_date - 29 FROM params)
                 AND first_apply_day BETWEEN 0 AND 29 THEN 1
                ELSE 0
            END
        ) AS D0至D29首次提单用户数
    FROM sample_flags
)

SELECT
    结清月份,
    D0至D29完整观察用户数,
    D0至D29首次提单用户数,
    ROUND(
        D0至D29首次提单用户数::numeric
        / NULLIF(D0至D29完整观察用户数, 0),
        4
    ) AS 结清当日及之后30天总提单率
FROM (
    SELECT * FROM monthly_d0_d29
    UNION ALL
    SELECT * FROM overall_d0_d29
) AS result
ORDER BY
    CASE WHEN 结清月份 = '总计' THEN 0 ELSE 1 END,
    结清月份;
