二、核心发现与提额效果分层

高复借用户的提额后表现最好。“6次及以上 × 上一笔接近100%/超过100%”的D+1～D+5放款率分别为47.81%和52.53%，下一笔到期盈利率分别为22.95%和21.35%；两组1—7月均为7/7个月正盈利。
提额后额度没有被等比例使用。全体用户上一笔使用率为86.37%，提额后下一笔使用率降至81.21%，下降5.16个百分点。上一笔接近100%和超过100%的用户，下一笔使用率也仅为85.75%至83.78%（高复借组示例），说明提额创造了可用空间，但并非全部转化为新增借款。
历史使用率与盈利不是简单正相关。上一笔低于80%的客群下一笔盈利率为19.35%，高于上一笔超过100%客群的13.51%；后者放款率却更高（44.90% 对 36.57%）。提额策略需要在转化和盈利之间做联合筛选。
整体D+1～D+5提单率明显下行。从1月的48.13%降至8月的37.34%，下降10.79个百分点。提额事件量增加并未带来同方向的短期转化提升，不能把“已提额”直接解释为提额产生了增量效果。
“1次 × 80%～100%”的高盈利属于小样本异常点。该组仅78笔到期订单，1—7月最低月盈利率-83.02%，波动很大，不应仅凭汇总盈利率28.26%扩大提额。

    SELECT user_id, credit_serial_id, vir_time, settled_serial_id, previous_credit_amt

    WHERE rn = 1
), loan_count_at_settlement AS (
    SELECT s.user_id, s.credit_serial_id, s.vir_time,
           COUNT(DISTINCT h.serial_id) AS successful_loan_n
    FROM strict_t0 s
    JOIN all_orders h
      ON h.user_id = s.user_id
     AND h.is_remit = 1
     AND COALESCE(h.remit_amt, 0) > 0
     AND h.remit_time IS NOT NULL
     AND h.remit_time <= s.repaid_time
    GROUP BY s.user_id, s.credit_serial_id, s.vir_time
), up_credit_snapshot AS (
    SELECT s.user_id, s.credit_serial_id, s.vir_date, s.vir_time,
           s.repaid_time, s.settled_serial_id,
           s.settled_remit_amt::numeric AS previous_remit_amt,
           p.previous_credit_amt,
           s.current_credit_amt,
           s.current_credit_amt - p.previous_credit_amt AS credit_change_amt,
           s.current_credit_amt / NULLIF(p.previous_credit_amt, 0) - 1 AS credit_change_rate,
           s.settled_remit_amt::numeric / NULLIF(p.previous_credit_amt, 0) AS previous_util_ratio,
           l.successful_loan_n
    FROM strict_t0 s
    JOIN previous_credit p
      ON p.user_id = s.user_id
     AND p.credit_serial_id = s.credit_serial_id
     AND p.vir_time = s.vir_time
     AND p.settled_serial_id = s.settled_serial_id
    JOIN loan_count_at_settlement l
      ON l.user_id = s.user_id
     AND l.credit_serial_id = s.credit_serial_id
     AND l.vir_time = s.vir_time
    WHERE s.current_credit_amt > 0
      AND p.previous_credit_amt > 0
      AND s.settled_remit_amt > 0
      AND s.current_credit_amt > p.previous_credit_amt
), eligible_snapshot AS (
    SELECT d.*
    FROM up_credit_snapshot d
    WHERE NOT EXISTS (
      SELECT 1
      FROM all_orders a
      WHERE a.user_id = d.user_id
        AND a.apply_time >= d.vir_time
        AND a.apply_date = d.vir_date
    )
), candidate_future_orders AS (
    SELECT e.user_id, e.credit_serial_id, e.vir_date, e.vir_time,
           o.serial_id, o.apply_time, o.apply_date,
           o.is_remit, o.remit_time, o.remit_amt, o.due_date,
      ROW_NUMBER() OVER (
        PARTITION BY o.serial_id
        ORDER BY e.vir_time DESC, e.credit_serial_id DESC
      ) AS event_match_rn
    FROM eligible_snapshot e
    JOIN all_orders o
      ON o.user_id = e.user_id
     AND o.apply_date BETWEEN e.vir_date + 1 AND e.vir_date + 5
     AND o.apply_time > e.vir_time
), assigned_future_orders AS (
    SELECT * FROM candidate_future_orders WHERE event_match_rn = 1
), first_d1_d5_apply AS (
    SELECT user_id, credit_serial_id, vir_time,
           serial_id AS next_apply_serial_id,
           apply_time AS next_apply_time
    FROM (
      SELECT o.*,
        ROW_NUMBER() OVER (
          PARTITION BY o.user_id, o.credit_serial_id, o.vir_time
          ORDER BY o.apply_time ASC NULLS LAST, o.serial_id ASC
        ) AS rn
      FROM assigned_future_orders o
    ) x
    WHERE rn = 1
), first_d1_d5_remit AS (
    SELECT user_id, credit_serial_id, vir_time,
           serial_id AS next_remit_serial_id,
           apply_time AS next_remit_apply_time,
           remit_time AS next_remit_time,
           remit_amt::numeric AS next_remit_amt,
           due_date AS next_due_date
    FROM (
      SELECT o.*,
        ROW_NUMBER() OVER (
          PARTITION BY o.user_id, o.credit_serial_id, o.vir_time
          ORDER BY o.apply_time ASC NULLS LAST, o.serial_id ASC
        ) AS rn
      FROM assigned_future_orders o
      WHERE o.is_remit = 1
        AND COALESCE(o.remit_amt, 0) > 0
    ) x
    WHERE rn = 1
), repayment_by_due AS (
    SELECT n.next_remit_serial_id,
           SUM(CASE WHEN r.payin_date <= n.next_due_date THEN r.amount ELSE 0 END) AS repaid_by_due_amt
    FROM first_d1_d5_remit n
    LEFT JOIN wangchuanliang.repayment_record_copy r
      ON r.serial_id = n.next_remit_serial_id
    WHERE n.next_due_date < CURRENT_DATE - 1
    GROUP BY n.next_remit_serial_id
)
SELECT
    DATE_TRUNC('month', e.vir_date)::date AS t0_vir_month,
    e.user_id,
    e.credit_serial_id,
    e.vir_date,
    e.vir_time,
    e.repaid_time,
    e.settled_serial_id,
    e.previous_credit_amt,
    e.previous_remit_amt,
    e.previous_util_ratio,
    CASE
      WHEN e.previous_util_ratio < 0.800 THEN '<80%'
      WHEN e.previous_util_ratio < 0.999 THEN '80%-100%'
      WHEN e.previous_util_ratio <= 1.001 THEN '接近100%'
      ELSE '>100%'
    END AS previous_util_band,
    e.current_credit_amt,
    e.credit_change_amt,
    e.credit_change_rate,
    e.successful_loan_n,
    CASE
      WHEN e.successful_loan_n = 1 THEN '1次'
      WHEN e.successful_loan_n = 2 THEN '2次'
      WHEN e.successful_loan_n = 3 THEN '3次'
      WHEN e.successful_loan_n = 4 THEN '4次'
      WHEN e.successful_loan_n = 5 THEN '5次'
      ELSE '6次及以上'
    END AS loan_count_bucket,
    CASE WHEN a.next_apply_serial_id IS NULL THEN 0 ELSE 1 END AS has_d1_d5_apply,
    a.next_apply_serial_id,
    a.next_apply_time,
    CASE WHEN n.next_remit_serial_id IS NULL THEN 0 ELSE 1 END AS has_d1_d5_remit,
    n.next_remit_serial_id,
    n.next_remit_apply_time,
    n.next_remit_time,
    n.next_remit_amt,
    n.next_due_date,
    n.next_remit_amt / NULLIF(e.current_credit_amt, 0) AS next_util_ratio,
    CASE WHEN p.next_remit_serial_id IS NULL THEN 0 ELSE 1 END AS is_matured,
    CASE WHEN p.next_remit_serial_id IS NULL THEN 0 ELSE n.next_remit_amt END AS matured_remit_amt,
    CASE WHEN p.next_remit_serial_id IS NULL THEN 0 ELSE COALESCE(p.repaid_by_due_amt, 0) END AS repaid_by_due_amt
FROM eligible_snapshot e
LEFT JOIN first_d1_d5_apply a
  ON a.user_id = e.user_id
 AND a.credit_serial_id = e.credit_serial_id
 AND a.vir_time = e.vir_time
LEFT JOIN first_d1_d5_remit n
  ON n.user_id = e.user_id
 AND n.credit_serial_id = e.credit_serial_id
 AND n.vir_time = e.vir_time
LEFT JOIN repayment_by_due p
  ON p.next_remit_serial_id = n.next_remit_serial_id;

WITH monthly AS (
          SELECT t0_vir_month, loan_count_bucket, previous_util_band,
            COUNT(*) AS eligible_event_n,
            SUM(has_d1_d5_apply) AS apply_n,
            SUM(has_d1_d5_remit) AS remit_n,

              AS maturity_profit_rate_pct
          FROM tmp_t0_up_util_loan_profit
          GROUP BY t0_vir_month, loan_count_bucket, previous_util_band
        ), total AS (
          SELECT loan_count_bucket, previous_util_band, 
  COUNT(*) AS eligible_event_n,
  COUNT(DISTINCT user_id) AS eligible_user_n,
  SUM(has_d1_d5_apply) AS d1_d5_apply_n,
  ROUND(100.0 * SUM(has_d1_d5_apply) / NULLIF(COUNT(*), 0), 2) AS d1_d5_apply_rate_pct,
  SUM(has_d1_d5_remit) AS d1_d5_remit_n,

  ROUND(100.0 * SUM(previous_remit_amt) / NULLIF(SUM(previous_credit_amt), 0), 2) AS previous_utilization_rate_pct,
  ROUND(100.0 * SUM(CASE WHEN has_d1_d5_remit=1 THEN next_remit_amt ELSE 0 END)
        / NULLIF(SUM(CASE WHEN has_d1_d5_remit=1 THEN current_credit_amt ELSE 0 END), 0), 2)
    AS next_utilization_rate_pct,
  SUM(is_matured) AS matured_order_n,
  SUM(matured_remit_amt) AS matured_remit_amt,
  SUM(repaid_by_due_amt) AS repaid_by_due_amt,
  ROUND(100.0 * SUM(repaid_by_due_amt) / NULLIF(SUM(matured_remit_amt), 0) - 100, 2)
    AS maturity_profit_rate_pct,
  ROUND(AVG(credit_change_amt), 2) AS avg_credit_change_amt,

          FROM tmp_t0_up_util_loan_profit
          GROUP BY loan_count_bucket, previous_util_band
        )
        SELECT t.*,
          COUNT(m.t0_vir_month) AS observed_month_n,
          SUM(CASE WHEN m.t0_vir_month < DATE '2026-08-01' THEN 1 ELSE 0 END) AS jan_jul_month_n,
          SUM(CASE WHEN m.t0_vir_month < DATE '2026-08-01'
                    AND m.maturity_profit_rate_pct > 0 THEN 1 ELSE 0 END) AS jan_jul_positive_profit_month_n,

                / NULLIF(SUM(CASE WHEN m.t0_vir_month < DATE '2026-08-01' THEN m.matured_remit_amt ELSE 0 END), 0) - 100, 2)
            AS jan_jul_profit_pct
        FROM total t
        JOIN monthly m USING (loan_count_bucket, previous_util_band)
        GROUP BY t.loan_count_bucket, t.previous_util_band,
          t.eligible_event_n, t.eligible_user_n,
          t.d1_d5_apply_n, t.d1_d5_apply_rate_pct,

          t.previous_utilization_rate_pct, t.next_utilization_rate_pct,
          t.matured_order_n, t.matured_remit_amt, t.repaid_by_due_amt,
