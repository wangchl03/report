  • 同人样本仅纳入分析期内至少两笔成功放款的存量用户；它验证“同一用户前后订单”的变化，不受用户进入/退出样本影响。
  • 大多数可比存量用户息费基本持平，说明“所有老用户普遍同步提价”的说法不成立；但息费上升用户多于下降用户，且上升者的平均变化为 14.23%。
  • 标准化结果显示，剔除用户年龄、历史次数、期限和金额档结构后的息费率仍上升，支持存在同类订单内部价格变化；该方法不等同于因果识别。

四、经营建议

  • 新增用户息费上调幅度更大，应优先核查新客产品、渠道、额度和期限结构是否发生了定价变化。
  • 存量用户整体也有上调，但同人结果存在明显异质性；后续应按历史贷款次数、风险档和用户年龄定位具体被提价的层级。
  • 将息费、T0 提单率、到期盈利和逾期率放在同一分层实验中评估；不能只看息费率上升判断经营改善。

模块 1：统一放款订单底表与用户分层
-- ============================================================
-- 2026-01-01 至 2026-08-31 放款订单息费变化拆解
-- 月度归属：apply_date（提单日期）；统计对象：成功放款订单 is_remit = 1。
-- 息费率：SUM(pre_amt + post_amt) / SUM(remit_amt)，不使用订单息费率简单平均。
-- ============================================================

DROP TABLE IF EXISTS tmp_fee_base;

CREATE TEMP TABLE tmp_fee_base AS
WITH all_remitted AS (
    SELECT
        o.serial_id,
        o.user_id,
        o.apply_date,
        o.apply_time,
        o.remit_date,
        o.remit_time,
        o.remit_amt,
        o.pre_amt,
        o.post_amt,
        o.loan_day,
        ROW_NUMBER() OVER (
            PARTITION BY o.user_id
            ORDER BY o.remit_time, o.serial_id
        ) AS 成功放款序号,
        FIRST_VALUE(o.apply_date) OVER (
            PARTITION BY o.user_id
            ORDER BY o.remit_time, o.serial_id
        ) AS 首次成功放款提单日,
        FIRST_VALUE(o.remit_date) OVER (
            PARTITION BY o.user_id
            ORDER BY o.remit_time, o.serial_id
        ) AS 首次成功放款日
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    WHERE o.is_remit = 1
      AND COALESCE(o.remit_amt, 0) > 0
), base AS (
    SELECT
        a.*,
        a.成功放款序号 - 1 AS 当前订单前成功放款次数,
        CASE
            WHEN DATE_TRUNC('month', a.首次成功放款提单日)
                 = DATE_TRUNC('month', a.apply_date)
                THEN '当月新增用户'
            ELSE '存量用户'
        END AS 用户新老分组,
        CASE
            WHEN a.remit_date = a.首次成功放款日 THEN '首贷（0天）'
            WHEN a.remit_date - a.首次成功放款日 BETWEEN 1 AND 30 THEN '新近复借（1-30天）'
            WHEN a.remit_date - a.首次成功放款日 BETWEEN 31 AND 90 THEN '成长期复借（31-90天）'
            WHEN a.remit_date - a.首次成功放款日 BETWEEN 91 AND 180 THEN '成熟复借（91-180天）'
            ELSE '长期存量复借（180天以上）'
        END AS 用户年龄分层,
        CASE
            WHEN a.成功放款序号 = 1 THEN '历史放款0次'
            WHEN a.成功放款序号 = 2 THEN '历史放款1次'
            WHEN a.成功放款序号 BETWEEN 3 AND 5 THEN '历史放款2-4次'
            ELSE '历史放款5次及以上'
        END AS 历史次数分层,
        CASE
            WHEN a.loan_day <= 7 THEN '期限≤7天'
            WHEN a.loan_day <= 14 THEN '期限8-14天'
            WHEN a.loan_day <= 21 THEN '期限15-21天'
            ELSE '期限22天以上'
        END AS 期限分层,
        CASE
            WHEN a.remit_amt < 500 THEN '金额<500'
            WHEN a.remit_amt < 1000 THEN '金额500-999'
            WHEN a.remit_amt < 1500 THEN '金额1000-1499'
            ELSE '金额1500及以上'
        END AS 金额分层
    FROM all_remitted AS a
    WHERE a.apply_date >= DATE '2026-01-01'
      AND a.apply_date < DATE '2026-09-01'
)
SELECT * FROM base;

-- ============================================================
模块 2：总体月度趋势
-- 模块 2：总体月度趋势（报告图表、汇总卡片）
-- ============================================================
SELECT
    TO_CHAR(apply_date, 'YYYY-MM') AS 统计月份,
    COUNT(DISTINCT user_id) AS 放款用户数,
    COUNT(DISTINCT serial_id) AS 放款订单数,
    SUM(remit_amt) AS 放款金额,
    SUM(pre_amt + post_amt) AS 息费金额,
    SUM(pre_amt + post_amt)::numeric / NULLIF(SUM(remit_amt), 0) AS 加权息费率,
    SUM(remit_amt)::numeric / NULLIF(COUNT(DISTINCT serial_id), 0) AS 平均放款金额,
    SUM(loan_day * remit_amt)::numeric / NULLIF(SUM(remit_amt), 0) AS 放款金额加权平均期限
FROM tmp_fee_base
GROUP BY TO_CHAR(apply_date, 'YYYY-MM')
ORDER BY 统计月份;

-- ============================================================
模块 3：新增 / 存量用户拆解
-- 模块 3：当月新增 / 存量用户拆解
-- ============================================================
WITH monthly_total AS (
    SELECT TO_CHAR(apply_date, 'YYYY-MM') AS 统计月份, SUM(remit_amt) AS 月总放款金额
    FROM tmp_fee_base
    GROUP BY TO_CHAR(apply_date, 'YYYY-MM')
)
SELECT
    TO_CHAR(b.apply_date, 'YYYY-MM') AS 统计月份,
    b.用户新老分组,
    COUNT(DISTINCT b.user_id) AS 放款用户数,
    COUNT(DISTINCT b.serial_id) AS 放款订单数,
    SUM(b.remit_amt) AS 放款金额,
    SUM(b.remit_amt)::numeric / NULLIF(t.月总放款金额, 0) AS 放款金额占比,
    SUM(b.pre_amt + b.post_amt) AS 息费金额,
    SUM(b.pre_amt + b.post_amt)::numeric / NULLIF(SUM(b.remit_amt), 0) AS 加权息费率,
    SUM(b.remit_amt)::numeric / NULLIF(COUNT(DISTINCT b.serial_id), 0) AS 平均放款金额,
    SUM(b.loan_day * b.remit_amt)::numeric / NULLIF(SUM(b.remit_amt), 0) AS 放款金额加权平均期限
FROM tmp_fee_base AS b
INNER JOIN monthly_total AS t
    ON t.统计月份 = TO_CHAR(b.apply_date, 'YYYY-MM')
GROUP BY TO_CHAR(b.apply_date, 'YYYY-MM'), b.用户新老分组, t.月总放款金额
ORDER BY 统计月份, b.用户新老分组;

-- ============================================================
模块 4：用户年龄分层
-- 模块 4：用户年龄分层（月度息费与金额结构）
-- ============================================================
WITH monthly_total AS (
    SELECT TO_CHAR(apply_date, 'YYYY-MM') AS 统计月份, SUM(remit_amt) AS 月总放款金额
    FROM tmp_fee_base GROUP BY TO_CHAR(apply_date, 'YYYY-MM')
)
SELECT
    TO_CHAR(b.apply_date, 'YYYY-MM') AS 统计月份,
    b.用户年龄分层,
    COUNT(DISTINCT b.user_id) AS 放款用户数,
    COUNT(DISTINCT b.serial_id) AS 放款订单数,
    SUM(b.remit_amt) AS 放款金额,
    SUM(b.remit_amt)::numeric / NULLIF(t.月总放款金额, 0) AS 放款金额占比,
    SUM(b.pre_amt + b.post_amt)::numeric / NULLIF(SUM(b.remit_amt), 0) AS 加权息费率
FROM tmp_fee_base AS b
JOIN monthly_total AS t ON t.统计月份 = TO_CHAR(b.apply_date, 'YYYY-MM')
GROUP BY TO_CHAR(b.apply_date, 'YYYY-MM'), b.用户年龄分层, t.月总放款金额
ORDER BY 统计月份, b.用户年龄分层;

-- ============================================================
模块 5：存量用户同人前后比较
-- 模块 5：存量用户同人前后比较
-- 样本：分析期内至少有两笔成功放款、且首笔不是当月新增的用户。
-- ============================================================
WITH ranked AS (
    SELECT
        b.*,
        ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY apply_time, serial_id) AS rn_first,
        ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY apply_time DESC, serial_id DESC) AS rn_last,
        COUNT(*) OVER (PARTITION BY user_id) AS 分析期放款订单数
    FROM tmp_fee_base AS b
), paired AS (
    SELECT
        user_id,
        MAX(CASE WHEN rn_first = 1 THEN (pre_amt + post_amt)::numeric / NULLIF(remit_amt, 0) END) AS 首笔息费率,
        MAX(CASE WHEN rn_last = 1 THEN (pre_amt + post_amt)::numeric / NULLIF(remit_amt, 0) END) AS 末笔息费率,
        MAX(CASE WHEN rn_first = 1 THEN apply_date END) AS 首笔提单日,
        MAX(CASE WHEN rn_last = 1 THEN apply_date END) AS 末笔提单日
    FROM ranked
    WHERE 用户新老分组 = '存量用户'
      AND 分析期放款订单数 >= 2
    GROUP BY user_id
)
SELECT
    CASE
        WHEN 末笔息费率 - 首笔息费率 > 0.005 THEN '息费上升（>0.5pp）'
        WHEN 末笔息费率 - 首笔息费率 < -0.005 THEN '息费下降（<-0.5pp）'
        ELSE '基本持平（±0.5pp）'
    END AS 同人息费变化分组,
    COUNT(*) AS 用户数,
    AVG(末笔息费率 - 首笔息费率) AS 人均息费率变化,
    PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY 末笔息费率 - 首笔息费率) AS 息费率变化中位数
FROM paired
GROUP BY 1
ORDER BY 1;

-- ============================================================
模块 6：三月结构标准化息费率
-- 模块 6：固定 2026-03 结构的标准化息费率
-- 结构维度：用户年龄 × 历史成功放款次数 × 期限 × 放款金额档。
-- 原始与标准化的差额用于描述结构贡献，非严格因果分解。
-- ============================================================
WITH strata_month AS (
    SELECT
        TO_CHAR(apply_date, 'YYYY-MM') AS 统计月份,
        用户年龄分层, 历史次数分层, 期限分层, 金额分层,
        SUM(remit_amt) AS 分层放款金额,
        SUM(pre_amt + post_amt)::numeric / NULLIF(SUM(remit_amt), 0) AS 分层加权息费率
    FROM tmp_fee_base
    GROUP BY 1,2,3,4,5
), march_weights AS (
    SELECT
        用户年龄分层, 历史次数分层, 期限分层, 金额分层,
        分层放款金额::numeric / NULLIF(SUM(分层放款金额) OVER (), 0) AS 三月固定结构权重
    FROM strata_month
    WHERE 统计月份 = '2026-03'
), raw_month AS (
    SELECT TO_CHAR(apply_date, 'YYYY-MM') AS 统计月份,
           SUM(pre_amt + post_amt)::numeric / NULLIF(SUM(remit_amt), 0) AS 原始加权息费率
    FROM tmp_fee_base GROUP BY 1
)
SELECT
    r.统计月份,
    r.原始加权息费率,
    SUM(w.三月固定结构权重 * s.分层加权息费率) AS 三月结构标准化息费率,
    r.原始加权息费率 - SUM(w.三月固定结构权重 * s.分层加权息费率) AS 结构变化近似贡献
FROM raw_month AS r
LEFT JOIN strata_month AS s ON s.统计月份 = r.统计月份
LEFT JOIN march_weights AS w
    ON w.用户年龄分层 = s.用户年龄分层
   AND w.历史次数分层 = s.历史次数分层
   AND w.期限分层 = s.期限分层
   AND w.金额分层 = s.金额分层
GROUP BY r.统计月份, r.原始加权息费率
ORDER BY r.统计月份;

-- ============================================================
模块 7：分层明细与分子分母
-- 模块 7：报告使用的分层明细（用于复核每个图表的分子、分母）
-- ============================================================
SELECT
    TO_CHAR(apply_date, 'YYYY-MM') AS 统计月份,
    用户新老分组,
    用户年龄分层,
    历史次数分层,
    期限分层,
    金额分层,
    COUNT(DISTINCT user_id) AS 放款用户数,
    COUNT(DISTINCT serial_id) AS 放款订单数,
    SUM(remit_amt) AS 放款金额分母,
    SUM(pre_amt + post_amt) AS 息费金额分子,
    SUM(pre_amt + post_amt)::numeric / NULLIF(SUM(remit_amt), 0) AS 加权息费率
FROM tmp_fee_base
GROUP BY 1,2,3,4,5,6
ORDER BY 1,2,3,4,5,6;
