一、目标：保留405，修正长期流失风险识别

保留评分基础

使用观测时刻前最近一次获额对应的405分；不求平均，不取最高，不混入363。

增加风险修正信息

重点加入评分×流失天数、历史多笔逾期与还款趋势，不人工指定“优质长期流失=高风险”。

二、样本、时间和标签

项目实际规则
流失起点整笔结清且无其他在贷，结清后当天通过获额，recycle_type=3；20秒链式去重
流失结束起点后首次用户提单，重新获额不重置流失天数
观测时刻历史：首次回流提单时；实时：2026-10-08 00:00
样本流失≥7天，最新405有效且最新获额通过为type3；与V1同一冻结样本
目标再次提单成功放款订单，due_date+D7结束仍未整笔结清=1，否则=0
成熟due_date+8≤下一阶段开始；未成熟、未放款和未提单不补正常标签
缺失最新405缺失的获额不纳入；其余特征缺失保留样本，树模型处理NaN
限制已放款样本有审批选择偏差；本次对照复用已被查看过的V1测试集，不是全新未见测试

三、特征开发与防泄漏

特征组细分方向
405与流失交互连续评分×log(1+流失天数)、训练期评分档位×15/30/60/90天标记
历史逾期最近1—5笔D0/D7结果；最近3/5/10笔逾期比例；90/180/365天成熟逾期比例
还款趋势最近1—5笔借还天数、提前结清天数；最近两笔/三笔差异；早还与借还天数波动
生命周期历史借款频率、循环贷占比、流失天数相对账期和历史借还速度
行为实际App与非还款日App均保留候选；7天、8—30天、31—90天不重叠窗口与活动强度
已有特征额度使用率、提降额、历史提单放款率、上一笔金额/账期/息费等沿用V1

实际候选323项，新增172项。目标订单逾期、目标订单当前状态、用户ID与serial_id不作特征。历史到期结果只在观测前已经成熟时使用。

观测时刻 t = 用户首次回流提单时刻
最新405 = 最新获额时间 ≤ t 对应的评分
流失天数 = DATE(t) - DATE(严格T0起点)
历史订单 = apply_time < t 且 remit_time < t，排除目标serial_id
历史D7可用 = 历史due_date + 8 ≤ DATE(t)
App可用 = active_date < DATE(t)
目标y = 目标订单到期D+7结束是否仍未结清
模型输出p = P(y=1 | 最新405, 流失, 历史借还, 行为)
重新评分 = 1000 × (1-p)，高分对应相对低风险

-- 流失>=7天405风险再评级：只创建本会话TEMP表，不改动源表。
-- 数据完整截止：2026-10-07，右开边界2026-10-08。
SET TIME ZONE 'America/Belize';
SET query_dop = 1;
-- STEP orders
CREATE TEMP TABLE tmp_c405_orders AS
SELECT serial_id,user_id,apply_time,apply_date,remit_time,remit_date,
       remit_amt,apply_amt,pre_amt,post_amt,loan_day,total_term,
       is_remit,repaid_time,repaid_date,is_repaid,due_date,
       loan_status_code,overdue_days,loan_type,remit_succ_rnb
FROM wangchuanliang.order_loan_f_v2_copy
WHERE apply_time >= TIMESTAMP '2000-01-01'
  AND apply_time < TIMESTAMP '2026-10-08 00:00:00';
-- STEP types
CREATE TEMP TABLE tmp_c405_types AS
SELECT serial_id,
       max(CASE WHEN recycle_type=3 THEN 1 ELSE 0 END) has_type3,
       max(CASE WHEN recycle_type=2 THEN 1 ELSE 0 END) has_type2
FROM wangchuanliang.side_recycle_type_copy GROUP BY serial_id;
-- STEP offers
CREATE TEMP TABLE tmp_c405_offers AS
SELECT v.user_id,v.serial_id,v.vir_unix,
       to_timestamp(v.vir_unix)::timestamp AS vir_time,v.vir_date,
       v.is_pass1,v.loan_type_code,v.max_credit_apply_amt,v.risk_product,
       CASE WHEN s.has_type3=1 AND s.has_type2=0 THEN 3
            WHEN s.has_type2=1 AND s.has_type3=0 THEN 2 ELSE 1 END AS recycle_type
FROM wangchuanliang.order_vir_f_copy v
LEFT JOIN tmp_c405_types s ON s.serial_id=v.serial_id
WHERE v.vir_unix>0 AND v.vir_date < DATE '2026-10-08';
-- STEP dedup
CREATE TEMP TABLE tmp_c405_anchor_candidates AS
WITH serial_latest AS (
 SELECT *,row_number() over(partition by user_id,serial_id
                           order by vir_time desc,serial_id desc) rn
 FROM tmp_c405_offers
 WHERE is_pass1=1 AND loan_type_code=2 AND recycle_type=3
), gaps AS (
 SELECT *,CASE WHEN vir_time-lag(vir_time) over(partition by user_id order by vir_time,serial_id)
                         <=interval '20 seconds' THEN 0 ELSE 1 END new_batch
 FROM serial_latest WHERE rn=1
), batches AS (
 SELECT *,sum(new_batch) over(partition by user_id order by vir_time,serial_id rows unbounded preceding) batch
 FROM gaps
), selected AS (
 SELECT *,row_number() over(partition by user_id,batch order by vir_time desc,serial_id desc) batch_rn
 FROM batches
)
SELECT user_id,serial_id,vir_time,vir_date FROM selected WHERE batch_rn=1;
-- STEP anchors
CREATE TEMP TABLE tmp_c405_anchors AS
SELECT user_id,anchor_serial,anchor_time,anchor_date,prev_serial
FROM (
 SELECT v.user_id,v.serial_id anchor_serial,v.vir_time anchor_time,v.vir_date anchor_date,
        o.serial_id prev_serial,
        row_number() over(partition by v.user_id,v.serial_id order by o.repaid_time desc,o.serial_id desc) rn
 FROM tmp_c405_anchor_candidates v
 JOIN tmp_c405_orders o ON o.user_id=v.user_id AND o.repaid_date=v.vir_date
      AND o.is_remit=1 AND o.remit_amt>0 AND (o.is_repaid=1 OR o.loan_status_code=7)
      AND o.repaid_time >= TIMESTAMP '2000-01-01' AND o.repaid_time<=v.vir_time
 WHERE NOT EXISTS (
  SELECT 1 FROM tmp_c405_orders x
  WHERE x.user_id=o.user_id AND x.serial_id<>o.serial_id
    AND x.is_remit=1 AND x.remit_amt>0 AND x.remit_time<=o.repaid_time
    AND (x.repaid_time IS NULL OR x.repaid_time<TIMESTAMP '2000-01-01' OR x.repaid_time>o.repaid_time)
 )
) z WHERE rn=1;
-- STEP spells
CREATE TEMP TABLE tmp_c405_spells AS
WITH windows AS (
 SELECT *,lead(anchor_time) over(partition by user_id order by anchor_time,anchor_serial) next_anchor_time
 FROM tmp_c405_anchors
), matched AS (
 SELECT a.*,o.serial_id target_serial,o.apply_time obs_time,
        o.apply_date obs_date,o.is_remit target_is_remit,o.remit_amt target_remit_amt,
        o.due_date target_due_date,o.repaid_date target_repaid_date,
        row_number() over(partition by a.user_id,a.anchor_serial order by o.apply_time,o.serial_id) rn
 FROM windows a LEFT JOIN tmp_c405_orders o
 ON o.user_id=a.user_id AND o.apply_time>=a.anchor_time
 AND (a.next_anchor_time IS NULL OR o.apply_time<a.next_anchor_time)
)
SELECT * FROM matched WHERE rn=1;

-- STEP base
CREATE TEMP TABLE tmp_c405_base AS
SELECT e.*,m.kabybxgboost405apd7crecyc risk_score_405,
       (e.obs_date-e.anchor_date)::float8 churn_days,
       extract(epoch from(e.obs_time-e.score_offer_time))/86400.0 score_age_days,
       CASE WHEN e.score_serial<>e.anchor_serial THEN 1 ELSE 0 END is_rescored,
       p.remit_succ_rnb::float8 previous_loan_count,
       p.remit_amt::float8 prev_remit_amt,p.apply_amt::float8 prev_apply_amt,
       p.loan_day::float8 prev_loan_day,p.total_term::float8 prev_total_term,
       ((p.pre_amt+p.post_amt)/nullif(p.remit_amt,0))::float8 prev_fee_rate,
       (p.due_date-p.repaid_date)::float8 prev_early_days,
       extract(epoch from(p.repaid_time-p.remit_time))/86400.0 prev_repay_days,
       coalesce(s.has_type2,0) prev_is_cycle
FROM tmp_c405_latest e
JOIN wangchuanliang.model_result_copy m ON m.serial_id=e.score_serial AND m.user_id=e.user_id
JOIN tmp_c405_orders p ON p.serial_id=e.prev_serial AND p.user_id=e.user_id
LEFT JOIN tmp_c405_types s ON s.serial_id=p.serial_id
WHERE m.kabybxgboost405apd7crecyc IS NOT NULL AND m.kabybxgboost405apd7crecyc<>-9999999
  AND e.current_pass1=1 AND e.current_recycle_type=3;

      GROUP BY e.event_id

        c.offer_count_since_anchor,c.pass_offer_rows_since_anchor

def metrics(y,p):
