价值判断：以 10% 大盘盈利率为标准，所有流失间隔层的平均两笔合计盈利率均达到高价值水平。用户即使流失较久，能够回归并在首笔有效期内再次形成循环贷，仍呈现较强的平均收益贡献。
风险边界：“平均高价值”不等于可无差别开放循环贷。流失达到 180 天后，循环贷第二笔盈利率开始明显下降，订单逾期率同步抬升；双当前逾期订单对始终是约 -44% 至 -54% 的明显亏损场景。

风险解读：各阈值下循环贷第二笔订单逾期率约 30%—33%，显著高于首笔的 15%—18%。超过 180 天后，双当前逾期订单对占成熟订单对的比例升至 17%—18%，且该组两笔合计盈利率均低于 -50%。因此，长流失用户的循环贷策略应从“是否整体有利润”转向“是否识别并隔离双逾期尾部风险”。

五、统计口径与使用边界

订单对定义：用户整笔结清进入 0 在贷后，取真实下一笔成功放款订单。该订单提单时间距结清时点超过设定阈值，即为回归首笔；若首笔已结清，则循环贷订单提单时间必须早于首笔结清时间；若首笔未结清，则必须早于首笔到期时间。循环贷订单要求 `recycle_type = 2` 且成功放款。
价值判定：10% 是大盘平均盈利率，不是单笔订单的风险阈值。报告中的“高价值”指该层的金额加权平均盈利率超过 10%；实际策略仍需以当前逾期、双逾期尾部及可接受风险预算共同约束。
样本边界：阈值越长，样本量越小。超过 240 天和 360 天的成熟订单对分别为 485 和 225 组，应重点观察趋势，但不宜仅凭小样本进行大幅策略调整。

以下代码一次运行即可输出报告所用的全部阈值指标，包括整体当前逾期率、成熟订单的首笔/循环贷/合计盈利与逾期指标，以及双当前逾期订单对表现。

/*
  流失间隔分层：回归首笔 + 首笔有效期内循环贷订单的价值与风险

  输出 3 个结果集：
  1) 各阈值全部符合条件订单对的规模与当前订单逾期率；
  2) 两笔均到期样本的首笔、循环贷第二笔、合计逾期率及盈利率；
  3) 两笔均当前逾期的订单对数量、占比及合计盈利率。

  口径：
  - 0 在贷：某笔成功放款订单结清时，用户不存在其他未结清的已放款订单；
  - 回归首笔：0 在贷后的真实下一笔成功放款订单，且提单间隔超过阈值；
  - 循环贷第二笔：recycle_type=2，且发生在首笔有效期内；
  - 首笔已结清：第二笔 apply_time < 首笔 repaid_time；
  - 首笔未结清：第二笔 apply_time < 首笔 due_time；
  - 当前逾期：loan_status_code=8；
  - 盈利率：截至运行日 (sum(repaid_amt)-sum(remit_amt))/sum(remit_amt)。
*/

SET TIME ZONE 'America/Belize';

DROP TABLE IF EXISTS tmp_long_churn_cycle_pairs_by_threshold;

CREATE TEMP TABLE tmp_long_churn_cycle_pairs_by_threshold AS
WITH thresholds AS (
    SELECT churn_days
    FROM (
        VALUES (7), (15), (30), (60), (90),
               (120), (150), (180), (240), (360)
    ) t(churn_days)
),
loan_orders AS (
    SELECT
        o.user_id::BIGINT AS user_id,
        o.serial_id,
        o.apply_time,
        o.remit_time,
        o.repaid_time,
        o.repaid_date,
        o.due_time,
        o.due_date,
        COALESCE(o.remit_amt, 0)::NUMERIC AS remit_amt,
        COALESCE(o.repaid_amt, 0)::NUMERIC AS repaid_amt,
        o.loan_status_code,
        COALESCE(r.recycle_type, 0) AS recycle_type
    FROM wangchuanliang.order_loan_f_v2_copy o
    LEFT JOIN wangchuanliang.side_recycle_type_copy r
        ON r.serial_id = o.serial_id
    WHERE o.is_remit = 1
      AND o.remit_time IS NOT NULL
),
zero_outstanding_events AS (
    SELECT DISTINCT
        a.user_id,
        a.repaid_time AS zero_outstanding_time
    FROM loan_orders a
    WHERE a.repaid_date > DATE '2000-01-01'
      AND NOT EXISTS (
          SELECT 1
          FROM loan_orders b
          WHERE b.user_id = a.user_id
            AND b.serial_id <> a.serial_id
            AND b.remit_time <= a.repaid_time
            AND (
                b.repaid_date IS NULL
                OR b.repaid_date < DATE '2000-01-01'
                OR b.repaid_time > a.repaid_time
            )
      )
),
next_funded_order_candidates AS (
    SELECT
        z.user_id,
        z.zero_outstanding_time,
        f.serial_id AS first_order_id,
        f.apply_time AS first_apply_time,
        f.remit_time AS first_remit_time,
        f.repaid_time AS first_repaid_time,
        f.repaid_date AS first_repaid_date,
        f.due_time AS first_due_time,
        f.due_date AS first_due_date,
        f.remit_amt AS first_remit_amt,
        f.repaid_amt AS first_repaid_amt,
        f.loan_status_code AS first_status,
        ROW_NUMBER() OVER (
            PARTITION BY z.user_id, z.zero_outstanding_time
            ORDER BY f.apply_time, f.serial_id
        ) AS next_funded_rn
    FROM zero_outstanding_events z
    JOIN loan_orders f
        ON f.user_id = z.user_id
       AND f.apply_time > z.zero_outstanding_time
),
long_churn_return_first AS (
    SELECT
        t.churn_days,
        n.*
    FROM next_funded_order_candidates n
    JOIN thresholds t ON n.first_apply_time > n.zero_outstanding_time
                                          + t.churn_days * INTERVAL '1 day'
    WHERE n.next_funded_rn = 1
),
qualified_cycle_pairs AS (
    SELECT
        f.churn_days,
        f.user_id,
        f.first_order_id,
        f.first_due_date,
        f.first_remit_amt,
        f.first_repaid_amt,
        f.first_status,
        s.serial_id AS cycle_order_id,
        s.due_date AS cycle_due_date,
        s.remit_amt AS cycle_remit_amt,
        s.repaid_amt AS cycle_repaid_amt,
        s.loan_status_code AS cycle_status
    FROM long_churn_return_first f
    JOIN loan_orders s
        ON s.user_id = f.user_id
       AND s.recycle_type = 2
       AND s.apply_time > f.first_apply_time
       AND s.remit_time > f.first_remit_time
    WHERE (
            f.first_repaid_date > DATE '2000-01-01'
            AND s.apply_time < f.first_repaid_time
          )
       OR (
            (f.first_repaid_date IS NULL
             OR f.first_repaid_date < DATE '2000-01-01')
            AND s.apply_time < f.first_due_time
          )
)
SELECT *
FROM qualified_cycle_pairs;


/* 结果集 1：所有符合条件订单对（包含尚未到期订单）的当前逾期率。 */
SELECT
    churn_days AS churn_threshold_days,
    COUNT(*) AS qualified_pair_count,
    COUNT(*) * 2 AS qualified_order_count,
    SUM(CASE WHEN first_status = 8 THEN 1 ELSE 0 END)
        + SUM(CASE WHEN cycle_status = 8 THEN 1 ELSE 0 END)
        AS current_overdue_order_count,
    (
        SUM(CASE WHEN first_status = 8 THEN 1 ELSE 0 END)
        + SUM(CASE WHEN cycle_status = 8 THEN 1 ELSE 0 END)
    )::NUMERIC / NULLIF(COUNT(*) * 2, 0) AS current_order_overdue_rate
FROM tmp_long_churn_cycle_pairs_by_threshold
GROUP BY churn_days
ORDER BY churn_days;


/* 结果集 2：两笔均已到期样本的首笔、循环贷第二笔、合计指标。 */
WITH matured_pairs AS (
    SELECT *
    FROM tmp_long_churn_cycle_pairs_by_threshold
    WHERE first_due_date <= CURRENT_DATE
      AND cycle_due_date <= CURRENT_DATE
),
metrics AS (
    SELECT
        churn_days,
        '首笔回归订单' AS metric_scope,
        COUNT(*) AS order_count,
        SUM(CASE WHEN first_status = 8 THEN 1 ELSE 0 END) AS current_overdue_order_count,
        SUM(CASE WHEN first_status = 8 THEN 1 ELSE 0 END)::NUMERIC
            / NULLIF(COUNT(*), 0) AS current_order_overdue_rate,
        SUM(first_remit_amt) AS remit_amt,
        SUM(first_repaid_amt) AS repaid_amt,
        SUM(first_repaid_amt - first_remit_amt)
            / NULLIF(SUM(first_remit_amt), 0) AS profit_rate
    FROM matured_pairs
    GROUP BY churn_days

    UNION ALL

    SELECT
        churn_days,
        '第二笔循环贷订单',
        COUNT(*),
        SUM(CASE WHEN cycle_status = 8 THEN 1 ELSE 0 END),
        SUM(CASE WHEN cycle_status = 8 THEN 1 ELSE 0 END)::NUMERIC
            / NULLIF(COUNT(*), 0),
        SUM(cycle_remit_amt),
        SUM(cycle_repaid_amt),
        SUM(cycle_repaid_amt - cycle_remit_amt)
            / NULLIF(SUM(cycle_remit_amt), 0)
    FROM matured_pairs
    GROUP BY churn_days

    UNION ALL

    SELECT
        churn_days,
        '两笔订单合计',
        COUNT(*) * 2,
        SUM(CASE WHEN first_status = 8 THEN 1 ELSE 0 END)
            + SUM(CASE WHEN cycle_status = 8 THEN 1 ELSE 0 END),
        (
            SUM(CASE WHEN first_status = 8 THEN 1 ELSE 0 END)
            + SUM(CASE WHEN cycle_status = 8 THEN 1 ELSE 0 END)
        )::NUMERIC / NULLIF(COUNT(*) * 2, 0),
        SUM(first_remit_amt + cycle_remit_amt),
        SUM(first_repaid_amt + cycle_repaid_amt),
        SUM(first_repaid_amt + cycle_repaid_amt - first_remit_amt - cycle_remit_amt)
            / NULLIF(SUM(first_remit_amt + cycle_remit_amt), 0)
    FROM matured_pairs
    GROUP BY churn_days
)
SELECT *
FROM metrics
ORDER BY churn_days,
         CASE metric_scope
             WHEN '首笔回归订单' THEN 1
             WHEN '第二笔循环贷订单' THEN 2
             WHEN '两笔订单合计' THEN 3
         END;


/* 结果集 3：两笔均当前逾期的订单对。 */
WITH matured_pairs AS (
    SELECT *
    FROM tmp_long_churn_cycle_pairs_by_threshold
    WHERE first_due_date <= CURRENT_DATE
      AND cycle_due_date <= CURRENT_DATE
)
SELECT
    churn_days AS churn_threshold_days,
    COUNT(*) AS matured_pair_count,
    SUM(CASE WHEN first_status = 8 AND cycle_status = 8 THEN 1 ELSE 0 END)
        AS both_current_overdue_pair_count,
    AVG(CASE WHEN first_status = 8 AND cycle_status = 8 THEN 1.0 ELSE 0.0 END)
        AS both_current_overdue_pair_share,
    SUM(
        CASE WHEN first_status = 8 AND cycle_status = 8
             THEN first_repaid_amt + cycle_repaid_amt
                - first_remit_amt - cycle_remit_amt
             ELSE 0 END
    ) / NULLIF(
        SUM(
            CASE WHEN first_status = 8 AND cycle_status = 8
                 THEN first_remit_amt + cycle_remit_amt
                 ELSE 0 END
        ),
        0
    ) AS both_current_overdue_pair_profit_rate
FROM matured_pairs
GROUP BY churn_days
ORDER BY churn_days;

DROP TABLE tmp_long_churn_cycle_pairs_by_threshold;
