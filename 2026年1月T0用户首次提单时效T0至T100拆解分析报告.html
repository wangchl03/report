五、附录：完整 PGSQL

三个模块需在同一数据库会话内按顺序执行。模块一创建会话级临时表；模块二和模块三使用该临时表输出诊断及 T0～T100+ 结果。

模块一：创建一月基础样本与首次提单时点
-- 模块一：创建 2026 年 1 月 T0 基础样本及首次提单时点临时表
-- 基础口径：干净结清 + 结清当日获额通过 + 获额时间大于或等于结清时间。
-- 粒度：用户 × 结清日；同一用户同日仅保留最早的合格结清订单。

DROP TABLE IF EXISTS tmp_t0_jan_first_apply;

CREATE TEMP TABLE tmp_t0_jan_first_apply AS
WITH params AS (
    SELECT
        DATE '2026-01-01' AS start_date,
        DATE '2026-01-31' AS end_date,
        CURRENT_DATE AS data_cutoff_date,
        CURRENT_TIMESTAMP AS data_cutoff_time
),
settled_orders AS (
    SELECT
        o.user_id,
        o.serial_id,
        o.repaid_time,
        o.repaid_date::date AS repaid_date
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    CROSS JOIN params AS p
    WHERE o.is_remit = 1
      AND (
            o.is_repaid = 1
         OR o.loan_status_code = 7
         OR o.repaid_date::date > DATE '2000-01-01'
      )
      AND o.repaid_time > TIMESTAMP '2000-01-01 00:00:00'
      AND o.repaid_date::date BETWEEN p.start_date AND p.end_date
),
candidate_users AS (
    SELECT DISTINCT user_id FROM settled_orders
),
all_orders AS (
    SELECT o.user_id, o.serial_id, o.apply_time, o.repaid_time, o.loan_status_code
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    INNER JOIN candidate_users AS u ON u.user_id = o.user_id
    CROSS JOIN params AS p
    WHERE o.apply_time < p.data_cutoff_time
),
clean_settles AS (
    -- 结清时点没有其他在途、逾期，或将在该时点之后才结清的订单。
    SELECT s.user_id, s.serial_id, s.repaid_time, s.repaid_date
    FROM settled_orders AS s
    WHERE NOT EXISTS (
        SELECT 1
        FROM all_orders AS o2
        WHERE o2.user_id = s.user_id
          AND o2.serial_id <> s.serial_id
          AND o2.apply_time < s.repaid_time
          AND (
                o2.loan_status_code IN (5, 8)
             OR (o2.loan_status_code = 7 AND o2.repaid_time > s.repaid_time)
          )
    )
),
same_day_credit_pass AS (
    SELECT
        v.user_id,
        v.vir_date::date AS vir_date,
        TO_TIMESTAMP(v.vir_unix) AS vir_time
    FROM wangchuanliang.order_vir_f_copy AS v
    CROSS JOIN params AS p
    WHERE v.is_pass1 = 1
      AND v.vir_date::date BETWEEN p.start_date AND p.end_date
),
eligible_settles AS (
    SELECT c.user_id, c.serial_id, c.repaid_time, c.repaid_date
    FROM clean_settles AS c
    WHERE EXISTS (
        SELECT 1
        FROM same_day_credit_pass AS cp
        WHERE cp.user_id = c.user_id
          AND cp.vir_date = c.repaid_date
          AND cp.vir_time >= c.repaid_time
    )
),
base_sample AS (
    -- 同一用户同日若有多笔合格结清，仅保留最早结清的一笔。
    SELECT DISTINCT ON (user_id, repaid_date)
        user_id, serial_id, repaid_time, repaid_date
    FROM eligible_settles
    ORDER BY user_id, repaid_date, repaid_time ASC
),
first_apply AS (
    SELECT
        b.user_id,
        b.serial_id AS settle_serial_id,
        b.repaid_time,
        b.repaid_date,
        MIN(a.apply_time) AS first_apply_time
    FROM base_sample AS b
    LEFT JOIN all_orders AS a
        ON a.user_id = b.user_id
       AND a.serial_id <> b.serial_id
       AND a.apply_time > b.repaid_time
    GROUP BY b.user_id, b.serial_id, b.repaid_time, b.repaid_date
)
SELECT
    f.user_id,
    f.settle_serial_id,
    f.repaid_time,
    f.repaid_date,
    f.first_apply_time,
    CASE
        WHEN f.first_apply_time IS NOT NULL
        THEN f.first_apply_time::date - f.repaid_date
    END AS first_apply_day
FROM first_apply AS f;
模块二：基础样本重复用户诊断
-- 模块二：基础样本的重复用户诊断
-- “重复用户”指同一 user_id 在 2026 年 1 月出现两条及以上合格结清样本。
WITH user_sample_cnt AS (
    SELECT user_id, COUNT(*) AS sample_cnt
    FROM tmp_t0_jan_first_apply
    GROUP BY user_id
)
SELECT
    (SELECT COUNT(*) FROM tmp_t0_jan_first_apply) AS 基础用户日期样本数,
    COUNT(*) AS 去重用户数,
    (SELECT COUNT(DISTINCT user_id)
     FROM tmp_t0_jan_first_apply
     WHERE first_apply_day IS NOT NULL) AS 有首次提单的去重用户数,
    ROUND(
        (SELECT COUNT(DISTINCT user_id)::numeric
         FROM tmp_t0_jan_first_apply
         WHERE first_apply_day IS NOT NULL)
        / NULLIF(COUNT(*), 0),
        4
    ) AS 去重用户首次提单率,
    SUM(CASE WHEN sample_cnt >= 2 THEN 1 ELSE 0 END) AS 有重复样本的用户数,
    SUM(CASE WHEN sample_cnt >= 2 THEN sample_cnt ELSE 0 END) AS 重复用户贡献的样本数,
    (SELECT COUNT(*) FROM tmp_t0_jan_first_apply) - COUNT(*) AS 去重后减少的样本数,
    MAX(sample_cnt) AS 单用户最大样本数
FROM user_sample_cnt;
模块三：T0～T100+ 时效分布
-- 模块三：T0 至 T100+ 首次提单时效分布
-- Tn：结清后第 n 天首次提单，且此前没有提单。
-- T100+：第 101 天及以后首次提单。各 T 分组在“用户日期样本”层面互斥。
WITH day_groups AS (
    SELECT
        n AS sort_order,
        'T' || n::text AS 提单时效分组,
        COUNT(*) AS 用户数,
        SUM(CASE WHEN first_apply_day = n THEN 1 ELSE 0 END) AS 提单用户数
    FROM generate_series(0, 100) AS n
    CROSS JOIN tmp_t0_jan_first_apply
    GROUP BY n
),
t100_plus AS (
    SELECT
        101 AS sort_order,
        'T100+' AS 提单时效分组,
        COUNT(*) AS 用户数,
        SUM(CASE WHEN first_apply_day >= 101 THEN 1 ELSE 0 END) AS 提单用户数
    FROM tmp_t0_jan_first_apply
),
all_groups AS (
    SELECT * FROM day_groups
    UNION ALL
    SELECT * FROM t100_plus
)
SELECT
    提单时效分组,
    用户数,
    提单用户数,
    ROUND(提单用户数::numeric / NULLIF(用户数, 0), 4) AS 提单率
FROM all_groups
ORDER BY sort_order;
