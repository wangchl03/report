五、结论与建议

1. T0 转化率下降是主结论

从 1 月 52.85% 到 7 月 41.63%，累计下降 11.22 个百分点。需将该趋势作为获额后即时转化效率的核心监测指标。

2. 先用周/月级别判断是否改善

日度波动可超过 20 个百分点，不应据此快速调整运营策略。周度连续改善、月度加权回升才构成更可靠的效果判断。

3. 监测分母结构，避免误判

提单率变化可能来自用户风险、额度、产品期次或渠道结构变化。后续应同步输出分母用户的历史借款次数、获额额度与产品结构，验证是否存在样本构成漂移。

4. 将 T0 与 T1–T15 共同观察

T0 降低未必全部等于流失，也可能部分转移为更晚提单。建议与 T1、T2–T7、T8–T15 的占比趋势联动监控，区分“时效后移”和“整体转化下降”。

附录：完整 PGSQL 查询代码（含口径注释）

以下代码完整覆盖本报告的干净结清、当日获额通过、T0 当天提单，以及按日、周、月累计统计逻辑。

/*
  T0 用户当天提单率：日、周、月统计

  分母：
  1. 统计日内最后一笔干净结清；
  2. 结清当日获额通过（is_pass1 = 1）；
  3. 获额通过时间晚于该笔结清时间。

  分子：
  满足分母条件，且在同一自然日、晚于结清时间发生提单的用户事件。

  周度、月度：
  直接汇总每日分子、每日分母后计算提单率，不对日提单率做简单平均。
*/

WITH all_orders AS (
    /* 订单基础表：保留干净结清和提单判断所需字段。 */
    SELECT
        o.user_id,
        o.serial_id,
        o.apply_time,
        o.repaid_time,
        o.repaid_date,
        o.loan_status_code,
        o.is_remit
    FROM wangchuanliang.order_loan_f_v2_copy AS o
    WHERE o.apply_time < TIMESTAMP '2026-08-16 00:00:00'
),

settled_orders AS (
    /*
      当前已结清订单。
      loan_status_code = 7 表示订单当前状态已结清；
      不限制订单在结清前是否曾为逾期状态。
    */
    SELECT
        o.user_id,
        o.serial_id,
        o.repaid_time,
        o.repaid_date::date AS repaid_date
    FROM all_orders AS o
    WHERE o.is_remit = 1
      AND o.loan_status_code = 7
      AND o.repaid_time IS NOT NULL
      AND o.repaid_date::date >= DATE '2026-01-01'
      AND o.repaid_date::date <= DATE '2026-08-15'
),

last_settle_of_day AS (
    /*
      同一用户同日可能多笔结清。
      仅取结清时间最后的一笔；repaid_time 不会相同，
      因此不以 serial_id 作为次级排序条件。
    */
    SELECT DISTINCT ON (
        s.user_id,
        s.repaid_date
    )
        s.user_id,
        s.serial_id,
        s.repaid_time,
        s.repaid_date
    FROM settled_orders AS s
    ORDER BY
        s.user_id,
        s.repaid_date,
        s.repaid_time DESC
),

clean_last_settles AS (
    /*
      干净结清：
      在目标订单结清的那个时刻，用户不存在其他在途订单。

      其他订单当前状态为 5 或 8，说明其仍未结清；
      其他订单当前虽为 7、但其结清时间晚于目标订单，
      说明在目标结清时点该订单仍在途，亦应排除。
    */
    SELECT
        s.user_id,
        s.serial_id,
        s.repaid_time,
        s.repaid_date
    FROM last_settle_of_day AS s
    WHERE NOT EXISTS (
        SELECT
            1
        FROM all_orders AS o2
        WHERE o2.user_id = s.user_id
          AND o2.serial_id <> s.serial_id
          AND o2.apply_time < s.repaid_time
          AND (
                o2.loan_status_code = 5
             OR o2.loan_status_code = 8
             OR (
                    o2.loan_status_code = 7
                AND o2.repaid_time > s.repaid_time
                )
          )
    )
),

credit_pass AS (
    /* 获额通过记录；vir_unix 转为可比较的时间戳。 */
    SELECT
        v.user_id,
        v.vir_date::date AS vir_date,
        to_timestamp(v.vir_unix) AS vir_time
    FROM wangchuanliang.order_vir_f_copy AS v
    WHERE v.is_pass1 = 1
      AND v.vir_date::date >= DATE '2026-01-01'
      AND v.vir_date::date <= DATE '2026-08-15'
),

t0_eligible AS (
    /*
      T0 分母：
      干净结清后，在结清当日获额通过，且获额通过时间晚于结清时间。
    */
    SELECT
        c.user_id,
        c.serial_id,
        c.repaid_time,
        c.repaid_date
    FROM clean_last_settles AS c
    WHERE EXISTS (
        SELECT
            1
        FROM credit_pass AS cp
        WHERE cp.user_id = c.user_id
          AND cp.vir_date = c.repaid_date
          AND cp.vir_time > c.repaid_time
    )
),

t0_event_result AS (
    /*
      每个合格“用户 × 结清日”事件标记是否当天提单。
      仅要求提单时间晚于结清时间，且提单日期等于结清日期。
    */
    SELECT
        e.user_id,
        e.repaid_date,
        CASE
            WHEN EXISTS (
                SELECT
                    1
                FROM all_orders AS a
                WHERE a.user_id = e.user_id
                  AND a.apply_time > e.repaid_time
                  AND a.apply_time::date = e.repaid_date
            )
            THEN 1
            ELSE 0
        END AS is_t0_apply
    FROM t0_eligible AS e
),

daily_stats AS (
    /*
      日度统计：
      每行均为当天 T0 分母、当天提单分子和当天提单率。
    */
    SELECT
        r.repaid_date AS period_start,
        COUNT(*) AS eligible_users,
        SUM(r.is_t0_apply) AS apply_users
    FROM t0_event_result AS r
    GROUP BY
        r.repaid_date
),

weekly_stats AS (
    /*
      周度统计：
      直接累计 daily_stats 的分子和分母。
      周起始日为周一。
    */
    SELECT
        date_trunc('week', d.period_start)::date AS period_start,
        SUM(d.eligible_users) AS eligible_users,
        SUM(d.apply_users) AS apply_users
    FROM daily_stats AS d
    GROUP BY
        date_trunc('week', d.period_start)::date
),

monthly_stats AS (
    /*
      月度统计：
      直接累计 daily_stats 的分子和分母。
    */
    SELECT
        date_trunc('month', d.period_start)::date AS period_start,
        SUM(d.eligible_users) AS eligible_users,
        SUM(d.apply_users) AS apply_users
    FROM daily_stats AS d
    GROUP BY
        date_trunc('month', d.period_start)::date
),

all_grains AS (
    /* 合并日、周、月三个粒度的统计结果。 */
    SELECT
        '日' AS period_type,
        d.period_start,
        d.eligible_users,
        d.apply_users
    FROM daily_stats AS d

    UNION ALL

    SELECT
        '周' AS period_type,
        w.period_start,
        w.eligible_users,
        w.apply_users
    FROM weekly_stats AS w

    UNION ALL

    SELECT
        '月' AS period_type,
        m.period_start,
        m.eligible_users,
        m.apply_users
    FROM monthly_stats AS m
)

SELECT
    g.period_type AS 统计粒度,
    g.period_start AS 结清周期起始日,
    g.eligible_users AS T0分母用户数,
    g.apply_users AS T0当天提单用户数,
    ROUND(
        g.apply_users::numeric
        / NULLIF(g.eligible_users, 0),
        6
    ) AS T0提单率
FROM all_grains AS g
ORDER BY
    CASE g.period_type
        WHEN '日' THEN 1
        WHEN '周' THEN 2
        WHEN '月' THEN 3
    END,
    g.period_start;
