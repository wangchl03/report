七、统计边界

  • 本报告只衡量获额后24小时内的首次提单，24小时后的提单不计入分子。
  • 模型分是当次获额订单的实时分，不以用户历史或当前其他订单的分数替代。
  • 结构图的统计单位是获额事件及其对应的首提事件；同一用户若在不同日期再次获额，可因实时分变化进入不同层。
  • 订单质量与盈利按整笔订单到期日观察。获额月份越新，到期覆盖率可能越低；比较月度风险和盈利时应同时参考覆盖率。
  • 各档月度提单率可用于比较行为趋势；但息费、额度、期限、渠道、准入和活动策略变化均可能影响结果，不能据此直接认定价格的因果作用。

附录：完整可复现 PGSQL

以下SQL可在同一数据库会话中顺序执行，依次输出实时模型分覆盖与阈值、提单时段、客群结构、整笔订单到期逾期率及盈利率。

展开完整 PGSQL
-- ============================================================================
-- T0 获额后 24 小时首次提单时效 × KabyBXgboost405Apd7CreCyc 实时资质分层
-- 统计期：2026-01-01（含）至 2026-09-01（不含），即覆盖至 2026-08-31。
-- 模型分关联：model_result_copy.serial_id = 当次 T0 获额订单 credit_serial_id。
-- 同一用户不同获额事件的模型分独立计算，禁止按 user_id 固定映射资质。
-- 分数越高代表资质越好；阈值为本统计期全部有分 T0 事件的 P30 / P60，固定应用到各月。
-- ============================================================================
SET search_path TO wangchuanliang, public;
SET statement_timeout = 900000;

DROP TABLE IF EXISTS tmp_t0_hour_params;
CREATE TEMP TABLE tmp_t0_hour_params AS
SELECT DATE '2026-01-01' AS start_date,
       DATE '2026-09-01' AS end_date;

-- 1. 候选获额订单：先取订单内最新获额记录。
DROP TABLE IF EXISTS tmp_t0_raw_offer;
CREATE TEMP TABLE tmp_t0_raw_offer AS
SELECT DISTINCT ON (v.user_id, v.serial_id)
       v.user_id,
       v.serial_id,
       v.vir_date::date AS vir_date,
       TO_TIMESTAMP(v.vir_unix) AS vir_time,
       r.recycle_type
FROM order_vir_f_copy AS v
INNER JOIN side_recycle_type_copy AS r
        ON r.serial_id = v.serial_id
CROSS JOIN tmp_t0_hour_params AS p
WHERE v.is_pass1 = 1
  AND v.loan_type_code = 2
  AND v.vir_date >= p.start_date - 1
  AND v.vir_date < p.end_date
ORDER BY v.user_id, v.serial_id, v.vir_unix DESC;

-- 2. 相邻获额间隔不超过 20 秒的记录属于同一批；每批取订单号最大的记录。
DROP TABLE IF EXISTS tmp_t0_dedup_offer;
CREATE TEMP TABLE tmp_t0_dedup_offer AS
WITH lagged AS (
    SELECT r.*,
           LAG(r.vir_time) OVER (
               PARTITION BY r.user_id ORDER BY r.vir_time, r.serial_id
           ) AS prev_vir_time
    FROM tmp_t0_raw_offer AS r
), batched AS (
    SELECT l.*,
           SUM(CASE WHEN l.prev_vir_time IS NULL
                         OR l.vir_time - l.prev_vir_time > INTERVAL '20 seconds'
                    THEN 1 ELSE 0 END)
           OVER (PARTITION BY l.user_id ORDER BY l.vir_time, l.serial_id
                 ROWS UNBOUNDED PRECEDING) AS batch_id
    FROM lagged AS l
), ranked AS (
    SELECT b.*,
           ROW_NUMBER() OVER (
               PARTITION BY b.user_id, b.batch_id
               ORDER BY b.serial_id DESC
           ) AS batch_rn
    FROM batched AS b
    CROSS JOIN tmp_t0_hour_params AS p
    WHERE b.recycle_type = 3
      AND b.vir_date >= p.start_date
      AND b.vir_date < p.end_date
)
SELECT user_id, serial_id, vir_date, vir_time
FROM ranked
WHERE batch_rn = 1;

-- 3. 候选用户的订单全集，供“结清时 0 在贷”判断及获额后首次提单查找使用。
DROP TABLE IF EXISTS tmp_t0_all_orders;
CREATE TEMP TABLE tmp_t0_all_orders AS
SELECT o.user_id, o.serial_id, o.apply_time, o.apply_date,
       o.repaid_time, o.repaid_date, o.is_remit, o.is_repaid,
       o.loan_status_code
FROM order_loan_f_v2_copy AS o
INNER JOIN (SELECT DISTINCT user_id FROM tmp_t0_dedup_offer) AS u
        ON u.user_id = o.user_id;

-- 4. 干净结清：成功放款且已结清，结清时不存在其他在途/在贷订单。
DROP TABLE IF EXISTS tmp_t0_clean_settlement;
CREATE TEMP TABLE tmp_t0_clean_settlement AS
SELECT o.user_id, o.serial_id, o.repaid_time, o.repaid_date::date AS repaid_date
FROM tmp_t0_all_orders AS o
WHERE o.is_remit = 1
  AND (o.is_repaid = 1 OR o.loan_status_code = 7)
  AND o.repaid_time > TIMESTAMP '2000-01-01 00:00:00'
  AND NOT EXISTS (
      SELECT 1
      FROM tmp_t0_all_orders AS x
      WHERE x.user_id = o.user_id
        AND x.serial_id <> o.serial_id
        AND x.apply_time < o.repaid_time
        AND (
             x.loan_status_code IN (5, 8)
          OR (x.loan_status_code = 7 AND x.repaid_time > o.repaid_time)
        )
  );

-- 5. 一个用户、一个结清日仅保留最早有效获额事件，形成严格 T0 获额样本。
DROP TABLE IF EXISTS tmp_t0_event;
CREATE TEMP TABLE tmp_t0_event AS
SELECT user_id, credit_serial_id, vir_date, vir_time, settlement_serial_id
FROM (
    SELECT e.user_id,
           e.serial_id AS credit_serial_id,
           e.vir_date,
           e.vir_time,
           s.serial_id AS settlement_serial_id,
           ROW_NUMBER() OVER (
               PARTITION BY e.user_id, e.vir_date
               ORDER BY e.vir_time ASC, e.serial_id DESC
           ) AS day_rn
    FROM tmp_t0_dedup_offer AS e
    INNER JOIN tmp_t0_clean_settlement AS s
            ON s.user_id = e.user_id
           AND s.repaid_date = e.vir_date
           AND e.vir_time >= s.repaid_time
) AS x
WHERE day_rn = 1;

-- 6. 关联当次获额订单的实时 405 模型分；-9999999 和空值均视为无有效分。
DROP TABLE IF EXISTS tmp_t0_scored;
CREATE TEMP TABLE tmp_t0_scored AS
SELECT e.*,
       m.kabybxgboost405apd7crecyc AS score
FROM tmp_t0_event AS e
LEFT JOIN model_result_copy AS m
       ON m.serial_id = e.credit_serial_id
WHERE m.kabybxgboost405apd7crecyc IS NOT NULL
  AND m.kabybxgboost405apd7crecyc <> -9999999;

-- 7. 用全统计期 T0 实时分数确定固定 P30/P60 阈值。
DROP TABLE IF EXISTS tmp_t0_score_cut;
CREATE TEMP TABLE tmp_t0_score_cut AS
SELECT PERCENTILE_CONT(0.30) WITHIN GROUP (ORDER BY score) AS p30,
       PERCENTILE_CONT(0.60) WITHIN GROUP (ORDER BY score) AS p60
FROM tmp_t0_scored;

-- 8. 每个获额事件的实时资质分层。
DROP TABLE IF EXISTS tmp_t0_tier;
CREATE TEMP TABLE tmp_t0_tier AS
SELECT s.*,
       CASE WHEN s.score <= c.p30 THEN '资质较差'
            WHEN s.score <= c.p60 THEN '资质一般'
            ELSE '资质较好' END AS quality_tier
FROM tmp_t0_scored AS s
CROSS JOIN tmp_t0_score_cut AS c;

-- 9. 仅取获额后 24 小时内最早一笔提单。每个 T0 事件最多出现一次。
DROP TABLE IF EXISTS tmp_t0_first_apply_24h;
CREATE TEMP TABLE tmp_t0_first_apply_24h AS
SELECT user_id, credit_serial_id, vir_date, vir_time,
       apply_serial_id, apply_time,
       FLOOR(EXTRACT(EPOCH FROM (apply_time - vir_time)) / 3600)::int AS hour_bucket
FROM (
    SELECT t.user_id, t.credit_serial_id, t.vir_date, t.vir_time,
           o.serial_id AS apply_serial_id, o.apply_time,
           ROW_NUMBER() OVER (
               PARTITION BY t.user_id, t.credit_serial_id, t.vir_time
               ORDER BY o.apply_time, o.serial_id
           ) AS apply_rn
    FROM tmp_t0_tier AS t
    INNER JOIN tmp_t0_all_orders AS o
            ON o.user_id = t.user_id
           AND o.apply_time >= t.vir_time
           AND o.apply_time < t.vir_time + INTERVAL '24 hours'
) AS x
WHERE apply_rn = 1;

-- 10. 完整事件底表：无24小时提单的 T0 事件仍保留，用于各层提单率分母。
DROP TABLE IF EXISTS tmp_t0_hour_base;
CREATE TEMP TABLE tmp_t0_hour_base AS
SELECT t.user_id, t.credit_serial_id, t.vir_date, t.vir_time,
       t.score, t.quality_tier,
       f.apply_serial_id, f.apply_time, f.hour_bucket
FROM tmp_t0_tier AS t
LEFT JOIN tmp_t0_first_apply_24h AS f
       ON f.user_id = t.user_id
      AND f.credit_serial_id = t.credit_serial_id
      AND f.vir_time = t.vir_time;

-- 11. 将每笔获额后首提订单只归因给最近一次此前的 T0 获额事件，防止跨窗口重复计入。
DROP TABLE IF EXISTS tmp_t0_apply_order_attributed;
CREATE TEMP TABLE tmp_t0_apply_order_attributed AS
SELECT user_id, credit_serial_id, vir_date, vir_time, quality_tier, apply_serial_id
FROM (
    SELECT b.user_id, b.credit_serial_id, b.vir_date, b.vir_time, b.quality_tier,
           b.apply_serial_id,
           ROW_NUMBER() OVER (
               PARTITION BY b.apply_serial_id
               ORDER BY b.vir_time DESC, b.credit_serial_id DESC
           ) AS attribution_rn
    FROM tmp_t0_hour_base AS b
    WHERE b.apply_serial_id IS NOT NULL
) AS x
WHERE attribution_rn = 1;

-- 12. 成功放款首提订单，以及已整笔到期的观察样本。
DROP TABLE IF EXISTS tmp_t0_remitted_order;
CREATE TEMP TABLE tmp_t0_remitted_order AS
SELECT a.*, o.remit_amt, o.due_date::date AS due_date, o.repaid_date::date AS repaid_date
FROM tmp_t0_apply_order_attributed AS a
INNER JOIN order_loan_f_v2_copy AS o
        ON o.serial_id = a.apply_serial_id
WHERE o.is_remit = 1
  AND o.remit_amt > 0;

DROP TABLE IF EXISTS tmp_t0_matured_order;
CREATE TEMP TABLE tmp_t0_matured_order AS
SELECT *
FROM tmp_t0_remitted_order
WHERE due_date < CURRENT_DATE - 1;

-- 13. 到期日前实际到账回款：只累计 payin_date <= due_date 的还款流水。
DROP TABLE IF EXISTS tmp_t0_due_repayment;
CREATE TEMP TABLE tmp_t0_due_repayment AS
SELECT m.apply_serial_id,
       COALESCE(SUM(CASE WHEN r.payin_date <= m.due_date THEN r.amount ELSE 0 END), 0) AS paid_by_due
FROM tmp_t0_matured_order AS m
LEFT JOIN repayment_record_copy AS r
       ON r.serial_id = m.apply_serial_id
GROUP BY m.apply_serial_id;

-- 14. 整笔订单到期质量：仅比较结清日与到期日；不使用 loan_status_code。
DROP TABLE IF EXISTS tmp_t0_due_order_detail;
CREATE TEMP TABLE tmp_t0_due_order_detail AS
SELECT m.*, p.paid_by_due,
       CASE WHEN m.repaid_date > DATE '2000-01-01'
                  AND m.repaid_date <= m.due_date
            THEN 0 ELSE 1 END AS is_overdue
FROM tmp_t0_matured_order AS m
INNER JOIN tmp_t0_due_repayment AS p
        ON p.apply_serial_id = m.apply_serial_id;


-- 结果 1：实时模型分覆盖率与固定分层阈值

SELECT (SELECT COUNT(*) FROM tmp_t0_event) AS all_t0_events,
       (SELECT COUNT(*) FROM tmp_t0_scored) AS scored_t0_events,
       (SELECT COUNT(*) FROM tmp_t0_tier) AS tiered_t0_events,
       p30, p60
FROM tmp_t0_score_cut;


-- 结果 2：月度 × 实时资质层 × 提单时段

SELECT TO_CHAR(vir_date, 'YYYY-MM') AS stat_month,
       quality_tier,
       COUNT(*) AS t0_events,
       SUM(CASE WHEN hour_bucket = 0 THEN 1 ELSE 0 END) AS apply_0_1h,
       SUM(CASE WHEN hour_bucket BETWEEN 1 AND 2 THEN 1 ELSE 0 END) AS apply_1_3h,
       SUM(CASE WHEN hour_bucket BETWEEN 3 AND 5 THEN 1 ELSE 0 END) AS apply_3_6h,
       SUM(CASE WHEN hour_bucket BETWEEN 6 AND 23 THEN 1 ELSE 0 END) AS apply_6_24h,
       SUM(CASE WHEN hour_bucket BETWEEN 0 AND 23 THEN 1 ELSE 0 END) AS apply_24h,
       AVG(score) AS avg_realtime_score
FROM tmp_t0_hour_base
GROUP BY TO_CHAR(vir_date, 'YYYY-MM'), quality_tier
ORDER BY stat_month,
         CASE quality_tier WHEN '资质较好' THEN 1 WHEN '资质一般' THEN 2 ELSE 3 END;


-- 结果 3：总计 24 个小时桶 × 实时资质层

SELECT quality_tier,
       hour_bucket,
       COUNT(*) AS first_apply_events
FROM tmp_t0_hour_base
WHERE hour_bucket BETWEEN 0 AND 23
GROUP BY quality_tier, hour_bucket
ORDER BY CASE quality_tier WHEN '资质较好' THEN 1 WHEN '资质一般' THEN 2 ELSE 3 END,
         hour_bucket;


-- 结果 4：月度实时资质结构（获额通过、24小时首提）

-- 月度实时资质结构：获额通过事件占比、获额后24小时首次提单事件占比。
WITH monthly_tier AS (
    SELECT TO_CHAR(vir_date, 'YYYY-MM') AS stat_month,
           quality_tier,
           COUNT(*) AS t0_events,
           SUM(CASE WHEN hour_bucket BETWEEN 0 AND 23 THEN 1 ELSE 0 END) AS apply_24h_events
    FROM tmp_t0_hour_base
    GROUP BY TO_CHAR(vir_date, 'YYYY-MM'), quality_tier
)
SELECT stat_month,
       quality_tier,
       t0_events,
       apply_24h_events,
       100.0 * t0_events / SUM(t0_events) OVER (PARTITION BY stat_month) AS t0_event_share,
       100.0 * apply_24h_events
             / NULLIF(SUM(apply_24h_events) OVER (PARTITION BY stat_month), 0) AS apply_24h_event_share
FROM monthly_tier
ORDER BY stat_month,
         CASE quality_tier WHEN '资质较好' THEN 1 WHEN '资质一般' THEN 2 ELSE 3 END;


-- 结果 5：月度整笔订单到期逾期率与盈利率

-- 月度整笔订单到期表现。月份按获额通过日期归属；分母为当月已整笔到期的成功放款订单。
WITH remitted AS (
    SELECT TO_CHAR(vir_date, 'YYYY-MM') AS cohort_month,
           quality_tier,
           COUNT(*) AS remitted_orders
    FROM tmp_t0_remitted_order
    GROUP BY TO_CHAR(vir_date, 'YYYY-MM'), quality_tier
), matured AS (
    SELECT TO_CHAR(vir_date, 'YYYY-MM') AS cohort_month,
           quality_tier,
           COUNT(*) AS matured_orders,
           SUM(is_overdue) AS overdue_orders,
           SUM(remit_amt) AS remit_amt,
           SUM(paid_by_due) AS paid_by_due
    FROM tmp_t0_due_order_detail
    GROUP BY TO_CHAR(vir_date, 'YYYY-MM'), quality_tier
)
SELECT r.cohort_month,
       r.quality_tier,
       r.remitted_orders,
       COALESCE(m.matured_orders, 0) AS matured_orders,
       ROUND(100.0 * COALESCE(m.matured_orders, 0) / NULLIF(r.remitted_orders, 0), 2) AS maturity_coverage_pct,
       COALESCE(m.overdue_orders, 0) AS overdue_orders,
       ROUND(100.0 * COALESCE(m.overdue_orders, 0) / NULLIF(m.matured_orders, 0), 2) AS overdue_rate_pct,
       COALESCE(m.remit_amt, 0) AS remit_amt,
       COALESCE(m.paid_by_due, 0) AS paid_by_due,
       COALESCE(m.paid_by_due, 0) - COALESCE(m.remit_amt, 0) AS profit_amt_by_due,
       ROUND(100.0 * (m.paid_by_due / NULLIF(m.remit_amt, 0) - 1), 2) AS profit_rate_pct
FROM remitted AS r
LEFT JOIN matured AS m
       ON m.cohort_month = r.cohort_month
      AND m.quality_tier = r.quality_tier
ORDER BY r.cohort_month,
         CASE r.quality_tier WHEN '资质较好' THEN 1 WHEN '资质一般' THEN 2 ELSE 3 END;


-- 结果 6：各实时资质层整体整笔订单到期逾期率与盈利率

-- 各实时资质层的整体整笔订单到期表现。
WITH remitted AS (
    SELECT quality_tier, COUNT(*) AS remitted_orders
    FROM tmp_t0_remitted_order
    GROUP BY quality_tier
), matured AS (
    SELECT quality_tier,
           COUNT(*) AS matured_orders,
           SUM(is_overdue) AS overdue_orders,
           SUM(remit_amt) AS remit_amt,
           SUM(paid_by_due) AS paid_by_due
    FROM tmp_t0_due_order_detail
    GROUP BY quality_tier
)
SELECT r.quality_tier,
       r.remitted_orders,
       COALESCE(m.matured_orders, 0) AS matured_orders,
       ROUND(100.0 * COALESCE(m.matured_orders, 0) / NULLIF(r.remitted_orders, 0), 2) AS maturity_coverage_pct,
       COALESCE(m.overdue_orders, 0) AS overdue_orders,
       ROUND(100.0 * COALESCE(m.overdue_orders, 0) / NULLIF(m.matured_orders, 0), 2) AS overdue_rate_pct,
       COALESCE(m.remit_amt, 0) AS remit_amt,
       COALESCE(m.paid_by_due, 0) AS paid_by_due,
       COALESCE(m.paid_by_due, 0) - COALESCE(m.remit_amt, 0) AS profit_amt_by_due,
       ROUND(100.0 * (m.paid_by_due / NULLIF(m.remit_amt, 0) - 1), 2) AS profit_rate_pct
FROM remitted AS r
LEFT JOIN matured AS m ON m.quality_tier = r.quality_tier
ORDER BY CASE r.quality_tier WHEN '资质较好' THEN 1 WHEN '资质一般' THEN 2 ELSE 3 END;
