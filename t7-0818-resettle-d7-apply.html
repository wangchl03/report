口径与 SQL

1,254 人

名单 t7recalllist_0818_0819，recall_date = 2026-08-18，11,585 人。召回后第一笔成功放款（提单日 ≥ 2026-08-18，is_remit = 1，放款额 > 0，按放款时间取最早一笔）且已结清：is_repaid = 1 或 loan_status_code = 7，还款时间、还款日有效。这一步 1,490 人。结清之后 is_pass1 = 1，获额时间不早于结清时间；同一用户 20 秒内多笔合成一批，批内留 serial_id 最大；每人留最早一批。这一步 1,416 人。获额日 ≤ 2026-09-30 的是 1,254 人，最早获额日是 2026-08-20。

三列和四类

流失天数是名单字段 churn_days。使用率是刚结清那笔 remit_amt / max_credit_apply_amt，获额取该 serial 最晚一条 vir_unix。提前还款天数 = due_date − repaid_date。账期 = loan_day。四类按上文顺序互斥。提单用获额 serial_id 对订单表，同一 serial 取最早一笔提单。

普通用户

获额日 2026-08-20 至 2026-09-30 的 T0 获额通过：loan_type_code = 2，is_pass1 = 1，recycle_type = 3，20 秒内多笔只留一批内最晚一笔，获额当天干净结清，同一用户同一天留最早一笔。去掉 8月18日召回名单后 28,699 人。提单窗口与 1,254 人相同。观察日 2026-10-07，这批获额的 7 日窗口都已走完。

SET statement_timeout = 0;

DROP TABLE IF EXISTS tmp_u;
CREATE TEMP TABLE tmp_u AS
SELECT DISTINCT churn_user_id::bigint AS user_id
FROM wangchuanliang.t7recalllist_0818_0819
WHERE recall_date = DATE '2026-08-18';

DROP TABLE IF EXISTS tmp_first_loan;
CREATE TEMP TABLE tmp_first_loan AS
SELECT user_id, serial_id, apply_date, remit_date, repaid_time, repaid_date, due_date
FROM (
    SELECT
        u.user_id,
        o.serial_id,
        o.apply_date::date AS apply_date,
        o.remit_date::date AS remit_date,
        o.repaid_time,
        o.repaid_date::date AS repaid_date,
        o.due_date::date AS due_date,
        o.is_repaid,
        o.loan_status_code,
        ROW_NUMBER() OVER (PARTITION BY u.user_id ORDER BY o.remit_time, o.serial_id) AS rn
    FROM tmp_u u
    INNER JOIN wangchuanliang.order_loan_f_v2_copy o ON o.user_id = u.user_id
    WHERE o.apply_date >= DATE '2026-08-18'
      AND o.is_remit = 1
      AND COALESCE(o.remit_amt, 0) > 0
) z
WHERE rn = 1
  AND (is_repaid = 1 OR loan_status_code = 7)
  AND repaid_time > TIMESTAMP '2000-01-01 00:00:00'
  AND repaid_date > DATE '2000-01-01';

DROP TABLE IF EXISTS tmp_credit_raw;

    f.user_id,
    f.serial_id AS settle_serial_id,
    f.repaid_time,
    f.repaid_date,
    v.serial_id,
    v.vir_date::date AS vir_date,
    TO_TIMESTAMP(v.vir_unix) AS vir_time,
    r.recycle_type
FROM tmp_first_loan f
INNER JOIN wangchuanliang.order_vir_f_copy v
    ON v.user_id = f.user_id
   AND v.is_pass1 = 1
   AND TO_TIMESTAMP(v.vir_unix) >= f.repaid_time
LEFT JOIN wangchuanliang.side_recycle_type_copy r ON r.serial_id = v.serial_id
ORDER BY v.user_id, v.serial_id, v.vir_unix DESC;

DROP TABLE IF EXISTS tmp_credit;
CREATE TEMP TABLE tmp_credit AS
WITH lagged AS (
    SELECT c.*,
           LAG(c.vir_time) OVER (PARTITION BY c.user_id ORDER BY c.vir_time, c.serial_id) AS prev_vir_time
    FROM tmp_credit_raw c
), batched AS (
    SELECT l.*,
           SUM(CASE WHEN l.prev_vir_time IS NULL OR l.vir_time - l.prev_vir_time > INTERVAL '20 seconds' THEN 1 ELSE 0 END)
             OVER (PARTITION BY l.user_id ORDER BY l.vir_time, l.serial_id ROWS UNBOUNDED PRECEDING) AS batch_id
    FROM lagged l
), one_batch AS (
    SELECT user_id, settle_serial_id, repaid_time, repaid_date, serial_id, vir_date, vir_time, recycle_type, batch_id
    FROM (
        SELECT b.*,
               ROW_NUMBER() OVER (PARTITION BY b.user_id, b.batch_id ORDER BY b.serial_id DESC) AS batch_rn
        FROM batched b
    ) x
    WHERE batch_rn = 1
)
SELECT user_id, settle_serial_id, repaid_date, serial_id, vir_date, vir_time, recycle_type
FROM (
    SELECT o.*,
           ROW_NUMBER() OVER (PARTITION BY o.user_id ORDER BY o.vir_time, o.serial_id DESC) AS user_rn
    FROM one_batch o
) z
WHERE user_rn = 1;

) z
WHERE rn = 1;

SELECT
    c.user_id,
    c.vir_date,
    d.churn_days,
    CASE
        WHEN d.churn_days >= 8 AND d.churn_days < 15 THEN 1
        WHEN d.churn_days >= 15 AND d.churn_days < 30 THEN 2
        WHEN d.churn_days >= 30 AND d.churn_days < 90 THEN 3
        WHEN d.churn_days >= 90 AND d.churn_days < 180 THEN 4
        ELSE 5
    END AS churn_ord,
    CASE
        WHEN d.churn_days >= 8 AND d.churn_days < 15 THEN '8-15天'
        WHEN d.churn_days >= 15 AND d.churn_days < 30 THEN '15-30天'
        WHEN d.churn_days >= 30 AND d.churn_days < 90 THEN '30-90天'
        WHEN d.churn_days >= 90 AND d.churn_days < 180 THEN '90-180天'
        ELSE '180天以上'
    END AS churn_grp,
    o.loan_day,
    o.remit_amt,
    v.max_credit_apply_amt AS credit_amt,
    o.remit_amt / v.max_credit_apply_amt AS util,
    CASE
        WHEN o.remit_amt / v.max_credit_apply_amt < 0.8 THEN 1
        WHEN o.remit_amt / v.max_credit_apply_amt < 0.999 THEN 2
        WHEN o.remit_amt / v.max_credit_apply_amt <= 1.001 THEN 3
        ELSE 4
    END AS util_ord,
    CASE
        WHEN o.remit_amt / v.max_credit_apply_amt < 0.8 THEN '80%以下'
        WHEN o.remit_amt / v.max_credit_apply_amt < 0.999 THEN '80%-100%'
        WHEN o.remit_amt / v.max_credit_apply_amt <= 1.001 THEN '接近100%'
        ELSE '100%以上'
    END AS util_grp,
    CASE WHEN o.remit_amt / v.max_credit_apply_amt < 0.8 THEN 1 ELSE 2 END AS util2_ord,
    CASE WHEN o.remit_amt / v.max_credit_apply_amt < 0.8 THEN '80%以下' ELSE '80%及以上' END AS util2_grp,
    (f.due_date - f.repaid_date) AS early_days,
    CASE
        WHEN (f.due_date - f.repaid_date) < 0 THEN 0
        WHEN (f.due_date - f.repaid_date) = 0 THEN 1
        WHEN (f.due_date - f.repaid_date) BETWEEN 1 AND 3 THEN 2
        WHEN (f.due_date - f.repaid_date) BETWEEN 4 AND 7 THEN 3
        WHEN (f.due_date - f.repaid_date) BETWEEN 8 AND 14 THEN 4
        ELSE 5
    END AS early_ord,
    CASE
        WHEN (f.due_date - f.repaid_date) < 0 THEN '逾期后还'
        WHEN (f.due_date - f.repaid_date) = 0 THEN '到期当天'
        WHEN (f.due_date - f.repaid_date) BETWEEN 1 AND 3 THEN '提前1-3天'
        WHEN (f.due_date - f.repaid_date) BETWEEN 4 AND 7 THEN '提前4-7天'
        WHEN (f.due_date - f.repaid_date) BETWEEN 8 AND 14 THEN '提前8-14天'
        ELSE '提前15天以上'
    END AS early_grp,
    CASE
        WHEN (f.due_date - f.repaid_date) <= 0 THEN 1
        WHEN (f.due_date - f.repaid_date) BETWEEN 1 AND 7 THEN 2
        ELSE 3
    END AS early3_ord,
    CASE
        WHEN (f.due_date - f.repaid_date) <= 0 THEN '未提前还'
        WHEN (f.due_date - f.repaid_date) BETWEEN 1 AND 7 THEN '提前1-7天'
        ELSE '提前8天以上'
    END AS early3_grp,
    CASE
        WHEN d.churn_days < 15 THEN 1
        WHEN d.churn_days >= 15 AND o.remit_amt / v.max_credit_apply_amt < 0.8 AND (f.due_date - f.repaid_date) <= 0 THEN 2
        WHEN d.churn_days >= 30 AND o.remit_amt / v.max_credit_apply_amt < 0.8 AND (f.due_date - f.repaid_date) BETWEEN 1 AND 7 THEN 3
        ELSE 4
    END AS seg_ord,
    CASE
        WHEN d.churn_days < 15 THEN '短流失'
        WHEN d.churn_days >= 15 AND o.remit_amt / v.max_credit_apply_amt < 0.8 AND (f.due_date - f.repaid_date) <= 0 THEN '低使用且未提前还'
        WHEN d.churn_days >= 30 AND o.remit_amt / v.max_credit_apply_amt < 0.8 AND (f.due_date - f.repaid_date) BETWEEN 1 AND 7 THEN '长流失低使用只提前几天'
        ELSE '其余'
    END AS seg_grp,
    CASE WHEN a.apply_date = c.vir_date THEN 1 ELSE 0 END AS d0,
    CASE WHEN a.apply_date >= c.vir_date AND a.apply_date <= c.vir_date + 2 THEN 1 ELSE 0 END AS d3,
    CASE WHEN a.apply_date >= c.vir_date AND a.apply_date <= c.vir_date + 6 THEN 1 ELSE 0 END AS d7

    SELECT churn_user_id::bigint AS user_id, churn_days
    FROM wangchuanliang.t7recalllist_0818_0819
    WHERE recall_date = DATE '2026-08-18'
) d ON d.user_id = c.user_id
INNER JOIN tmp_first_loan f ON f.user_id = c.user_id
INNER JOIN wangchuanliang.order_loan_f_v2_copy o ON o.serial_id = f.serial_id
INNER JOIN (
    SELECT DISTINCT ON (serial_id) serial_id, max_credit_apply_amt
    FROM wangchuanliang.order_vir_f_copy
    WHERE serial_id IN (SELECT serial_id FROM tmp_first_loan)
    ORDER BY serial_id, vir_unix DESC
) v ON v.serial_id = f.serial_id AND COALESCE(v.max_credit_apply_amt, 0) > 0

DROP TABLE IF EXISTS tmp_m_origin;
CREATE TEMP TABLE tmp_m_origin AS
WITH params AS (
    SELECT DATE '2026-08-20' AS start_date, DATE '2026-10-01' AS end_date
), raw_offer AS (
    SELECT DISTINCT ON (v.user_id, v.serial_id)
        v.user_id, v.serial_id, v.vir_date::date AS vir_date, TO_TIMESTAMP(v.vir_unix) AS vir_time
    FROM wangchuanliang.order_vir_f_copy v
    INNER JOIN wangchuanliang.side_recycle_type_copy r ON r.serial_id = v.serial_id
    CROSS JOIN params p
    WHERE v.is_pass1 = 1 AND v.loan_type_code = 2 AND r.recycle_type = 3
      AND v.vir_date >= p.start_date - 1 AND v.vir_date < p.end_date
    ORDER BY v.user_id, v.serial_id, v.vir_unix DESC
), lagged AS (
    SELECT r.*, LAG(r.vir_time) OVER (PARTITION BY r.user_id ORDER BY r.vir_time, r.serial_id) AS prev_vir_time
    FROM raw_offer r
), batched AS (
    SELECT l.*, SUM(CASE WHEN l.prev_vir_time IS NULL OR l.vir_time - l.prev_vir_time > INTERVAL '20 seconds' THEN 1 ELSE 0 END)
      OVER (PARTITION BY l.user_id ORDER BY l.vir_time, l.serial_id ROWS UNBOUNDED PRECEDING) AS batch_id
    FROM lagged l
), dedup_offer AS (
    SELECT user_id, serial_id, vir_date, vir_time
    FROM (
        SELECT b.*, ROW_NUMBER() OVER (PARTITION BY b.user_id, b.batch_id ORDER BY b.vir_time DESC, b.serial_id DESC) AS batch_rn
        FROM batched b
    ) x
    CROSS JOIN params p
    WHERE batch_rn = 1 AND vir_date >= p.start_date AND vir_date < p.end_date
), candidate_users AS (SELECT DISTINCT user_id FROM dedup_offer),
all_orders AS (
    SELECT o.user_id, o.serial_id, o.apply_time, o.repaid_time, o.repaid_date::date AS repaid_date,
           o.loan_status_code, o.is_remit, o.is_repaid
    FROM wangchuanliang.order_loan_f_v2_copy o
    INNER JOIN candidate_users u ON u.user_id = o.user_id
), clean_settlement AS (
    SELECT o.user_id, o.serial_id, o.repaid_time, o.repaid_date
    FROM all_orders o
    WHERE o.is_remit = 1 AND (o.is_repaid = 1 OR o.loan_status_code = 7)
      AND o.repaid_time > TIMESTAMP '2000-01-01 00:00:00'
      AND NOT EXISTS (
          SELECT 1 FROM all_orders x
          WHERE x.user_id = o.user_id AND x.serial_id <> o.serial_id AND x.apply_time < o.repaid_time
            AND (x.loan_status_code IN (5, 8) OR (x.loan_status_code = 7 AND x.repaid_time > o.repaid_time))
      )
), paired AS (
    SELECT e.user_id, e.serial_id AS credit_serial_id, e.vir_date, e.vir_time,
           ROW_NUMBER() OVER (PARTITION BY e.user_id, e.vir_date ORDER BY e.vir_time ASC, e.serial_id DESC) AS day_rn
    FROM dedup_offer e
    INNER JOIN clean_settlement s
        ON s.user_id = e.user_id AND s.repaid_date = e.vir_date AND e.vir_time >= s.repaid_time
)
SELECT user_id, credit_serial_id, vir_date, vir_time FROM paired WHERE day_rn = 1;

        INNER JOIN tmp_m_origin t ON t.credit_serial_id = a.serial_id
    ) z WHERE rn = 1
) o ON o.serial_id = t.credit_serial_id
