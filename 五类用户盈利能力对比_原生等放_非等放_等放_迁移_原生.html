卡点 1 — 首贷质量：等放和原生非等放的首贷 s8 率都超过 40%。等放首贷亏 618 万，原生非等放首贷亏 83 万。迁移首贷赚 2,126 万，拉高了非等放整体的首贷盈利。
卡点 2 — 留存黑洞：等放 47.4% 的人只借 1 次，盈利率 −61.3%。迁移仅 32.7% 的人只借 1 次，且单次盈利率高达 +53.9%。
卡点 3 — 原生非等放困境：被风控拒绝但仍首贷 s8 率 40.9%，件均仅 778 元。风控拒绝策略既不准（没筛掉坏人），又太严（限制了件均）。盈利率仅 3.74%。

五、结论

第一层（优质）：迁移用户，34.71%。逾期 → 恢复 → 被拒 → 再申请。三重筛选后存活，首贷 s8 率 0.4%。贡献 50.1% 的利润，但只占 14.5% 的放款。

第二层（尚可）：非等放整体，25.76%。被迁移用户拉高。原生非等放实际盈利率仅 3.74%。

第三层（及格）：原生等放整体，10.4%。88.8% 的用户。贡献 97.8% 的利润。

第四层（不及格）：等放，6.02%。75.2% 用户。43.5% 首贷 s8 率。利润贡献仅 47.7%。

第五层（最差）：原生非等放，3.74%。11.2% 用户。利润贡献 2.2%。

行动建议：如果等放首贷 s8 率从 43.5% 降到 30%，约 2.6 万用户从 s8 转为 s7，进入复借盈利区间。等放整体盈利率预计从 6% 提升到 12-15%。

附录 A：PGSQL 查询代码

-- ============================================================================
-- 五类用户盈利能力分析 — 完整 PGSQL 查询
-- ============================================================================
-- 数据范围：2025.01 ~ 2026.05 注册用户
-- 表：order_loan_f_v2_copy, repayment_plan_v2_copy
-- ============================================================================

-- 第一步：用户分类
WITH user_type AS (
    SELECT user_id,
        CASE 
            WHEN MIN(CASE WHEN is_remit = 0 THEN apply_time END) IS NULL THEN '等放'
            WHEN MIN(CASE WHEN is_remit = 1 THEN apply_time END)
               < MIN(CASE WHEN is_remit = 0 THEN apply_time END) THEN '迁移'
            ELSE '原生非等放'
        END AS utype
    FROM wangchuanliang.order_loan_f_v2_copy
    WHERE reg_date >= '2025-01-01' 
      AND reg_date <= '2026-05-31'
    GROUP BY user_id
    HAVING SUM(CASE WHEN is_remit = 1 THEN 1 ELSE 0 END) > 0
),
-- 原生等放 = 等放 + 迁移
-- 非等放 = 迁移 + 原生非等放

-- 第二步：放款序列（按 apply_time 编号）
loan_seq AS (
    SELECT 
        o.user_id,
        ut.utype,
        CASE 
            WHEN ut.utype IN ('等放','迁移') THEN '原生等放' 
            ELSE '原生非等放' 
        END AS big_type,
        o.remit_amt,
        o.repaid_amt,
        o.apply_amt,
        o.overdue_days,
        o.loan_status_code,
        o.loan_day,
        ROW_NUMBER() OVER (PARTITION BY o.user_id ORDER BY o.apply_time) AS seq
    FROM wangchuanliang.order_loan_f_v2_copy o
    JOIN user_type ut ON o.user_id = ut.user_id
    WHERE o.is_remit = 1
      AND o.reg_date >= '2025-01-01' 
      AND o.reg_date <= '2026-05-31'
),

-- 第三步：按用户类型 + 放款次序聚合
seq_agg AS (
    SELECT 
        big_type,
        utype,
        CASE 
            WHEN seq = 1 THEN '首贷'
            WHEN seq = 2 THEN '第2次放款'
            WHEN seq = 3 THEN '第3次放款'
            ELSE '第4次+放款'
        END AS stage,
        COUNT(*)                                                AS n_orders,
        COUNT(DISTINCT user_id)                                 AS n_users,
        SUM(CASE WHEN overdue_days > 0 THEN 1 ELSE 0 END)       AS od_orders,
        SUM(CASE WHEN loan_status_code = 8 THEN 1 ELSE 0 END)   AS s8_orders,
        SUM(CASE WHEN loan_status_code = 7 THEN 1 ELSE 0 END)   AS s7_orders,
        SUM(CASE WHEN loan_status_code = 5 THEN 1 ELSE 0 END)   AS s5_orders,
        COALESCE(SUM(CASE WHEN loan_status_code IN (7,8) THEN remit_amt END), 0) AS end_remit,
        COALESCE(SUM(CASE WHEN loan_status_code IN (7,8) THEN repaid_amt END), 0) AS end_repaid,
        COALESCE(SUM(CASE WHEN loan_status_code = 8 THEN remit_amt END), 0)       AS s8_remit,
        COALESCE(SUM(CASE WHEN loan_status_code = 8 THEN repaid_amt END), 0)      AS s8_repaid,
        SUM(remit_amt)                                          AS total_remit,
        SUM(apply_amt)                                          AS total_apply,
        AVG(remit_amt)                                          AS avg_remit,
        AVG(loan_day)                                           AS avg_loan_day
    FROM loan_seq
    GROUP BY big_type, utype, stage
)

-- 第四步：最终输出
SELECT 
    big_type   AS "大类",
    utype      AS "子类",
    stage      AS "阶段",
    n_orders   AS "订单数",
    n_users    AS "用户数",
    -- 订单级逾期率
    ROUND(od_orders::numeric / NULLIF(n_orders, 0) * 100, 1)     AS "订单级od>0%",
    ROUND(s8_orders::numeric / NULLIF(n_orders, 0) * 100, 1)     AS "订单级s8%",
    -- 盈利率
    ROUND((end_repaid - end_remit) / NULLIF(end_remit, 0) * 100, 2) AS "盈利率%",
    -- 逾期回款率
    ROUND(s8_repaid / NULLIF(s8_remit, 0) * 100, 2)               AS "逾期回款率%",
    -- 件均
    ROUND(total_remit / NULLIF(n_orders, 0))                     AS "件均放款额",
    ROUND(total_apply / NULLIF(n_orders, 0))                     AS "件均申请额",
    ROUND(avg_remit)                                              AS "平均放款额",
    ROUND(avg_loan_day)                                           AS "平均期限(天)"
FROM seq_agg
ORDER BY big_type, utype, stage;


-- ============================================================================
-- 用户维度：总放款次数分布 + 单次/多次盈利拆解
-- ============================================================================
WITH user_type AS (
    -- 同上
    SELECT user_id,
        CASE 
            WHEN MIN(CASE WHEN is_remit = 0 THEN apply_time END) IS NULL THEN '等放'
            WHEN MIN(CASE WHEN is_remit = 1 THEN apply_time END)
               < MIN(CASE WHEN is_remit = 0 THEN apply_time END) THEN '迁移'
            ELSE '原生非等放'
        END AS utype
    FROM wangchuanliang.order_loan_f_v2_copy
    WHERE reg_date >= '2025-01-01' 
      AND reg_date <= '2026-05-31'
    GROUP BY user_id
    HAVING SUM(CASE WHEN is_remit = 1 THEN 1 ELSE 0 END) > 0
),
user_loan_cnt AS (
    SELECT 
        ut.utype,
        ut.user_id,
        SUM(CASE WHEN o.is_remit = 1 THEN 1 ELSE 0 END) AS total_loans,
        -- 单次用户
        COALESCE(SUM(CASE WHEN o.is_remit = 1 AND o.loan_status_code IN (7,8) THEN o.remit_amt END), 0) AS s1_remit,
        COALESCE(SUM(CASE WHEN o.is_remit = 1 AND o.loan_status_code IN (7,8) THEN o.repaid_amt END), 0) AS s1_repaid,
        -- 多次用户
        COALESCE(SUM(CASE WHEN o.is_remit = 1 AND o.loan_status_code IN (7,8) THEN o.remit_amt END), 0) AS m_remit,
        COALESCE(SUM(CASE WHEN o.is_remit = 1 AND o.loan_status_code IN (7,8) THEN o.repaid_amt END), 0) AS m_repaid
    FROM user_type ut
    JOIN wangchuanliang.order_loan_f_v2_copy o ON ut.user_id = o.user_id
    WHERE o.reg_date >= '2025-01-01' 
      AND o.reg_date <= '2026-05-31'
    GROUP BY ut.utype, ut.user_id
)
SELECT 
    utype,
    COUNT(*)                                                              AS n_users,
    SUM(CASE WHEN total_loans = 1 THEN 1 ELSE 0 END)                      AS single_users,
    SUM(CASE WHEN total_loans >= 2 THEN 1 ELSE 0 END)                     AS multi_users,
    -- 单次用户盈利率
    ROUND((SUM(CASE WHEN total_loans = 1 THEN s1_repaid ELSE 0 END) 
         - SUM(CASE WHEN total_loans = 1 THEN s1_remit ELSE 0 END))
         / NULLIF(SUM(CASE WHEN total_loans = 1 THEN s1_remit ELSE 0 END), 0) * 100, 1) AS single_profit_pct,
    -- 多次用户盈利率
    ROUND((SUM(CASE WHEN total_loans >= 2 THEN m_repaid ELSE 0 END) 
         - SUM(CASE WHEN total_loans >= 2 THEN m_remit ELSE 0 END))
         / NULLIF(SUM(CASE WHEN total_loans >= 2 THEN m_remit ELSE 0 END), 0) * 100, 1) AS multi_profit_pct
FROM user_loan_cnt
GROUP BY utype
ORDER BY utype;

#!/usr/bin/env python3
"""五类用户盈利能力分析 — Python 数据提取脚本"""
import psycopg2, json

DB = {
    "host": "47.89.225.85", "port": "8000", "dbname": "kaby_dw",
    "user": "wangchuanliang_readonly", "password": "gMOZQxb3spunr"
}

conn = psycopg2.connect(**DB)
cur = conn.cursor()

# === 查询1：用户分类 + 放款序列 + 订单级指标 ===
q1 = """
WITH user_type AS (
    SELECT user_id,
        CASE 
            WHEN MIN(CASE WHEN is_remit=0 THEN apply_time END) IS NULL THEN '等放'
            WHEN MIN(CASE WHEN is_remit=1 THEN apply_time END)
               < MIN(CASE WHEN is_remit=0 THEN apply_time END) THEN '迁移'
            ELSE '原生非等放'
        END AS utype
    FROM wangchuanliang.order_loan_f_v2_copy
    WHERE reg_date>='2025-01-01' AND reg_date<='2026-05-31'
    GROUP BY user_id HAVING SUM(CASE WHEN is_remit=1 THEN 1 ELSE 0 END)>0
),
loan_seq AS (
    SELECT o.user_id, ut.utype,
        o.remit_amt, o.repaid_amt, o.apply_amt,
        o.overdue_days, o.loan_status_code, o.loan_day,
        ROW_NUMBER() OVER (PARTITION BY o.user_id ORDER BY o.apply_time) AS seq
    FROM wangchuanliang.order_loan_f_v2_copy o
    JOIN user_type ut ON o.user_id=ut.user_id
    WHERE o.is_remit=1 AND o.reg_date>='2025-01-01' AND o.reg_date<='2026-05-31'
)
SELECT 
    CASE WHEN utype IN ('等放','迁移') THEN '原生等放' ELSE '原生非等放' END AS big_type,
    utype,
    CASE WHEN seq=1 THEN '首贷' WHEN seq=2 THEN '第2次' WHEN seq=3 THEN '第3次' ELSE '第4次+' END AS stage,
    COUNT(*),
    SUM(remit_amt),
    SUM(CASE WHEN loan_status_code IN (7,8) THEN remit_amt END),
    SUM(CASE WHEN loan_status_code IN (7,8) THEN repaid_amt END),
    AVG(remit_amt)
FROM loan_seq GROUP BY 1,2,3 ORDER BY 1,2,3
"""

cur.execute(q1)
rows = cur.fetchall()
# 按 (大类, 子类, 阶段) 整理数据
results = {}
for r in rows:
    bt, ut, stage, n, tr, er, ep, avgr = r
    profit = float(ep or 0) - float(er or 0)
    key = (bt, ut)
    if key not in results: results[key] = {}
    results[key][stage] = {
        'n_orders': n, 'total_remit': float(tr or 0),
        'profit': profit, 'profit_pct': profit / float(er or 1) * 100
    }

# 聚合五类指标
categories = {
    '原生等放': {'等放', '迁移'},
    '非等放': {'迁移', '原生非等放'},
    '等放': {'等放'},
    '迁移': {'迁移'},
    '原生非等放': {'原生非等放'},
}

for cat, subs in categories.items():
    total_orders = 0; total_remit = 0; total_profit = 0
    for bt, ut in results:
        if ut in subs:
            for stage, d in results[(bt, ut)].items():
                total_orders += d['n_orders']
                total_remit += d['total_remit']
                total_profit += d['profit']
    print(f"{cat}: 订单={total_orders:,}, 放款={total_remit:,.0f}, 利润={total_profit:,.0f}, 盈利率={total_profit/total_remit*100:.1f}%")

cur.close()
conn.close()
