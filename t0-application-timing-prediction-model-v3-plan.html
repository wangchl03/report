二、定义与时点合同

对象口径
循环贷可提单资格is_pass1=1且recycle_type=2；同用户/serial在预测前取最新记录，再20秒链式去重。不是所有用户都有。
最早贷款阶段资格对应潜在第二笔及以后的贷款：已有一笔成功放款也可能获得资格。不按成功放款次数≥2过滤。首贷本身不是循环贷订单。
历史循环贷提单获额serial_id与订单serial_id匹配，apply_time严格早于当前T0；不要求成功放款，不用其他订单回填。
历史循环贷结果分别保存截至时点匹配率、已结束机会提单率、过期/覆盖未提单、提单间隔；无资格/无有效分母比率NaN，不填0。
覆盖/失效min(获额时点+15×24小时、后续已知type2/3获额时点、当前T0)作为机会结束边界；没有权威expire_at，用业务规则代理并披露。
历史T0逐用户恢复全可观察历史的严格type3事件：同日整笔结清、0在贷、批准、续贷、20秒批次。不是所有结清后续借都算T0。
历史T0转化口径1任意订单首次后续提单归最近前序T0；结果只用apply<当前T0。窗口未成熟或下一T0删失未知，不当负样本。
历史时点分档D0、1–3、4–7、8–15、>15自然日；同时保留实际经过天数。D30仅作历史习惯特征，不新增本次预测标签。

四、模型结构与实施流程

模型结构｜离散时间条件风险

沿用V2的单个LightGBM条件风险模型；对未发生首次提单且可观察的事件展开D0、D1–3、D4–7、D8–15风险行。区间编号/长度是结构输入。

累计概率｜不独立训练四个分类器

各区间条件概率经校准后组合：P(Dh内提单)=1−此前各区间(1−hazard)的连乘；因此D0≤D3≤D7≤D15。未完成或被下一T0删失的未知观察不填0。

输出字段含义
p_apply_d0/d3/d7/d15截至当天、D+3、D+7、D+15的累计提单概率，含当天1/4/8/16自然日
p_d0、p_d1_3、p_d4_7、p_d8_15互不重叠的首次提单时点区间概率
p_no_apply_d15D+15仍未提单概率；与四个时点区间概率合计1

-- T0 model V1. All tables created here are session-local TEMP tables.
-- Business timezone: America/Belize. Labels: user follow-up (definition 1).
-- Source data complete through 2026-10-06. Events: Jan 1 through Sep 30.
-- Model scores are keyed by serial_id; source has no score generation timestamp.

-- STEP params
CREATE TEMP TABLE tmp_t0ml_params AS
SELECT DATE '2026-01-01' AS start_date, DATE '2026-10-01' AS end_date,
       TIMESTAMP '2026-10-07 00:00:00' AS data_cutoff_ts;

-- STEP offer
CREATE TEMP TABLE tmp_t0ml_offer AS
SELECT DISTINCT ON (v.user_id,v.serial_id)
       v.user_id,v.serial_id,v.vir_date::date AS vir_date,
       (TO_TIMESTAMP(v.vir_unix) AT TIME ZONE 'America/Belize') AS vir_time,
       v.max_credit_apply_amt::float8 AS current_credit_amt,
       v.total_term::float8 AS current_total_term,
       v.risk_product,v.vir_version
FROM wangchuanliang.order_vir_f_copy v CROSS JOIN tmp_t0ml_params p
WHERE v.is_pass1=1 AND v.loan_type_code=2
  AND v.vir_date>=p.start_date-1 AND v.vir_date<p.data_cutoff_ts::date+1
  AND EXISTS (SELECT 1 FROM wangchuanliang.side_recycle_type_copy r
              WHERE r.serial_id=v.serial_id AND r.recycle_type=3)
ORDER BY v.user_id,v.serial_id,v.vir_unix DESC;

-- STEP dedup
CREATE TEMP TABLE tmp_t0ml_dedup AS
WITH lagged AS (
 SELECT v.*,LAG(vir_time) OVER(PARTITION BY user_id ORDER BY vir_time,serial_id) AS previous_time
 FROM tmp_t0ml_offer v
), batched AS (
 SELECT l.*,SUM(CASE WHEN previous_time IS NULL OR vir_time-previous_time>INTERVAL '20 seconds'
                    THEN 1 ELSE 0 END)
 OVER(PARTITION BY user_id ORDER BY vir_time,serial_id ROWS UNBOUNDED PRECEDING) AS batch_id
 FROM lagged l
), ranked AS (
 SELECT b.*,ROW_NUMBER() OVER(PARTITION BY user_id,batch_id ORDER BY vir_time DESC,serial_id DESC) AS rn
 FROM batched b
)
SELECT r.* FROM ranked r CROSS JOIN tmp_t0ml_params p
WHERE rn=1 AND vir_date>=p.start_date AND vir_time<p.data_cutoff_ts;

-- STEP orders
CREATE TEMP TABLE tmp_t0ml_orders AS
SELECT DISTINCT ON(o.user_id,o.serial_id)
 o.user_id,o.serial_id,o.apply_time,o.apply_date,o.apply_amt::float8,
 o.remit_time,o.remit_date,o.remit_amt::float8,o.is_remit,
 o.repaid_time,o.repaid_date,o.is_repaid,o.loan_status_code,
 o.due_date,o.total_term::float8,o.loan_day::float8,
 o.pre_amt::float8,o.post_amt::float8,o.remit_succ_rnb,
 o.reg_date,o.product_no,o.loan_type,o.repaid_amt::float8
FROM wangchuanliang.order_loan_f_v2_copy o
JOIN (SELECT DISTINCT user_id FROM tmp_t0ml_dedup) u ON u.user_id=o.user_id
ORDER BY o.user_id,o.serial_id,o.apply_time ASC;

-- STEP settlement
CREATE TEMP TABLE tmp_t0ml_settlement AS
SELECT o.* FROM tmp_t0ml_orders o CROSS JOIN tmp_t0ml_params p
WHERE o.is_remit=1 AND o.remit_amt>0
 AND (o.is_repaid=1 OR o.loan_status_code=7)
 AND o.repaid_time> TIMESTAMP '2000-01-01'
 AND o.repaid_date>=p.start_date AND o.repaid_time<p.data_cutoff_ts
 AND NOT EXISTS (
  SELECT 1 FROM tmp_t0ml_orders x
  WHERE x.user_id=o.user_id AND x.serial_id<>o.serial_id
    AND x.is_remit=1 AND x.remit_amt>0 AND x.remit_time<=o.repaid_time
    AND (x.repaid_time IS NULL OR x.repaid_time<=TIMESTAMP '2000-01-01'
         OR x.repaid_time>o.repaid_time)
 );

-- STEP events
CREATE TEMP TABLE tmp_t0ml_events_all AS
WITH matched AS (
 SELECT e.*,s.serial_id AS settled_serial_id,s.repaid_time,
  ROW_NUMBER() OVER(PARTITION BY e.user_id,e.serial_id ORDER BY s.repaid_time DESC,s.serial_id DESC) AS srn
 FROM tmp_t0ml_dedup e JOIN tmp_t0ml_settlement s
 ON s.user_id=e.user_id AND s.repaid_date=e.vir_date AND s.repaid_time<=e.vir_time
)
SELECT m.*,LEAD(vir_time) OVER(PARTITION BY user_id ORDER BY vir_time,serial_id) AS next_t0_time
FROM matched m WHERE srn=1;

-- STEP labels
CREATE TEMP TABLE tmp_t0ml_events AS
SELECT e.*,MIN(o.apply_time) AS first_apply_time,
 LEAST(COALESCE(e.next_t0_time,p.data_cutoff_ts),p.data_cutoff_ts,
       e.vir_date+INTERVAL '16 days') AS followup_end,
 p.data_cutoff_ts
FROM tmp_t0ml_events_all e CROSS JOIN tmp_t0ml_params p
LEFT JOIN tmp_t0ml_orders o ON o.user_id=e.user_id
 AND o.apply_time>=e.vir_time AND o.apply_time<p.data_cutoff_ts
 AND o.apply_time<e.vir_date+INTERVAL '16 days'
 AND (e.next_t0_time IS NULL OR o.apply_time<e.next_t0_time)
WHERE e.vir_date<p.end_date
GROUP BY e.user_id,e.serial_id,e.vir_date,e.vir_time,e.current_credit_amt,
 e.current_total_term,e.risk_product,e.vir_version,e.previous_time,e.batch_id,
 e.rn,e.settled_serial_id,e.repaid_time,e.srn,e.next_t0_time,p.data_cutoff_ts;

-- STEP previous_offer
CREATE TEMP TABLE tmp_t0ml_order_credit AS
SELECT DISTINCT ON (o.user_id,o.serial_id)
 o.user_id,o.serial_id,v.max_credit_apply_amt::float8 AS credit_amt,
 (TO_TIMESTAMP(v.vir_unix) AT TIME ZONE 'America/Belize') AS credit_time
FROM tmp_t0ml_orders o JOIN wangchuanliang.order_vir_f_copy v
 ON v.user_id=o.user_id AND v.serial_id=o.serial_id AND v.is_pass1=1
 AND (TO_TIMESTAMP(v.vir_unix) AT TIME ZONE 'America/Belize')<=o.apply_time
ORDER BY o.user_id,o.serial_id,v.vir_unix DESC;

-- STEP history
CREATE TEMP TABLE tmp_t0ml_order_history AS
WITH ranked AS (
 SELECT e.serial_id AS event_id,o.*,
  v.credit_amt,v.credit_time,
  ROW_NUMBER() OVER(PARTITION BY e.serial_id ORDER BY o.apply_time DESC,o.serial_id DESC) AS arn,
  ROW_NUMBER() OVER(PARTITION BY e.serial_id
    ORDER BY CASE WHEN o.is_remit=1 AND o.remit_time<=e.vir_time THEN o.remit_time END DESC NULLS LAST,
             o.serial_id DESC) AS rrn,
  CASE WHEN o.is_remit=1 AND o.remit_amt>0 AND o.remit_time<=e.vir_time THEN 1 ELSE 0 END AS funded,
  CASE WHEN o.repaid_time>TIMESTAMP '2000-01-01' AND o.repaid_time<=e.vir_time THEN 1 ELSE 0 END AS settled
 FROM tmp_t0ml_events e JOIN tmp_t0ml_orders o ON o.user_id=e.user_id AND o.apply_time<e.vir_time
 LEFT JOIN tmp_t0ml_order_credit v ON v.user_id=o.user_id AND v.serial_id=o.serial_id
), calculated AS (
 SELECT r.*,
  CASE WHEN apply_amt>0 AND credit_amt>0 THEN apply_amt/credit_amt END AS utilization,
  CASE WHEN credit_time IS NOT NULL THEN EXTRACT(EPOCH FROM (apply_time-credit_time))/86400.0 END AS credit_apply_gap_days
 FROM ranked r
)
SELECT event_id,COUNT(*)::float8 AS historical_apply_count,
 SUM(funded)::float8 AS successful_loan_count,
 SUM(CASE WHEN funded=1 THEN remit_amt ELSE 0 END)::float8 AS total_remit_amt,
 SUM(CASE WHEN settled=1 AND repaid_date<due_date THEN 1 ELSE 0 END)::float8 AS early_settle_count,
 SUM(CASE WHEN settled=1 AND repaid_date>due_date THEN 1 ELSE 0 END)::float8 AS late_settle_count,
 MAX(CASE WHEN settled=1 THEN GREATEST(repaid_date-due_date,0) END)::float8 AS max_past_late_days,
 MIN(reg_date) AS registration_date,
 MAX(CASE WHEN arn=1 THEN apply_amt END) AS apply_amt_l1,
 MAX(CASE WHEN arn=2 THEN apply_amt END) AS apply_amt_l2,
 MAX(CASE WHEN arn=3 THEN apply_amt END) AS apply_amt_l3,
 MAX(CASE WHEN arn=1 THEN utilization END) AS apply_util_l1,
 MAX(CASE WHEN arn=2 THEN utilization END) AS apply_util_l2,
 MAX(CASE WHEN arn=3 THEN utilization END) AS apply_util_l3,
 MAX(CASE WHEN arn=1 THEN credit_apply_gap_days END) AS apply_gap_l1,
 MAX(CASE WHEN arn=2 THEN credit_apply_gap_days END) AS apply_gap_l2,
 MAX(CASE WHEN arn=3 THEN credit_apply_gap_days END) AS apply_gap_l3,
 MAX(CASE WHEN funded=1 AND rrn=1 THEN remit_amt END) AS remit_amt_l1,
 MAX(CASE WHEN funded=1 AND rrn=2 THEN remit_amt END) AS remit_amt_l2,
 MAX(CASE WHEN funded=1 AND rrn=3 THEN remit_amt END) AS remit_amt_l3,
 MAX(apply_time) AS last_apply_time,
 MAX(CASE WHEN funded=1 THEN remit_time END) AS last_remit_time
FROM calculated GROUP BY event_id;

-- STEP credit_history
CREATE TEMP TABLE tmp_t0ml_credit_history AS

), ranked AS (

SELECT event_id,COUNT(*)::float8 AS credit_event_count,

FROM ranked GROUP BY event_id;

def prepare(frame):

    for c in ['as_of_time','vir_date','first_apply_time','followup_end','next_t0_time','data_cutoff_ts']:
        if c in df: df[c]=pd.to_datetime(df[c],format='mixed')

            df[c]=pd.to_numeric(df[c],errors='coerce')
    df['credit_change_amt']=df.current_credit_amt-df.previous_credit_amt
    df['credit_change_ratio']=df.credit_change_amt/df.previous_credit_amt.where(df.previous_credit_amt>0)
    df['credit_direction']=np.sign(df.credit_change_amt)

def expand(frame):
    chunks=[]
    for i,(start,end) in enumerate(zip(START_DAYS,END_DAYS)):
        left=frame.as_of_time if i==0 else frame.vir_date+pd.to_timedelta(start,unit='D')
        right=frame.vir_date+pd.to_timedelta(end,unit='D')
        in_risk=(frame.followup_end>left)&(frame.first_apply_time.isna()|(frame.first_apply_time>=left))
        hit=frame.first_apply_time.notna()&(frame.first_apply_time>=left)&(frame.first_apply_time<right)
        complete=frame.followup_end>=right
        selected=in_risk&(hit|complete)
        part=frame.loc[selected].copy()
        part['interval_id']=i
        part['interval_start_day']=start
        part['interval_length']=end-start
        part['hazard_label']=hit.loc[selected].astype(int)
        chunks.append(part)
    return pd.concat(chunks,ignore_index=True)

def make_preprocessor(columns):
    numeric=[c for c in columns if c not in CATEGORIES]
    categorical=[c for c in columns if c in CATEGORIES]
    return ColumnTransformer([
        ('numeric',Pipeline([('impute',SimpleImputer(strategy='median',add_indicator=True,keep_empty_features=True)),
                             ('scale',StandardScaler())]),numeric),
        ('categorical',OneHotEncoder(handle_unknown='ignore',min_frequency=100,
                                    max_categories=60,sparse_output=False),categorical),
    ],sparse_threshold=0)


def predict_raw(model,x):
    if hasattr(model,'booster_'):
        p=model.booster_.predict(x,num_threads=4)
    else:
        with np.errstate(over='ignore',invalid='ignore',divide='ignore'):
            p=model.predict_proba(x)[:,1]
    if not np.isfinite(p).all():raise AssertionError('Non-finite model probabilities')
    return p


def predict_cumulative(bundle,frame):
    hazards=[]
    for i,(start,end) in enumerate(zip(START_DAYS,END_DAYS)):
        current=frame.copy();current['interval_id']=i
        current['interval_start_day']=start;current['interval_length']=end-start
        x=bundle['preprocessor'].transform(current[bundle['features']].replace([np.inf,-np.inf],np.nan))
        p=predict_raw(bundle['model'],x)
        calibrator=bundle.get('calibrators',{}).get(i)
        if calibrator:
            if calibrator['kind']=='sigmoid':
                z=np.log(np.clip(p,1e-6,1-1e-6)/np.clip(1-p,1e-6,1))
                p=calibrator['model'].predict_proba(z.reshape(-1,1))[:,1]
            elif calibrator['kind']=='isotonic':p=calibrator['model'].predict(p)
        hazards.append(np.clip(p,1e-7,1-1e-7))
    hazards=np.column_stack(hazards)
    cumulative=1-np.cumprod(1-hazards,axis=1)
    if not np.isfinite(cumulative).all():raise AssertionError('Non-finite cumulative probabilities')
    if np.any(np.diff(cumulative,axis=1)<-1e-12):raise AssertionError('Probabilities are not monotone')
    return cumulative,hazards

def metric(y,p):
    valid=np.isfinite(y);y=np.asarray(y)[valid].astype(int);p=np.asarray(p)[valid]
    if len(y)<2 or len(np.unique(y))<2:return {'n':int(len(y))}
    top=max(1,int(np.ceil(len(y)*.2)))
    bins=pd.qcut(pd.Series(p),q=10,duplicates='drop') if len(np.unique(p))>1 else pd.Series(0,index=range(len(p)))
    cal=pd.DataFrame({'y':y,'p':p,'bin':bins}).groupby('bin',observed=True).agg(n=('y','size'),actual=('y','mean'),predicted=('p','mean'))
    fpr,tpr,_=roc_curve(y,p)
    top10=max(1,int(np.ceil(len(y)*.1)))
    def captured(k):
        cut=np.partition(p,len(p)-k)[len(p)-k]
        stronger=p>cut;ties=p==cut
        return float(y[stronger].sum()+(k-stronger.sum())/ties.sum()*y[ties].sum())
    captured20=captured(top);captured10=captured(top10)
    return {'n':int(len(y)),'positive':int(y.sum()),'actual_rate':float(y.mean()),'predicted_rate':float(p.mean()),
            'auc':float(roc_auc_score(y,p)),'pr_auc':float(average_precision_score(y,p)),
            'ks':float(np.max(tpr-fpr)),
            'brier':float(brier_score_loss(y,p)),'log_loss':float(log_loss(y,p)),
            'top10_lift':float(captured10/top10/y.mean()),
            'top10_coverage':float(captured10/y.sum()),
            'top20_lift':float(captured20/top/y.mean()),
            'top20_coverage':float(captured20/y.sum()),
            'ece':float((cal.n*(cal.actual-cal.predicted).abs()).sum()/len(y))}

def drift_metrics(training,test,columns):
    records=[]
    for c in columns:
        if c.startswith('interval_'):continue
        a=training[c].replace([np.inf,-np.inf],np.nan);b=test[c].replace([np.inf,-np.inf],np.nan)
        if c in CATEGORIES:
            common=set(a.value_counts().head(30).index)
            aa=a.fillna('MISSING').map(lambda v:v if v in common else 'OTHER')
            bb=b.fillna('MISSING').map(lambda v:v if v in common else 'OTHER')
            categories=list(common)+['OTHER']
        else:
            values=pd.to_numeric(a,errors='coerce').dropna()
            breaks=np.unique(values.quantile(np.linspace(.1,.9,9)).to_numpy()) if len(values) else np.array([])
            edges=np.r_[-np.inf,breaks,np.inf]
            aa=pd.cut(a,edges,labels=False,include_lowest=True).fillna(-1)
            bb=pd.cut(b,edges,labels=False,include_lowest=True).fillna(-1)
            categories=list(range(len(edges)-1))+[-1]
        pa=aa.value_counts(normalize=True).reindex(categories,fill_value=0).to_numpy()+1e-4
        pb=bb.value_counts(normalize=True).reindex(categories,fill_value=0).to_numpy()+1e-4
        pa/=pa.sum();pb/=pb.sum()
        records.append({'feature':c,'psi':float(np.sum((pb-pa)*np.log(pb/pa))),
            'train_missing':float(a.isna().mean()),'test_missing':float(b.isna().mean())})
    return sorted(records,key=lambda r:r['psi'],reverse=True)

def dump_query(cur,sql,path):
    # Compress with the database's existing function before crossing the slow
    # network. The random transport key is temporary, is not a database
    # password, and is never saved. No extension or persistent DDL is created.
    import warnings
    from cryptography.utils import CryptographyDeprecationWarning
    with warnings.catch_warnings():
        warnings.simplefilter('ignore',CryptographyDeprecationWarning)
        import pgpy
    partial=path.with_name(path.name+'.partial')
    n=0;key=secrets.token_hex(24)
    cur.execute('SELECT * FROM ('+sql+') contract LIMIT 0')
    columns=[d[0] for d in cur.description]
    quoted=[]
    for col in columns:
        ident='exported."'+col.replace('"','""')+'"'
        quoted.append("CASE WHEN "+ident+" IS NULL THEN '' ELSE chr(34)||replace("+ident+"::text,chr(34),chr(34)||chr(34))||chr(34) END")
    csv_line='concat_ws(chr(44),'+','.join(quoted)+')'
    with gzip.open(partial,'wt',encoding='utf-8',newline='') as handle:
        writer=csv.writer(handle)
        writer.writerow(columns)
        while True:
            packet='SELECT * FROM ('+sql+') ordered_rows ORDER BY 1 LIMIT 10000 OFFSET '+str(n)
            compressed="SELECT encode(pgp_sym_encrypt(COALESCE(string_agg("+csv_line+",chr(10)),''),%s,'compress-algo=1,compress-level=6,cipher-algo=aes128'),'base64') FROM ("+packet+') exported'
            cur.execute(compressed,(key,))
            blob=base64.b64decode(cur.fetchone()[0])
            with warnings.catch_warnings():
                warnings.simplefilter('ignore',CryptographyDeprecationWarning)
                decoded=pgpy.PGPMessage.from_blob(blob).decrypt(key).message
            rows=list(csv.reader(io.StringIO(decoded,newline='')))
            if not rows:break
            if any(len(r)!=len(columns) for r in rows):raise AssertionError('CSV column count mismatch')
            writer.writerows(rows);n+=len(rows)
            print('EXPORT',path.name,n,flush=True)
    partial.replace(path)
    return n

def known_orders(orders,t):
    return orders.loc[orders.apply_time.lt(t)].sort_values(['apply_time','serial_id'])

def t0_history(offers,orders,t):
    candidates=dedup(offers.loc[offers.has_type3.eq(1)&offers.loan_type_code.eq(2)])
    funded=orders.loc[orders.is_remit.eq(1)&orders.remit_amt.gt(0)&orders.remit_time.le(t)]
    closed=funded.loc[((funded.is_repaid==1)|(funded.loan_status_code==7))&funded.repaid_time.gt(pd.Timestamp('2000-01-01'))&funded.repaid_time.lt(t)]
    dates={d:g.sort_values(['repaid_time','serial_id']) for d,g in closed.groupby('repaid_date',sort=False)}
    good=[]
    for c in candidates.itertuples():
        g=dates.get(c.vir_date)
        if g is None:continue
        g=g.loc[g.repaid_time.le(c.credit_time)]
        eligible=[]
        for o in g.itertuples():
            active=funded.loc[funded.serial_id.ne(o.serial_id)&funded.remit_time.le(o.repaid_time)&
                  (funded.repaid_time.isna()|funded.repaid_time.le(pd.Timestamp('2000-01-01'))|funded.repaid_time.gt(o.repaid_time))]
            if len(active)==0:eligible.append(o)
        if eligible:
            p=eligible[-1];good.append({'serial_id':c.serial_id,'credit_time':c.credit_time,'vir_date':c.vir_date,'settled_serial_id':p.serial_id})
    return pd.DataFrame(good,columns=['serial_id','credit_time','vir_date','settled_serial_id'])

def digest(p):
    h=hashlib.sha256()
    with p.open('rb') as f:
        for b in iter(lambda:f.read(1048576),b''):h.update(b)
    return h.hexdigest()

def predict(bundle,x,rows=None):
    hazards=[];cols=bundle['column_indices'];rows=np.arange(len(x)) if rows is None else np.asarray(rows)
    at={n:i for i,n in enumerate(bundle['features'])}

        pred=[]

            for c,v in [('interval_id',j),('interval_length',end-start),('interval_start_day',start)]:batch[:,at[c]]=v
            p=bundle['model'].booster_.predict(batch,num_threads=4)
            cal=bundle.get('calibrators',{}).get(j)
            if cal:
                if cal['kind']=='sigmoid':
                    logit=np.log(np.clip(p,1e-6,1-1e-6)/np.clip(1-p,1e-6,1));p=cal['model'].predict_proba(logit[:,None])[:,1]
                else:p=cal['model'].predict(p)
            pred.append(np.clip(p,1e-7,1-1e-7))
        hazards.append(np.concatenate(pred))
    hazard=np.column_stack(hazards);cumulative=1-np.cumprod(1-hazard,axis=1)
    if not np.isfinite(cumulative).all() or (np.diff(cumulative,axis=1)<-1e-12).any():raise AssertionError('Invalid prediction')
    return cumulative,hazard

def psi(a,b):
    a=np.asarray(a,float);b=np.asarray(b,float);av=a[np.isfinite(a)]
    if len(av)==0:return np.nan
    cuts=np.unique(np.quantile(av,np.linspace(0,1,11)))[1:-1]
    def counts(v):
        finite=np.isfinite(v);bins=np.searchsorted(cuts,v[finite],side='right')
        return np.r_[np.bincount(bins,minlength=len(cuts)+1),np.sum(~finite)]/len(v)
    p=np.clip(counts(a),1e-6,1);q=np.clip(counts(b),1e-6,1)
    return float(np.sum((q-p)*np.log(q/p)))
