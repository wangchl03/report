D0～D3：v1 71.93%，v2 73.85%（上次 v2 为 73.85%），仍高 1.92pp。v2 全窗前 1/3 占比 68.09%，随中低分后续提单进来会略低于短窗，但仍明显高于随机 33%。

v2 AUC 0.749（v1 0.708）。S 档提单率 25.26%，约是全体 7.84% 的 3.2 倍。经营上仍适合按 v2 分做触达排序和额度/券倾斜。

-- =============================================================================
-- T7 召回 · 提单意愿模型 V2 复验
-- 库：kaby_dw · schema wangchuanliang
-- 名单：v1 t7recall_active0910（触达 2026-09-10）
--       v2 t7recalllist_active0918（触达 2026-09-18）
-- 提单：order_loan_f_v2_copy，触达日起至 as_of（CURRENT_DATE）首次 apply_date
-- 主指标：score 前 1/3 流失用户中的提单人数 / 该批全部提单人数
-- 对齐窗：D0～D3；以及触达后同长 12 天（D0～D11）
-- 五档：S 前10% / A 10–20% / B 20–40% / C 40–70% / D 70–100%
-- =============================================================================

SET search_path TO wangchuanliang, public;
SET statement_timeout = 0;

DROP TABLE IF EXISTS tmp_p;
CREATE TEMP TABLE tmp_p AS
SELECT
    DATE '2026-09-10' AS v1_start,
    DATE '2026-09-18' AS v2_start,
    CURRENT_DATE AS as_of;

SELECT as_of FROM tmp_p;

-- ---------- v1 ----------
DROP TABLE IF EXISTS tmp_u_v1;
CREATE TEMP TABLE tmp_u_v1 AS
SELECT DISTINCT ON (churn_user_id)
    churn_user_id::bigint AS user_id,
    score::numeric AS score
FROM wangchuanliang.t7recall_active0910
WHERE churn_user_id IS NOT NULL
ORDER BY churn_user_id, score DESC NULLS LAST;

DROP TABLE IF EXISTS tmp_r_v1;
CREATE TEMP TABLE tmp_r_v1 AS
SELECT u.*, NTILE(3) OVER (ORDER BY score DESC NULLS LAST, user_id) AS tertile
FROM tmp_u_v1 u;

DROP TABLE IF EXISTS tmp_a_v1;
CREATE TEMP TABLE tmp_a_v1 AS
SELECT o.user_id, MIN(o.apply_date) AS first_apply
FROM tmp_u_v1 u
INNER JOIN wangchuanliang.order_loan_f_v2_copy o ON o.user_id = u.user_id
CROSS JOIN tmp_p p
WHERE o.apply_date >= p.v1_start AND o.apply_date <= p.as_of
GROUP BY o.user_id;

-- R1 v1 总览
SELECT
    'v1_0910' AS batch,
    (SELECT as_of::text FROM tmp_p) AS as_of,
    (SELECT COUNT(*) FROM tmp_r_v1) AS n_churn,
    (SELECT COUNT(*) FROM tmp_r_v1 WHERE tertile = 1) AS n_top,
    (SELECT MIN(score) FROM tmp_r_v1 WHERE tertile = 1) AS top_score_min,
    (SELECT MAX(score) FROM tmp_r_v1 WHERE tertile = 1) AS top_score_max,
    (SELECT COUNT(*) FROM tmp_a_v1) AS n_apply,
    (SELECT COUNT(*) FROM tmp_a_v1 a INNER JOIN tmp_r_v1 r ON r.user_id = a.user_id AND r.tertile = 1) AS n_apply_top,
    ROUND(100.0 * (SELECT COUNT(*) FROM tmp_a_v1 a INNER JOIN tmp_r_v1 r ON r.user_id = a.user_id AND r.tertile = 1)
        / NULLIF((SELECT COUNT(*) FROM tmp_a_v1), 0), 2) AS share_t1,
    (SELECT COUNT(*) FROM tmp_a_v1 CROSS JOIN tmp_p p WHERE first_apply <= p.v1_start + 3) AS n_apply_d3,
    (SELECT COUNT(*) FROM tmp_a_v1 a INNER JOIN tmp_r_v1 r ON r.user_id = a.user_id AND r.tertile = 1
        CROSS JOIN tmp_p p WHERE a.first_apply <= p.v1_start + 3) AS n_apply_top_d3,
    (SELECT COUNT(*) FROM tmp_a_v1 CROSS JOIN tmp_p p WHERE first_apply <= p.v1_start + 11) AS n_apply_d11,
    (SELECT COUNT(*) FROM tmp_a_v1 a INNER JOIN tmp_r_v1 r ON r.user_id = a.user_id AND r.tertile = 1
        CROSS JOIN tmp_p p WHERE a.first_apply <= p.v1_start + 11) AS n_apply_top_d11;

-- R2 v1 三分位
SELECT r.tertile,
       COUNT(*) AS n_churn,
       COUNT(a.user_id) AS n_apply,
       ROUND(100.0 * COUNT(a.user_id) / NULLIF(COUNT(*), 0), 2) AS apply_rate_pct
FROM tmp_r_v1 r
LEFT JOIN tmp_a_v1 a ON a.user_id = r.user_id
GROUP BY r.tertile
ORDER BY r.tertile;

-- R3 v1 按首次提单日
SELECT a.first_apply::text AS d,
       (a.first_apply - (SELECT v1_start FROM tmp_p)) AS day_idx,
       COUNT(*) AS n_apply,
       SUM(CASE WHEN r.tertile = 1 THEN 1 ELSE 0 END) AS n_apply_top
FROM tmp_a_v1 a
INNER JOIN tmp_r_v1 r ON r.user_id = a.user_id
GROUP BY 1, 2
ORDER BY 1;

-- ---------- v2 ----------
DROP TABLE IF EXISTS tmp_u_v2;
CREATE TEMP TABLE tmp_u_v2 AS
SELECT DISTINCT ON (churn_user_id)
    churn_user_id::bigint AS user_id,
    score::numeric AS score
FROM wangchuanliang.t7recalllist_active0918
WHERE churn_user_id IS NOT NULL
ORDER BY churn_user_id, score DESC NULLS LAST;

DROP TABLE IF EXISTS tmp_r_v2;
CREATE TEMP TABLE tmp_r_v2 AS
SELECT u.*, NTILE(3) OVER (ORDER BY score DESC NULLS LAST, user_id) AS tertile
FROM tmp_u_v2 u;

DROP TABLE IF EXISTS tmp_a_v2;
CREATE TEMP TABLE tmp_a_v2 AS
SELECT o.user_id, MIN(o.apply_date) AS first_apply
FROM tmp_u_v2 u
INNER JOIN wangchuanliang.order_loan_f_v2_copy o ON o.user_id = u.user_id
CROSS JOIN tmp_p p
WHERE o.apply_date >= p.v2_start AND o.apply_date <= p.as_of
GROUP BY o.user_id;

-- R4 v2 总览
SELECT
    'v2_0918' AS batch,
    (SELECT as_of::text FROM tmp_p) AS as_of,
    (SELECT COUNT(*) FROM tmp_r_v2) AS n_churn,
    (SELECT COUNT(*) FROM tmp_r_v2 WHERE tertile = 1) AS n_top,
    (SELECT MIN(score) FROM tmp_r_v2 WHERE tertile = 1) AS top_score_min,
    (SELECT MAX(score) FROM tmp_r_v2 WHERE tertile = 1) AS top_score_max,
    (SELECT COUNT(*) FROM tmp_a_v2) AS n_apply,
    (SELECT COUNT(*) FROM tmp_a_v2 a INNER JOIN tmp_r_v2 r ON r.user_id = a.user_id AND r.tertile = 1) AS n_apply_top,
    ROUND(100.0 * (SELECT COUNT(*) FROM tmp_a_v2 a INNER JOIN tmp_r_v2 r ON r.user_id = a.user_id AND r.tertile = 1)
        / NULLIF((SELECT COUNT(*) FROM tmp_a_v2), 0), 2) AS share_t1,
    (SELECT COUNT(*) FROM tmp_a_v2 CROSS JOIN tmp_p p WHERE first_apply <= p.v2_start + 3) AS n_apply_d3,
    (SELECT COUNT(*) FROM tmp_a_v2 a INNER JOIN tmp_r_v2 r ON r.user_id = a.user_id AND r.tertile = 1
        CROSS JOIN tmp_p p WHERE a.first_apply <= p.v2_start + 3) AS n_apply_top_d3,
    (SELECT COUNT(*) FROM tmp_a_v2 CROSS JOIN tmp_p p WHERE first_apply <= p.v2_start + 11) AS n_apply_d11,
    (SELECT COUNT(*) FROM tmp_a_v2 a INNER JOIN tmp_r_v2 r ON r.user_id = a.user_id AND r.tertile = 1
        CROSS JOIN tmp_p p WHERE a.first_apply <= p.v2_start + 11) AS n_apply_top_d11;

-- R5 v2 三分位
SELECT r.tertile,
       COUNT(*) AS n_churn,
       COUNT(a.user_id) AS n_apply,
       ROUND(100.0 * COUNT(a.user_id) / NULLIF(COUNT(*), 0), 2) AS apply_rate_pct
FROM tmp_r_v2 r
LEFT JOIN tmp_a_v2 a ON a.user_id = r.user_id
GROUP BY r.tertile
ORDER BY r.tertile;

-- R6 v2 按首次提单日
SELECT a.first_apply::text AS d,
       (a.first_apply - (SELECT v2_start FROM tmp_p)) AS day_idx,
       COUNT(*) AS n_apply,
       SUM(CASE WHEN r.tertile = 1 THEN 1 ELSE 0 END) AS n_apply_top
FROM tmp_a_v2 a
INNER JOIN tmp_r_v2 r ON r.user_id = a.user_id
GROUP BY 1, 2
ORDER BY 1;

-- R7 v2 五档
DROP TABLE IF EXISTS tmp_g_v2;
CREATE TEMP TABLE tmp_g_v2 AS
SELECT
    u.user_id,
    u.score,
    CASE
        WHEN pct < 0.10 THEN 'S'
        WHEN pct < 0.20 THEN 'A'
        WHEN pct < 0.40 THEN 'B'
        WHEN pct < 0.70 THEN 'C'
        ELSE 'D'
    END AS grp
FROM (
    SELECT
        user_id,
        score,
        (ROW_NUMBER() OVER (ORDER BY score DESC NULLS LAST, user_id) - 1)::numeric
            / NULLIF(COUNT(*) OVER (), 0) AS pct
    FROM tmp_u_v2
) u;

SELECT
    g.grp,
    COUNT(*) AS n_churn,
    COUNT(a.user_id) AS n_apply,
    ROUND(100.0 * COUNT(a.user_id) / NULLIF(COUNT(*), 0), 2) AS apply_rate_pct,
    MIN(g.score) AS score_min,
    MAX(g.score) AS score_max
FROM tmp_g_v2 g
LEFT JOIN tmp_a_v2 a ON a.user_id = g.user_id
GROUP BY g.grp
ORDER BY CASE g.grp WHEN 'S' THEN 1 WHEN 'A' THEN 2 WHEN 'B' THEN 3 WHEN 'C' THEN 4 WHEN 'D' THEN 5 END;

-- R8 用户级打分（算 AUC）v1
SELECT r.user_id, r.score, r.tertile, CASE WHEN a.user_id IS NULL THEN 0 ELSE 1 END AS y
FROM tmp_r_v1 r
LEFT JOIN tmp_a_v1 a ON a.user_id = r.user_id;

-- R9 用户级打分（算 AUC）v2
SELECT r.user_id, r.score, r.tertile, CASE WHEN a.user_id IS NULL THEN 0 ELSE 1 END AS y
FROM tmp_r_v2 r
LEFT JOIN tmp_a_v2 a ON a.user_id = r.user_id;
