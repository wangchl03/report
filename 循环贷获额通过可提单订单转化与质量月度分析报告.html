七、口径说明

可提单循环贷订单获额通过（is_pass1=1）且 recycle_type=2 的获额订单；获额月份按 vir_date 分组。
提单率与时间间隔提单率=提单订单数÷可提单订单数；时间间隔=提单时间−获额时间，以天计，取提单订单平均值。
降息券使用率使用 type=3 降息券的提单订单数÷提单订单数；按 user_id、serial_id 去重，避免分期多行重复。
逾期率仅统计已放款且已到期（is_due=1）的提单订单；逾期率=状态为 8 的订单数÷到期放款订单数。
盈利率仅统计已放款、已到期且状态为 7 或 8 的提单订单：SUM(repaid_amt)÷SUM(remit_amt)−1；排除状态为 5 的在途订单。
成熟度提示8 月统计截止至 8 月 18 日，且到期样本显著更少；其逾期率和盈利率仅作早期观察，不与完整月份做经营结论。

以下代码覆盖本报告的总体、月度趋势及按对应在途订单成功借款次数的分层指标。统计周期：2026-01-01 至 2026-08-18；降息券仅统计 type=3。

WITH cycle_credit AS (
    SELECT DISTINCT ON (v.user_id, v.serial_id)
        v.user_id,
        v.serial_id,
        to_timestamp(v.vir_unix) AS vir_time,
        v.vir_date::date AS vir_date
    FROM wangchuanliang.order_vir_f_copy v
    INNER JOIN wangchuanliang.side_recycle_type_copy r
        ON r.serial_id = v.serial_id
    WHERE v.is_pass1 = 1
      AND r.recycle_type = 2
      AND v.vir_date >= DATE '2026-01-01'
      AND v.vir_date < DATE '2026-08-19'
    ORDER BY v.user_id, v.serial_id, v.vir_unix DESC
),
applied_order AS (
    SELECT
        c.user_id, c.serial_id, c.vir_time, c.vir_date,
        o.apply_time, o.is_remit, o.is_due, o.loan_status_code,
        o.remit_amt, o.repaid_amt
    FROM cycle_credit c
    LEFT JOIN wangchuanliang.order_loan_f_v2_copy o
        ON o.user_id = c.user_id
       AND o.serial_id = c.serial_id
       AND o.apply_date IS NOT NULL
       AND o.apply_time >= c.vir_time
),
coupon_3 AS (
    SELECT DISTINCT user_id, serial_id
    FROM wangchuanliang.coupon_user_detail_copy
    WHERE type = 3
),
base AS (
    SELECT a.*,
           CASE WHEN c.serial_id IS NOT NULL THEN 1 ELSE 0 END AS has_coupon_3
    FROM applied_order a
    LEFT JOIN coupon_3 c
      ON c.user_id = a.user_id AND c.serial_id = a.serial_id
),
metrics AS (
    SELECT
        to_char(date_trunc('month', vir_date), 'YYYY-MM') AS month,
        COUNT(*) AS eligible_offer_count,
        COUNT(apply_time) AS applied_order_count,
        COUNT(apply_time)::numeric / NULLIF(COUNT(*), 0) AS apply_rate,
        AVG(CASE WHEN apply_time IS NOT NULL
            THEN EXTRACT(EPOCH FROM (apply_time - vir_time)) / 86400.0 END) AS avg_apply_delay_days,
        SUM(CASE WHEN apply_time IS NOT NULL AND has_coupon_3 = 1 THEN 1 ELSE 0 END) AS coupon_apply_count,
        SUM(CASE WHEN apply_time IS NOT NULL AND has_coupon_3 = 1 THEN 1 ELSE 0 END)::numeric
            / NULLIF(COUNT(apply_time), 0) AS coupon_use_rate,
        SUM(CASE WHEN apply_time IS NOT NULL AND is_remit = 1 AND is_due = 1 THEN 1 ELSE 0 END) AS due_order_count,
        SUM(CASE WHEN apply_time IS NOT NULL AND is_remit = 1 AND is_due = 1
                      AND loan_status_code = 8 THEN 1 ELSE 0 END) AS overdue_order_count,
        SUM(CASE WHEN apply_time IS NOT NULL AND is_remit = 1 AND is_due = 1
                      AND loan_status_code = 8 THEN 1 ELSE 0 END)::numeric
            / NULLIF(SUM(CASE WHEN apply_time IS NOT NULL AND is_remit = 1 AND is_due = 1
                               THEN 1 ELSE 0 END), 0) AS overdue_rate,
        SUM(CASE WHEN apply_time IS NOT NULL AND is_remit = 1 AND is_due = 1
                      AND loan_status_code IN (7, 8) THEN repaid_amt ELSE 0 END)
            / NULLIF(SUM(CASE WHEN apply_time IS NOT NULL AND is_remit = 1 AND is_due = 1
                               AND loan_status_code IN (7, 8) THEN remit_amt ELSE 0 END), 0) - 1 AS profit_rate
    FROM base
    GROUP BY GROUPING SETS ((to_char(date_trunc('month', vir_date), 'YYYY-MM')), ())
)
SELECT
    COALESCE(month, '总计') AS period,
    eligible_offer_count AS 可提单循环贷获额订单数,
    applied_order_count AS 提单订单数,
    ROUND(apply_rate * 100, 2) AS 提单率_pct,
    ROUND(avg_apply_delay_days, 4) AS 平均提单间隔_天,
    coupon_apply_count AS 使用降息券提单订单数,
    ROUND(coupon_use_rate * 100, 2) AS 降息券使用率_pct,
    due_order_count AS 到期放款订单数,
    overdue_order_count AS 逾期订单数,
    ROUND(overdue_rate * 100, 2) AS 逾期率_pct,
    ROUND(profit_rate * 100, 2) AS 盈利率_pct
FROM metrics
ORDER BY CASE WHEN month IS NULL THEN 0 ELSE 1 END, month;

all_success_loans AS (
    SELECT o.user_id, o.serial_id, o.remit_time, o.repaid_time, o.loan_status_code,

             PARTITION BY o.user_id ORDER BY o.remit_time, o.serial_id

SELECT
    CASE WHEN active_loan_success_no >= 11 THEN '第11次及以上'
         ELSE '第' || active_loan_success_no || '次' END AS 对应在途订单成功借款次数,
    COUNT(*) AS 可提单订单数,
    ROUND(COUNT(apply_time)::numeric / NULLIF(COUNT(*), 0) * 100, 2) AS 提单率_pct,

        / NULLIF(COUNT(apply_time), 0) * 100, 2) AS 降息券使用率_pct,
    ROUND(SUM(CASE WHEN apply_time IS NOT NULL AND is_remit = 1 AND is_due = 1
                      AND loan_status_code = 8 THEN 1 ELSE 0 END)::numeric

WITH eligible_cycle_offer AS (
    SELECT DISTINCT ON (v.user_id, v.serial_id)
        v.user_id,
        v.serial_id,
        v.vir_date::date AS vir_date,
        to_timestamp(v.vir_unix) AS vir_time,
        v.is_pass1,
        r.recycle_type
    FROM wangchuanliang.order_vir_f_copy v
    INNER JOIN wangchuanliang.side_recycle_type_copy r
      ON r.serial_id = v.serial_id
    WHERE v.is_pass1 = 1
      AND r.recycle_type = 2
      AND v.vir_date >= DATE '2026-01-01'
      AND v.vir_date < DATE '2026-08-19'
    ORDER BY v.user_id, v.serial_id, v.vir_unix DESC
),
interest_coupon AS (
    SELECT
        user_id,
        serial_id,
        MAX(coupon_no) AS coupon_no,
        SUM(coupon_amount) AS coupon_amt
    FROM wangchuanliang.coupon_user_detail_copy
    WHERE type = 3
    GROUP BY user_id, serial_id
)
SELECT
    e.user_id,
    e.serial_id AS 获额订单号,
    e.vir_date AS 获额日期,
    e.vir_time AS 获额时间,
    o.apply_date AS 提单日期,
    o.apply_time AS 提单时间,
    EXTRACT(EPOCH FROM (o.apply_time - e.vir_time)) / 86400.0 AS 提单间隔_天,
    o.is_remit,
    o.is_due,
    o.loan_status_code,
    o.remit_amt,
    o.repaid_amt,
    CASE WHEN c.serial_id IS NOT NULL THEN 1 ELSE 0 END AS 是否使用降息券,
    c.coupon_no,
    c.coupon_amt
FROM eligible_cycle_offer e
LEFT JOIN wangchuanliang.order_loan_f_v2_copy o
  ON o.user_id = e.user_id
 AND o.serial_id = e.serial_id
 AND o.apply_date IS NOT NULL
 AND o.apply_time >= e.vir_time
LEFT JOIN interest_coupon c
  ON c.user_id = e.user_id
 AND c.serial_id = e.serial_id
ORDER BY e.vir_time, e.user_id, e.serial_id;

SELECT
    COALESCE(to_char(date_trunc('month', vir_date), 'YYYY-MM'), '总计') AS 获额月份,
    SUM(CASE WHEN is_remit = 1 AND is_due = 1 THEN 1 ELSE 0 END) AS 到期放款订单数,
    SUM(CASE WHEN is_remit = 1 AND is_due = 1 AND loan_status_code = 8
             THEN 1 ELSE 0 END) AS 逾期订单数,
    ROUND(
      SUM(CASE WHEN is_remit = 1 AND is_due = 1 AND loan_status_code = 8 THEN 1 ELSE 0 END)::numeric
      / NULLIF(SUM(CASE WHEN is_remit = 1 AND is_due = 1 THEN 1 ELSE 0 END), 0) * 100, 2
    ) AS 逾期率_pct,
    ROUND((
      SUM(CASE WHEN is_remit = 1 AND is_due = 1 AND loan_status_code IN (7, 8)
               THEN repaid_amt ELSE 0 END)
      / NULLIF(SUM(CASE WHEN is_remit = 1 AND is_due = 1 AND loan_status_code IN (7, 8)
                         THEN remit_amt ELSE 0 END), 0)
      - 1
    ) * 100, 2) AS 盈利率_pct
FROM applied_order
GROUP BY GROUPING SETS ((to_char(date_trunc('month', vir_date), 'YYYY-MM')), ())
ORDER BY CASE WHEN to_char(date_trunc('month', vir_date), 'YYYY-MM') IS NULL THEN 0 ELSE 1 END,
         to_char(date_trunc('month', vir_date), 'YYYY-MM');
