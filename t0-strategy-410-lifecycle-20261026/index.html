"""Descriptive fixed-weight comparisons of T0 timing cohorts.

This is standardization of observed book profit, not a treatment-effect model.
All cell weights use the same pooled D0+delayed principal distribution.
"""
from pathlib import Path
from decimal import Decimal
import json, math
import pandas as pd

ROOT=Path(__file__).resolve().parent
VALID_BANDS=['HIGH20','MID30','LOW50']
LIFE=['1','2','3','4','5','6-10','11-20','21+']
def js(v):
    if isinstance(v,dict):return {str(k):js(x) for k,x in v.items()}
    if isinstance(v,(tuple,list)):return [js(x) for x in v]
    if isinstance(v,Decimal):return str(v)
    if pd.isna(v):return None
    if hasattr(v,'item'):return js(v.item())
    return v
def total(s):return sum((Decimal(str(v)) for v in s if pd.notna(v)),Decimal(0))

def standardized(df,scope,delayed,keys,min_orders=30):
    selected_scope='main' if scope=='jan_sep' and 'period' in keys else scope
    grain=df[df.scope.eq(selected_scope)&df.grain.eq('event')&df.bucket.isin(['D0',delayed])]
    if scope=='jan_sep' and 'period' in keys:
        grain=grain[grain.period.between('2026-01','2026-09')]
    if 'period' in keys and not (~grain.period.eq('ALL')).any():
        return {'scope':scope,'delayed':delayed,'dimensions':keys,'min_orders_each_cell':min_orders,
                'cells':None,'estimable':False,'unavailable_reason':'Monthly cells were not extracted for this maturity scope; no zero-support inference.'}
    if keys==['score_band']:
        grain=grain[grain.level.eq('risk')&grain.period.eq('ALL')&grain.score_band.isin(VALID_BANDS)]
    elif keys==['lifecycle_band']:
        grain=grain[grain.level.eq('lifecycle')&grain.period.eq('ALL')&grain.lifecycle_band.isin(LIFE)]
    else:
        grain=grain[grain.level.eq('joint')&grain.score_band.isin(VALID_BANDS)&grain.lifecycle_band.isin(LIFE)]
        grain=grain[~grain.period.eq('ALL') if 'period' in keys else grain.period.eq('ALL')]
    a=grain[grain.bucket.eq('D0')].set_index(keys)
    b=grain[grain.bucket.eq(delayed)].set_index(keys)
    assert not a.index.duplicated().any() and not b.index.duplicated().any()
    shared=a.index.intersection(b.index)
    valid=[k for k in shared if float(a.loc[k,'principal'])>0 and float(b.loc[k,'principal'])>0
           and int(a.loc[k,'mature_orders'])>=min_orders and int(b.loc[k,'mature_orders'])>=min_orders]
    full={x:{'principal':total(grain[grain.bucket.eq(x)].principal),'profit':total(grain[grain.bucket.eq(x)].profit),'orders':int(grain[grain.bucket.eq(x)].mature_orders.sum())} for x in ['D0',delayed]}
    if not valid:return {'scope':scope,'delayed':delayed,'dimensions':keys,'min_orders_each_cell':min_orders,'cells':0,'estimable':False}
    aa=a.loc[valid];bb=b.loc[valid]
    pa=aa.principal.astype(float).to_numpy();pb=bb.principal.astype(float).to_numpy()
    ra=aa.profit.astype(float).to_numpy()/pa;rb=bb.profit.astype(float).to_numpy()/pb
    weights=(pa+pb)/(pa+pb).sum()
    pointa=float((weights*ra).sum());pointb=float((weights*rb).sum())
    out={'scope':scope,'delayed':delayed,'dimensions':keys,'min_orders_each_cell':min_orders,'cells':len(valid),'estimable':True,
         'D0_standardized_profit_rate':pointa,'delayed_standardized_profit_rate':pointb,'delayed_minus_D0_pp':100*(pointb-pointa),
         'weight_method':'same pooled principal weights on common cells; no causal effect or remaining-unmeasured-confounder adjustment',
         'D0_common_principal':float(pa.sum()),'delayed_common_principal':float(pb.sum()),
         'D0_common_raw_profit_rate':float(aa.profit.astype(float).sum()/pa.sum()),
         'delayed_common_raw_profit_rate':float(bb.profit.astype(float).sum()/pb.sum()),
         'D0_principal_coverage':float(pa.sum()/float(full['D0']['principal'])) if full['D0']['principal'] else None,
         'delayed_principal_coverage':float(pb.sum()/float(full[delayed]['principal'])) if full[delayed]['principal'] else None,
         'D0_common_orders':int(aa.mature_orders.sum()),'delayed_common_orders':int(bb.mature_orders.sum()),
         'coverage_denominator':'same dimensions valid-score/known-lifecycle sample before minimum-cell restriction; not all T0',
         'cell_weights':[{**dict(zip(keys,(k if isinstance(k,tuple) else (k,)))),'weight':float(w),'D0_orders':int(a.loc[k,'mature_orders']),'delayed_orders':int(b.loc[k,'mature_orders'])} for k,w in zip(valid,weights)]}
    return out

def main():
    df=pd.read_csv(ROOT/'cells.csv',dtype={'period':'string','lifecycle_band':'string','score_band':'string','principal':'string','repaid':'string','profit':'string'})
    results=[]
    for scope in ['main','jan_sep','recovery30']:
      for delayed in ['D1+','D4+']:
        for dims in [['score_band'],['lifecycle_band'],['score_band','lifecycle_band'],['period','score_band','lifecycle_band']]:
            results.append(standardized(df,scope,delayed,dims))
    summary={
      'method':'common-support descriptive standardization, no treatment effect; same D0 and delayed weights',
      'predeclared_min_orders_per_arm_cell':30,
      'results':results,
      'warnings':['Financing and maturity are post-offer selections; results conditional on funded/overdue-date-complete first applications.',
                  'User lifecycle is the settled order successful funding sequence at this T0, never current lifetime total.',
                  'Percentile cuts fixed on all complete-day valid-score T0 offer events; ties are not split.',
                  'Coverage is conditional on observed valid dimensions. Missing-score users are separately reported, not reallocated.']}
    (ROOT/'composition_analysis.json').write_text(json.dumps(js(summary),ensure_ascii=False,indent=2,allow_nan=False))
    flat=[{k:v for k,v in x.items() if k!='cell_weights'} for x in results]
    for x in flat:x['dimensions']=' + '.join(x['dimensions'])
    pd.DataFrame(flat).to_csv(ROOT/'composition_standardized.csv',index=False)
    print(json.dumps({'comparisons':len(results),'estimable':sum(x['estimable'] for x in results)},ensure_ascii=False))

if __name__=='__main__':main()

def analyze_rules(d):
    lowp=d.p_v4_d3.to_numpy(float)<.5
    codes,users=pd.factorize(d.user_id,sort=True)
    u=len(users);rng=np.random.default_rng(SEED)
    boot=rng.multinomial(u,np.full(u,1/u),size=REPLICATES)
    lowp_user=np.bincount(codes,weights=lowp,minlength=u)
    boot_lowp=boot@lowp_user
    rows=[]
    for name,label,eligible in rule_specs(d):
        selected=eligible&lowp
        arrays=count_arrays(d,eligible,selected)
        counts={k:float(np.sum(v)) for k,v in arrays.items()}
        point=derive(counts,int(lowp.sum()))
        keys=list(arrays)
        user_matrix=np.column_stack([np.bincount(codes,weights=arrays[k],minlength=u) for k in keys])
        samples=boot@user_matrix
        bc={k:samples[:,i] for i,k in enumerate(keys)}
        bmetrics=derive(bc,boot_lowp)
        row={'rule':name,'description':label,**{k:int(v) for k,v in counts.items()},
             'cohort_users':u,'eligible_users':int(d.loc[eligible,'user_id'].nunique()),
             'selected_all_users':int(d.loc[selected,'user_id'].nunique()),
             'selected_d0_evaluable_users':int(d.loc[selected&d.y_d0.notna().to_numpy(),'user_id'].nunique()),
             'selected_d3_evaluable_users':int(d.loc[selected&d.y_d3.notna().to_numpy(),'user_id'].nunique()),
             **{k:float(v) for k,v in point.items()}}
        for k,v in bmetrics.items():
            f=np.asarray(v)[np.isfinite(v)]
            row[k+'_ci95_low']=float(np.quantile(f,.025)) if len(f) else None
            row[k+'_ci95_high']=float(np.quantile(f,.975)) if len(f) else None
            row[k+'_bootstrap_valid']=len(f)
        row['zero_observed_d0_error_warning']=bool(counts['selected_d0_known_events']>0 and counts['selected_d0_apply_events']==0)
        rows.append(row)
    return rows

 ), r410_profiles_unbinned AS (

   COALESCE(m.score_null_rows,0) AS score_null_rows,COALESCE(m.score_valid_rows,0) AS score_valid_rows,
   COALESCE(m.score_placeholder_rows,0) AS score_placeholder_rows,
   COALESCE(m.score_nonpositive_rows,0) AS score_nonpositive_rows,
   COALESCE(m.score_above3000_rows,0) AS score_above3000_rows,
   COALESCE(m.score_other_invalid_rows,0) AS score_other_invalid_rows,
   CASE WHEN m.serial_id IS NULL THEN 'NO_RECORD'
        WHEN m.distinct_valid_scores>1 THEN 'CONFLICT'
        WHEN m.distinct_valid_scores=1 THEN 'VALID'
        WHEN m.score_null_rows=m.score_source_rows THEN 'NULL_SCORE' ELSE 'INVALID' END AS score_status,
   CASE WHEN s.history_bad_remit_clock_n=0 AND s.remit_time<=e.clean_settle_time
             AND s.remit_time<=e.vir_time THEN s.settled_loan_seq END AS settled_loan_seq,
   CASE WHEN s.serial_id IS NULL THEN 'SETTLED_ORDER_UNMATCHED'
        WHEN s.history_bad_remit_clock_n>0 THEN 'HISTORY_REMIT_CLOCK_MISSING'
        WHEN s.remit_time>e.clean_settle_time OR s.remit_time>e.vir_time THEN 'INVALID_CHRONOLOGY'
        ELSE 'VALID' END AS sequence_status,
   s.history_bad_remit_clock_n,
   e.apply_time::date-e.vir_date AS apply_day,
   CASE WHEN e.is_remit=1 AND e.remit_amt>0 THEN 1 ELSE 0 END AS funded,
   e.data_cutoff_ts::date-e.due_date AS recovery_age_days
 FROM r410_events e LEFT JOIN r410_model m
   ON m.user_id=e.user_id AND m.serial_id=e.credit_serial_id
 LEFT JOIN r410_fundseq s ON s.user_id=e.user_id AND s.serial_id=e.settled_serial_id
 ), r410_ordered_scores AS (
 SELECT score_410,ROW_NUMBER() OVER(ORDER BY score_410) AS score_rn,COUNT(*) OVER() AS valid_n
 FROM r410_profiles_unbinned WHERE score_status='VALID'
 ), r410_thresholds AS (
 SELECT MAX(valid_n) AS valid_n,
   MIN(CASE WHEN score_rn>=CEIL(valid_n*0.5) THEN score_410 END) AS q50,
   MIN(CASE WHEN score_rn>=CEIL(valid_n*0.8) THEN score_410 END) AS q80
 FROM r410_ordered_scores
 ), r410_profiles AS (
 SELECT e.*,
   CASE WHEN score_status<>'VALID' THEN score_status
        WHEN score_410>=t.q80 THEN 'HIGH20'
        WHEN score_410>=t.q50 THEN 'MID30' ELSE 'LOW50' END AS score_band,
   CASE WHEN settled_loan_seq IS NULL THEN 'UNKNOWN'
        WHEN settled_loan_seq BETWEEN 1 AND 5 THEN settled_loan_seq::text
        WHEN settled_loan_seq BETWEEN 6 AND 10 THEN '6-10'
        WHEN settled_loan_seq BETWEEN 11 AND 20 THEN '11-20' ELSE '21+' END AS lifecycle_band
 FROM r410_profiles_unbinned e CROSS JOIN r410_thresholds t
 ), r410_bucket_defs AS (
 SELECT 'D0'::text AS bucket UNION ALL SELECT 'D1' UNION ALL SELECT 'D2'
 UNION ALL SELECT 'D3' UNION ALL SELECT 'D4+' UNION ALL SELECT 'D4-D15'
 UNION ALL SELECT 'D16+' UNION ALL SELECT 'D1-D3' UNION ALL SELECT 'D1+'
 UNION ALL SELECT 'ALL' UNION ALL SELECT 'no_apply'
 ), r410_staged AS (
 SELECT e.*,b.bucket
 FROM r410_profiles e CROSS JOIN r410_bucket_defs b
 WHERE b.bucket='ALL' OR (b.bucket='no_apply' AND apply_serial_id IS NULL)
    OR (b.bucket='D0' AND apply_day=0) OR (b.bucket='D1' AND apply_day=1)
    OR (b.bucket='D2' AND apply_day=2) OR (b.bucket='D3' AND apply_day=3)
    OR (b.bucket='D4+' AND apply_day>=4) OR (b.bucket='D4-D15' AND apply_day BETWEEN 4 AND 15)
    OR (b.bucket='D16+' AND apply_day>=16) OR (b.bucket='D1-D3' AND apply_day BETWEEN 1 AND 3)
    OR (b.bucket='D1+' AND apply_day>=1)
 ), r410_grained AS (
 SELECT e.*,'event'::text AS grain FROM r410_staged e
 ), r410_scopes AS (
 SELECT 'main'::text AS scope,'mature_complete_days'::text AS scenario
 UNION ALL SELECT 'jan_sep','mature_complete_days'
 UNION ALL SELECT 'recovery30','mature_30d'
 UNION ALL SELECT 'all_funded','all_funded'
 ), r410_scoped AS (
 SELECT e.*,s.scope,s.scenario,
   CASE WHEN funded=1 AND ((s.scenario='all_funded')
     OR (s.scenario='mature_complete_days' AND due_date>=DATE '2000-01-01' AND due_date<data_cutoff_ts::date)
     OR (s.scenario='mature_30d' AND due_date>=DATE '2000-01-01' AND recovery_age_days>=30)) THEN 1 ELSE 0 END AS eligible
 FROM r410_grained e CROSS JOIN r410_scopes s
 WHERE s.scope<>'jan_sep' OR vir_date<DATE '2026-10-01'
 ), r410_periods AS (
 SELECT e.*,'ALL'::text AS period FROM r410_scoped e
 UNION ALL SELECT e.*,TO_CHAR(vir_date,'YYYY-MM') FROM r410_scoped e WHERE scope='main'
 ), r410_cells AS (
 SELECT scope,grain,period,scenario,bucket,
   CASE WHEN GROUPING(score_band)=0 AND GROUPING(lifecycle_band)=0 THEN 'joint'
        WHEN GROUPING(score_band)=0 THEN 'risk' WHEN GROUPING(lifecycle_band)=0 THEN 'lifecycle'
        ELSE 'overall' END AS level,
   CASE WHEN GROUPING(score_band)=1 THEN 'ALL' ELSE score_band END AS score_band,
   CASE WHEN GROUPING(lifecycle_band)=1 THEN 'ALL' ELSE lifecycle_band END AS lifecycle_band,
   COUNT(*) AS event_count,COUNT(DISTINCT user_id) AS users,COUNT(apply_serial_id) AS first_apply_orders,
   SUM(funded) AS funded_orders,SUM(eligible) AS mature_orders,
   SUM(CASE WHEN eligible=1 AND repaid_amt IS NOT NULL THEN remit_amt ELSE 0 END) AS principal,
   SUM(CASE WHEN eligible=1 AND repaid_amt IS NOT NULL THEN repaid_amt ELSE 0 END) AS repaid,
   SUM(CASE WHEN eligible=1 AND repaid_amt IS NULL THEN 1 ELSE 0 END) AS missing_repaid_orders,
   SUM(CASE WHEN eligible=1 AND repaid_amt IS NULL THEN remit_amt ELSE 0 END) AS missing_repaid_principal,
   SUM(CASE WHEN eligible=1 AND loan_status_code=8 THEN 1 ELSE 0 END) AS state8_orders,
   SUM(CASE WHEN score_status='VALID' THEN 1 ELSE 0 END) AS score_valid_events,
   SUM(CASE WHEN score_status='CONFLICT' THEN 1 ELSE 0 END) AS score_conflict_events,
   SUM(CASE WHEN score_status='INVALID' THEN 1 ELSE 0 END) AS score_invalid_events,
   SUM(CASE WHEN score_status='NO_RECORD' THEN 1 ELSE 0 END) AS score_no_record_events,
   SUM(CASE WHEN score_status='NULL_SCORE' THEN 1 ELSE 0 END) AS score_null_events,
   SUM(CASE WHEN sequence_status<>'VALID' THEN 1 ELSE 0 END) AS lifetime_unknown_events,
   SUM(CASE WHEN eligible=1 AND loan_day IS NOT NULL THEN loan_day ELSE 0 END) AS loan_day_sum,
   SUM(CASE WHEN eligible=1 AND loan_day IS NOT NULL THEN 1 ELSE 0 END) AS loan_day_n,
   SUM(CASE WHEN eligible=1 AND recovery_age_days IS NOT NULL THEN recovery_age_days ELSE 0 END) AS recovery_age_sum,
   SUM(CASE WHEN eligible=1 AND recovery_age_days IS NOT NULL THEN 1 ELSE 0 END) AS recovery_age_n
 FROM r410_periods
 GROUP BY GROUPING SETS (
  (scope,grain,period,scenario,bucket,score_band,lifecycle_band),
  (scope,grain,period,scenario,bucket,score_band),
  (scope,grain,period,scenario,bucket,lifecycle_band),
  (scope,grain,period,scenario,bucket))
 ), r410_exact_lifetime AS (
 SELECT grain,bucket,settled_loan_seq,COUNT(*) AS event_count,COUNT(DISTINCT user_id) AS users,
   SUM(funded) AS funded_orders,
   SUM(CASE WHEN funded=1 AND due_date>=DATE '2000-01-01' AND due_date<data_cutoff_ts::date THEN 1 ELSE 0 END) AS mature_orders
 FROM r410_grained GROUP BY grain,bucket,settled_loan_seq
 ), r410_quality AS (
 SELECT COUNT(*) AS events,COUNT(DISTINCT user_id) AS users,
   COUNT(*)-COUNT(DISTINCT user_id::text||':'||credit_serial_id::text) AS duplicate_events,
   COUNT(apply_serial_id)-COUNT(DISTINCT CASE WHEN apply_serial_id IS NOT NULL THEN user_id::text||':'||apply_serial_id::text END) AS duplicate_first_order_assignments,
   SUM(CASE WHEN source_order_rows>1 THEN 1 ELSE 0 END) AS duplicate_financial_source_orders,
   SUM(CASE WHEN clean_settle_time>vir_time OR vir_time::date<>vir_date OR apply_time<vir_time OR apply_time>=next_t0_time OR apply_time>=data_cutoff_ts THEN 1 ELSE 0 END) AS invalid_time_events,
   SUM(CASE WHEN score_status='VALID' THEN 1 ELSE 0 END) AS score_valid_events,
   SUM(CASE WHEN score_status='CONFLICT' THEN 1 ELSE 0 END) AS score_conflict_events,
   SUM(CASE WHEN score_status='INVALID' THEN 1 ELSE 0 END) AS score_invalid_events,
   SUM(CASE WHEN score_status='NO_RECORD' THEN 1 ELSE 0 END) AS score_no_record_events,
   SUM(CASE WHEN score_status='NULL_SCORE' THEN 1 ELSE 0 END) AS score_null_events,
   SUM(COALESCE(score_source_rows,0)) AS score_source_rows,
   SUM(score_null_rows) AS score_null_rows,SUM(score_valid_rows) AS score_valid_rows,
   SUM(score_placeholder_rows) AS score_placeholder_rows,SUM(score_nonpositive_rows) AS score_nonpositive_rows,
   SUM(score_above3000_rows) AS score_above3000_rows,SUM(score_other_invalid_rows) AS score_other_invalid_rows,
   SUM(CASE WHEN score_source_rows>1 THEN 1 ELSE 0 END) AS model_duplicate_row_events,
   SUM(CASE WHEN score_status='VALID' AND (score_placeholder_rows+score_nonpositive_rows+score_above3000_rows+score_other_invalid_rows)>0 THEN 1 ELSE 0 END) AS valid_with_invalid_companion_events,
   SUM(CASE WHEN sequence_status<>'VALID' THEN 1 ELSE 0 END) AS lifetime_unknown_events,
   MIN(settled_loan_seq) AS min_settled_loan_seq,MAX(settled_loan_seq) AS max_settled_loan_seq,
   MIN(vir_date) AS first_event_date,MAX(vir_date) AS last_event_date,
   MAX(data_cutoff_ts) AS observation_cutoff
 FROM r410_profiles
 ), r410_validation_mapping AS (
 SELECT credit_serial_id AS event_id,credit_serial_id,user_id,vir_time,TO_CHAR(vir_date,'YYYY-MM') AS month,
   score_410,score_status,score_band,NULL::timestamp AS score_time,
   settled_serial_id,settled_loan_seq,lifecycle_band,sequence_status,score_source_rows,distinct_valid_scores
 FROM r410_profiles WHERE credit_serial_id IN (
       /* 核验订单编号由运行方填入；公开版不携带个人订单编号。 */
       NULL::bigint
     )
 )
 SELECT 'cell'::text AS record_type,row_to_json(x)::text AS payload FROM r410_cells x
 UNION ALL SELECT 'exact_lifecycle',row_to_json(x)::text FROM r410_exact_lifetime x
 UNION ALL SELECT 'quality',row_to_json(x)::text FROM r410_quality x
 UNION ALL SELECT 'thresholds',row_to_json(x)::text FROM r410_thresholds x
 UNION ALL SELECT 'validation_mapping',row_to_json(x)::text FROM r410_validation_mapping x
;
