口径定义

def percentile_cont(values,counts,p):
    """PostgreSQL percentile_cont: linear interpolation at (N-1)*p."""
    pairs=sorted(zip(values,counts));n=sum(int(c) for _,c in pairs)
    if not n:return None
    rank=(n-1)*p;lo=math.floor(rank);hi=math.ceil(rank)
    def at(k):
        running=0
        for v,c in pairs:
            running+=int(c)
            if k<running:return float(v)
        raise AssertionError('rank out of range')
    return at(lo)+(at(hi)-at(lo))*(rank-lo)

def segment_hits_rect(x1, y1, x2, y2, rect):
    """Liang–Barsky segment/rectangle intersection, including boundaries."""
    left, top, right, bottom = rect
    dx, dy = x2-x1, y2-y1
    p, q = [-dx, dx, -dy, dy], [x1-left, right-x1, y1-top, bottom-y1]
    lo, hi = 0.0, 1.0
    for pi, qi in zip(p, q):
        if pi == 0:
            if qi < 0:
                return False
        elif pi < 0:
            lo = max(lo, qi/pi)
        else:
            hi = min(hi, qi/pi)
    return lo <= hi

              if r[f'{prefix}_{tier}_pct'] is not None]
    if not values:
        return 0, 100
    lo, hi = min(values), max(values)
    pad = max(2.0, (hi-lo)*0.10)
    lower = max(0, int(math.floor((lo-pad)/5)*5))
    upper = min(100, int(math.ceil((hi+pad)/5)*5))
    if upper-lower < 10:
        lower = max(0, lower-5)
        upper = min(100, upper+5)
    return lower, upper


def chart(model, prefix, cid):
    """Line composition chart with all series and collision-checked good labels."""
    rows = model['rows']
    W, H = 1000, 480
    left, right, top, bottom = 56, 947, 26, 410
    # Horizontal inset keeps first/last data labels clear of the y-axis labels.
    xs = [left+35+(right-left-70)*i/max(1, len(rows)-1) for i in range(len(rows))]
    axis_min, axis_max = axis_domain(model)
    y = lambda v: bottom - (bottom-top)*(v-axis_min)/(axis_max-axis_min)
    out = [f'<svg id="{cid}" class="composition-chart" data-axis-min="{axis_min}" data-axis-max="{axis_max}" viewBox="0 0 {W} {H}" role="img" aria-labelledby="{cid}-title {cid}-desc">']
    chartname = 'T0获额事件的分档构成' if prefix == 't0' else 'T0当天首次提单事件的分档构成'
    out += [f'<title id="{cid}-title">{model["id"]}模型：{chartname}</title>',
            f'<desc id="{cid}-desc">每月最好档、中间档、较差档在有效模型分事件中的占比；最好档直接显示数字。纵轴范围{axis_min}%—{axis_max}%，同一模型两图使用相同纵轴。连续折线和实心圆点表示实际观测值，缺分月份不补数值。星号表示有效模型分覆盖率低于99%，仅供参考。</desc>']
    for v in range(axis_min,axis_max+1,5):
        yy = y(v)
        out += [f'<line x1="{left}" y1="{yy:.2f}" x2="{right}" y2="{yy:.2f}" stroke="#263b52" stroke-width="1"/>',
                f'<text x="{left-11}" y="{yy+4:.2f}" text-anchor="end" class="axis">{v}%</text>']
    segments = []
    points = []
    for tier in ['poor','mid','good']:
        vals = [r.get(f'{prefix}_{tier}_pct') for r in rows]
        observed = [(xs[i],y(v)) for i,v in enumerate(vals) if v is not None]
        segments.extend((a[0],a[1],b[0],b[1]) for a,b in zip(observed,observed[1:]))
        if observed:
            coordinates = ' '.join(f'{px:.2f},{py:.2f}' for px,py in observed)
            out.append(f'<polyline class="data-series" data-tier="{tier}" data-point-count="{len(observed)}" points="{coordinates}" fill="none" stroke="{COLORS[tier]}" stroke-width="2.8" stroke-linecap="round" stroke-linejoin="round"/>')
        for i, (r,v) in enumerate(zip(rows, vals)):
            if v is None:
                continue
            den = r['valid_n'] if prefix == 't0' else r['valid_apply_n']
            numerator = r[f'{prefix}_{tier}_n']
            title = f'{month_label(r)} · {TIER_NAMES[tier]}：{num(numerator)} / {num(den)}次 = {pct(v)}；该月获额模型分覆盖率{pct(r["coverage"],scale=100)}'
            out.append(f'<circle class="data-marker" data-tier="{tier}" cx="{xs[i]:.2f}" cy="{y(v):.2f}" r="5" fill="{COLORS[tier]}"><title>{esc(title)}</title></circle>')
            points.append((xs[i],y(v)))
    labels = []
    for i,r in enumerate(rows):
        v = r.get(f'{prefix}_good_pct')
        if v is None:
            out.append(f'<text x="{xs[i]:.2f}" y="{(top+bottom)/2:.2f}" class="missing" text-anchor="middle">—</text>')
            continue
        center = y(v)
        label_min,label_max=max(14,top-15),bottom+8
        candidates = [center+d for d in [-19,19,-34,34,-49,49,-64,64,-79,79]] + [top-5,bottom+5]
        candidates = [cy for cy in candidates if label_min <= cy <= label_max]
        best = None
        for cy in candidates:
            box = (xs[i]-30,cy-10,xs[i]+30,cy+10)
            overlaps = sum(segment_hits_rect(*s,(box[0]-3,box[1]-3,box[2]+3,box[3]+3)) for s in segments)
            overlaps += sum(box[0]-7<=px<=box[2]+7 and box[1]-7<=py<=box[3]+7 for px,py in points)
            overlaps += sum(not(box[2]+4<a or box[0]-4>c or box[3]+4<b or box[1]-4>d) for a,b,c,d in labels)
            score = overlaps*1000 + abs(cy-center)
            if best is None or score < best[0]:
                best = (score,cy,box,overlaps)
        _,cy,box,overlaps = best
        if overlaps:
            raise ValueError(f'Cannot place chart label without collision: {cid}, {r["month"]}')
        labels.append(box)
        text = f'{v:.2f}%'
        out.append(f'<rect x="{box[0]:.2f}" y="{box[1]:.2f}" width="60" height="20" rx="4" fill="#101e30"/>')
        out.append(f'<text x="{xs[i]:.2f}" y="{cy+4:.2f}" text-anchor="middle" class="data-label" fill="{COLORS["good"]}">{text}</text>')
    for i,r in enumerate(rows):
        main = f'{int(r["month"][-2:])}月' + ('*' if r['coverage'] < .99 else '')
        out.append(f'<text x="{xs[i]:.2f}" y="{H-35}" text-anchor="middle" class="axis">{main}</text>')
        if r['month']=='2026-10':
            out.append(f'<text x="{xs[i]:.2f}" y="{H-17}" text-anchor="middle" class="axis small">1—20日</text>')
    out.append('</svg>')
    return ''.join(out)


def month_change(model):
    jan,octo = model['rows'][0],model['rows'][-1]
    prefix_note = ''
    if model['id']=='407' and jan['coverage']<.99:
        usable=[r for r in model['rows'] if r['coverage']>=.99 and r['t0_good_pct'] is not None]
        if usable:
            jan=usable[0]
            prefix_note=f'1—3月无有效分，4月覆盖率仅{pct(model["rows"][3]["coverage"],scale=100)}，不能作为比较起点。以首个覆盖率不低于99%的{int(jan["month"][-2:])}月对照10月：'
    if jan['coverage'] < .99 or octo['coverage'] < .99:
        return '<strong>1月与10月不能直接判定整体变化。</strong> 至少一端的有效分覆盖率低于99%，图中保留数据并加“*”；结果只描述已匹配有效分的事件。'
    parts=[]
    for prefix,title in [('t0','获额'),('apply','当天首次提单')]:
        values = [jan.get(f'{prefix}_good_pct'),octo.get(f'{prefix}_good_pct')]
        poor = [jan.get(f'{prefix}_poor_pct'),octo.get(f'{prefix}_poor_pct')]
        if any(v is None for v in values+poor):
            parts.append(f'{title}存在零分母，不能比较首尾月。')
            continue
        d=values[1]-values[0]
        dp=poor[1]-poor[0]
        parts.append(f'{title}最好档{pct(values[0])}→{pct(values[1])}（{d:+.2f}个百分点），较差档{pct(poor[0])}→{pct(poor[1])}（{dp:+.2f}个百分点）。')
    return prefix_note+''.join(parts) + f' 10月仅统计1—20日，与{int(jan["month"][-2:])}月完整月的比较属于阶段性结构对照。'

def validate(data):
    assert [m['id'] for m in data['models']] == ORDER, 'Four models must be in frozen order'
    months=[f'2026-{i:02d}' for i in range(1,11)]
    for m in data['models']:
        assert [r['month'] for r in m['rows']] == months
        for r in m['rows']:
            assert r['valid_n'] <= r['raw_n']
            assert r['valid_apply_n'] <= r['valid_n']
            assert sum(r[f't0_{t}_n'] for t in TIER_NAMES)==r['valid_n']
            assert sum(r[f'apply_{t}_n'] for t in TIER_NAMES)==r['valid_apply_n']
            for prefix,den in [('t0',r['valid_n']),('apply',r['valid_apply_n'])]:
                for tier in TIER_NAMES:
                    v=r[f'{prefix}_{tier}_pct']
                    assert (v is None) == (den==0), (m['id'],r['month'],prefix)
                    if v is not None:
                        assert math.isclose(v,100*r[f'{prefix}_{tier}_n']/den,abs_tol=.00001)
