怎么读这五批

首期状态 8 的金额逾期率五批都在 14.60%–17.74%，差距不大。
0818 是 17.74%，0902 是 17.55%，0930 是 16.63%，0910 是 15.90%，0917 是 14.60%。这一指标只要求首期到期，所以后几批也有样本：0818 有 5,225 笔，0930 也有 278 笔。
首期 7+ 会随批次变晚下降，不能直接读成后期批次风险更低。
0818 的首期 7+ 金额逾期率是 15.36%，0902 是 13.25%，0910 是 12.03%，0917 是 7.17%，0930 是 0.00%。0930 从 9 月 30 日到 2026-10-11 只有 11 天，首期里逾期超过 7 天的放款金额为 0。0917 的首期状态 8 金额逾期率还有 14.60%，但 7+ 只有 7.17%，差出来的是已经状态为 8、逾期天数还没超过 7 天的首期。
整笔到期占比从 0818 的 57.30% 降到 0930 的 2.50%。整笔盈利率和订单逾期率要看到期样本够不够。
0818 到期 3,353 笔，盈利率 12.78%，订单逾期率 34.75%，金额逾期率 26.91%。0902 到期 907 笔，盈利率 16.95%，订单逾期率 31.31%。0910 到期 416 笔，盈利率 24.54%，订单逾期率 29.09%。0917 只有 119 笔整笔到期。0930 只有 14 笔，盈利率 -36.92%，订单逾期率 50.00%，这两格现在不能当作成熟结论。提单率方面，0902 是 40.69%，0910 是 40.05%，0917 是 37.37%，0818 是 27.43%，0930 是 21.38%，0930 的窗口还没走完。

五批对照

提单率

分母是各批名单人数。0930 的观察窗最短。

四类逾期率

首期两条不要求整笔到期。订单逾期率和金额逾期率的分母是整笔已到期订单，后两批样本少。

口径

项目定义
0818t7recalllist_0818_0819，recall_date = 2026-08-18，提单从这一天起
0902 / 0910 / 0917 / 0930各用整张表，提单分别从 2026-09-02、09-10、09-17、09-30 起
提单订单数apply_date 在召回日和 2026-10-11 之间的订单笔数，不要求放款
提单率召回日起至少有一笔提单的人数 / 召回用户数
放款订单量is_remit = 1 且 remit_amt > 0。同一用户多笔都计入
整笔到期due_date > 2000-01-01 且 due_date < 2026-10-11
到期订单占比整笔已到期放款订单数 / 放款订单量
整期盈利率整笔已到期订单的 Σ(当前 repaid_amt − remit_amt) / Σ(remit_amt)
首期金额逾期率current_term = 1 且首期已到期。状态 8 的 remit_amt_period / 全部已到期首期的 remit_amt_period。整笔不必到期
首期7+金额逾期率同一批已到期首期里，overdue_days_period > 7 的 remit_amt_period / 同一分母。不要求状态为 8
订单逾期率整笔已到期且 loan_status_code = 8 的订单数 / 整笔已到期放款订单数
金额逾期率整笔已到期订单上，分期状态 8 的 remit_amt_period / 这些订单的 remit_amt
合计用户去重。同一人取最早召回日，每笔订单只计一次

appendix_t7_recall_five_batch_quality.sql 
-- =============================================================================
-- T7 五批召回：提单、放款、整笔到期盈利、首期逾期、整笔逾期
-- 库：kaby_dw · schema wangchuanliang
-- 批次：
--   0818  t7recalllist_0818_0819 且 recall_date = 2026-08-18，提单从 2026-08-18
--   0902  t7recalllist_0902，提单从 2026-09-02
--   0910  t7recalllist_0910，提单从 2026-09-10
--   0917  t7recalllist_0917，提单从 2026-09-17
--   0930  t7recalllist_0930，提单从 2026-09-30
-- 订单：apply_date >= 该批召回日，且 apply_date <= CURRENT_DATE
-- 放款：is_remit = 1 且 remit_amt > 0
-- 整笔到期：due_date > 2000-01-01 且 due_date < CURRENT_DATE
-- 整期盈利率：整笔已到期订单 Σ(repaid_amt - remit_amt) / Σ(remit_amt)，订单表现值
-- 首期：current_term = 1 且 due_date_period > 2000-01-01 且 due_date_period < CURRENT_DATE
--       不要求整笔到期。金额用 remit_amt_period
-- 首期金额逾期率：首期 loan_status_code_period = 8 的 remit_amt_period / 已到期首期 remit_amt_period
-- 首期7+金额逾期率：首期 overdue_days_period > 7 的 remit_amt_period / 同一分母，不要求状态 8
-- 订单逾期率：整笔已到期且 loan_status_code = 8 / 整笔已到期放款订单数
-- 金额逾期率：整笔已到期订单上，分期状态 8 的 remit_amt_period / 这些订单的 remit_amt
-- 合计：用户去重；同一用户取最早召回日，订单只计一次
-- =============================================================================
SET search_path TO wangchuanliang, public;
SET statement_timeout = 0;

DROP TABLE IF EXISTS tmp_batch_user;
CREATE TEMP TABLE tmp_batch_user AS
SELECT 1 AS batch_ord, '0818'::text AS batch_code, '8月18日'::text AS batch_name,
       DATE '2026-08-18' AS start_dt, churn_user_id::bigint AS user_id
FROM wangchuanliang.t7recalllist_0818_0819
WHERE recall_date = DATE '2026-08-18' AND churn_user_id IS NOT NULL
UNION
SELECT 2, '0902', '9月2日', DATE '2026-09-02', churn_user_id::bigint
FROM wangchuanliang.t7recalllist_0902
WHERE churn_user_id IS NOT NULL
UNION
SELECT 3, '0910', '9月10日', DATE '2026-09-10', churn_user_id::bigint
FROM wangchuanliang.t7recalllist_0910
WHERE churn_user_id IS NOT NULL
UNION
SELECT 4, '0917', '9月17日', DATE '2026-09-17', churn_user_id::bigint
FROM wangchuanliang.t7recalllist_0917
WHERE churn_user_id IS NOT NULL
UNION
SELECT 5, '0930', '9月30日', DATE '2026-09-30', churn_user_id::bigint
FROM wangchuanliang.t7recalllist_0930
WHERE churn_user_id IS NOT NULL;

INSERT INTO tmp_batch_user (batch_ord, batch_code, batch_name, start_dt, user_id)
SELECT 6, 'ALL', '合计', MIN(start_dt), user_id
FROM tmp_batch_user
GROUP BY user_id;

CREATE INDEX tmp_batch_user_uid ON tmp_batch_user (user_id);
CREATE INDEX tmp_batch_user_code ON tmp_batch_user (batch_code);

DROP TABLE IF EXISTS tmp_ord;
CREATE TEMP TABLE tmp_ord AS
SELECT
    o.user_id,
    o.serial_id,
    o.apply_date::date AS apply_date,
    CASE WHEN o.is_remit = 1 AND COALESCE(o.remit_amt, 0) > 0 THEN 1 ELSE 0 END AS is_remit,
    COALESCE(o.remit_amt, 0) AS remit_amt,
    COALESCE(o.repaid_amt, 0) AS repaid_amt,
    o.due_date::date AS due_date,
    o.loan_status_code
FROM wangchuanliang.order_loan_f_v2_copy o
WHERE o.apply_date >= DATE '2026-08-18'
  AND o.apply_date <= CURRENT_DATE
  AND o.user_id IN (SELECT user_id FROM tmp_batch_user);

CREATE INDEX tmp_ord_uid ON tmp_ord (user_id);
CREATE INDEX tmp_ord_sid ON tmp_ord (serial_id);

DROP TABLE IF EXISTS tmp_fpd;
CREATE TEMP TABLE tmp_fpd AS
SELECT serial_id, due_date_period, loan_status_code_period, overdue_days_period, remit_amt_period
FROM (
    SELECT
        p.serial_id,
        p.due_date_period::date AS due_date_period,
        p.loan_status_code_period,
        COALESCE(p.overdue_days_period, 0) AS overdue_days_period,
        COALESCE(p.remit_amt_period, 0) AS remit_amt_period,
        ROW_NUMBER() OVER (PARTITION BY p.serial_id ORDER BY p.due_date_period, p.current_term) AS rn
    FROM wangchuanliang.repayment_plan_v2_copy p
    WHERE p.is_remit = 1
      AND p.current_term = 1
      AND p.serial_id IN (SELECT serial_id FROM tmp_ord WHERE is_remit = 1)
) z
WHERE rn = 1;

CREATE INDEX tmp_fpd_sid ON tmp_fpd (serial_id);

DROP TABLE IF EXISTS tmp_st8p;
CREATE TEMP TABLE tmp_st8p AS
SELECT p.serial_id,
       SUM(CASE WHEN p.loan_status_code_period = 8 THEN COALESCE(p.remit_amt_period, 0) ELSE 0 END) AS st8_amt
FROM wangchuanliang.repayment_plan_v2_copy p
WHERE p.is_remit = 1
  AND p.serial_id IN (SELECT serial_id FROM tmp_ord WHERE is_remit = 1)
GROUP BY p.serial_id;

CREATE INDEX tmp_st8p_sid ON tmp_st8p (serial_id);

DROP TABLE IF EXISTS tmp_hit;
CREATE TEMP TABLE tmp_hit AS
SELECT
    u.batch_ord,
    u.batch_code,
    u.batch_name,
    u.start_dt,
    u.user_id,
    o.serial_id,
    o.is_remit,
    o.remit_amt,
    o.repaid_amt,
    o.loan_status_code,
    CASE
        WHEN o.is_remit = 1
         AND o.due_date > DATE '2000-01-01'
         AND o.due_date < CURRENT_DATE
        THEN 1 ELSE 0
    END AS is_due,
    CASE
        WHEN o.is_remit = 1
         AND f.due_date_period > DATE '2000-01-01'
         AND f.due_date_period < CURRENT_DATE
        THEN 1 ELSE 0
    END AS is_fpd_due,
    CASE WHEN f.loan_status_code_period = 8 THEN 1 ELSE 0 END AS fpd_st8,
    COALESCE(f.overdue_days_period, 0) AS fpd_od_days,
    COALESCE(f.remit_amt_period, 0) AS fpd_amt,
    COALESCE(s.st8_amt, 0) AS st8_period_amt
FROM tmp_batch_user u
LEFT JOIN tmp_ord o
  ON o.user_id = u.user_id
 AND o.apply_date >= u.start_dt
LEFT JOIN tmp_fpd f ON f.serial_id = o.serial_id AND o.is_remit = 1
LEFT JOIN tmp_st8p s ON s.serial_id = o.serial_id AND o.is_remit = 1;

-- R1 数据日
SELECT CURRENT_DATE::text AS as_of;

-- R2 五批 + 合计
SELECT
    batch_ord,
    batch_code,
    batch_name,
    MIN(start_dt)::text AS start_dt,
    COUNT(DISTINCT user_id) AS n_user,
    COUNT(DISTINCT serial_id) AS n_apply_order,
    COUNT(DISTINCT CASE WHEN serial_id IS NOT NULL THEN user_id END) AS n_apply_user,
    COUNT(DISTINCT CASE WHEN is_remit = 1 THEN serial_id END) AS n_remit_order,
    COUNT(DISTINCT CASE WHEN is_remit = 1 THEN user_id END) AS n_remit_user,
    COUNT(DISTINCT CASE WHEN is_due = 1 THEN serial_id END) AS n_due_order,
    SUM(CASE WHEN is_due = 1 THEN remit_amt ELSE 0 END) AS due_remit_amt,
    SUM(CASE WHEN is_due = 1 THEN repaid_amt ELSE 0 END) AS due_repaid_amt,
    SUM(CASE WHEN is_due = 1 THEN repaid_amt - remit_amt ELSE 0 END) AS profit_amt,
    COUNT(DISTINCT CASE WHEN is_fpd_due = 1 THEN serial_id END) AS n_fpd_due,
    SUM(CASE WHEN is_fpd_due = 1 THEN fpd_amt ELSE 0 END) AS fpd_due_amt,
    SUM(CASE WHEN is_fpd_due = 1 AND fpd_st8 = 1 THEN fpd_amt ELSE 0 END) AS fpd_st8_amt,
    SUM(CASE WHEN is_fpd_due = 1 AND fpd_od_days > 7 THEN fpd_amt ELSE 0 END) AS fpd_d7_amt,
    COUNT(DISTINCT CASE WHEN is_due = 1 AND loan_status_code = 8 THEN serial_id END) AS n_od_order,
    SUM(CASE WHEN is_due = 1 THEN st8_period_amt ELSE 0 END) AS od_period_amt
FROM tmp_hit
GROUP BY batch_ord, batch_code, batch_name
ORDER BY batch_ord;

-- R3 一人出现在几批名单
SELECT n_batch, COUNT(*) AS n_user
FROM (
    SELECT user_id, COUNT(*) AS n_batch
    FROM tmp_batch_user
    WHERE batch_code <> 'ALL'
    GROUP BY user_id
) z
GROUP BY n_batch
ORDER BY n_batch;
