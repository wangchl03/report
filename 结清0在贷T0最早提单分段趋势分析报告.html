主要原因：整体提单率下滑主要来自当天最早提单率下降；其从 1 月的 52.90% 降至 8 月的 39.20%。

经营含义：应优先定位获额后当日未提单的用户为何未转化，并针对其后续 1—15 天的可经营窗口优化触达、产品和体验，而不能仅用全量让利追求当天转化。

四、统计口径

对象口径
T0 分母2026-01-01 至 2026-08-31 正常结清的订单中，结清时不存在其他在途或逾期订单；且用户在结清当日、结清后获得通过额度。
获额去重同一用户连续 20 秒内多次获额，通过业务本地时间排序后仅保留 serial_id 最大的一笔；同一用户同日存在多个有效结清获额事件时，保留结清后的最早有效获额。
提单定义获额时间之后 31 天内的最早一笔 apply_time。不要求该订单最终放款成功；被拒订单也属于提单。
分段定义按最早提单日期与结清获额日期的自然日差：0 天、1—7 天、8—15 天、16—30 天；四段互斥。
分母与分子各段提单率 = 该段最早提单用户数 ÷ 当月 T0 样本数。按结清获额月份分组，而不是按后续提单月份分组。

五、附录：可复现 PGSQL

以下代码按模块构建 T0 样本并输出报告的全部月度指标。执行前请确认表名、字段权限及业务时区与当前环境一致。

模块 1：构建结清 0 在贷样本与获额去重
SET TIME ZONE 'America/Belize';

CREATE TEMP TABLE settled_orders AS
SELECT o.user_id, o.serial_id, o.apply_time, o.repaid_time, o.repaid_date::date AS repaid_date
FROM wangchuanliang.order_loan_f_v2_copy o
WHERE o.is_remit = 1
  AND o.loan_status_code = 7
  AND o.repaid_time > TIMESTAMP '2000-01-01 00:00:00'
  AND o.repaid_date >= DATE '2026-01-01'
  AND o.repaid_date <= DATE '2026-08-31';

CREATE TEMP TABLE candidate_users AS SELECT DISTINCT user_id FROM settled_orders;

CREATE TEMP TABLE all_orders AS
SELECT o.user_id, o.serial_id, o.apply_time, o.repaid_time, o.loan_status_code
FROM wangchuanliang.order_loan_f_v2_copy o
JOIN candidate_users u ON u.user_id = o.user_id;

CREATE TEMP TABLE clean_settles AS
SELECT s.*
FROM settled_orders s
WHERE NOT EXISTS (
  SELECT 1 FROM all_orders o
  WHERE o.user_id = s.user_id
    AND o.serial_id <> s.serial_id
    AND o.apply_time < s.repaid_time
    AND (o.loan_status_code IN (5, 8)
         OR (o.loan_status_code = 7 AND o.repaid_time > s.repaid_time))
);

CREATE TEMP TABLE credit_days AS
SELECT DISTINCT user_id, repaid_date - 1 AS vir_date FROM clean_settles
UNION
SELECT DISTINCT user_id, repaid_date AS vir_date FROM clean_settles;

CREATE TEMP TABLE raw_credit_pass AS
SELECT v.user_id, v.serial_id, v.vir_date::date AS vir_date,
       (TO_TIMESTAMP(v.vir_unix) AT TIME ZONE 'America/Belize') AS vir_time
FROM wangchuanliang.order_vir_f_copy v
JOIN credit_days d ON d.user_id = v.user_id AND d.vir_date = v.vir_date
WHERE v.is_pass1 = 1;

CREATE TEMP TABLE dedup_credit_pass AS
WITH ordered AS (
  SELECT *, LAG(vir_time) OVER(PARTITION BY user_id ORDER BY vir_time, serial_id) AS prev_vir_time
  FROM raw_credit_pass
), batched AS (
  SELECT *, SUM(CASE WHEN prev_vir_time IS NULL OR vir_time-prev_vir_time > INTERVAL '20 seconds'
                       THEN 1 ELSE 0 END)
           OVER(PARTITION BY user_id ORDER BY vir_time, serial_id ROWS UNBOUNDED PRECEDING) AS batch_id
  FROM ordered
)
SELECT user_id, serial_id, vir_date, vir_time
FROM (
  SELECT *, ROW_NUMBER() OVER(PARTITION BY user_id, batch_id ORDER BY serial_id DESC) AS rn
  FROM batched
) x WHERE rn = 1;
模块 2：匹配同日有效获额及获额后最早提单
CREATE TEMP TABLE t0_samples AS
SELECT user_id, repaid_date, credit_vir_time
FROM (
  SELECT c.user_id, c.repaid_date, cp.vir_time AS credit_vir_time,
         ROW_NUMBER() OVER(
           PARTITION BY c.user_id, c.repaid_date
           ORDER BY cp.vir_time, cp.serial_id DESC, c.repaid_time
         ) AS rn
  FROM clean_settles c
  JOIN dedup_credit_pass cp
    ON cp.user_id = c.user_id
   AND cp.vir_date = c.repaid_date
   AND cp.vir_time >= c.repaid_time
) x WHERE rn = 1;

CREATE TEMP TABLE first_apply AS
SELECT user_id, repaid_date, first_apply_date
FROM (
  SELECT t.user_id, t.repaid_date, a.apply_time::date AS first_apply_date,
         ROW_NUMBER() OVER(
           PARTITION BY t.user_id, t.repaid_date ORDER BY a.apply_time, a.serial_id
         ) AS rn
  FROM t0_samples t
  JOIN all_orders a
    ON a.user_id = t.user_id
   AND a.apply_time >= t.credit_vir_time
   AND a.apply_time < t.credit_vir_time + INTERVAL '31 days'
) x WHERE rn = 1;
模块 3：输出报告月度指标
WITH monthly AS (
  SELECT TO_CHAR(DATE_TRUNC('month', t.repaid_date), 'YYYY-MM') AS cohort_month,
         COUNT(*) AS t0_sample_cnt,
         SUM(CASE WHEN f.first_apply_date = t.repaid_date THEN 1 ELSE 0 END) AS day0_apply_cnt,
         SUM(CASE WHEN f.first_apply_date - t.repaid_date BETWEEN 1 AND 7 THEN 1 ELSE 0 END) AS day1_7_apply_cnt,
         SUM(CASE WHEN f.first_apply_date - t.repaid_date BETWEEN 8 AND 15 THEN 1 ELSE 0 END) AS day8_15_apply_cnt,
         SUM(CASE WHEN f.first_apply_date - t.repaid_date BETWEEN 16 AND 30 THEN 1 ELSE 0 END) AS day16_30_apply_cnt
  FROM t0_samples t
  LEFT JOIN first_apply f ON f.user_id = t.user_id AND f.repaid_date = t.repaid_date
  GROUP BY 1
), report AS (
  SELECT * FROM monthly
  UNION ALL
  SELECT '总计', SUM(t0_sample_cnt), SUM(day0_apply_cnt), SUM(day1_7_apply_cnt),
         SUM(day8_15_apply_cnt), SUM(day16_30_apply_cnt)
  FROM monthly
)
SELECT cohort_month AS 结清获额月份,
       t0_sample_cnt AS 结清0在贷获额用户数,
       day0_apply_cnt AS 当天最早提单用户数,
       ROUND(day0_apply_cnt::numeric / NULLIF(t0_sample_cnt,0) * 100,2) AS 当天提单率_pct,
       day1_7_apply_cnt AS 一至七天最早提单用户数,
       ROUND(day1_7_apply_cnt::numeric / NULLIF(t0_sample_cnt,0) * 100,2) AS 一至七天提单率_pct,
       day8_15_apply_cnt AS 八至十五天最早提单用户数,
       ROUND(day8_15_apply_cnt::numeric / NULLIF(t0_sample_cnt,0) * 100,2) AS 八至十五天提单率_pct,
       day16_30_apply_cnt AS 十六至三十天最早提单用户数,
       ROUND(day16_30_apply_cnt::numeric / NULLIF(t0_sample_cnt,0) * 100,2) AS 十六至三十天提单率_pct
FROM report
ORDER BY CASE WHEN cohort_month='总计' THEN 0 ELSE 1 END, cohort_month;
本报告只讨论 T0 样本的最早提单率及时间分布；未纳入盈利率、逾期率或资质模型指标。
