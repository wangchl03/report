六、提升结清0在贷提单率的建议

获额后即时触达与页面承接

重点针对 T0 未提单订单，在获额后 0–30 分钟内触达，并打标“是否进入提单页、进入后是否查看价格、是否提交失败”。当前最大缺口位于即时提单阶段，应先定位需求、价格或链路卡点。

按意愿分层使用优惠

不建议对全部 T0 未提单订单同力度发券。优先覆盖已进入提单页、查看价格、反复点击但未提交的用户；低行为意愿用户采用低成本提醒，避免扩大无效营销成本。

建立当日转化看板与实验

以 T0 提单率作为首要指标，同时观察 T1–T5+ 是否承接。通过随机对照比较即时券、价格展示、触达时间与页面优化的增量，避免仅以延迟提单增加误判为当日链路改善。

八、统计口径

20 秒批次去重

先对获额通过、续贷获额订单按用户和获额时间排序；相邻获额时间差不超过 20 秒的记录划入同一批，仅保留 serial_id 最大的一笔。随后才判断 recycle_type。

全部可提单订单

20 秒去重后，保留 recycle_type IN (2,3) 的订单；提单时间需不早于获额时间。

结清0在贷订单

在上述基础样本中限定 recycle_type=3；用户某笔订单结清时无任何其他在途订单；获额日期与结清日期相同，且获额时间大于等于结清时间。跨日获额不纳入，即使只相差数秒。

其他结清续贷订单

完成 20 秒去重后的结清续贷可提单订单中，未被识别为结清0在贷的订单；包含主动获额及边界场景。

模块一：全部、循环贷、结清续贷的总计与月度提单率
WITH eligible_offer AS (
    SELECT DISTINCT ON (v.user_id, v.serial_id)
        v.user_id, v.serial_id, v.vir_date::date AS vir_date,
        to_timestamp(v.vir_unix) AS vir_time, r.recycle_type
    FROM wangchuanliang.order_vir_f_copy v
    INNER JOIN wangchuanliang.side_recycle_type_copy r ON r.serial_id = v.serial_id
    WHERE v.is_pass1 = 1
      AND v.loan_type_code = 2
      AND r.recycle_type IN (2, 3)
      AND v.vir_date >= DATE '2026-01-01'
      AND v.vir_date < DATE '2026-08-16'
    ORDER BY v.user_id, v.serial_id, v.vir_unix DESC
),
base AS (
    SELECT e.*, CASE WHEN o.serial_id IS NOT NULL THEN 1 ELSE 0 END AS is_applied
    FROM eligible_offer e
    LEFT JOIN wangchuanliang.order_loan_f_v2_copy o
      ON o.user_id = e.user_id AND o.serial_id = e.serial_id
     AND o.apply_date IS NOT NULL AND o.apply_time >= e.vir_time
),
metrics AS (
    SELECT to_char(date_trunc('month', vir_date), 'YYYY-MM') AS month,
           COUNT(*) AS eligible_count,
           SUM(CASE WHEN recycle_type = 2 THEN 1 ELSE 0 END) AS cycle_count,
           SUM(CASE WHEN recycle_type = 3 THEN 1 ELSE 0 END) AS type3_count,
           SUM(is_applied) AS applied_count,
           SUM(CASE WHEN recycle_type = 2 THEN is_applied ELSE 0 END) AS cycle_applied,
           SUM(CASE WHEN recycle_type = 3 THEN is_applied ELSE 0 END) AS type3_applied
    FROM base
    GROUP BY GROUPING SETS ((to_char(date_trunc('month', vir_date), 'YYYY-MM')), ())
)
SELECT COALESCE(month, '总计') AS 获额月份,
       eligible_count AS 可提单订单数,
       cycle_count AS 循环贷可提单订单数,
       type3_count AS 结清续贷可提单订单数,
       applied_count AS 可提单提单订单数,
       ROUND(applied_count::numeric / NULLIF(eligible_count, 0) * 100, 2) AS 可提单订单提单率_pct,
       ROUND(cycle_applied::numeric / NULLIF(cycle_count, 0) * 100, 2) AS 循环贷提单率_pct,
       ROUND(type3_applied::numeric / NULLIF(type3_count, 0) * 100, 2) AS 结清续贷提单率_pct
FROM metrics
ORDER BY CASE WHEN month IS NULL THEN 0 ELSE 1 END, month;
模块二：建立结清0在贷与主动获额的严格分类临时表
/* 本模块须在同一数据库会话中顺序执行。 */
CREATE TEMP TABLE tmp_eligible_type3 AS
SELECT DISTINCT ON (v.user_id, v.serial_id)
    v.user_id, v.serial_id, v.vir_date::date AS vir_date,
    to_timestamp(v.vir_unix) AS vir_time
FROM wangchuanliang.order_vir_f_copy v
INNER JOIN wangchuanliang.side_recycle_type_copy r ON r.serial_id = v.serial_id
WHERE v.is_pass1 = 1
  AND v.loan_type_code = 2
  AND r.recycle_type = 3
  AND v.vir_date >= DATE '2026-01-01'
  AND v.vir_date < DATE '2026-08-16'
ORDER BY v.user_id, v.serial_id, v.vir_unix DESC;
CREATE INDEX ON tmp_eligible_type3 (user_id, vir_date, vir_time);

CREATE TEMP TABLE tmp_candidate_users AS
SELECT DISTINCT user_id FROM tmp_eligible_type3;
CREATE INDEX ON tmp_candidate_users (user_id);

CREATE TEMP TABLE tmp_orders AS
SELECT o.user_id, o.serial_id, o.apply_time, o.repaid_time,
       o.repaid_date::date AS repaid_date, o.loan_status_code,
       o.is_remit, o.is_repaid
FROM wangchuanliang.order_loan_f_v2_copy o
INNER JOIN tmp_candidate_users u ON u.user_id = o.user_id;
CREATE INDEX ON tmp_orders (user_id, apply_time);
CREATE INDEX ON tmp_orders (user_id, repaid_date);

CREATE TEMP TABLE tmp_settled AS
SELECT user_id, serial_id, repaid_time, repaid_date
FROM tmp_orders
WHERE is_remit = 1
  AND (is_repaid = 1 OR loan_status_code = 7 OR repaid_date > DATE '2000-01-01')
  AND repaid_time > TIMESTAMP '2000-01-01 00:00:00'
  AND repaid_date >= DATE '2026-01-01'
  AND repaid_date < DATE '2026-08-16';
CREATE INDEX ON tmp_settled (user_id, repaid_date, repaid_time);

/* 在结清瞬间，无其他在途订单。 */
CREATE TEMP TABLE tmp_clean_settles AS
SELECT s.user_id, s.serial_id, s.repaid_time, s.repaid_date
FROM tmp_settled s
WHERE NOT EXISTS (
    SELECT 1 FROM tmp_orders o2
    WHERE o2.user_id = s.user_id
      AND o2.serial_id <> s.serial_id
      AND o2.apply_time < s.repaid_time
      AND (o2.loan_status_code IN (5, 8)
           OR (o2.loan_status_code = 7 AND o2.repaid_time > s.repaid_time))
);
CREATE INDEX ON tmp_clean_settles (user_id, repaid_date, repaid_time);

CREATE TEMP TABLE tmp_applied AS
SELECT DISTINCT e.user_id, e.serial_id
FROM tmp_eligible_type3 e
INNER JOIN wangchuanliang.order_loan_f_v2_copy a
  ON a.user_id = e.user_id AND a.serial_id = e.serial_id
 AND a.apply_date IS NOT NULL AND a.apply_time >= e.vir_time;
CREATE INDEX ON tmp_applied (user_id, serial_id);

/* 结清0在贷：同日、结清后（含同一秒）获额。 */
CREATE TEMP TABLE tmp_t0_offer AS
SELECT DISTINCT e.user_id, e.serial_id
FROM tmp_eligible_type3 e
INNER JOIN tmp_clean_settles c
  ON c.user_id = e.user_id
 AND c.repaid_date = e.vir_date
 AND e.vir_time >= c.repaid_time;
CREATE INDEX ON tmp_t0_offer (user_id, serial_id);

CREATE TEMP TABLE tmp_settle_date AS
SELECT DISTINCT user_id, repaid_date FROM tmp_settled;
CREATE INDEX ON tmp_settle_date (user_id, repaid_date);

CREATE TEMP TABLE tmp_labeled AS
SELECT e.user_id, e.serial_id, e.vir_date, e.vir_time,
       CASE WHEN a.serial_id IS NOT NULL THEN 1 ELSE 0 END AS is_applied,
       CASE WHEN t.serial_id IS NOT NULL THEN 1 ELSE 0 END AS is_t0,
       CASE WHEN d.user_id IS NOT NULL THEN 1 ELSE 0 END AS has_settle_on_vir_date
FROM tmp_eligible_type3 e
LEFT JOIN tmp_applied a ON a.user_id = e.user_id AND a.serial_id = e.serial_id
LEFT JOIN tmp_t0_offer t ON t.user_id = e.user_id AND t.serial_id = e.serial_id
LEFT JOIN tmp_settle_date d ON d.user_id = e.user_id AND d.repaid_date = e.vir_date;
CREATE INDEX ON tmp_labeled (user_id, vir_time);

/* 获额时点是否仍存在任何其他在途订单。 */
CREATE TEMP TABLE tmp_inflight_at_vir AS
SELECT l.user_id, l.serial_id,
       CASE WHEN COUNT(o2.serial_id) > 0 THEN 1 ELSE 0 END AS has_inflight
FROM tmp_labeled l
LEFT JOIN tmp_orders o2
  ON o2.user_id = l.user_id
 AND o2.serial_id <> l.serial_id
 AND o2.apply_time < l.vir_time
 AND (o2.loan_status_code IN (5, 8)
      OR (o2.loan_status_code = 7 AND o2.repaid_time > l.vir_time))
GROUP BY l.user_id, l.serial_id;
CREATE INDEX ON tmp_inflight_at_vir (user_id, serial_id);

CREATE TEMP TABLE tmp_classified AS
SELECT l.*,
       CASE
         WHEN l.is_t0 = 1 THEN '结清0在贷订单'
         WHEN l.has_settle_on_vir_date = 0 AND f.has_inflight = 0 THEN '主动获额订单'
         ELSE '未归入两类订单'
       END AS order_type

WITH monthly_base AS (
    SELECT
        to_char(date_trunc('month', vir_date), 'YYYY-MM') AS stat_month,
        COUNT(*) AS all_cnt,
        SUM(CASE WHEN apply_time IS NOT NULL THEN 1 ELSE 0 END) AS all_applied,
        SUM(CASE WHEN recycle_type = 2 THEN 1 ELSE 0 END) AS cycle_cnt,
        SUM(CASE WHEN recycle_type = 2 AND apply_time IS NOT NULL THEN 1 ELSE 0 END) AS cycle_applied,
        SUM(CASE WHEN recycle_type = 3 THEN 1 ELSE 0 END) AS type3_cnt,
        SUM(CASE WHEN recycle_type = 3 AND apply_time IS NOT NULL THEN 1 ELSE 0 END) AS type3_applied
    FROM tmp_eligible_offer
    GROUP BY to_char(date_trunc('month', vir_date), 'YYYY-MM')
),
monthly_t0 AS (
    SELECT
        to_char(date_trunc('month', t.vir_date), 'YYYY-MM') AS stat_month,
        COUNT(*) AS t0_cnt,
        SUM(CASE WHEN e.apply_time IS NOT NULL THEN 1 ELSE 0 END) AS t0_applied,
        SUM(CASE WHEN e.apply_time IS NOT NULL
                  AND e.apply_time::date = t.vir_date
                 THEN 1 ELSE 0 END) AS t0_d0
    FROM tmp_t0_offer t
    JOIN tmp_eligible_offer e
      ON e.user_id = t.user_id
     AND e.serial_id = t.serial_id
    GROUP BY to_char(date_trunc('month', t.vir_date), 'YYYY-MM')
),
rates AS (
    SELECT
        b.*,
        t.t0_cnt,
        t.t0_applied,
        t.t0_d0,
        b.type3_cnt - t.t0_cnt AS other_cnt,
        b.type3_applied - t.t0_applied AS other_applied,
        b.all_applied::numeric / NULLIF(b.all_cnt, 0) AS all_rate,
        b.cycle_applied::numeric / NULLIF(b.cycle_cnt, 0) AS cycle_rate,
        b.type3_applied::numeric / NULLIF(b.type3_cnt, 0) AS type3_rate,
        t.t0_applied::numeric / NULLIF(t.t0_cnt, 0) AS t0_rate,
        t.t0_d0::numeric / NULLIF(t.t0_cnt, 0) AS t0_d0_rate,
        (b.type3_applied - t.t0_applied)::numeric
          / NULLIF(b.type3_cnt - t.t0_cnt, 0) AS other_rate
    FROM monthly_base b
    JOIN monthly_t0 t ON t.stat_month = b.stat_month
),
jan AS (
    SELECT * FROM rates WHERE stat_month = '2026-01'
),
jul AS (
    SELECT * FROM rates WHERE stat_month = '2026-07'
)

    '月度图表底表' AS 输出类型,

    cycle_cnt AS 循环贷订单数,
    ROUND(cycle_rate * 100, 2) AS 循环贷提单率_pct,

本模块只用于复现报告中唯一延长至 8 月 28 日的指标：8 月结清0在贷订单的 T0 当日提单率。公共样本、20 秒批次去重、干净结清与同日获额条件均与附录 B 一致；仅将获额日期范围改为 2026-08-01 至 2026-08-28。

模块：8 月 T0 当日提单率（截至 8 月 28 日）
/* 在附录 B 的公共临时表逻辑基础上执行；将获额样本范围改为 [2026-08-01, 2026-08-29)。
   最终仅取 8 月的结清0在贷订单，计算 apply_time::date = vir_date 的占比。 */
WITH f AS (
  SELECT
    t.vir_date,
    e.apply_time
  FROM tmp_t0_offer t
  JOIN tmp_eligible_offer e
    ON e.user_id = t.user_id
   AND e.serial_id = t.serial_id
  WHERE t.vir_date >= DATE '2026-08-01'
    AND t.vir_date < DATE '2026-08-29'
)
SELECT
  COUNT(*) AS 结清0在贷订单数,
  SUM(CASE WHEN apply_time::date = vir_date THEN 1 ELSE 0 END) AS T0提单订单数,
  ROUND(
    SUM(CASE WHEN apply_time::date = vir_date THEN 1 ELSE 0 END)::numeric
    / NULLIF(COUNT(*), 0) * 100,
    2
  ) AS T0当日提单率_pct
FROM f;
