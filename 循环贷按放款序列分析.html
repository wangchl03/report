-- ============================================================================
-- 循环贷用户：按放款序列分析 — 完整 PGSQL 查询
-- 维度：循环贷 vs 非循环贷 双维度（逾期率 + 盈利率）
-- 数据源: wangchuanliang.order_loan_f_v2_copy（订单主表）
--         wangchuanliang.side_recycle_type_copy（循环贷标签）
-- 范围: 2025.01 ~ 2026.05 注册用户, 全量
-- 终态口径: 仅保留 loan_status_code IN (7,8)，排除 s5（在途）
-- 
-- 用户分类:
--   等放        = 从未被拒(is_remit=0 为 NULL)
--   迁移        = 先被拒后获批(is_remit=0 MIN < is_remit=1 MIN)
--   原生非等放  = 先获批后被拒
-- ============================================================================

WITH user_type AS (
    -- =====================================================================
    -- Step 1: 五类用户分类
    -- 通过 MIN(apply_time WHERE is_remit=0) 和 MIN(apply_time WHERE is_remit=1)
    -- 的先后关系判断用户类型
    -- =====================================================================
    SELECT user_id,
        CASE 
            WHEN MIN(CASE WHEN is_remit = 0 THEN apply_time END) IS NULL THEN '等放'
            WHEN MIN(CASE WHEN is_remit = 1 THEN apply_time END)
               < MIN(CASE WHEN is_remit = 0 THEN apply_time END) THEN '迁移'
            ELSE '原生非等放'
        END AS utype
    FROM wangchuanliang.order_loan_f_v2_copy
    WHERE reg_date >= '2025-01-01' AND reg_date <= '2026-05-31'
    GROUP BY user_id
    HAVING SUM(CASE WHEN is_remit = 1 THEN 1 ELSE 0 END) > 0
),
cycle_user AS (
    -- =====================================================================
    -- Step 2: 筛选循环贷用户
    -- 定义：生命周期中至少有一笔 recycle_type=2 订单的用户
    -- =====================================================================
    SELECT DISTINCT o.user_id
    FROM wangchuanliang.order_loan_f_v2_copy o
    JOIN wangchuanliang.side_recycle_type_copy r ON o.serial_id = r.serial_id
    WHERE o.reg_date >= '2025-01-01' AND o.reg_date <= '2026-05-31'
      AND o.is_remit = 1
      AND r.recycle_type = 2
),
loan_seq AS (
    -- =====================================================================
    -- Step 3: 放款序列编号
    -- 仅取 is_remit=1 的放款记录, 按 apply_time 排序编号
    -- 仅关联为"循环贷用户"的 user_id
    -- =====================================================================
    SELECT 
        o.user_id, ut.utype,
        o.serial_id, o.remit_amt, o.repaid_amt, 
        o.loan_status_code,
        ROW_NUMBER() OVER (PARTITION BY o.user_id ORDER BY o.apply_time) AS seq
    FROM wangchuanliang.order_loan_f_v2_copy o
    JOIN user_type ut ON o.user_id = ut.user_id
    WHERE o.is_remit = 1
      AND o.reg_date >= '2025-01-01' AND o.reg_date <= '2026-05-31'
      AND o.user_id IN (SELECT user_id FROM cycle_user)
),
recycle_tag AS (
    -- =====================================================================
    -- Step 4: 循环贷标签关联
    -- recycle_type=2: 循环贷获额
    -- recycle_type=3: 非循环贷获额
    -- 未匹配: 0（可能是未标记的普通订单）
    -- =====================================================================
    SELECT serial_id, recycle_type
    FROM wangchuanliang.side_recycle_type_copy
    WHERE recycle_type IN (2, 3)
),
loan_tag AS (
    -- =====================================================================
    -- Step 5: 终态过滤 + 循环贷标记
    -- 仅保留 loan_status_code IN (7,8)，排除 s5（在途）
    -- =====================================================================
    SELECT ls.*, COALESCE(rt.recycle_type, 0) AS cycle_type
    FROM loan_seq ls
    LEFT JOIN recycle_tag rt ON ls.serial_id = rt.serial_id
    WHERE ls.loan_status_code IN (7, 8)       -- 排除 s5 在途, 仅保留终态
      AND ls.seq <= 10                         -- 只取前 10 次放款
)
SELECT 
    utype                                                          AS "用户类型",
    seq                                                             AS "放款序列",
    COUNT(*)                                                        AS "总订单数",
    -- 循环贷
    SUM(CASE WHEN cycle_type = 2 THEN 1 ELSE 0 END)                AS "循环贷订单数",
    ROUND(SUM(CASE WHEN cycle_type = 2 THEN 1 ELSE 0 END) * 100.0 
        / COUNT(*), 1)                                              AS "循环贷占比%",
    -- 逾期率
    SUM(CASE WHEN loan_status_code = 8 THEN 1 ELSE 0 END)          AS "逾期订单数(s8)",
    ROUND(SUM(CASE WHEN loan_status_code = 8 THEN 1 ELSE 0 END) * 100.0 
        / COUNT(*), 1)                                              AS "总逾期率%",
    SUM(CASE WHEN cycle_type = 2 AND loan_status_code = 8 
        THEN 1 ELSE 0 END)                                          AS "循环贷逾期数",
    ROUND(CASE WHEN SUM(CASE WHEN cycle_type = 2 THEN 1 ELSE 0 END) > 0 
        THEN SUM(CASE WHEN cycle_type = 2 AND loan_status_code = 8 THEN 1 ELSE 0 END) * 100.0 
        / SUM(CASE WHEN cycle_type = 2 THEN 1 ELSE 0 END) 
        ELSE NULL END, 1)                                           AS "循环贷逾期率%",
    -- 非循环贷
    SUM(CASE WHEN cycle_type != 2 THEN 1 ELSE 0 END)               AS "非循环贷订单数",
    SUM(CASE WHEN cycle_type != 2 AND loan_status_code = 8 
        THEN 1 ELSE 0 END)                                          AS "非循环贷逾期数",
    ROUND(CASE WHEN SUM(CASE WHEN cycle_type != 2 THEN 1 ELSE 0 END) > 0 
        THEN SUM(CASE WHEN cycle_type != 2 AND loan_status_code = 8 THEN 1 ELSE 0 END) * 100.0 
        / SUM(CASE WHEN cycle_type != 2 THEN 1 ELSE 0 END) 
        ELSE NULL END, 1)                                           AS "非循环贷逾期率%",
    -- 放款/还款
    SUM(remit_amt)                                                  AS "总放款额",
    SUM(repaid_amt)                                                 AS "总还款额",
    ROUND((SUM(repaid_amt) - SUM(remit_amt)) * 100.0 
        / NULLIF(SUM(remit_amt), 0), 2)                             AS "总盈利率%",
    -- 循环贷盈利率
    SUM(CASE WHEN cycle_type = 2 THEN remit_amt ELSE 0 END)        AS "循环贷放款额",
    SUM(CASE WHEN cycle_type = 2 THEN repaid_amt ELSE 0 END)       AS "循环贷还款额",
    ROUND((SUM(CASE WHEN cycle_type = 2 THEN repaid_amt ELSE 0 END) 
         - SUM(CASE WHEN cycle_type = 2 THEN remit_amt ELSE 0 END)) * 100.0 
        / NULLIF(SUM(CASE WHEN cycle_type = 2 THEN remit_amt ELSE 0 END), 0), 2) AS "循环贷盈利率%",
    -- 非循环贷盈利率
    SUM(CASE WHEN cycle_type != 2 THEN remit_amt ELSE 0 END)       AS "非循环贷放款额",
    SUM(CASE WHEN cycle_type != 2 THEN repaid_amt ELSE 0 END)      AS "非循环贷还款额",
    ROUND((SUM(CASE WHEN cycle_type != 2 THEN repaid_amt ELSE 0 END) 
         - SUM(CASE WHEN cycle_type != 2 THEN remit_amt ELSE 0 END)) * 100.0 
        / NULLIF(SUM(CASE WHEN cycle_type != 2 THEN remit_amt ELSE 0 END), 0), 2) AS "非循环贷盈利率%"
FROM loan_tag
GROUP BY utype, seq
ORDER BY utype, seq;
